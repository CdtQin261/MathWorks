\chapter*{解析几何-椭圆}
\subsection*{椭圆的定义}
我们把\textbf{平面上到两个不同定点 ${F}_{1}\text{ 、 }{F}_{2}$ 的距离之和等于常数 (大于 $\left| {{F}_{1}{F}_{2}}\right|$ ) 的点的轨迹叫做椭圆}. 这两个定点 ${F}_{1}\text{ 、 }{F}_{2}$ 叫做椭圆的\textbf{焦点},两个焦点之间的距离 $\left| {{F}_{1}{F}_{2}}\right|$ 叫做椭圆的\textbf{焦距}.
\begin{example}
    取一段长为 ${2a}$ 的绳子,将它的两端都固定在图板的一点 $O$ , 将铅笔尖套在绳子里并拉紧绳子,使笔尖 $P$ 移动一周,那么笔尖点 $P$ 就画出了一个圆心为 $O$ 、 半径为 $a$ 的圆.

    把这段长为 ${2a}$ 的绳子,将它的两端分别固定在图板上不同的两点 ${F}_{1}$ 和 ${F}_{2}$ 处 $( | {{F}_{1}{F}_{2}|   < {2a}} )$ ,将铅笔尖套在绳子里并拉紧绳子,使笔尖 $P$ 顺势移动一周, 那么笔尖点 $P$ 就画出了一个椭圆.

\end{example}

\subsection*{椭圆的标准方程}
\subsubsection*{椭圆标准方程的建立}
\begin{center}
\includegraphics[width=0.25\linewidth]{bodyxb/src/0196a0c6-e677-7dd6-8032-0188357e8906_1_1084_118_391_283_0.jpg}
\end{center}
\hspace*{2em} \\

如图所示,以线段 ${F}_{1}{F}_{2}$ 的中点为坐标原点,以 $\vv{{F}_{1}{F}_{2}}$ 的方向为 $x$ 轴正方向,建立平面直角坐标系. 设椭圆的焦距 $\left| {{F}_{1}{F}_{2}}\right|  = {2c}\left( {c > 0}\right)$ ,则点 ${F}_{1}\text{ 、 }{F}_{2}$ 的坐标分别为$(-c,0)$, $(c,0)$.

设动点 $M\left( {x,y}\right)$ 是椭圆上任意一点,则点 $M$ 到两个定点 ${F}_{1}$ 、 ${F}_{2}$ 的距离之和等于常数,设为 ${2a}\left( {a > 0}\right)$，那么椭圆上点的集合为：$\left\{  {M\left| \right| M{F}_{1}\left| +\right| M{F}_{2} \mid   = {2a},{2a} > {2c} > 0}\right\}  .$

所以,由 $\left| {M{F}_{1}}\right|  + \left| {M{F}_{2}}\right|  = {2a},{2a} > {2c} > 0$ ,得$\sqrt{{\left( x + c\right) }^{2} + {y}^{2}} + \sqrt{{\left( x - c\right) }^{2} + {y}^{2}} = {2a}$\\
移项可得$\sqrt{{\left( x + c\right) }^{2} + {y}^{2}} = {2a} - \sqrt{{\left( x - c\right) }^{2} + {y}^{2}}$ .\\
两边平方得 ${\left( x + c\right) }^{2} + {y}^{2} = 4{a}^{2} - {4a}\sqrt{{\left( x - c\right) }^{2} + {y}^{2}} + {\left( x - c\right) }^{2} + {y}^{2}$ .
整理得${a}^{2} - {cx} = a\sqrt{{\left( x - c\right) }^{2} + {y}^{2}}.$ 两边再平方得$ {a}^{4} - 2{a}^{2}{cx} + {c}^{2}{x}^{2} = {a}^{2}{\left( x - c\right) }^{2} + {a}^{2}{y}^{2}.$
整理, 得$ \left( {{a}^{2} - {c}^{2}}\right) {x}^{2} + {a}^{2}{y}^{2} = {a}^{2}\left( {{a}^{2} - {c}^{2}}\right). $ \\
因为 ${2a} > {2c} > 0$ ,即 $a > c > 0$ . 所以 ${a}^{2} - {c}^{2} > 0$ ,设 ${b}^{2} = {a}^{2} - {c}^{2}$ ,且 $b > 0$，则可以得到表示焦点在 $x$ 轴上的椭圆的标准方程：$\dfrac{{x}^{2}}{{a}^{2}} + \dfrac{{y}^{2}}{{b}^{2}} = 1\left( {a > b > 0}\right).$

\begin{center}
\includegraphics[width=0.2\linewidth]{bodyxb/src/0196a0c6-e677-7dd6-8032-0188357e8906_1_1119_1264_360_377_0.jpg}
\end{center}

类似地,如果以焦点 ${F}_{1}\text{ 、 }{F}_{2}$ 所在直线为 $y$ 轴，线段 ${F}_{1}{F}_{2}$ 的垂直平分线为 $x$ 轴,建立平面直角坐标系，则可得到焦点在 $y$ 轴上的椭圆的方程： $\dfrac{{y}^{2}}{{a}^{2}} + \dfrac{{x}^{2}}{{b}^{2}} = 1\left( {a > b > 0}\right)$,其中焦点是 ${F}_{1}\left( {0, - c}\right) \text{ 、 }{F}_{2}\left( {0,c}\right)$ ,且 ${b}^{2} = {a}^{2} - {c}^{2}\left( {b > 0}\right)$.

\subsubsection*{求动点轨迹方程的一般步骤}
回顾一下求椭圆标准方程的过程, 其步骤概括如下：
\begin{dingautolist}{172}
    \item 建系:建立适当的平面直角坐标系；
    \item 设点:设出曲线上任意一点的坐标；
    \item 找等量关系:曲线上任一点满足的等量关系(几何关系)；
    \item 坐标化:把相关点的坐标代入等量关系；
    \item 化简:把曲线方程化到最简形式；
    \item 检验:检验以方程的解为坐标的点在曲线上.
\end{dingautolist}
例题：
\begin{enumerate}
    \item 椭圆 $\dfrac{{x}^{2}}{16} + \dfrac{{y}^{2}}{25} = 1$ 上点 $P$ 到上焦点的距离为 4,则点 $P$ 到下焦点的距离为\dotfill{(\kh{A})}
    \xx{6}{3}{4}{2}
    \begin{jieda}
        由椭圆方程 $\dfrac{{x}^{2}}{16} + \dfrac{{y}^{2}}{25} = 1$ ,得 ${a}^{2} = {25}$ ,即 $a = 5$ ,设下焦点为 ${F}_{1}$ ,上焦点为 ${F}_{2}$ ,则 $\left| {P{F}_{1}}\right|$  $+ \left| {P{F}_{2}}\right|  = {2a} = {10}$ ,因为 $\left| {P{F}_{2}}\right|  = 4$ ,所以 $\left| {P{F}_{1}}\right|  = 6$ ,即点 $P$ 到下焦点的距离为 6 .
    \end{jieda}
    \item 已知定点 ${F}_{1}\text{ 、 }{F}_{2}$ ,其中 ${F}_{1}\left( {-4,0}\right) ,{F}_{2}\left( {4,0}\right)$ ,动点 $P$ 满足 $\left| {P{F}_{1}}\right|  + \left| {P{F}_{2}}\right|  = 8$ ,则动点 $P$ 的轨迹是\dotfill{(\kh{D})}
    \xx{椭圆}{圆}{直线}{线段}
    \begin{jieda}
        动点$P$到两定点的距离之和等于两定点之间的距离.
    \end{jieda}

    \item
    \begin{enumerate}
        \item 若方程 $\dfrac{{x}^{2}}{{16} - m} + \dfrac{{y}^{2}}{m + 9} = 1$ 表示焦点在 $y$ 轴上的椭圆,则实数 $m$ 的取值范围是\blkda{$\left(\dfrac{7}{2}, 16\right)$}
        \item 若方程 ${x}^{2} + k{y}^{2} = 3$ 表示焦点在 $x$ 轴上的椭圆,则实数 $k$ 的取值范围是\blkda{$(1, +\infty)$}
    \end{enumerate}
    \begin{jieda}
        \begin{enumerate}
            \item 焦点在$y$轴上$\Longleftrightarrow m+9 > 16-m > 0$，因此$m \in \left(\dfrac{7}{2}, 16\right)$
            \item ${x}^{2} + k{y}^{2} = 3$即$\dfrac{x^2}{3} + \dfrac{y^2}{\dfrac{3}{k}} = 1$，焦点在 $x$ 轴上，故$k>1$
        \end{enumerate}
    \end{jieda}
    \item 已知定点 ${F}_{1}\left( {-4,0}\right) \text{ 、 }{F}_{2}\left( {4,0}\right)$ 和动点 $M\left( {x,y}\right)$ ,求满足 $\left| {M{F}_{1}}\right|  + \left| {M{F}_{2}}\right|  =$  ${2a}\left( {a > 0}\right)$ 的动点 $M$ 的轨迹.
    \begin{jieda}
        当 ${2a} > \left| {{F}_{1}{F}_{2}}\right|$ ,即 $a > 4$ 时,轨迹是以 ${F}_{1}\text{ 、 }{F}_{2}$ 为焦点的椭圆,因为 $c = 4,{b}^{2} = {a}^{2} -$  ${c}^{2} = {a}^{2} - {16}$ ,所以椭圆的方程为 $\dfrac{{x}^{2}}{{a}^{2}} + \dfrac{{y}^{2}}{{a}^{2} - {16}} = 1$ .
        
        当 ${2a} = \left| {{F}_{1}{F}_{2}}\right|$ ,即 $a = 4$ 时,轨迹是线段 ${F}_{1}{F}_{2}$ ,轨迹方程为 $y = 0\left( {-4 \leq  x \leq  4}\right)$ .
        
        当 ${2a} < \left| {{F}_{1}{F}_{2}}\right|$ ,即 $0 < a < 4$ 时,轨迹不存在. \\
        \ps 只说平面内到两点的距离为定值是不足以判断出是点的轨迹是椭圆，一定要加上对于定值的约束.
    \end{jieda}
    \item 求满足下列条件的椭圆的标准方程.
    \begin{enumerate}
        \item 焦距是 12,椭圆上一点到两个焦点的距离之和等于 16；
        \item 焦点坐标分别为 $\left( {0, - 2}\right) ,\left( {0,2}\right)$ ,且经过点 $\left( {4,3\sqrt{2}}\right)$ ；
        \item 经过两点 $\left( {\sqrt{3}, - 2}\right) ,\left( {-2\sqrt{3},1}\right)$ .
    \end{enumerate}
    \begin{jieda}
        \begin{enumerate}
            \item 由椭圆的定义可知: ${2a} = {16},{2c} = {12}$，即$a = 8, c = 6$，所以 ${b}^{2} = {a}^{2} - {c}^{2} = {64} - {36} = {28}$. 
            \begin{dingautolist}{172}
                \item 若焦点在 $x$ 轴上, 椭圆的标准方程为 $\dfrac{{x}^{2}}{64} + \dfrac{{y}^{2}}{28} = 1$
                \item 若焦点在 $y$ 轴上，椭圆的标准方程为 $\dfrac{{y}^{2}}{64} + \dfrac{{x}^{2}}{28} = 1$
            \end{dingautolist}
        \item 
            \begin{enumerate}
                \item 定义法 \\
                首先根据点$\left( {4,3\sqrt{2}}\right)$到两焦点的距离可以得到：\\
                ${2a} = \sqrt{{\left( 4 - 0\right) }^{2} + {\left( 3\sqrt{2} + 2\right) }^{2}} + \sqrt{{\left( 4 - 0\right) }^{2} + {\left( 3\sqrt{2} - 2\right) }^{2}} = \left( {6 + \sqrt{2}}\right)  + \left( {6 - \sqrt{2}}\right)  = {12}$，即$a = 6$ . 又 $c = 2$ ,所以 ${b}^{2} = {a}^{2} - {c}^{2} = {32}$. \\
                故所求椭圆的标准方程为 $\dfrac{{y}^{2}}{36} + \dfrac{{x}^{2}}{32} = 1$.
                \item 待定系数法 \\
                因为椭圆的焦点在 $y$ 轴上,设所求椭圆的标准方程为 $\dfrac{{y}^{2}}{{a}^{2}} + \dfrac{{x}^{2}}{{b}^{2}} = 1\left( {a > b > 0}\right)$ . 由题意,得 $\left\{  \begin{array}{l} \dfrac{16}{{b}^{2}} + \dfrac{18}{{a}^{2}} = 1, \\  {a}^{2} - {b}^{2} = 4, \end{array}\right.$ 解得 $\left\{  \begin{array}{l} {a}^{2} = {36}, \\  {b}^{2} = {32}. \end{array}\right.$ 故所求椭圆的标准方程为 $\dfrac{{y}^{2}}{36} + \dfrac{{x}^{2}}{32} = 1$ . \\
                \ps 定义法好处在于思路直接，但问题在于求距离开方不利于手算，有计算器的情况下，困难不大；而待定系数法运算步骤更多一些.
            \end{enumerate}
        \item 设所求椭圆的方程为 $A{x}^{2} + B{y}^{2} = 1$ ,其中 $A > 0,B > 0,A \neq  B$ . \\
        代入题干中两个点的坐标可得 $\left\{  \begin{array}{l} {3A} + {4B} = 1, \\  {12A} + B = 1, \end{array}\right.$ 解得 $\left\{  \begin{array}{l} A = \dfrac{1}{15}, \\  B = \dfrac{1}{5}. \end{array}\right.$ 故所求椭圆的标准方程为 $\dfrac{{x}^{2}}{15} + \dfrac{{y}^{2}}{5} = 1$ \\
        \ps 当不知道椭圆焦点在$x$轴还是$y$轴上时，总是可以将椭圆方程设为$A{x}^{2} + B{y}^{2} = 1$ 
        \end{enumerate}
    \end{jieda}
    \item 求与两圆 ${C}_{1} : {\left( x + 2\right) }^{2} + {y}^{2} = 1$ 和 ${C}_{2} : {x}^{2} - {4x} + {y}^{2} - {45} = 0$ 都相切的动圆圆心 $P$ 的轨迹方程.
    \begin{jieda}
        $C_1$为圆心为$O_1(-2, 0)$，半径为1的圆，$C_2$化为标准方程为：$(x-2)^2+y^2 = 49$，代表圆心为$O_2(2, 0)$，半径为$7$的圆，易知$C_1$内含在$C_2$中
        \begin{dingautolist}{172}
            \item 若圆$P$与$C_1$外切，与$C_2$内切，因此满足$|PO_1| + |PO_2| = 8$，满足椭圆的定义，因此$a = 4, c = 2$，故椭圆的标准方程为$\dfrac{x^2}{16} + \dfrac{y^2}{12} = 1$
            \item 若圆$P$与$C_1$内切，与$C_2$内切，因此满足$|PO_1| + |PO_2| = 6$，满足椭圆的定义，因此$a = 3, c = 2$，故椭圆的标准方程为$\dfrac{x^2}{9} + \dfrac{y^2}{5} = 1$
        \end{dingautolist}
    \end{jieda}
    \item 椭圆有这样的光学性质: 从椭圆的一个焦点出发的光线,经椭圆反射后,反射光线必经过椭圆的另一个焦点. 今有一个水平放置的椭圆形台球盘,点 $A$ 、 $B$ 是它的两个焦点,长轴长为 ${2a}$ ,焦距为 ${2c}$ . 当放在点 $A$ 的小球 (小球的半径不计),从点 $A$ 沿直线击出,经椭圆壁反弹后再回到点 $A$ 时,小球经过的路程是\dotfill{(\kh{A})}
    \xx{${4a}$}{$2\left( {a - c}\right)$}{$2\left( {a + c}\right)$}{以上三种情况都有可能}
    \begin{jieda}
        \ps 小球的轨迹形成了两个焦点三角形，每个焦点三角形除底边外的两边之和为$2a$，因此总路程为$4a$
    \end{jieda}
    \item 已知点 ${F}_{1}$ 、 ${F}_{2}$ 为椭圆 $\dfrac{{x}^{2}}{100} + \dfrac{{y}^{2}}{{b}^{2}} = 1\left( {0 < b < {10}}\right)$ 的左、右焦点, $P$ 是椭圆上一点.
    \begin{enumerate}
        \item 若 $\angle {F}_{1}P{F}_{2} = {60}^{ \circ  }$ 且 $\bigtriangleup {F}_{1}P{F}_{2}$ 的面积为 $\dfrac{{64}\sqrt{3}}{3}$ ,求实数 $b$ 的值
        \item 求 $\left| {P{F}_{1}}\right|  \cdot  \left| {P{F}_{2}}\right|$ 的最大值
    \end{enumerate}
    \begin{jieda}
        \begin{enumerate}
            \item 先推导一般结论：\\
            因为点 $P$ 在椭圆上，设 $\left| {P{F}_{1}}\right|  = {t}_{1},\left| {P{F}_{2}}\right|  = {t}_{2},\left| {{F}_{1}{F}_{2}}\right|  = {2c}\left( {c > 0}\right), \angle{F_1PF_2} = \theta$ ,则$S_{\triangle F_1PF_2} = \dfrac{1}{2}t_1t_2sin\theta$，根据椭圆的定义和余弦定理可得: \\
            $\left \{\begin{array}{l}
                 t_1 + t_2 = 2a \\
                 {t}_{1}^{2} + {t}_{2}^{2} - 2{t}_{1}{t}_{2} \cdot  \cos\theta = 4{c}^{2}
            \end{array} \right.$，即$\left \{\begin{array}{l}
                 t_1 + t_2 = 2a \\
                 (t_1+t_2)^2 -2t_1t_2\cdot \left(1+cos\theta\right) = 4{c}^{2}
            \end{array} \right.$，将上式代入下式可得$4a^2 - 2t_1t_2\cdot \left(1+cos\theta\right) = 4{c}^{2}$，因此$t_1t_2 = \dfrac{2b^2}{1+cos\theta}$，故$S_{\triangle F_1PF_2} = \dfrac{b^2sin\theta}{1+cos\theta} = b^2 \cdot tan\dfrac{\theta}{2}$ \\
            对于本题代入$\theta = 60^{\circ}$，得到$b^2 \cdot \dfrac{\sqrt{3}}{3} = \dfrac{64\sqrt{3}}{3}$，因此$b = 8$
            \item 
            \begin{enumerate}
                \item 利用上一问的一般结论 \\
                $\left| {P{F}_{1}}\right|  \cdot  \left| {P{F}_{2}}\right| = \dfrac{2b^2}{1+cos\theta}$，因为$\theta \in (0, \pi)$，因此$\theta$最大时，目标式最大. \\
                根据$S_{\triangle F_1PF_2} =  b^2 \cdot tan\dfrac{\theta}{2}$可知$S_{\triangle F_1PF_2}$取最大值时，$\theta$取最大值，即目标式最大.\\
                而根据三角形底边不变可以判断出$S_{\triangle F_1PF_2}$取最大值的时候，$P$位于短轴顶点，因此此时$\left| {P{F}_{1}}\right|  \cdot  \left| {P{F}_{2}}\right|$最大，为$a^2 = 100$ \\
                \ps $P$位于短轴顶点时，三角形面积$S_{\triangle F_1PF_2}$最大，夹角$\angle{F_1PF_2}$最大，两边乘积$\left| {P{F}_{1}}\right|  \cdot  \left| {P{F}_{2}}\right|$最大
                \item 利用均值不等式 \\
                根据平均值不等式 $\left| {P{F}_{1}}\right|  \cdot  \left| {P{F}_{2}}\right|  \leq  {\left( \dfrac{\left| {P{F}_{1}}\right|  + \left| {P{F}_{2}}\right| }{2}\right) }^{2} = 100$ .
            \end{enumerate}
        \end{enumerate}
    \end{jieda}
\end{enumerate}
\subsection*{椭圆的性质}
在解析几何中, 可利用曲线的方程研究曲线的几何性质. 我们从椭圆的标准方程$\dfrac{{x}^{2}}{{a}^{2}} + \dfrac{{y}^{2}}{{b}^{2}} = 1\left( {a > b > 0}\right)$出发,可以得到椭圆的下列性质.
\begin{enumerate}
    \item 对称性 \\
    在标准方程中,用 $- x$ 代换 $x$ ,或用 $- y$ 代换 $y$ ,或用 $- x$ 、 $- y$ 代换 $x$ 、 $y$，标准方程依然成立. 这就是说,若点 ${M}_{1}\left( {x,y}\right)$ 的坐标满足方程①,则它关于 $y$ 轴、 $x$ 轴、坐标原点的对称点 ${M}_{2}\left( {-x,y}\right) \text{ 、 }{M}_{3}\left( {x, - y}\right) \text{ 、 }{M}_{4}\left( {-x, - y}\right)$ 也都满足方程，所以椭圆关于 $y$ 轴、 $x$ 轴和坐标原点对称. 因此,椭圆既是\textbf{轴对称}图形, 有两条互相垂直的对称轴; 也是\textbf{中心对称}图形, 其唯一的对称中心叫做椭圆的\textbf{中心}. 在椭圆标准方程的条件下,椭圆的对称轴就是坐标轴,对称中心就是坐标原点.
    \item 顶点 \\
    椭圆与它的对称轴的交点叫做椭圆的顶点.
    
    \begin{center}
    \includegraphics[width=0.4\linewidth]{bodyxb/src/0196a0c6-e677-7dd6-8032-0188357e8906_6_1135_556_354_312_0.jpg}
    \end{center}
    
    如图所示, 对于焦点在$x$轴的椭圆，在其的标准方程中,令 $x = 0$ ,得 $y =  \pm  b$ ；令 $y = 0$ ,得 $x =$  $\pm  a$ . 这样我们就得到了椭圆与坐标轴 (对称轴) 的交点坐标: ${A}_{1}\left( {-a,0}\right) \text{ 、 }{A}_{2}\left( {a,0}\right) \text{ 、 }{B}_{1}\left( {0, - b}\right) \text{ 、 }{B}_{2}\left( {0,b}\right)$ ,这四个点都是椭圆的顶点. \\
    因为 $a > b > 0$ ,我们把线段 ${A}_{1}{A}_{2}\text{ 、 }{B}_{1}{B}_{2}$ 分别叫做椭圆的\textbf{长轴}和\textbf{短轴},长轴长等于 ${2a}$ ,短轴长等于 ${2b}$ ；而把 $a$ 和 $b$ 分别叫做椭圆的长半轴的长和短半轴的长. 椭圆的焦距为 ${2c}$ ,椭圆的半焦距为 $c$ ,三者之间的数量关系为 ${a}^{2} = {b}^{2} + {c}^{2}$.\\
    椭圆的两个焦点都在它的长轴上, 而椭圆的长轴和短轴分别在椭圆的两条相互垂直的对称轴上.
    \item 范围 \\
    由椭圆的标准方程可知,椭圆上任一点的坐标$(x, y)$都满足不等式 $\dfrac{{x}^{2}}{{a}^{2}} \leq$  $1,\dfrac{{y}^{2}}{{b}^{2}} \leq  1$ ,即 $- a \leq  x \leq  a, - b \leq  y \leq  b$ .
    
    \begin{center}
    \includegraphics[width=0.4\linewidth]{bodyxb/src/0196a0c6-e677-7dd6-8032-0188357e8906_6_1160_1180_332_240_0.jpg}
    \end{center}
    
    这就是说椭圆位于直线 $x =  \pm  a$ 和直线 $y =  \pm  b$ 所围成的矩形内.
    \item 离心率 \\

\subsection*{椭圆的第二定义}
平面内到定点的距离和定直线的距离之比为定值$e$($0 < e < 1$)的点的轨迹为椭圆. 这个定点就是椭圆的一个焦点, 这条定直线叫做椭圆的\textbf{准线},而常数 $e\left( {0 < e < 1}\right)$ 是椭圆的离心率.
        
由椭圆的对称性可知,椭圆 $\dfrac{{x}^{2}}{{a}^{2}} + \dfrac{{y}^{2}}{{b}^{2}} = 1\left( {a > b > 0}\right)$ 有两条准线: 与焦点 ${F}_{1}\left( {-c,0}\right)$ 的对应的准线方程是 $x =  - \dfrac{{a}^{2}}{c}$ ; 与焦点 ${F}_{2}\left( {c,0}\right)$ 对应的准线方程是 $x = \dfrac{{a}^{2}}{c}$ .

            易知直线$AB$的斜率不为0，因此可设$AB$为$x = my+1$，记 $A\left( {{x}_{1},{y}_{1}}\right) ,B\left( {{x}_{2},{y}_{2}}\right)$ ,则 $C\left( {2,{y}_{2}}\right)$
            
            联立直线与椭圆的方程 $\left\{  \begin{array}{l} x = my+1 \\  \dfrac{{x}^{2}}{2} + {y}^{2} = 1 \end{array}\right.$ ,则 $(my+1)^2 + 2y^2 = 2$，即$(2+m^2)y^2+2my-1 = 0$，易知$\Delta>0$
    
            根据韦达定理有$\left\{\begin{array}{l}
                 y_1+y_2 = \dfrac{-2m}{2+m^2} \\
                 y_1y_2 = -\dfrac{1}{2+m^2}
            \end{array}\right.$
    
            $\vv{NA} = \left(x_1-\dfrac{3}{2}, y_1\right) = \left(my_1-\dfrac{1}{2}, y_1\right)$，$\vv{NC} = \left(\dfrac{1}{2}, y_2\right)$，因为$\left(my_1-\dfrac{1}{2}\right)\cdot y_2 - \dfrac{1}{2}y_2 = my_1y_2 + \dfrac{y_1+y_2}{2} = 0$，所以$N,A,C$三点共线，即直线 ${AC}$ 经过线段 ${EF}$ 的中点.
        \end{jieda}

    \item 椭圆 $\dfrac{x^2}{4} + \dfrac{{y}^{2}}{3} = 1$ 的上有两点 $A\text{ 、 }B$ ,点 $B$ 在直线 $x = 4$ 上的射影是点 ${B}^{\prime }$ ,若直线 ${AB}$ 过右焦点,则直线 $A{B}^{\prime }$ 必定经过的定点坐标为\blkda{$\left(\dfrac{5}{2}, 0\right)$}.
    \begin{jieda}
        设$A(x_1, y_1), B(x_2, y_2)， B'\left(4, y_2\right)$，右焦点$F(1, 0)$

        设$AB: y = k(x-1), k = \dfrac{y_2 - y_1}{x_2 - x_1}$，设$AB': y - y_1 = k'(x - x_1), k' = \dfrac{y_2-y_1}{4 - x_1} = \dfrac{x_2-x_1}{4 - x}k$，故$AB'$可以表示为$y - k(x_1-1) = \dfrac{x_2-x_1}{4 - x_1}k(x-x_1)$

        易知$x_1 \neq 4$，故欲证$AB'$过定点，可以先去分母，即证方程$\left(4 - x_1\right)y + k(x_1-1)\left(x_1-4\right) = (x_2 - x_1)kx -x_1x_2k + kx_1^2$存在一组对任意的参数恒成立的解$(x_0, y_0)$.

        易知该方程对于$x_1, x_2$应是轮换对称的，故可以将其整理成$x_1, x_2$为主元的式子，即$\left(-y -5k+kx\right)x_1  - (kx)x_2 + \left(4y+4k+x_1x_2k\right) = 0$. 最后一项对于$x_1, x_2$是轮换对称的，因此必有前两项的系数相等，即$y + 5k - kx = kx$

        若$AB'$过定点$(x_0, y_0)$，则对任意的$x_1, x_2$，等式代入$(x_0, y_0)$均成立，则应有$y_0 + 5k - kx_0 = kx_0$恒成立，而$k$并非定值，因此$x_0 = \dfrac{5}{2}, y_0 = 0$可以使得等式恒成立，故若存在满足题干要求的顶点，则必为$\left(\dfrac{5}{2}, 0\right)$

        下证$A{B}^{\prime }$ 必定经过的定点$\left(\dfrac{5}{2}, 0\right)$：

        将$\left(\dfrac{5}{2}, 0\right)$代入$\left(-y -5k+kx\right)x_1  - (kx)x_2 + \left(4y+4k+x_1x_2k\right) = 0$，可得$LHS = -\dfrac{5}{2}k(x_1+x_2)+(4+x_1x_2)k = k\left(4+x_1x_2-\dfrac{5}{2}(x_1+x_2)\right)$
        
        联立$\left\{\begin{array}{l}
                y = k(x-1) \\
                \dfrac{x^2}{4} + \dfrac{{y}^{2}}{3} = 1
        \end{array}\right.$，消去$y$可得$3x^2+4k^2(x-1)^2 = 12$，整理得$(3+4k^2)x^2-8k^2x+(4k^2-12) = 0$，根据韦达定理有$\left\{\begin{array}{l}
                x_1+x_2 = \dfrac{8k^2}{4k^2+3} \\
                x_1x_2 = \dfrac{4k^2-12}{4k^2+3}
        \end{array}\right.$，代入可得$LHS = k\left(4+x_1x_2-\dfrac{5}{2}(x_1+x_2)\right) = 0 = RHS$
    \end{jieda}
\end{enumerate}

\subsection*{点和和椭圆的位置关系}
对于点$(x_0, y_0)$和椭圆$\dfrac{x^2}{a^2}+\dfrac{y^2}{b^2} = 1$
\begin{dingautolist}{172}
        \item $\dfrac{x_0^2}{a^2}+\dfrac{y_0^2}{b^2} < 1 \Longleftrightarrow$ 点在椭圆内
        \item $\dfrac{x_0^2}{a^2}+\dfrac{y_0^2}{b^2} = 1 \Longleftrightarrow$ 点在椭圆上
        \item $\dfrac{x_0^2}{a^2}+\dfrac{y_0^2}{b^2} > 1 \Longleftrightarrow$ 点在椭圆外
    \end{dingautolist}

椭圆上点的三角换元：对于椭圆$\dfrac{x^2}{m^2}+\dfrac{y^2}{n^2} = 1$，其上的点$P(x_0, y_0)$可表示为$\left\{\begin{array}{l}
     x_0 = mcos\theta \\
     y_0 = nsin\theta
\end{array}\right.$，此处三角换元的系数与椭圆的焦点位置无关，且$\theta$和$OP$倾斜角除特殊情况外，不是同一个角.

    \item 已知 $A,B,C$ 是椭圆 $\dfrac{{x}^{2}}{12} + \dfrac{{y}^{2}}{4} = 1$ 上三个不同的点, $\bigtriangleup  {ABC}$ 的重心为原点 $O$ ,那么 $\bigtriangleup  {ABC}$ 的面积是\blkda{9}.

    \end{jieda}
    
        \item 已知 $A\left( {0,3}\right)$ 和 $P\left( {3,\dfrac{3}{2}}\right)$ 为椭圆 $C : \dfrac{{x}^{2}}{{a}^{2}} + \dfrac{{y}^{2}}{{b}^{2}} = 1\left( {a > b > 0}\right)$ 上两点.
        \begin{enumerate}
            \item 求 $C$ 的离心率
            \item 过 $P$ 的直线 $l$ 交 $C$ 于另一点 $B$
            \begin{dingautolist}{172}
                \item 求$\bigtriangleup  {ABP}$ 的面积最大值.
                \item 若$\bigtriangleup  {ABP}$ 的面积为 9,求 $l$ 的方程
            \end{dingautolist}
        \end{enumerate}
        \begin{jieda}
            \begin{enumerate}

                \item ${AP}$ 的方程为 $x + {2y} - 6 = 0$，$\left| {AP}\right|  = \sqrt{{\left( 0 - 3\right) }^{2} + {\left( 3 - \dfrac{3}{2}\right) }^{2}} = \dfrac{3\sqrt{5}}{2}$
                
                设点 $B$ 到直线 ${AP}$ 的距离为 $d$，设$B(2\sqrt{3}cos\theta, 3sin\theta)$，则$d = \dfrac{|2\sqrt{3}cos\theta+6sin\theta-6|}{\sqrt{5}}$
                \begin{dingautolist}{172}
                    \item $d \leq \dfrac{6+4\sqrt{3}}{\sqrt{5}}$，故$S \leq \dfrac{9+6\sqrt{3}}{2}$

\subsection*{直线和椭圆的位置关系}
直线与椭圆的位置关系可以通过代数方法进行判断.
\begin{enumerate}
    \item 联立:把直线与椭圆的方程联立,构成二元二次方程组
    \item 消元:消去未知数 $x$ (或 $y$ ),得到关于 $y$ (或 $x$ )的一元二次方程（二次项系数必不为0）
    \item 计算该一元二次方程的判别式 $\Delta$ ,并得出结论:
    \begin{dingautolist}{172}
        \item $\Delta  > 0 \Longleftrightarrow$ 有两个公共点
        \item $\Delta  = 0 \Longleftrightarrow$ 有且只有一个公共点
        \item $\Delta  < 0 \Longleftrightarrow$ 没有公共点
    \end{dingautolist}
\end{enumerate}
直线与圆锥曲线的位置关系问题,一般都按照上述步骤处理,并能进一步利用根与系数的关系和其他几何等量关系来解决一些问题. 

弦长公式

已知 $A\left( {{x}_{1},{y}_{1}}\right) ,B\left( {{x}_{2},{y}_{2}}\right)$ ,直线 ${AB}$ 的斜率为 $k\left( {k \neq  0}\right)$，则 $\left| {AB}\right|  = \sqrt{{\left( {x}_{1} - {x}_{2}\right) }^{2} + {\left( {y}_{1} - {y}_{2}\right) }^{2}} = \sqrt{1 + {k}^{2}}\left| {{x}_{1} - {x}_{2}}\right|$

例题：
\begin{enumerate}
    \item 已知椭圆 $C$ 的两个焦点分别是 ${F}_{1}\left( {-1,0}\right)$ 、 ${F}_{2}\left( {1,0}\right)$ ,且椭圆 $C$ 经过点 $\left( {1,\dfrac{3}{2}}\right)$ .
    \begin{enumerate}
        \item 求椭圆 $C$ 的标准方程
        \item 当实数 $m$ 取何值时,直线 $l : y = x + m$ 与椭圆 $C$ 满足下列条件:
        \begin{dingautolist}{172}
            \item 有两个公共点
            \item 只有一个公共点
            \item 没有公共点
        \end{dingautolist}
    \end{enumerate}
    \begin{jieda}
        \begin{enumerate}
            \item 设椭圆 $C$ 的标准方程为 $\dfrac{{x}^{2}}{{a}^{2}} + \dfrac{{y}^{2}}{{b}^{2}} = 1\left( {a > b > 0}\right)$ ,由题意可得 $\left\{  \begin{array}{l} c = 1, \\ 
            \dfrac{9}{{a}^{2}} + \dfrac{4}{{b}^{2}} = 1, \end{array}\right.$ 解得

            $\left\{  \begin{array}{l} {a}^{2} = 4, \\  {b}^{2} = 3. \end{array}\right.$ 所以椭圆 $C$ 的标准方程为 $\dfrac{{x}^{2}}{4} + \dfrac{{y}^{2}}{3} = 1$ .
            
            \item 联立方程组 $\left\{  \begin{array}{l} 
                y = x + m, \\  \dfrac{{x}^{2}}{4} + \dfrac{{y}^{2}}{3} = 1. \end{array}\right.$ 消去 $y$ ,得$7{x}^{2} + {8mx} + 4{m}^{2} - {12} = 0$， \\
                则有$\Delta  = {64}{m}^{2} - 4 \times  7 \times  \left( {4{m}^{2} - {12}}\right)  = {336} - {48}{m}^{2} = {48}\left( {7 - {m}^{2}}\right)$，所以
                \begin{dingautolist}{172}
                    \item 当 $\Delta  > 0$ ,即 $- \sqrt{7} < m < \sqrt{7}$ 时,方程有两个不同的实数根,则直线 $y = x + m$ 与椭圆 $C$ 有两个公共点
                    \item 当 $\Delta  = 0$ ,即 $m =  \pm  \sqrt{7}$ 时,方程有两个相等的实数根,则直线 $y = x + m$ 与椭圆 $C$ 只有一个公共点
                    \item 当 $\Delta  < 0$ ,即 $m <  - \sqrt{7}$ 或 $m > \sqrt{7}$ 时,方程无实数根,则直线 $y = x + m$ 与椭圆 $C$ 没有公共点
                \end{dingautolist}
                综上所述,直线 $l : y = x + m$ 与椭圆 $C$ : \ding{172} 当 $- \sqrt{7} < m < \sqrt{7}$ 时有两个公共点; \ding{173} 当 $m =$  $\pm  \sqrt{7}$ 时,有一个公共点; \ding{174} 当 $m <  - \sqrt{7}$ 或 $m > \sqrt{7}$ ,没有公共点.
        \end{enumerate}
    \end{jieda}

    \item 已知 $O$ 为坐标原点,椭圆 $C : \dfrac{{x}^{2}}{{a}^{2}} + \dfrac{{y}^{2}}{{b}^{2}} = 1\left( {a > b > 0}\right)$ 的离心率为 $\dfrac{\sqrt{6}}{3}$ ,且经过点 $P\left( {\sqrt{6},1}\right)$ .
    \begin{enumerate}
        \item 求椭圆 $C$ 的方程
        \item 直线 $l$ 与椭圆 $C$ 交于 $A,B$ 两点,直线 ${OA}$ 的斜率为 ${k}_{1}$ ,直线 ${OB}$ 的斜率为 ${k}_{2}$ ,且 ${k}_{1}{k}_{2} =  - \dfrac{1}{3}$ ,求 $\vv{OA} \cdot  \vv{OB}$ 的取值范围.
    \end{enumerate}
    \begin{jieda}
        \begin{enumerate}
            \item 由题意可得 $\left\{  \begin{array}{l} \dfrac{c}{a} = \dfrac{\sqrt{6}}{3}, \\  \dfrac{6}{{a}^{2}} + \dfrac{1}{{b}^{2}} = 1, \end{array}\right.$，因为${a}^{2} = {b}^{2} + {c}^{2}$，解得 $a = 3,b = \sqrt{3}$ .
            
            所以椭圆 $C$ 的方程为 $\dfrac{{x}^{2}}{9} + \dfrac{{y}^{2}}{3} = 1$ .
            \item 设 $A\left( {{x}_{1},{y}_{1}}\right) ,B\left( {{x}_{2},{y}_{2}}\right)$
            \begin{dingautolist}{172}
                \item 当直线 $l$ 的斜率不存在时, ${x}_{2} = {x}_{1},{y}_{2} =  - {y}_{1}$ ,则 ${k}_{1}{k}_{2} =  - \dfrac{{y}_{1}^{2}}{{x}_{1}^{2}} =  - \dfrac{1}{3}$ ,又 $\dfrac{{x}_{1}^{2}}{9} + \dfrac{{y}_{1}^{2}}{3} = 1$ ,解得 ${x}_{1}^{2} = \dfrac{9}{2}$ ,则 $\vv{OA} \cdot  \vv{OB} = {x}_{1}^{2} - {y}_{1}^{2} = \dfrac{2}{3}{x}_{1}^{2} = 3$ .
                \item 当直线 $l$ 的斜率存在时,设 $l : y = {kx} + t$，联立直线和椭圆方程有 $\left\{  \begin{array}{l} y = {kx} + t, \\  \dfrac{x^2}{9} + \dfrac{{y}^{2}}{3} = 1, \end{array}\right.$
                
                消去 $y$ 得 $\left( {1 + 3{k}^{2}}\right) {x}^{2} + {6ktx} + 3{t}^{2} - 9 = 0$，根据题意可得$\Delta  = {12}\left( {3 + 9{k}^{2} - {t}^{2}}\right)  > 0$ ,
                
                根据韦达定理有$\left\{  {\begin{array}{l} {x}_{1} + {x}_{2} = \dfrac{-{6kt}}{1 + 3{k}^{2}}, \\  {x}_{1}{x}_{2} = \dfrac{3{t}^{2} - 9}{1 + 3{k}^{2}}, \end{array}}\right.$
                
                ${k}_{1}{k}_{2} = \dfrac{{y}_{1}{y}_{2}}{{x}_{1}{x}_{2}} =  - \dfrac{1}{3}$，代入直线方程可得$\dfrac{{y}_{1}{y}_{2}}{{x}_{1}{x}_{2}} = \dfrac{\left( {k{x}_{1} + t}\right) \left( {k{x}_{2} + t}\right) }{{x}_{1}{x}_{2}} = {k}^{2} + \dfrac{{kt}\left( {{x}_{1} + {x}_{2}}\right)  + {t}^{2}}{{x}_{1}{x}_{2}} = {k}^{2} + \dfrac{\dfrac{-6{k}^{2}{t}^{2}}{1 + 3{k}^{2}} + {t}^{2}}{\dfrac{3{t}^{2} - 9}{1 + 3{k}^{2}}} = \dfrac{{t}^{2} - 9{k}^{2}}{3{t}^{2} - 9} =  - \dfrac{1}{3},$
                
                整理得 $2{t}^{2} = 9{k}^{2} + 3$且$t^2 \neq 3$，故${t}^{2} \geq \dfrac{3}{2}$且$t^2 \neq 3$，经检验$\Delta  > 0$ 恒成立.
                
                $\vv{OA} \cdot  \vv{OB} = {x}_{1}{x}_{2} + {y}_{1}{y}_{2} = {x}_{1}{x}_{2} - \dfrac{1}{3}{x}_{1}{x}_{2} = \dfrac{2}{3}{x}_{1}{x}_{2} = \dfrac{2}{3} \cdot  \dfrac{3{t}^{2} - 9}{1 + 3{k}^{2}} = 3 \cdot  \dfrac{{t}^{2} - 3}{{t}^{2}} = 3\left( {1 - \dfrac{3}{{t}^{2}}}\right) \in  \lbrack  - 3,0) \cup  \left( {0,3}\right)$ .
            \end{dingautolist}
            综上, $\vv{OA} \cdot  \vv{OB}$ 的取值范围为 $\left\lbrack  {-3,0)\cup (0,3}\right\rbrack$ .
        \end{enumerate}
    \end{jieda}


    
    \item 已知椭圆 $C : \dfrac{{x}^{2}}{{a}^{2}} + \dfrac{{y}^{2}}{{b}^{2}} = 1$ 的右焦点为 $\left( {1,0}\right)$ ,且经过点 $A\left( {0,1}\right)$ .
    \begin{enumerate}
        \item 求椭圆 $C$ 的方程
        \item 设 $O$ 为原点,直线 $l : y = {kx} + t\left( {t \neq   \pm  1}\right)$ 与椭圆 $C$ 交于两个不同点 $P,Q$ ,直线 ${AP}$ 与 $x$ 轴交于点 $M$ ,直线 ${AQ}$ 与 $x$ 轴交于点 $N$ ,若 $\left| {OM}\right|  \cdot  \left| {ON}\right|  = 2$ ,求证: 直线 $l$ 经过定点.
    \end{enumerate}
    \begin{jieda}
    \begin{enumerate}
        \item 因为椭圆的右焦点为 $\left( {1,0}\right)$ ,所以 $c = 1$
        
        因为椭圆经过点 $A\left( {0,1}\right)$ ,所以 $b = 1$ ,所以 ${a}^{2} = {b}^{2} + {c}^{2} = 2$ ,故椭圆的方程为 $\dfrac{{x}^{2}}{2} + {y}^{2} = 1$ .
        \item 设 $P\left( {{x}_{1},{y}_{1}}\right) ,Q\left( {{x}_{2},{y}_{2}}\right)$

        直线 ${AP} : y - 1 = \dfrac{{y}_{1} - 1}{{x}_{1}}x$ ,令 $y = 0$ 得 $x = \dfrac{-{x}_{1}}{{y}_{1} - 1}$ ,即 $\left| {OM}\right|  = \left| \dfrac{-{x}_{1}}{{y}_{1} - 1}\right|$，同理可得 $\left| {ON}\right|  = \left| \dfrac{-{x}_{2}}{{y}_{2} - 1}\right|$

        因为 $\left| {OM}\right| \left| {ON}\right|  = 2$ ,所以 $\left| \dfrac{-{x}_{1}}{{y}_{1} - 1}\right| \left| \dfrac{-{x}_{2}}{{y}_{2} - 1}\right|  = \left| \dfrac{{x}_{1}{x}_{2}}{{y}_{1}{y}_{2} - \left( {{y}_{1} + {y}_{2}}\right)  + 1}\right|  = 2$

        联立 $\left\{  \begin{matrix} \dfrac{{x}^{2}}{2} + {y}^{2} = 1 \\  y = {kx} + t\left( {t \neq  1}\right)  \end{matrix}\right.$ 得 $\left( {1 + 2{k}^{2}}\right) {x}^{2} + {4ktx} + 2{t}^{2} - 2 = 0$ ,

        $\Delta  > 0 \Longleftrightarrow 16k^2t^2-4(1+2k^2)(2t^2-2) > 0$
        
        由韦达定理：$\left\{\begin{array}{l}
                {x}_{1} + {x}_{2} =  - \dfrac{4kt}{1 + 2{k}^{2}}  \\
                {x}_{1}{x}_{2} = \dfrac{2{t}^{2} - 2}{1 + 2{k}^{2}}

                设椭圆$E$关于$M$的对称椭圆$E'$（不是以原点为对称中心的椭圆）上的点$P'$坐标为$(x, y)$，则椭圆$E$上$P'$的对称点坐标为$P(2x_0 - x, 2y_0-y)$，又因为$P$满足圆$E$的方程因此有$\dfrac{(2x_0 - x)^2}{a^2}+\dfrac{(2y_0-y)^2}{b^2} = 1$，即圆$E'$的方程（并非标准方程）为$\dfrac{(x-2x_0)^2}{a^2}+\dfrac{(y-2y_0)^2}{b^2} = 1$
                椭圆$E$和椭圆$E'$外切，两方程相减可得其具有公共点的公切线方程$\dfrac{4x_0^2-4x_0x}{a^2}+\dfrac{4y_0^2-4y_0y}{b^2} = 0$，又因为$\dfrac{x_0^2}{a^2}+\dfrac{y_0^2}{b^2} = 1$，所以切线方程为$\dfrac{x_0x}{a^2}+\dfrac{y_0y}{b^2} = 1$
                \item 方法2 \\
                首先代入$M$的坐标可知$l$过$M$，联立方程组$\left\{  \begin{array}{l} 
                \dfrac{x_0x}{a^2}+\dfrac{y_0y}{b^2} = 1, \\  
                \dfrac{x^2}{a^2}+\dfrac{y^2}{b^2} = 1. \end{array}\right.$ 上式移项平方有$\dfrac{y_0^2y^2}{b^4} = 1 + \dfrac{x_0^2x^2}{a^4}-\dfrac{2x_0x}{a^2}$，下式移项同乘后有$\dfrac{y_0^2y^2}{b^4} = \dfrac{y_0^2}{b^2}+\dfrac{y_0^2x^2}{a^2b^2}$，两式消去$\dfrac{y_0^2y^2}{b^4}$可得$1 + \dfrac{x_0^2x^2}{a^4} - \dfrac{2x_0x}{a^2} = \dfrac{y_0^2}{b^2}+\dfrac{y_0^2x^2}{a^2b^2}$，整理可得$\left(\dfrac{x_0^2}{a^4}-\dfrac{y_0^2}{a^2b^2}\right)x^2-\dfrac{2x_0}{a^2}x + \left(1-\dfrac{y_0^2}{b^2}\right) = 0$，利用$\dfrac{x_0^2}{a^2}+\dfrac{y_0^2}{b^2} = 1$化简有$\dfrac{1}{a^2}x^2-\dfrac{2x_0}{a^2}x + \dfrac{x_0^2}{a^2} = 0$，即$x^2-2x_0x + x_0^2 = 0$，容易判断判别式$\Delta = 4x_0^2 - 4x_0^2 = 0$，得证.
            \end{enumerate}
            \item 设$A(x_1, y_1), B(x_2, y_2)$
            
            $A$处切线方程为$\dfrac{x_1x}{a^2}+\dfrac{y_1y}{b^2} = 1$，$B$处切线方程为$\dfrac{x_2x}{a^2}+\dfrac{y_2y}{b^2} = 1$，两者交点为$M(x_0, y_0)$，因此有$\left\{\begin{array}{l}
                 \dfrac{x_0x_1}{a^2}+\dfrac{y_0y_1}{b^2} = 1 \\
                 \dfrac{x_0x_2}{a^2}+\dfrac{y_0y_2}{b^2} = 1
            \end{array}\right.$，故$A,B$两点坐标均满足方程$\dfrac{x_0x}{a^2}+\dfrac{y_0y}{b^2} = 1$，因此切点弦$AB$的方程为$\dfrac{x_0x}{a^2}+\dfrac{y_0y}{b^2} = 1$
            \item 设$M(k_0acos\alpha, k_0bcos\alpha)$设直线$\dfrac{x_0x}{a^2}+\dfrac{y_0y}{b^2} = 1$上的点为$(kacos\beta, kbsin\beta)$，则有$k_0kcos(\alpha-\beta) = 1$，若$k_0\in(0, 1)$，则$k > 1$，即直线与椭圆相离.
            \end{enumerate}
    \end{jieda}

    
    \item 已知椭圆 $\Gamma  : \dfrac{{x}^{2}}{6} + \dfrac{{y}^{2}}{3} = 1,O$ 为坐标原点
    \begin{enumerate}
        \item 求$\Gamma$的离心率 $e$
        \item 设点 $N\left( {1,0}\right)$ ，点 $M$ 在 $\Gamma$ 上，求 $\left| {MN}\right|$ 的最大值和最小值
        \item 点 $T\left( {2,1}\right)$ ，点 $P$ 在直线 $x + y = 3$ 上，过点 $P$ 且与 ${OT}$ 平行的直线 $l$ 与 $\Gamma$ 交于 $A,B$ 两点；试探究: 是否存在常数 $\lambda$ ,使得 $\left| {\vv{PA} \cdot  \vv{PB}}\right|  = \lambda {\left| \vv{PT}\right| }^{2}$ 恒成立; 若存在,求出该常数的值; 若不存在. 说明理由
    \end{enumerate}
    \begin{jieda}
        \begin{enumerate}
            \item 设 $\Gamma$ 的半长轴长为 $a$ ，半短轴长为 $b$ ，半焦距为 $c$ ，则 $a = \sqrt{6},b = \sqrt{3}$ ，则 $c = \sqrt{3}$ ，所以 $e = \dfrac{c}{a} = \dfrac{\sqrt{3}}{\sqrt{6}} = \dfrac{\sqrt{2}}{2}$ .
            \item 依题意，设 $M\left( {x,y}\right)$ ，则 $- \sqrt{6} \leq  x \leq  \sqrt{6}$ ， $\dfrac{{x}^{2}}{6} + \dfrac{{y}^{2}}{3} = 1$ ，故 ${y}^{2} = 3 - \dfrac{{x}^{2}}{2}$ ，则 $\left| {MN}\right|  = \sqrt{{\left( x - 1\right) }^{2} + {y}^{2}} = \sqrt{{\left( x - 1\right) }^{2} + 3 - \dfrac{{x}^{2}}{2}} = \sqrt{\dfrac{1}{2}{x}^{2} - {2x} + 4} = \sqrt{\dfrac{1}{2}{\left( x - 2\right) }^{2} + 2}$ .
            
            所以由二次函数的性质可知，当 $x = 2$ 时， $\left| {MN}\right|$ 取得最小值为 $\sqrt{2}$ .当 $x =  - \sqrt{6}$ 时. $\left| {MN}\right|$ 取得最大值为 $\sqrt{\dfrac{1}{2}{\left( -\sqrt{6} - 2\right) }^{2} + 2} = 1 + \sqrt{6}$ .
            \item 设 $P\left( {a,3 - a}\right) ,A\left( {{x}_{1},{y}_{1}}\right) ,B\left( {{x}_{2},{y}_{2}}\right)$ ，又 $T\left( {2,1}\right)$ ，易得 ${k}_{OT} = \dfrac{1}{2}$， $x+y = 3$为椭圆在$T$处切线.

    \item 已知椭圆 $C : \dfrac{{y}^{2}}{{a}^{2}} + \dfrac{{x}^{2}}{{b}^{2}} = 1\left( {a > b > 0}\right)$ 的右顶点为 $P\left( {1,0}\right) ,C$ 的两个焦点为 ${F}_{1},{F}_{2}$ ,且 $\sin \angle P{F}_{1}{F}_{2} + \sin \angle P{F}_{2}{F}_{1} = \dfrac{2\sqrt{3}}{3}\sin \angle {F}_{1}P{F}_{2}.$

        \item 设直线 $l : y = {kx} + \sqrt{3}$ 与 $C$ 相交于 $A,B$ 两点， $B$ 关于 $y$ 轴的对称点为 $D$
