\chapter*{导数及其应用}
\subsection*{函数的极限与连续性}
\subsubsection*{函数的极限}
\begin{enumerate}
    \item $x \rightarrow x_0$时的极限
    \begin{enumerate}
        \item 邻域：设$x_0, \delta \in \mathbf{R}$，且$\delta > 0$，则开区间$(x_0 - \delta, x_0 + \delta)$称为点$x_0$的$\delta$\textbf{邻域}，记作$U(x_0, \delta)$ \\
        即$U(x_0, \delta) =  \{x | |x-x_0| < \delta\} = \{x | x_0 - \delta < x < x_0 + \delta \} = (x_0 - \delta, x_0 + \delta)$，\\
        其中$x_0$称为邻域中心，$\delta$称为邻域半径.
        \item 去心邻域：$U^{\circ}(x_0, \delta) =  \{x | 0 < |x-x_0| < \delta\}$
        \item $x \rightarrow x_0$时的极限\\
        直观理解：在自变量$x \rightarrow x_0$的过程中，对应的函数值$f(x)$无限趋近于某确定的数值$A$，则称$A$是函数$f(x)$当$x \rightarrow x_0$时的极限. \\
        定义：设函数$f(x)$在点$x_0$的某一\textbf{去心邻域}内有定义，如果存在常数$A$，对于\textbf{任意给定的正数}$\epsilon$（不论它多么小），总\textbf{存在正数}$\delta$使得当$x$满足不等式$0 < |x-x_0| < \delta$时，对应的函数值$f(x)$都满足不等式$|f(x)-A| < \epsilon$，那么常数$A$就叫做函数$f(x)$当$x \rightarrow x_0$时的极限，记作$\lim\limits_{x \to x_0}f(x) = A$. \\
        \ps 根据定义可知$f(x)$在$x_0$处是否有极限与$f(x)$在$x_0$处是否有定义并无关系. \\
        符号语言表述：$\lim\limits_{x \to x_0}f(x) = A \Longleftrightarrow \forall \epsilon > 0 , \exists \delta > 0$，当$0 < |x-x_0| < \delta$时，有$|f(x)-A| < \epsilon$ \\
        几何解释：任意给定一正数$\epsilon$，做平行于$x$轴的两条直线$y = A + \epsilon, y = A - \epsilon$，都存在点$x_0$的一个去心$\delta$邻域，满足该段上的函数图像夹在两直线间的横条区域内.
        \begin{figure}[h]
            \centering
            \includegraphics[width=0.2\linewidth]{bodyxb//src/jixian1.png}
        \end{figure}
        \begin{example}
            \begin{dingautolist}{172}
                \item 证明：$\lim\limits_{x \to 1}(2x-1) = 1$ \\
                由于$|(2x-1)-1| = 2|x-1|$，所以对于$\forall \epsilon > 0$，故$\exists \delta = \dfrac{\epsilon}{2}$使得当$0 < |x-1| < \delta$时，满足$|(2x-1)-1| = 2|x-1| < \epsilon$
                \item 证明：$\lim\limits_{x \to 1}\left(\dfrac{x^2-1}{x-1}\right) = 2$ \\
                对于$x \in U^{\circ}(1, \delta)$，$\left|\dfrac{x^2-1}{x-1} - 2\right| = \left|x-1\right|$，故$\exists \delta = \epsilon$使得当$0 < |x-1| < \delta$时，满足$\left|\dfrac{x^2-1}{x-1} - 2\right| = \left|x-1\right| < \epsilon$
            \end{dingautolist}
        \end{example}
        
    \end{enumerate}
    \item $x \rightarrow +\infty$时的极限（$-\infty$同理）
    \begin{enumerate}
        \item $x \rightarrow +\infty$时的极限 \\
        直观理解：在自变量$x \rightarrow +\infty$的过程中，对应的函数值$f(x)$无限趋近于某确定的数值$A$，则称$A$是函数$f(x)$当$x \rightarrow +\infty$时的极限.  \\
        定义：设函数$f(x)$当$x$大于某一正数时有定义，如果存在常数$A$，对于\textbf{任意给定的正数}$\epsilon$（不论它多么小），总\textbf{存在正数}$X$使得当$x$满足不等式$x > X$时，对应的函数值$f(x)$都满足不等式$|f(x)-A| < \epsilon$，那么常数$A$就叫做函数$f(x)$当$x \rightarrow +\infty$时的极限，记作$\lim\limits_{x \to +\infty}f(x) = A$. \\
        符号语言表述：$\lim\limits_{x \to +\infty}f(x) = A \Longleftrightarrow \forall \epsilon > 0 , \exists X > 0$，当$x > X$时，有$|f(x)-A| < \epsilon$ \\
        几何解释：任意给定一正数$\epsilon$，做平行于$x$轴的两条直线$y = A + \epsilon, y = A - \epsilon$，都存在正数$X$，满足$(X, +\infty)$上的函数图像夹在两直线间的横条区域内，即$y = A$是$y = f(x)$在$x \rightarrow +\infty$时的水平渐近线.
        \begin{figure}[h]
            \centering
            \includegraphics[width=0.2\linewidth]{bodyxb//src/jixian2.png}
        \end{figure}
        \end{enumerate}
        \begin{example}
            证明：$\lim\limits_{x \to +\infty}\dfrac{1}{x} = 0$ \\
            即证对$\forall \epsilon > 0, \exists X$，使得$x > X$时 有$\left|\dfrac{1}{x} - 0\right| < \epsilon$，可取$X = \dfrac{1}{\epsilon}$满足要求.
        \end{example}
\end{enumerate}
\subsubsection*{函数的极限的运算法则}
我们不加证明地给出以下运算法则和一些正例：
\begin{dingautolist}{172}
    \item $lim[f(x) \pm g(x)] = lim[f(x)] \pm lim[g(x)]$
    \item $lim[f(x) \cdot g(x)] = lim[f(x)] \cdot lim[g(x)]$
    \item $lim\left[\dfrac{f(x)}{g(x)}\right] = \dfrac{f(x)}{g(x)}$
    \item 推论1：$lim[C \cdot f(x)] = C \cdot lim[f(x)]$
    \item 推论2：$lim[f(x)]^n = [limf(x)]^n$
\end{dingautolist}
\begin{example}    
    \begin{dingautolist}{172}
        \item $\lim\limits_{x \to 2} [(x^2 + 3x) - (2x - 1)] = \lim\limits_{x \to 2} (x^2 + 3x) - \lim\limits_{x \to 2} (2x - 1) = 10 - 3 = 7$

        \item $\lim\limits_{x \to 1} [x \cdot (x + 2)] = \lim\limits_{x \to 1} x \cdot \lim\limits_{x \to 1} (x + 2) = 1 \cdot 3 = 3$
        
        \item $\lim\limits_{x \to 4} \left(\dfrac{x^2}{2x}\right) = \dfrac{\lim\limits_{x \to 4} x^2}{\lim\limits_{x \to 4} 2x} = \dfrac{16}{8} = 2$
        
        \item 推论1：$\lim\limits_{x \to 3} [5 \cdot x^2] = 5 \cdot \lim\limits_{x \to 3} x^2 = 5 \cdot 9 = 45$
        
        \item 推论2：$\lim\limits_{x \to 2} (x + 1)^3 = \left[\lim\limits_{x \to 2} (x + 1)\right]^3 = 3^3 = 27$
    \end{dingautolist}
\end{example}
\subsubsection*{函数的连续性}
\begin{enumerate}
    \item 增量 \\
    设函数$y = f(x)$在点$x_0$的某个邻域内有定义，当自变量由$x_0$变到$x$（仍然在邻域内）时，称差$x - x_0$为自变量在$x_0$处的增量，记为$\Delta x$；相应地函数值由$f(x_0)$变化到$f(x)$，称差$f(x) - f(x_0)$为函数$f(x)$在$x_0$处的增量，记为$\Delta y$ \\
    \ps $\Delta x, \Delta y$均可正可负 
    \item 连续的定义：设函数$y = f(x)$在点$x_0$的某个邻域内有定义，若$\lim\limits_{\Delta x \rightarrow 0} \Delta y = \lim\limits_{\Delta x \rightarrow 0} [f(x_0+\Delta x) - f(x_0)]= 0$，则称函数$y = f(x)$在$x_0$处连续，$x_0$称为函数$f(x)$的连续点 \\
    函数$f(x)$在点$x_0$处连续的定义又可表示为：设函数$y = f(x)$在点$x_0$的某个邻域内有定义，若$\lim\limits_{x \to x_0} f(x) = f(x_0)$，则称函数$y = f(x)$在$x_0$处连续 \\
    直观理解：函数$f(x)$在点$x_0$处连续需同时满足三个条件：\ding{172}函数在该点邻域内有定义，\ding{173}函数在该点的极限存在，\ding{174}函数在该点的极限等于该点函数值.
    \item 若函数在区间$(a, b)$内的每一点都连续，则称函数$f(x)$在$(a, b)$内连续，函数$f(x)$是区间$(a, b)$上的连续函数
    \item 连续函数常用定理
    \begin{dingautolist}{172}
        \item 若函数$f(x), g(x)$均在点$x_0$处连续，则$f(x) \pm g(x), f(x) \cdot g(x), \dfrac{f(x)}{g(x)}(g(x_0) \neq 0)$均在点$x_0$处连续
        \item 设函数$u = g(x)$在点$x_0$处连续，$y = f(u)$在$u_0 = g(x_0)$处连续. 若复合函数$y = f(g(x))$在点$x_0$的某邻域内有定义，则$y = f(g(x))$在点$x_0$处连续
        \item 基本初等函数在其定义区间（包含在定义域内的开区间）上都是连续函数
        \item 【最值定理】闭区间上的连续函数在该区间上一定存在最大值和最小值
        \item 【零点存在定理】闭区间$[a, b]$上连续的函数$f(x)$满足$f(a) \cdot f(b) < 0$，则至少存在一点$\xi$使得$f(\xi) = 0$
    \end{dingautolist}
\end{enumerate}
例题：
\begin{enumerate}
    \item 讨论函数$f(x) = \left\{\begin{array}{ll}
         \dfrac{x^2-5x+6}{x-3} & x \neq 3\\
         1 & x = 3
    \end{array}\right.$在$x = 3$处是否连续
    \begin{jieda}
        $\lim\limits_{x \to 3}f(x) = \lim\limits_{x \to 3}\dfrac{(x-3)(x-2)}{x-3} = \lim\limits_{x \to 3}(x-2) = 1$，因此$\lim\limits_{x \to 3}f(x) = f(3)$，故$f(x)$在$x = 3$处连续.
    \end{jieda}
    \item 设函数$y = f(x)$在闭区间$[a, b]$上连续，且$f(a) \neq f(b)$，$C$为介于$f(a)$与$f(b)$之间的任何数，证明至少存在一点$\xi$使得$f(\xi) = C$
    \begin{jieda}
        构造函数$g(x) = f(x) - C$，故根据零点存在定理可知$g(x)$在$[a, b]$上至少存在一个零点$\xi$，故存在$\xi$使得$g(\xi) = f(\xi) - C = 0$，即至少存在一点$\xi$使得$f(\xi) = C$.
    \end{jieda}
\end{enumerate}

\subsection*{导数的概念与意义}
\subsubsection*{导数的概念}
\begin{enumerate}
    \item 在物理学中，速度（瞬时速度）是衡量位移变化快慢的量，加速度是衡量速度变化快慢的量.位移对时间的导数是速度，速度对时间的导数是加速度.
    \begin{example}
        设物体的位移函数为 $ s(t) = t^2 + 3t $
        
        平均速度定义为位移变化量与时间变化量的比值：
        \[
        \bar{v} = \frac{\Delta s}{\Delta t} = \frac{s(t+\Delta t) - s(t)}{\Delta t}
        \]
        代入位移函数：
        \[
        \bar{v} = \frac{(t+\Delta t)^2 + 3(t+\Delta t) - (t^2 + 3t)}{\Delta t} = 2t + 3 + \Delta t
        \]
        
        当时间间隔 $\Delta t \to 0$ 时，平均速度趋近于瞬时速度：
        \[
        v(t) = \lim_{\Delta t \to 0} \frac{\Delta s}{\Delta t} = \lim_{\Delta t \to 0} (2t + 3 + \Delta t) = 2t + 3
        \]
        例如，在 $ t = 2 $ 时，瞬时速度为：$v(2) = 2 \times 2 + 3 = 7$
    \end{example}
    \item 一般地，设函数$f(x)$在$x_0$的邻域内有定义且当$\Delta x \rightarrow 0$时，$\dfrac{\Delta y}{\Delta x}$存在极限，则将此极限值称为$y = f(x)$在$x = x_0$处的\textbf{导数}，记作$f'(x_0) = \lim\limits_{\Delta x \rightarrow 0}\dfrac{\Delta y}{\Delta x} = \lim\limits_{\Delta x \rightarrow 0}\dfrac{f(x_0+\Delta x) - f(x_0)}{\Delta x}$，并称函数$y = f(x)$在$x = x_0$处\textbf{可导}；若$\lim\limits_{\Delta x \rightarrow 0}\dfrac{f(x_0+\Delta x) - f(x_0)}{\Delta x}$不存在极限，则称函数在在该点\textbf{不可导}. \\
    \ps 导数的记号也可以用$y'|_{x = x_0}, \left.\dfrac{dy}{dx}\right|_{x = x_0}, \left.\dfrac{df(x)}{dx}\right|_{x = x_0}$来表示.
    \item 一般地，$f'(x_0)$就是函数$f(x)$在$x = x_0$处的\textbf{瞬时变化率}，反应了该函数$y = f(x)$在某个自变量$x = x_0$处的瞬时变化情况.
    \item 以下函数在$x = 0$处均不可导：
    \begin{enumerate}
        \item $y = |x|$
        \item $y = \sqrt{x}$
        \item $y = \left\{\begin{array}{ll}
             x, & x \leq 0 \\
             x+1, & x > 0
        \end{array}\right.$
        \item $y = \left\{\begin{array}{ll}
             xsin \dfrac{1}{x}, & x \neq 0 \\
             0, & x = 0
        \end{array}\right.$
    \end{enumerate}
\end{enumerate}
例题：
\begin{enumerate}
    \item “函数$f(x)$在$x = x_0$处连续”是“函数$f(x)$在$x = x_0$处可导”的\blkda[3]{必要非充分}条件.
    \begin{jieda}
        必要性：$\lim\limits_{\Delta x \rightarrow 0}\dfrac{f(x_0+\Delta x) - f(x_0)}{\Delta x}$存在必然有$\lim\limits_{\Delta x \rightarrow 0}f(x_0+\Delta x) - f(x_0) = 0$ \\
        非充分性：$y = |x|$在$x = 0$处连续，但不可导.
    \end{jieda}
    \item 已知函数$f(x) = ax^2+bx+c (a, b, c \in \mathbf{R}, a \neq 0)$，试求：
    \begin{enumerate}
        \item 函数$y = f(x)$在区间$[1, 3]$上的平均变化率
        \item 函数$y = f(x)$在$x = 2$处的导数值
    \end{enumerate}
    \begin{jieda}
        根据平均变化率的定义可以得到 \\
        $\dfrac{f(x+\Delta x) - f(x)}{\Delta x} = \dfrac{a(x+\Delta x)^2+b(x+\Delta x) + c - (ax^2+bx+c)}{\Delta x} = 2ax + a(\Delta x) + b$
        \begin{enumerate}
            \item 根据上式，代入$x = 1, \Delta x = 2$，可以得到$[1, 3]$上的平均变化率为$\dfrac{f(3) - f(1)}{2} = 4a+b$
            \item 根据导数定义可得$f'(2) = \lim\limits_{\Delta x \to 0}\dfrac{f(2+\Delta x)-f(2)}{\Delta x} = \lim\limits_{\Delta x \to 0}[4a+a(\Delta x)+b] = 4a+b$
        \end{enumerate}
    \end{jieda}
\end{enumerate}
\subsubsection*{导数的几何意义}
\begin{enumerate}
    \item 切线的定义\\

    \item 导数的几何意义\\
    如图所示，不难发现，平均变化率就是函数$y = f(x)$上两点连线的斜率，而瞬时变化率就是当$\Delta x \rightarrow 0$时这两点连线的斜率，即切线的斜率.

    就像瞬时速度是物体在给定时刻运动状态的最好描述一样，在$P$处的切线就是曲线在$P$附近性质的最好的描述.
\end{enumerate}
例题：
\begin{enumerate}
    \item 求函数$f(x) = -x^2+x$的图像上点$A(2, -2)$，$B(1, 0)$两处的切线方程.
    \begin{jieda}
        由导数的几何意义可知$A$处的切线斜率为$f'(2) = \lim\limits_{\Delta x \to 0}\dfrac{f(2+\Delta x)-f(2)}{\Delta x} = \lim\limits_{\Delta x \to 0}(-3-\Delta x) = -3$，故所求切线方程为$y = -3(x-2)-2$，即$3x+y-4=0$

        由导数的几何意义可知$B$处的切线斜率为$f'(1) = \lim\limits_{\Delta x \to 0}\dfrac{f(1+\Delta x)-f(1)}{\Delta x} = \lim\limits_{\Delta x \to 0}(-1-\Delta x) = -1$，故所求切线方程为$y = -(x-1)$，即$x+y-1=0$
    \end{jieda}
    \item 证明：设函数$y=f(x)$满足$f(x_0) = 0$，在$x = x_0$处的切线为$y=l(x)$（斜率不为0），则$\lim\limits_{x \to x_0}\dfrac{f(x)}{l(x)} = 1$
    \begin{jieda}
        $\lim\limits_{x \to x_0}\dfrac{f(x)}{l(x)} = \lim\limits_{x \to x_0}\dfrac{f(x)}{f(x_0)+f'(x_0)(x-x_0)} = \lim\limits_{x \to x_0}\dfrac{f(x)}{f(x_0)+\dfrac{f(x)-f(x_0)}{x-x_0} \cdot (x-x_0)} = 1$
        
        \ps 故有：
        \begin{dingautolist}{172}
            \item $\lim\limits_{x \to 0}\dfrac{sinx}{x} = 1$
            \item $\lim\limits_{x \to 0}\dfrac{e^x-1}{x} = 1$
            \item $\lim\limits_{x \to 0}\dfrac{ln(x+1)}{x} = 1$
        \end{dingautolist}
    \end{jieda}
\end{enumerate}
\subsection*{导数的运算}
\subsubsection*{导函数的概念}
\begin{enumerate}
    \item 对于给定函数$y = f(x)$，在导数存在的前提下，自变量$x$取不同的值时，都有位移确定的导数值$f'(x)$与之对应，这样就确定了一个新的函数，称为$y = f(x)$的\textbf{导函数}，也简称\textbf{导数}，记作$f'(x)$（或$y'$、$\dfrac{dy}{dx}$、$\dfrac{df(x)}{dx}$）
    \item 求一个函数的函数的导（函）数的过程称为求导，一般步骤为：
    \begin{dingautolist}{172}
        \item 求增量：$\Delta y = f(x+\Delta x) - f(x)$
        \item 算比值：$\dfrac{\Delta y}{\Delta x} = \dfrac{f(x+\Delta x) - f(x)}{\Delta x}$
        \item 求极限：$f'(x) = \lim\limits_{\Delta x \to 0}\dfrac{\Delta y}{\Delta x} = \lim\limits_{\Delta x \to 0}\dfrac{f(x+\Delta x) - f(x)}{\Delta x}$
    \end{dingautolist}
\end{enumerate}
\subsubsection*{基本初等函数的导数公式}
\begin{dingautolist}{172}
    \item $C' = 0$
    \item $(x^a)' = ax^{a-1}$
    \item $(sinx)' = cosx$
    \item $(cosx)' = -sinx$
    \item $(a^x)' = lna(a^x)$
    \item $(log_ax)' = \dfrac{1}{(lna)x}$
\end{dingautolist}
\subsubsection*{导数的四则运算}
\begin{dingautolist}{172}
    \item $[f(x) \pm g(x)]' = f'(x) \pm g'(x)$
    \item $[f(x) \cdot g(x)]' = f'(x)g(x) + g'(x)f(x)$
    \item $\left[\dfrac{f(x)}{g(x)}\right]' = \dfrac{f'(x)g(x) - f(x)g'(x)}{g^2(x)} (g(x) \neq 0)$
\end{dingautolist}
证明如下：
\begin{dingautolist}{172}
    \item \begin{align*}
        & [f(x) \pm g(x)]'  \\
        & = \lim\limits_{\Delta x \to 0}\dfrac{[f(x+\Delta x) \pm g(x+\Delta x)] - [f(x)\pm g(x)]}{\Delta x} \\
        & = \lim\limits_{\Delta x \to 0}\dfrac{[f(x+\Delta x) - f(x)] \pm [g(x+\Delta x) - g(x)]}{\Delta x} \\
        & = f'(x) \pm g'(x)
    \end{align*}
    \item \begin{align*}
        & [f(x) \cdot g(x)]' \\
        & = \lim\limits_{\Delta x \to 0}\dfrac{[f(x+\Delta x) \cdot g(x+\Delta x)] - [f(x) \cdot g(x)]}{\Delta x} \\
        & = \lim\limits_{\Delta x \to 0}\dfrac{[f(x+\Delta x) \cdot g(x+\Delta x) - f(x) \cdot g(x+\Delta x)] + [f(x) \cdot g(x+\Delta x) - f(x) \cdot g(x)]}{\Delta x} \\
        & = f'(x)g(x) + g'(x)f(x)
    \end{align*}
    \item \begin{align*}
        & \left[\dfrac{f(x)}{g(x)}\right]' (g(x) \neq 0) \\
        & = \lim\limits_{\Delta x \to 0}\dfrac{\dfrac{f(x + \Delta x)}{g(x+\Delta x)} - \dfrac{f(x)}{g(x)}}{\Delta x} \\
        & = \lim\limits_{\Delta x \to 0}\dfrac{\dfrac{f(x + \Delta x)g(x) - f(x)g(x+\Delta x)}{\Delta x}}{g(x+\Delta x)g(x)} \\
        & = \lim\limits_{\Delta x \to 0}\dfrac{\dfrac{[f(x + \Delta x)g(x) - f(x)g(x)] - [f(x)g(x+\Delta x)-f(x)g(x)]}{\Delta x}}{g(x+\Delta x)g(x)} \\
        & = \lim\limits_{\Delta x \to 0}\dfrac{\dfrac{[f(x + \Delta x) - f(x)]}{\Delta x} \cdot g(x) - \dfrac{[g(x+\Delta x)-g(x)]}{\Delta x} \cdot f(x)}{g(x+\Delta x)g(x)} \\
        & = \dfrac{f'(x)g(x) - f(x)g'(x)}{g^2(x)} 
        \end{align*}
\end{dingautolist}

隐函数求导与奇偶性

若$f(x)$在定义域内均可导，则有以下性质：
\begin{enumerate}
    \item 原函数为奇函数，导函数为偶函数：$f(x) = -f(-x)\Longrightarrow f'(x) = f'(-x)$
    \item 原函数为偶函数，导函数为奇函数：$f(x) = f(-x) \Longrightarrow f'(x) = -f'(-x)$
    \item 导函数为奇函数，原函数为偶函数：$f'(x) = -f'(-x)\Longrightarrow f(x) + C_1 = f(-x) + C_2$

    代入对称横坐标$x_0$与$-x_0$，则$\left\{\begin{array}{l}
         f(x_0) + C_1 = f(-x_0) + C_2 \\
         f(-x_0) + C_1 = f(x_0) + C_2 
    \end{array}\right. \Longrightarrow C_1 = C_2$，故$f(x) = f(-x) $
    
    \item 导函数为偶函数，原函数对称中心横坐标为0：$f'(x) = f'(-x)\Longrightarrow f(x) + C_1 = -f(-x) + C_2 \Longleftrightarrow f(x) + f(-x) = C$
\end{enumerate}
总结：
\begin{dingautolist}{172}
    \item 原函数为偶函数$\Longleftrightarrow$导函数为奇函数
    
    实例：$f(x) = x^2+c, c\in \mathbf{R}, f'(x) = x$
    \item 原函数为奇函数$\Longrightarrow$导函数为偶函数，原函数对称中心横坐标为0$\Longleftrightarrow$导函数为偶函数
    
    实例：$f_1(x) = x^3, f_1'(x) = 3x^2, f_2(x) = x^3+c, c\in \mathbf{R}, f_2'(x) = 3x^2$

    \item 核心差别：函数上下平移不影响求导结果，偶函数上下平移仍为偶函数，奇函数上下平移不再是奇函数.
\end{dingautolist}



隐函数求导与反函数：\\
如果两个函数 $ f $ 和 $ g $ 互为反函数，即 $ f(g(x)) = x $ 和 $ g(f(x)) = x $，并且它们在相应的点都可导，那么它们的导数之间满足$f'(g(x)) = \dfrac{1}{g'(x)}$ \quad \text{或等价地} \quad $g'(x) = \dfrac{1}{f'(g(x))}$
这个关系表明：$ f' $ 在 $ g(x) $ 处的值等于 $ g'(x) $ 的倒数. \\
推导过程：
\begin{enumerate}
    \item 从反函数的定义出发，对 $ f(g(x)) = x $ 两边关于 $ x $ 求导：$\dfrac{d}{dx} [f(g(x))] = \dfrac{d}{dx} [x]$
    \item 使用链式法则计算：$f'(g(x)) \cdot g'(x) = 1$
\end{enumerate}
\begin{example}
    对于指数函数和对数函数：\\
    设 $ f(x) = e^x $，则其反函数为 $ g(x) = \ln x $; $ f'(x) = e^x $，$ g'(x) = \dfrac{1}{x} $; $ f'(g(x)) = e^{\ln x} = x $，而 $ \dfrac{1}{g'(x)} = x $，确实满足 $ f'(g(x)) = \dfrac{1}{g'(x)} $
\end{example}

例题：
\begin{enumerate}
    \item $f(x) = 2f(2-x)-x^2+8x-8$，求$f(x)$在$x = 1$处的切线.
    \begin{jieda}
        $f'(x) = -2f'(2-x)-2x+8$，$f'(1) = -2f'(1)-2x+8$，故$f'(1) = 2$

        $f(1) = 1$

        切线方程$y = 2(x-1)+1$
    \end{jieda}

    \item $f(x) = ln\left(e^{2x}+1\right)-ax$是偶函数，求$f(x)$在$x = 0$处切线
    \begin{jieda}
        $f(1) = f(-1) \Longrightarrow a = 1$，$f(0) = ln2$，偶函数可导可知$f'(0) = 0$，故切线方程为$y = ln2$.
    \end{jieda}
\end{enumerate}

\subsection*{导数的应用}
\subsubsection*{利用导数研究函数的单调性}
导数是函数的瞬时变化率,即函数的平均变化率的极限情形,它反映了该函数在自变量发生变化的瞬间,函数值随之发生变化的情况.若导数大于零,则说明在自变量发生变化的瞬间,函数值随自变量的增加而增大;若导数小于零,则说明在自变量发生变化的瞬间,函数值随自变量的增加而减小,从而\textbf{根据导数的正负就可以直接判断函数的单调性}. \\
函数的单调性研究的就是在一个区间内函数是否严格增或严格减,即函数值是否随着自变量的增大而增大,或者随着自变量的增大而减小.因此导数是研究函数单调性的最佳工具.
\begin{enumerate}
    \item 设函数$y = f(x)$在区间$I$上可导，若$f'(x) > 0$则函数$y = f(x)$在该区间严格增；若$f'(x) < 0$则函数$y = f(x)$在该区间严格减.
    \begin{example}
        \begin{dingautolist}{172}
            \item $y = e^x$，$y' = e^x$；原函数在定义域内严格单调增，导函数在定义域内恒大于零
            \item $y = -x^2, y' = -2x$；原函数在$(-\infty, 0)$严格增，导函数在$(-\infty, 0)$恒大于零；原函数在$(0, +\infty)$严格减，导函数在$(0, +\infty)$恒小于零；
        \end{dingautolist}
    \end{example}
    \item 在导数值都存在的情况下，导数值由正变负或者由负变正的过程中可能会出现导数值为零的点，即导函数的零点，称为原函数的驻点.因此要把函数的单调区间划分出来就要先找到函数的\textbf{驻点}，找到驻点后再看驻点两侧的导数值的符号是否发生变化，如果发生变化，此驻点就是函数单调增区间和单调减区间的\textbf{分界点}. \\
    \ps 导数值为零的点可能是分界点，也可能不是分界点，不可导或没有定义的点也可能是分界点.
    \begin{example}
        \begin{dingautolist}{172}
            \item $y = e^x - x - 1, y' = e^x-1$；$x = 0$\textbf{是}原函数单调增区间和单调减区间的分界点.
            \item $y = x^3, y' = 3x^2$；$x = 0$\textbf{不是}原函数单调增区间和单调减区间的分界点.
            \item $y = \sqrt[3]{x^2}, y' = \dfrac{2}{3} \cdot x^{-\frac{1}{3}}$，$x = 0$\textbf{是}原函数单调增区间和单调减区间的分界点.
            \item $y = \dfrac{1}{x}, y' = -\dfrac{1}{x^2}$，$x = 0$\textbf{是}原函数两个不同的单调减区间的分界点.
        \end{dingautolist}
    \end{example}
\end{enumerate}

\subsubsection*{凹凸性}
设 $f\left( x\right)$ 在区间 $I$ 上连续,

如果对 $I$ 上任意两点 ${x}_{1},{x}_{2}$ 恒有
$f\left( \dfrac{{x}_{1} + {x}_{2}}{2}\right)  < \dfrac{f\left( {x}_{1}\right)  + f\left( {x}_{2}\right) }{2},$那么称 $f\left( x\right)$ 在 $I$ 上的图形是 (向上) 凹的 (或凹弧); 

如果恒有$f\left( \dfrac{{x}_{1} + {x}_{2}}{2}\right)  > \dfrac{f\left( {x}_{1}\right)  + f\left( {x}_{2}\right) }{2},$那么称 $f\left( x\right)$ 在 $I$ 上的图形是 (向上) 凸的 (或凸弧).


如果函数 $f\left( x\right)$ 在 $I$ 内具有二阶导数,那么可以利用二阶导数的符号来判定曲线的凹凸性：设 $f\left( x\right)$ 在 $\left\lbrack  {a,b}\right\rbrack$ 上连续,在$(a,b)$内具有一阶和二阶导数,那么
\begin{dingautolist}{172}
    \item 若在$(a,b)$内 ${f}^{\prime \prime }\left( x\right)  > 0$ ,则 $f\left( x\right)$ 在 $\left\lbrack  {a,b}\right\rbrack$ 上的图形是凹的
    \item 若在$(a,b)$内 ${f}^{\prime \prime }\left( x\right)  < 0$ ,则 $f\left( x\right)$ 在 $\left\lbrack  {a,b}\right\rbrack$ 上的图形是凸的
\end{dingautolist}

\ps 凸函数(concave function)，凹函数（convex function）在翻译上是混乱的
\begin{dingautolist}{172}
    \item 通常concave翻译成凹，而convex翻译成凸，在定义convex function时根据的是函数图像上方的区域是凸集(即任意两点连线完全位于区域内)，反之为concave function.
    \item 中文的"凸"常被理解为"向外凸起",而“开口向上”的曲线在视觉上可能更像“凹坑”，故定义为凹函数，反之视觉上向上凸起，定义为凸函数.
\end{dingautolist}

\subsubsection*{利用导数研究函数的极值}
\begin{enumerate}
    \item 观察图函数$y = f(x)$的图像，其中有一些特殊的点，在这些特殊点的左右两侧附近，函数的单调性发生了改变． 例如，函数在点$(x_1, f(x_1))$的左侧附近严格增，右侧附近严格减，此处出现了一个“山峰”；又如，函数在点$(x_2, f(x_2))$的左侧附近严格减，右侧附近严格增，此处出现了一个“山谷”．\\
    \begin{center}
        \includegraphics[width=0.5\linewidth]{bodyxb//src/daoshu4.png}
    \end{center}
    换句话说，存在$x_1$的一个邻域，$f(x_1)$是该邻域内最大的函数值，此时，就说函数$y = f(x)$在$x = x_1$处取得\textbf{极大值}（$f(x_1)$），而$x_1$称为函数$y = f(x)$的\textbf{极大值点}．类似地， 存在$x_2$的一个邻域，$f(x_2)$是该邻域内最小的函数值，此时，就说函数$y = f(x)$在$x = x_2$处取得\textbf{极大值}（$f(x_2)$），而$x_2$称为函数$y = f(x)$的\textbf{极大值点}．因此，图中函数有三个极大值点$x_1, x_3, x_5$， 还有三个极小值点$x_2, x_4, x_6$． \\
    极大值和极小值统称为\textbf{极值}，而极大值点和极小值点则统称为\textbf{极值点}．\\
    \ps 根据定义，端点不是极值点；对$y = c$来说每一个实数都既是极大值点又是极小值点.
    \item 设函数$y = f(x)$的导函数为$y = f'(x)$，$x = x_0$是函数$y = f(x)$的一个驻点
    \begin{dingautolist}{172}
        \item 若在$x_0$的左侧附近有$f'(x) > 0$且在$x_0$的右侧附近有$f'(x) < 0$，则函数$y = f(x)$在$x_0$处取得极大值.
        \item 若在$x_0$的左侧附近有$f'(x) < 0$且在$x_0$的右侧附近有$f'(x) > 0$，则函数$y = f(x)$在$x_0$处取得极小值.
    \end{dingautolist}
    \ps “$x = x_0$处导数值为零”是“$x = x_0$为极值点”的\blkda[4]{既非充分也非必要}条件.
    \item 一般地，求函数$f(x)$极值的步骤为：
    \begin{dingautolist}{172}
        \item 确定函数定义域
        \item 求出驻点和不可导的点作为极值点的候选点，利用这些点将函数的定义域分称若干个小区间
        \item 在每个小区间上确定原函数的单调性，根据候选点两侧原函数的单调性确定极值点，并求出相应的极值
    \end{dingautolist}
\end{enumerate}
例题：
\begin{enumerate}
    \item 求下列函数的极值
    \begin{enumerate}
        \item $f(x) = x^3 - 3x$
        \item $f(x) = e^x - 2x + 2a$
        \item $f(x) = \dfrac{2ax-a^2+1}{x^2+1}$，（$a \neq 0$）
        \item $f(x) = e^x \cdot cosx$
    \end{enumerate}
    \begin{jieda}
        \begin{enumerate}
            \item $f(x)$定义域为$\mathbf{R}$，$f'(x) = 3x^2-3$，故$x = \pm 1$时$f'(x) = 0$，$f(x)$在$(-\infty, -1)$和$(1, +\infty)$上单调递增，在$(-1, 1)$单调递减，因此$f(x)$有极大值$f(-1) = 2$和极小值$f(1) = -2$.
            \item $f(x)$定义域为$\mathbf{R}$，$f'(x) = e^x-2$，故$x = ln2$时$f'(x) = 0$，$f(x)$在$(ln2, +\infty)$上单调递增，在$(-\infty, ln2)$单调递减，因此$f(x)$有极小值$f(ln2) = 2(1-ln2+a)$，无极大值.
            \item $f(x)$定义域为$\mathbf{R}$，$f'(x) = \dfrac{-2(x-a)(ax+1)}{(x^2+1)^2}$，故$x = a$或$-\dfrac{1}{a}$时$f'(x) = 0$
            \begin{dingautolist}{172}
                \item 若$a > 0$，则$f(x)$在$\left(-\infty, -\dfrac{1}{a}\right)$和$(a, +\infty)$上单调递减，在$\left(-\dfrac{1}{a}, a\right)$单调递增，因此$f(x)$有极大值$f(a) = 1$和极小值$f\left(-\dfrac{1}{a}\right) = -a^2$
                \item 若$a < 0$，则$f(x)$在$\left(-\infty, a\right)$和$\left(-\dfrac{1}{a}, +\infty\right)$上单调递增，在$\left(a, -\dfrac{1}{a}\right)$单调递减，因此$f(x)$有极大值$f(a) = 1$和极小值$f\left(-\dfrac{1}{a}\right) = -a^2$
            \end{dingautolist}
            综上，$f(x)$有极大值$f(a) = 1$和极小值$f\left(-\dfrac{1}{a}\right) = -a^2$.
            \item $f(x)$定义域为$\mathbf{R}$，$f'(x) = e^x(cosx-sinx) = -\sqrt{2}e^x\cdot sin\left(x - \dfrac{\pi}{4}\right)$，当$x = k\pi + \dfrac{\pi}{4}, k \in \mathbf{Z}$时$f'(x) = 0$，$f(x)$在$\left(2k\pi+\dfrac{\pi}{4}, 2k\pi+\dfrac{5\pi}{4}\right), k \in \mathbf{Z}$单调递减，在$\left(2k\pi+\dfrac{5\pi}{4}, 2k\pi+\dfrac{9\pi}{4}\right), k \in \mathbf{Z}$单调递增，故$f(x)$有极小值$f\left(2k\pi+\dfrac{5\pi}{4}\right) = -\dfrac{\sqrt{2}}{2}e^{2k\pi+\frac{5\pi}{4}}, k \in \mathbf{Z}$和极大值$f\left(2k\pi+\dfrac{\pi}{4}\right) = \dfrac{\sqrt{2}}{2}e^{2k\pi+\frac{\pi}{4}}, k \in \mathbf{Z}$
        \end{enumerate}
        \ps 因为有令导函数为$0$的操作，所以当出现分式型的导函数的时候应考虑通分，将导函数为零的问题转化为其分子为零进行研究.
    \end{jieda}
    \item 已知函数$f(x) = \dfrac{a}{3}x^3+bx^2+cx+d (a>0)$，且方程$f'(x) - 9x = 0$的两个根分别是$1, 4$.若$f(x)$无极值点，求实数$a$的取值范围.
    \begin{jieda}
        $f'(x) = ax^2+2bx+c$，根据题意有$\left \{ \begin{array}{l}
             f'(1) = 9 \\
             f'(4) = 36 \\
             \Delta = 4b^2 - 4ac \leq 0
        \end{array}\right.$，前两式解得$\left \{ \begin{array}{l}
             c = 4a \\
             2b = 9 - 5a
        \end{array}\right.$，代入第三式，结合$a > 0$，解得$a \in [1, 9]$
    \end{jieda}
    \item 已知函数$f(x) = sin2x - ln(1+x)$，$f'(x)$为$y = f(x)$的导数
    \begin{enumerate}
        \item 证明$f'(x)$在区间$\left(-1, \dfrac{\pi}{4}\right)$上存在唯一极大值点.
        \item 讨论$y = f(x)$的零点个数.
    \end{enumerate}
    \begin{jieda}
        $f'(x) = 2cos2x - \dfrac{1}{1+x}$，$f''(x) = -4sin2x + \dfrac{1}{(1+x)^2}$
        \begin{enumerate}
            \item 
            \begin{dingautolist}{172}
            \item $x \in (-1, 0]$时，$-4sin2x \geq 0, \dfrac{1}{(1+x)^2} > 0$，故$f''(x) > 0$，$f'(x)$单调递增.
            \item $x \in \left(0, \dfrac{\pi}{4}\right)$时，$f'''(x) = -8cos2x - \dfrac{2}{(1+x)^3} < 0$，故$f''(x)$单调递减，$f''\left(\dfrac{\pi}{4}\right) < 0, f''(0) > 0$，因此$f''(x)$在$\left(0, \dfrac{\pi}{4}\right)$内存在唯一零点，记为$a$，故$f'(x)$在$(-1, a)$内单调递增，在$\left(a, \dfrac{\pi}{4}\right)$内单调递减.
            \end{dingautolist}
            综上，$f'(x)$在区间$\left(-1, \dfrac{\pi}{4}\right)$上存在唯一极大值点.
            \item 易知$f(0) = 0$，$f(x)$在$[e-1, +\infty)$内无零点.
            \begin{dingautolist}{172}
            \item $x \in (-1, 0)$时，由上一问可知$f'(x)$单调递增，$f'(0) = 1$，$f\left(-\dfrac{\pi}{4}\right) = \dfrac{1}{\frac{\pi}{4}-1} < 0$，因此$f'(x)$在$(-1, 0)$内存在唯一零点$b \left(b \in \left(\dfrac{\pi}{4}, 0\right)\right)$，故$f(x)$在$(-1, b)$单调递减，在$(b, 0)$单调递增，即$b$为$f(x)$的极小值点，且$f(b) < 0$.\\
            又因为$f\left(-\dfrac{\pi}{4}\right) = -1 - ln\left(1-\dfrac{\pi}{4}\right) > 0$，因此$f(x)$在$\left(-\dfrac{\pi}{4}, b\right)$上存在唯一零点.
            \item $x \in \left(0, \dfrac{\pi}{4}\right]$时，结合上一问可知$f''(x)$在$\left(0, \dfrac{\pi}{4}\right)$单调递减，故$f'(x)$在$(0, a)$内单调递增，在$\left(a, \dfrac{\pi}{4}\right]$单调递减，又因为$f'\left(\dfrac{\pi}{4}\right) < 0, f'(0) = 1 > 0$，因此$f'(x)$在$\left(0, \dfrac{\pi}{4}\right]$存在唯一零点$c$，即$f(x)$在$(0, c)$单调递增，在$\left(c, \dfrac{\pi}{4}\right]$单调递减，又因为$f\left(\dfrac{\pi}{4}\right) > 0$，所以$f(x)$在$\left(c, \dfrac{\pi}{4}\right]$内无零点.
            \item 当$x \in \left(\dfrac{\pi}{4}, e-1\right)$时$f'(x) < 0$，故$f(x)$单调递减，$f\left(\dfrac{\pi}{4}\right) > 0, f(e-1) < 0$，故$f(x)$在$\left(\dfrac{\pi}{4}, e-1\right)$内有唯一零点.
            \end{dingautolist}
            \ps 第二小问通过数形结合可以确定出零点个数和区间划分.
        \end{enumerate}
        综上，$f(x)$共有三个零点.
    \end{jieda}
\end{enumerate}
\subsubsection*{利用导数研究函数的最值}
极值是反应函数在某一点附近的局部性质，而不是函数在整个定义域内或某特定区间内的性质。而最值是反应定义域内（或某区间内）函数的整体性质 \\
\ps “一个函数在定义域内有极值”是“一个函数在定义域内存在最值”的\blkda[4]{既不充分也不必要}条件.
    \begin{jieda}
        非充分性反例：$y = x^3 - x$在定义域内不存在最值，但存在两个极值点 \\
        非必要性反例：$y = x, x\in [0, 1]$，在定义域内存在最值，但是不存在极值点
    \end{jieda}
例题：
\begin{enumerate}
    \item 已知三次函数$f(x) = x^3 + ax^2-6x+b, (a, b \in \mathbf{R})$，$f(0) = 1$，曲线$y = f(x)$在点$(1, f(1))$处切线斜率为$-6$.
    \begin{enumerate}
        \item 求出$f(x)$解析式
        \item 求出$y = f(x)$在区间$[-2, 4]$上的最值
    \end{enumerate}
    \begin{jieda}
        \begin{enumerate}
            \item $f'(x) = 3x^2+2ax-6$，根据题意有$\left\{\begin{array}{l}
                 f'(1) = -6 \\
                 f(0) = 1
            \end{array}\right.$，解得$a = -\dfrac{3}{2}, b = 1$，故$f(x) = x^3 - \dfrac{3}{2}x^2-6x+1$
            \item $f'(x) = 3x^2-3x-6$，当$x = -1$或$2$时，$f'(x) = 0$，故$f(x)$在$[-2, -1]$和$[2, 4]$单调递增，在$(-1, 2)$单调递减 \\
            $f(-2) = -1, f(2) = -9$，故$f(x)$在$[-2, 4]$上最小值为$f(2) = -9$ \\
            $f(-1) = \dfrac{9}{2}, f(4) = 17$，故$f(x)$在$[-2, 4]$上最大值为$f(4) = 17$
        \end{enumerate}
    \end{jieda}
    \item 已知函数$f(x) = x  + lnx + \dfrac{a}{x}, (a\in \mathbf{R})$
    \begin{enumerate}
        \item 当$a = 2$时，求$f(x)$在区间$\left[\dfrac{1}{2}, 3\right]$上的最值
        \item 讨论$g(x) = f'(x) - x$的零点个数
    \end{enumerate}
    \begin{jieda}
        \begin{enumerate}
            \item 当$a = 2$时，$f(x) = x + \dfrac{2}{x} + lnx$，$f'(x) = 1 - \dfrac{2}{x^2} + \dfrac{1}{x} = \dfrac{x^2+x-2}{x^2} (x > 0)$，令$f'(x)=0$，$f'(1) = 0$，$f'(x)$在$\left[\dfrac{1}{2}, 1\right]$小于等于0，在$(1, 3]$大于0，故$f(x)$在$\left[\dfrac{1}{2}, 1\right]$单调减，在$(1, 3]$严格增，$f\left(\dfrac{1}{2}\right) = \dfrac{9}{2} - ln2, f(1) = 3, f(3) = \dfrac{11}{3} + ln3$ \\
            因此$f(x)$在区间$\left[\dfrac{1}{2}, 3\right]$最大值为$f(3) = \dfrac{11}{3} + ln3$，最小值为$f(1) = 3$
            \item $f'(x) = \dfrac{x^2+x-a}{x^2}$，$g(x) = \dfrac{-x^3+x^2+x-a}{x^2}$，定义域为$(0, +\infty)$ \\
            $g(x)=0$即$a=-x^3+x^2+x$，故问题转化为$y=a$与$h(x)=-x^3+x^2+x$的交点个数.\\
            $h'(x)=-3x^2+2x+1$根据$h'(x)$符号可知$h(x)$在$(1,+\infty)$严格减，在$(0,1]$严格增. \\
            因此当$a\in (-\infty,0] \cup \{1\}$时，$g(x)$有一个零点；当$a\in (0, 1)$时，$g(x)$有两个零点；当$a\in (1, +\infty)$时，$g(x)$没有零点；
        \end{enumerate}
    \end{jieda}
\end{enumerate}
