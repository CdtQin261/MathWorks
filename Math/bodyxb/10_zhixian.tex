\chapter*{解析几何-直线}
\subsection*{直线的倾斜角与斜率}
为了用数量来刻画直线这种“陡”的程度，需要引入直线的倾斜角和斜率的概念.
\begin{enumerate}
    \item 倾斜角与斜率的定义\\
    在平面直角坐标系中,设直线$l$与$x$轴交于点$M$,将$x$轴绕点$M$按逆时针方向旋转至与直线$l$重合时所形成的最小正角$\alpha$叫做直线$l$的\textbf{倾斜角}. 特别地，当直线$l$与$x$轴平行或重合时，规定其倾斜角$\alpha = 0$,因此直线的倾斜角$\alpha$的\textbf{范围}是$[0, \pi)$.当直线的倾斜角$\alpha \neq \dfrac{\pi}{2}$时,把角$\alpha$的正切值$tan \alpha$叫做直线$l$的\textbf{斜率},记作$k = tan\alpha$.
    \item 倾斜角与斜率的关系\\
    直线的斜率 $k$ 与倾斜角 $\alpha$ 之间的关系
    
    \begin{center}
    \adjustbox{width=0.5\linewidth}{
    \begin{tabular}{|c|c|c|c|c|}
    \hline
    $\alpha$ & ${0}^{ \circ  }$ & ${0}^{ \circ  } < \alpha  < {90}^{ \circ  }$ & ${90}^{ \circ  }$ & ${90}^{ \circ  } < \alpha  < {180}^{ \circ  }$ \\
    \cline{1-5}
    $k$ & 0 & $k > 0$ & 不存在 & $k < 0$ \\
    \cline{1-5}
    \hline
    \end{tabular}
    }
    \end{center}
    由正切函数的性质，不难得出以下结论：
    \begin{dingautolist}{172}
    \item 当$\alpha = 0$时,斜率$k = 0$
    \item 当$0 < \alpha < \dfrac{\pi}{2}$时，$k > 0$，$k$随$\alpha$增大而增大
    \item 当$\alpha = \dfrac{\pi}{2}$时,直线的斜率$k$不存在(或称趋向于无穷大)
    \item 当$\dfrac{\pi}{2} < \alpha < \pi$时，$k < 0$，$k$随$\alpha$增
    
    大而增大
    \end{dingautolist}
    \item 已知两点坐标：$A(x_1, y_1), B(x_2, y_2) (x_1 \neq x_2)$，可求直线$AB$的斜率：$k = \dfrac{y_1 - y_2}{x_1 - x_2}$
\end{enumerate}

    \end{dingautolist}

\subsection*{直线的方向向量和法向量}
\begin{enumerate}
    \item 方向向量与法向量的定义
    \begin{dingautolist}{172}
        \item 我们把与直线$l$平行的非零向量$\Vec{d} = (u,v)$叫做直线$l$的\textbf{方向向量},所有与$\Vec{d}$平行的非零向量都是直线$l$的方向向量
        \item 与直线$l$垂直的非零向量$\Vec{n} = (a,b)$叫做直线$l$的\textbf{法向量}，所有与$\Vec{n}$平行的非零向量都是直线$l$的法向量.
    \end{dingautolist}
    \item 方向向量、法向量、倾斜角、斜率之间的转化
    \begin{dingautolist}{172}
        \item 如果已知方向向量$\Vec{d} = (u, v)$，那么法向量可以是$\Vec{n} = (v, -u)$或者$\Vec{n} = (-v, u)$
        \item 如果已知方向向量$\Vec{d} = (u, v)$，当$u \neq 0$时，斜率$k = \dfrac{v}{u}$，当$u = 0$时，斜率不存在
        \item 如果已知斜率$k$，那么方向向量可以是$\vec{d} = (1, k)$
        \item 如果已知倾斜角$\alpha$，那么方向向量可以是$\vec{d} = (cos\alpha, sin\alpha)$
    \end{dingautolist}
\end{enumerate}

\subsection*{直线的点斜式方程}
给定直线上的一个点，再给定一个方向（倾斜程度）就可以在平面直角坐标系内确定这一条直线
\begin{enumerate}
    \item 点斜式方程：给定斜率即可确定直线的倾斜程度 \\
    经过点$P_0(x_0, y_0)$，且斜率为$k$的直线的方程的推导：\\
    设$P(x, y)$是直线上异于$P_0$的一个点，因此可以确定斜率为$k = \dfrac{y - y_0}{x - x_0}$，故直线方程为：$y - y_0 = k(x-x_0)$ 
    \begin{dingautolist}{172}
        \item 由推导过程可知，直线上的每一个点都满足直线的方程
        \item 反过来，可以验证满足方程的点都在直线上：若$P_1(x_1, y_1)$满足直线方程$y - y_0 = k(x-x_0)$
        
        若$x_1 = x_0$，则$P_1$与$P_0$重合，满足要求
        
        若$x_1 \neq x_0$，则有直线$P_0P_1$的斜率也为$k$，故两直线重合，因此$P_1$在直线上.
    \end{dingautolist}
    \textbf{点斜式方程式处于直线方程中的核心位置，点斜式方程只能表示斜率存在的直线.} \\
    \ps \textbf{解析几何的核心要义：满足方程的点在曲线(直线)上，曲线（直线）上的点满足方程.}
    \item 斜截式方程：点斜式的特例 \\
    当已知的点是直线与$y$轴交点$(0, b)$的时候，点斜式方程就变成了斜截式方程：$y = kx + b$，$b$是直线与$y$轴交点的纵坐标，称为\textbf{截距}. \\
    \ps 截距不是距离，截距可以为负数.
    \item 点法式方程：给定法向量也可以确定出直线的方向 \\
    经过点$P_0(x_0, y_0)$，且法向量为$\vec{n} = (a, b)$的直线的方程的推导： \\
    设$P(x, y)$是直线上异于$P_0$的一个点，容易写出直线的方向向量$\vec{d} = (x-x_0, y - y_0)$，由于方向向量和法向量的数量积为0，因此有$a(x-x_0)+b(y-y_0) = 0$
\end{enumerate}
\subsection*{直线的两点式方程}
两点可以确定一条直线
\begin{enumerate}
    \item 两点式方程 \\
    经过点$P_1(x_1, y_1)， P_2(x_2, y_2)$的直线方程的推导：
    当直线斜率存在的时候，根据两个点的坐标可以求出直线的斜率$k = \dfrac{y_2 - y_1}{x_2 - x_1}$，对于直线上异于$P_1, P_2$的点$P(x, y)$，可以得到斜率的表示为$k = \dfrac{y - y_1}{x - x_1}$，因此有$\dfrac{y_2 - y_1}{x_2 - x_1} = \dfrac{y - y_1}{x - x_1}$，变形可以得到$\dfrac{y - y_1}{y_2 - y_1} = \dfrac{x - x_1}{x_2 - x_1}$，称为直线的两点式方程. \\
    \textbf{从两点式方程的形式可知，两点式方程不能表示与坐标轴（$x$轴、$y$轴）平行的直线.} \\
    \ps \textbf{从两点式的推导过程中可以看出斜率的桥梁作用，斜率在直线方程的建立过程中起到至关重要的作用.} 
    \item 点方向式方程：两点式的推广 \\
    特别地，当已知过$(x_0, y_0)$的直线的方向向量$\vv{d} =  (a, b)$时，方向向量的两个坐标可以看做是该直线上两点之间的坐标差，因此可以写出对应的直线方程为：$\dfrac{y-y_0}{b} = \dfrac{x-x_0}{a}$
    \item 截距式方程：两点式的特例 \\
    若$P_1, P_2$两点是直线与坐标轴的交点，$P_1(a, 0), P_2(0, b)$，则两点式直线方程为$\dfrac{y-0}{b-0} = \dfrac{x-a}{0-a}$，变形有$\dfrac{y}{b}+\dfrac{x}{a} = 1$，称作是直线的截距式方程.
\end{enumerate}
\subsection*{直线的一般式方程}
任意的一个二元一次方程$ax+by+c = 0(a^2+b^2 \neq 0)$都是直线方程，任意一条直线的方程都是一个二元一次方程$ax+by+c = 0(a^2+b^2 \neq 0)$.
\begin{enumerate}
    \item 对于任意的$ax+by+c = 0(a^2+b^2 \neq 0)$，当$b \neq 0$时，可以整理成斜截式方程，对应一条直线，当$b = 0$的时候，对应直线$x = -\dfrac{c}{a}$
    \item 对于任意一条直线，如果斜率存在，则其点斜式方程可以整理成$ax+by+c = 0(a^2+b^2 \neq 0)$的形式；如果斜率不存在，设其与$x$轴交点为$(x_0, 0)$，则直线方程为：$x = x_0$，也符合$ax+by+c = 0(a^2+b^2 \neq 0)$的形式
\end{enumerate}

\begin{enumerate}
    \item 下列命题中为真命题的是\blkda{\ding{173}\ding{176}\ding{177}}
    \begin{dingautolist}{172}
        \item 过点$P(x_0, y_0)$的直线都可以用$y - y_0 = k(x-x_0)$表示
        \item 过任意两点$P_1(x_1, y_1)$、$P_2(x_2, y_2)$的直线都可以用方程$(y-y_1)(x_2-x_1) = (x-x_1)(y_2-y_1)$表示
        \item 不过原点的直线都可以用方程$\dfrac{x}{a} + \dfrac{y}{b} = 1$表示
        \item 经过$A(0, b)$的直线都可以用方程$y = kx+b$表示
        \item 若直线 $l$ 与 $x,y$ 轴的交点分别为 $A,B$ 且 ${AB}$ 的中点为(4,1),则直线 $l$ 的方程为 $\dfrac{x}{8} + \dfrac{y}{2} = 1$
        \item 过点(1,1)且在两坐标轴上截距相等的直线方程为 $y = x$ 或 $x + y = 2$
        \item 直线 ${3x} - {2y} = 4$ 的截距式方程为 $\dfrac{x}{\frac{4}{3}} + \dfrac{y}{2} = 1$

        \ding{172}\ding{175}是斜率不存在的直线表示不了，而\ding{174}是表示不了和坐标轴垂直的直线

        \ding{176} ${AB}$ 的中点为(4,1), 那么 $A\left( {8,0}\right) ,B\left( {0,2}\right)$ ,则直线 $l$ 的方程为 $\dfrac{x}{8} + \dfrac{y}{2} = 1$

        \ding{177} 直线过原点时方程为 $y = x$ , 不过原点时方程为 $x + y = 2$

        \ding{178} 方程 ${3x} - {2y} = 4$ 可化为 $\dfrac{x}{\frac{4}{3}} + \dfrac{y}{-2} = 1$
    \end{jieda}
    
    \item 下列说法正确的有\blkda{\ding{172}\ding{173}\ding{174}}
    \begin{dingautolist}{172}
        \item 若直线 $y = {kx} + b$ 经过第一、二、四象限,则$(k, b)$在第二象限
        \item 直线 $y = {ax} - {3a} + 2$ 过定点$(3,2)$
        \item 过点 $(2, - 1)$,斜率为 $- \sqrt{3}$ 的直线的点斜式方程为 $y + 1 =  - \sqrt{3}\left( {x - 2}\right)$
        \item 斜率为 $-2$,在 $y$ 轴上截距为 3 的直线方程为 $y =  - {2x} \pm  3$
    \end{dingautolist}

        \begin{dingautolist}{172}
            \item 直线 $y = {kx} + b$ 经过第一、二、四象限,则 $k < 0,b > 0$ ,所以$(k, b)$在第二象限，故正确
            \item 直线可写为 $y - 2 = a\left( {x - 3}\right)$ ,所以直线过定点(3,2)，故正确
            \item 根据直线的点斜式方程知正确
            \item 由直线的斜截式方程得 $y =  - {2x} + 3$，故错误. 
        \end{dingautolist}
    \end{jieda}
    
    \item 已知直线 $l$ 过点$(1,1)$,且倾斜角为 $\ang{90}$ ,则直线 $l$ 的方程为\blkda{$x = 1$}
    \begin{jieda}
        因为直线 $l$ 的倾斜角为 $\ang{90}$, 所以该直线无斜率,与 $x$ 轴垂直,又因为直线 $l$ 过点(1,1), 所以直线 $l$ 的方程为 $x = 1$ .
    \end{jieda} 
    \item 已知直线 $l$ 的一个方向向量为 $\vv{d} = \left( {2,3}\right)$ ,若 $l$ 过点 $A\left( {-4,3}\right)$ ,则直线 $l$ 的方程为\blkda[3]{$3x-2y+18=0$}
    \begin{jieda}
        根据题干条件可以写出直线的点方向式方程：$\dfrac{x+4}{2} = \dfrac{y-3}{3}$
        
        故直线 $l$ 的方程为 $3x-2y+18=0$ .
    \end{jieda}
    \item 过点 $P\left( {2,3}\right)$ 且在两坐标轴上截距相等的直线方程为\blkda[5]{${3x} - {2y} = 0$ 或 $x + y - 5 = 0$}.
    \begin{jieda}
        \begin{dingautolist}{172}
            \item 当截距为 0 时，直线方程为 ${3x} - {2y} = 0$

\subsection*{两直线位置关系}
对于两直线$l_1:a_1 x + b_1y + c_1 = 0$和$l_2:a_2x+b_2y+c_2 = 0$：
\begin{dingautolist}{172}
    \item $l_1$与$l_2$平行$\Longleftrightarrow \left \{ \begin{array}{l}
         a_1 x + b_1y + c_1 = 0  \\
         a_2x + b_2y + c_2 = 0 
    \end{array} \right.$ 无解 $\Longleftrightarrow$ 斜率相等或均不存在
    \item $l_1$与$l_2$重合$\Longleftrightarrow \left \{ \begin{array}{l}
         a_1 x + b_1y + c_1 = 0  \\
         a_2x + b_2y + c_2 = 0 
    \end{array} \right.$ 有无数组解 $\Longleftrightarrow $斜率相等或均不存在，且存在公共点
    \item $l_1$与$l_2$相交$\Longleftrightarrow \left \{ \begin{array}{l}
         a_1 x + b_1y + c_1 = 0  \\
         a_2x + b_2y + c_2 = 0 
    \end{array} \right.$ 有唯一解 $\Longleftrightarrow$ 斜率不相等
    \item $l_1$与$l_2$垂直 $\Longleftrightarrow$ 斜率互为负倒数或者一条直线斜率为0另一条直线斜率不存在
\end{dingautolist}

\subsection*{两直线的夹角}
\begin{enumerate}
    \item 两直线相交时所形成的锐角或直角称为是两直线的夹角，当两直线平行或重合时，规定其夹角为0，因此两直线夹角的取值范围是$\left[0, \dfrac{\pi}{2}\right]$
    \item 对于两直线$l_1:a_1 x + b_1y + c_1 = 0$和$l_2:a_2x+b_2y+c_2 = 0$，其方向向量为$\vv{d_1} = (b_1, -a_1), \vv{d_2} = (b_2, -a_2)$，两直线夹角$\alpha$的余弦值即两方向向量夹角$\theta$的余弦值的绝对值，由此可得两直线夹角的余弦值为$cos\alpha = |cos\theta| = \left|\dfrac{\vv{d_1} \cdot \vv{d_2}}{|\vv{d_1}| \cdot |\vv{d_2}|}\right| = \dfrac{|a_1a_2 + b_1b_2|}{\sqrt{a_1^2+b_1^2} \cdot \sqrt{a_2^2+b_2^2}}$
    \item 两直线垂直的充要条件是$a_1a_2 + b_1b_2 = 0$
    \item 若两直线斜率$k_1, k_2$均存在，则当两直线垂直时有$k_1k_2 = -1$，当两直线不垂直时，根据两角差的正切公式有$tan\alpha = \left|\dfrac{k_1 - k_2}{1+k_1k_2}\right|$
\end{enumerate}

\begin{enumerate}
    \item $l$过$M$：
    
    根据$M$在$l_1$上可知$A_1x_0+B_1y_0+C_1=0$，同理根据$M$在$l_2$上可知$A_2x_0+B_2y_0+C_2=0$，故有$A_1x_0+B_1y_0+C_1 + \lambda(A_2x_0+B_2y_0+C_2) = 0$，即$l$过点$M$.

    \item $l$与$l_2$不重合
    \begin{enumerate}
        \item 相交直线不重合的充要条件（$l_1$与$l_2$不重合的充要条件）

        首先将两直线写成点法式$l_1: A_1(x-x_0)+B_1(y-y_0) = 0$，$l_2: A_2(x-x_0)+B_2(y-y_0) = 0$，故有$\vv{n_1} = (A_1, B_1)$，$\vv{n_2} = (A_2, B_2)$，$l_1$与$l_2$不重合$\Longleftrightarrow \vv{n_1}$与$\vv{n_2}$不平行$\Longleftrightarrow A_1B_2 \neq A_2B_1$.

        \item $l$与$l_2$不重合

        将$l$变形：$l: (A_1+\lambda A_2)x+(B_1+\lambda B_2)y + (C_1+C_2) = 0$
        
        写出点法式方程：$l: (A_1+\lambda A_2)(x-x_0)+(B_1+\lambda B_2)(y-y_0) = 0$

        $\vv{n} = (A_1+\lambda A_2, B_1+\lambda B_2)$，$\vv{n_2} = (A_2, B_2)$
        
        $(A_1+\lambda A_2)B_2 - (B_1+\lambda B_2)A_2 = A_1B_2-B_1A_2 \neq 0$，故$\vv{n}$与$\vv{n_2}$不平行$\Longleftrightarrow l$与$l_2$不重合
    \end{enumerate}
    \item 除$l_2$外，一切过$M$的直线均可以用$l$来表示

    除$l_2$外，过$M$点的直线可以表示为$l': A(x-x_0)+B(y-y_0) = 0 (AB_2 \neq BA_2)$，若$l$与$l'$重合则应有法向量平行
    
    即$\vv{n'} = (A, B)$与$\vv{n} = (A_1+\lambda A_2, B_1+\lambda B_2)$平行
    
    即$A(B_1+\lambda B_2) = B(A_1+\lambda A_2)$，解得$\lambda = \dfrac{AB_1-BA_1}{BA_2-AB_2}$

    \item 基向量视角

    根据前面的推导可知，当两直线有交点时，法向量平行则两直线重合
    
    将$\vv{e_1} = (A_1, B_1), \vv{e_2} = (A_2, B_2)$看做是一组基来表示向量$\vv{n'} = (A, B)$，即$\vv{n'} = p \cdot \vv{e_1} + q \cdot \vv{e_2}$
    
    当且仅当$p \neq 0$时，存在$\vv{m'} = \dfrac{\vv{n'}}{p}$与$\vv{n'}$平行，且有$\vv{m'} = \vv{e_1} + \dfrac{q}{p} \cdot \vv{e_2}$
    
    故存在$\lambda = \dfrac{q}{p}$使得$l$的法向量$\vv{n} = \vv{m'}$与$l'$的法向量$\vv{n'}$平行
\end{enumerate}

\subsubsection*{平行直线系与垂直直线系}

设直线$l: Ax+By+C=0$

直线$m: Ax+By+p= 0$可以表示与直线$l$平行或重合的直线系.

直线$n: -Bx+Ay+q = 0$可以表示与直线$l$垂直的直线系.

例题：
\begin{enumerate}
    \item 设直线 ${2x} + \left( {k - 3}\right) y - {2k} + 6 = 0$ 过定点 $P$ ,则点 $P$ 的坐标为\blkda{$\left( {0,2}\right)$}.
    \begin{jieda}
        由直线方程 ${2x} + \left( {k - 3}\right) y - {2k} + 6 = 0$ ,可化简为 $\left( {{2x} - {3y} + 6}\right)  + k\left( {y - 2}\right)  = 0$ ,
    
        又由 $\left\{  \begin{array}{l} {2x} - {3y} + 6 = 0 \\  y - 2 = 0 \end{array}\right.$ ,解得 $x = 0,y = 2$，即直线恒经过定点 $P\left( {0,2}\right)$ .

        \ps 解决直线过定点问题的问题中，实际就是将过两直线交点的直线系的方程表示出来，并求得两个方程的交点的过程.
    \end{jieda}
    \item 经过两直线 ${l}_{1} : {2x} - y + 3 = 0$ 与 ${l}_{2} : x + {2y} - 1 = 0$ 的交点,且平行于直线 ${3x} + {2y} + 7 = 0$ 的直线方程是\blkda[3]{${3x} + {2y} + 1 = 0$}
    \begin{jieda}
        \begin{enumerate}
            \item 方法1

            由 $\left\{  {\begin{array}{l} {2x} - y + 3 = 0, \\  x + {2y} - 1 = 0, \end{array}\;\text{ 解得 }\left\{  \begin{array}{l} x =  - 1, \\  y = 1, \end{array}\right. }\right.$ 所以直线 ${l}_{1}$ 与 ${l}_{2}$ 的交点为$(-1,1)$,设与直线 ${3x} + {2y} + 7 = 0$ 平行的直线为 ${3x} + {2y} + m = 0\left( {m \neq  7}\right)$ ,所以 $3 \times  \left( {-1}\right)  + 2 \times  1 + m = 0$ ,解得 $m = 1$ ,所以所求直线方程为 ${3x} + {2y} + 1 = 0$ .

            \item 方法2

            经过两直线 ${l}_{1} : {2x} - y + 3 = 0$ 与 ${l}_{2} : x + {2y} - 1 = 0$ 的交点的直线可以表示为$({2x} - y + 3) + \lambda (x + {2y} - 1) = 0$，即$(2+\lambda)x + (-1+2\lambda)y + (3-\lambda) = 0$，与直线 ${3x} + {2y} + 7 = 0$ 平行则应有$2(2+\lambda) = 3(-1+2\lambda)$，解得$\lambda = \dfrac{7}{4}$
            
            故直线方程为$\dfrac{15}{4}x+\dfrac{10}{4}y+\dfrac{5}{4} = 0$，即${3x} + {2y} + 1 = 0$
        \end{enumerate}
        
    \end{jieda}
    \item 若直线 ${2x} - y - {10} = 0$ 经过直线 ${4x} + {3y} - {10} = 0$ 和 ${ax} + {2y} + 8 = 0$ 的交点,则实数 $a =$ \blkda{$-1$}.
    \begin{jieda}
        \begin{enumerate}
            \item 方法1 \\
            直线 ${2x} - y - {10} = 0$ 和直线 ${4x} + {3y} - {10} = 0$交于$(4, -2)$，因此直线${ax} + {2y} + 8 = 0$过该点，故$a = -1$
            \item 方法2 \\
            ${2x} - y - {10} = \mu({4x} + {3y} - {10} + \lambda({ax} + {2y} + 8))$

            则有$\left\{\begin{array}{l}
                 2 = 4\mu +\mu \cdot a \cdot \lambda \\
                 -1 = 3\mu + 2\mu\lambda \\
                 -10 = -10 \mu + 8 \mu \lambda
            \end{array}\right.$，解得$\left\{\begin{array}{l}
                 a = -1  \\
                 \mu = \dfrac{3}{11} \\
                 \lambda = -\dfrac{10}{3}
            \end{array}\right.$
        \end{enumerate}
    \end{jieda}
\end{enumerate}

\subsection*{点到直线的距离}
\begin{enumerate}
    \item 点到直线的距离\\
    平面上一点$P(x_0, y_0)$到直线$l: ax+by+c=0$的距离为$d = \dfrac{|ax_0+by_0+c|}{\sqrt{a^2+b^2}}$，推导如下：\\
    设点$P$在直线$l$上的投影为点$Q(x_1, y_1)$，则点到直线的距离即$\vv{PQ} = (x_1-x_0, y_1-y_0)$的模，直线的法向量为$\vv{n} = (a, b)$，由于$\vv{PQ}$与$\vv{n}$共线，因此$|\vv{PQ}| = \dfrac{|\vv{PQ} \cdot \vv{n}|}{|\vv{n}|} = \dfrac{|a(x_1-x_0)+b(y_1-y_0)|}{\sqrt{a^2+b^2}} = \dfrac{|-ax_0-by_0-c|}{\sqrt{a^2+b^2}} = \dfrac{|ax_0+by_0+c|}{\sqrt{a^2+b^2}}$
    \item 点到直线距离公式的推论 \\
    在推导点到直线距离公式的过程中，若$\vv{PQ}$与$\vv{n}$同向共线，则$ax_0+by_0+c > 0$；反之，若$\vv{PQ}$与$\vv{n}$反向共线，则$ax_0+by_0+c < 0$，因此有推论：设$\delta = ax_0+by_0+c$，对于直线同侧的点，$\delta$是同号的，对于直线异侧的点，$\delta$是异号的.
    \item 两直线间距离\\
    根据点到直线距离公式可以进一步推得两平行直线$l_1: ax+by+c_1=0$与$l_2: ax+by+c_2=0$间的距离公式为$d = \dfrac{|c_1-c_2|}{\sqrt{a^2+b^2}}$, 推导如下：\\
    在直线$l_1$上取点$P(x_1, y_1)$，故根据点到直线距离公式可知$P$到$l_2$的距离为$\dfrac{|ax_1+by_2+c|}{\sqrt{a^2+b^2}}$，代入$l_1$的方程即有$d = \dfrac{|c_1-c_2|}{\sqrt{a^2+b^2}}$
\end{enumerate}
例题：

    \begin{jieda}
        点 $A\left( {2,5}\right)$ 到直线 $l : x - {2y} + 3 = 0$ 的距离为 $d = \dfrac{\left| 2 - {10} + 3\right| }{\sqrt{1 + 4}} = \sqrt{5}$ .
    \end{jieda}

    \item 已知点$P(2, -1)$，求：
    \begin{enumerate}
        \item 过点$P$，且与原点距离为2的直线方程
        \item 过点$P$，且与原点距离最大的直线方程，并求出距离最大值
    \end{enumerate}
    \begin{jieda}
        \begin{enumerate}
            \item 若直线斜率不存在，则直线方程为$x = 2$，符合条件 \\
            若直线斜率存在，则设直线方程为$y = k(x-2)-1$，代入点到直线距离公式有$d = \dfrac{|2k+1|}{\sqrt{1+k^2}} = 2$，去分母平方后有$4k^2+4k+1 = 4(1+k^2)$，解得$k = \dfrac{3}{4}$，因此存在直线$y = \dfrac{3(x-2)}{4}-1$符合条件
            \item 若直线斜率存在，则设直线方程为$y = k(x-2)-1$，代入点到直线距离公式有$d = \dfrac{|2k+1|}{\sqrt{1+k^2}}$，距离$d$取最大值即$d^2 = \dfrac{4k^2+4k+1}{1+k^2} = 4+\dfrac{4k-3}{1+k^2}$取最大值，设$t = 4k-3$，则$\dfrac{4k-3}{1+k^2} = \dfrac{t}{1+\left(\dfrac{t+3}{4}\right)^2} = \dfrac{16t}{t^2+6t+25} = \dfrac{16}{t+\dfrac{25}{t}+6} \leq 1$，当$k = 2$时取得最大值.因此$d$的最大值为$\sqrt{5}$，此时直线方程为$y = 2x - 5$
        \end{enumerate}
    \end{jieda}

    \begin{jieda}

                    $\therefore$ 由两点式得直线 ${m}^{\prime }$ 的方程为 ${9x} - {46y} + {102} = 0$ .

                    解得 $C =  - 9$ ,
                    $\therefore {l}^{\prime }$ 的方程为 ${2x} - {3y} - 9 = 0$ .
