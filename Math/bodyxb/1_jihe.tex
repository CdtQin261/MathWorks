\chapter*{预备知识-集合}
\section*{元素与集合}
\begin{enumerate}
    \item 定义：一些确定对象的全体叫做\textbf{集合}, 集合中确定的对象称为\textbf{元素}
    \begin{example}
        \begin{dingautolist}{172}
            \item 1$\sim$10之间的所有偶数；
            \item 立德中学今年入学的全体高一学生；
            \item 所有的正方形；
            \item 到直线$l$的距离等于定长$d$的所有点；
            \item 方程$x^2-3x+2=0$的所有实数根；
            \item 地球上的四大洋.
        \end{dingautolist}
    \end{example}
    \item 符号表示：通常用小写字母表示元素，用大写字母表示集合
    \item \textbf{元素与集合的关系}
            \begin{enumerate}
                \item 元素$a$属于集合A，记做$a \in A$
                \item 元素$b$不属于集合A，记做$b \notin A$
            \end{enumerate}
        \item \textbf{集合中元素的性质}
            \ding{172}确定性 \ding{173}互异性 \ding{174} 无序性
        \item 集合的分类 \\
            \ding{172}有限集: 元素个数有限的集合称为有限集；\ding{173}无限集：元素个数无限的集合称为无限集；\ding{174}不含有任何元素的集合称为空集，用$\varnothing$表示。\\
            数学中常见的集合：点集、数集
        \item 常见数集符号：
            \begin{center}
            \setlength{\tabcolsep}{7mm}{
            \begin{tabular}{cc}
            \hline
                 数集& 符号 \\
            \hline
                 自然数集& $\mathbf{N}$ \\
                 正整数集& $\mathbf{N}^*$\\
                 整数集& $\mathbf{Z}$\\
                 有理数集& $\mathbf{Q}$\\
                 实数集& $\mathbf{R}$\\
             \hline
            \end{tabular}
            }
            \end{center}
            有时也用数集符号加正负角标的表示形式，例如:$\mathbf{R^+}$表示正实数集，$\mathbf{Z^-}$表示负整数集。
        \item 集合的表示方法：
            \begin{enumerate}
            \item 列举法：把集合的所有元素不重复地一一列举出来，并用花括号括起来表示集合的方法叫做列举法。\\
            例如：\{太平洋，大西洋，印度洋，北冰洋\}
            \item 描述法：在大括号内先写出这个集合中元素的一个记号，再画一条竖线，并在竖线的右边写上集合中所有元素具有的共同特征。“集合中所有元素具有的共同特征”是指：在该集合内的元素都具有这个特征，而不在这个集合内的元素都不具有这个特征。\\
            例如：$\{x|x>0\}$
            \item 区间：数学中常常需要表示满足一些不等式的全体实数所组成的集合，由此引入了区间的概念。区间可以在数轴上表示出来
                \begin{dingautolist}{172}
                    \item 满足不等式$a \leq x \leq b$的全体实数$x$组成的集合称为一个\textbf{闭区间}，记作$[a, b]$
                    \item 满足不等式$a < x < b$的全体实数$x$组成的集合称为一个\textbf{开区间}，记作$(a, b)$
                    \item 满足不等式$a \leq x < b$或$a < x \leq b$的全体实数$x$所组成的集合称为一个\textbf{半开半闭区间}，分别记作 $[a, b)$或 $(a, b]$
                    \item 用符号$+\infty$代表正无穷，满足不等式$x > a$的全体实数$x$组成的区间即是$(a, +\infty)$, 符号$-\infty$代表负无穷，$(-\infty, +\infty)$就等价于实数集$\mathbf{R}$。
                \end{dingautolist}
                \ps 特别注意，规定区间的左端点应小于右端点.
            \item 文氏图（Venn图）：用平面上封闭曲线的内部代表集合
            \end{enumerate}
            列举法只适合有限集，并且常用来表示元素个数不多的有限集，例如列举法表示不了不等式$x-3 < 7$的解集。 \\ 
            描述法只能描述集合中元素具有共同特征的集合，例如描述法表示不了集合$\{1, \mbox{太阳}, \mbox{上海}\}$。\\
            区间对应的集合一定是实数集的子集。
        \item 特殊的元素与集合：一个集合可以作为另一个集合的元素（集合集）
            \begin{example}
                \begin{dingautolist}{172}
                    \item 家庭这个集合可以包含家庭成员、宠物等元素，同时家庭这个集合也可以看作是社会这个集合中的一个元素。
                    \item $\{\varnothing\}$代表一个以空集为一个元素的集合。
                \end{dingautolist}
            \end{example}
\end{enumerate}

\section*{集合间的关系}
    \begin{enumerate}
        \item 包含关系：
            \begin{enumerate}
                \item $\mbox{任意} a \in A$, 满足 $a \in B$， 则称集合A包含于集合B(集合B包含集合A)，记做$A \subseteq B$，也称集合A是集合B的\textbf{子集}。规定空集是任意集合的子集。
                \item 若 $A \subseteq B, \mbox{存在}  b \in B, b \notin A$, 则称集合A真包含于集合B(集合B真包含集合A)，记做$A \subset B$(或者$A \subsetneqq B$)，也称集合A是集合B的\textbf{真子集}。
            \end{enumerate}
        注：任意两集合间\textbf{未必}存在包含关系，例如：$\{1,2\}$和$\{2,3\}$。
        \item 相等关系：$A \subseteq B$，$B \subseteq A$ 则 $A = B$
    \end{enumerate}
    相关结论：
    \begin{enumerate}
        \item 空集是一切集合的子集，是一切非空集合的真子集
        \item 含有$n$个元素的有限集$A$的子集有$2^n$个，其中$2^n-1$个为真子集，其中$2^n-2$个为非空真子集
        \item $A \subseteq B, B \subseteq C \Longrightarrow A \subseteq C$
    \end{enumerate}

\begin{enumerate}
\item 由\textbf{所有既属于集合$A$又属于集合$B$}的元素组成的集合，叫做集合$A$与$B$的\textbf{交集}(intersection)，记作$A \cap B$（读作“$A$交$B$”），即有$A \cap B = \{x|x \in A \mbox{且} x \in B\}$ \\
运算前后集合间的关系：$A \cap B \subseteq B, A \cap B \subseteq A$
\item 由\textbf{所有属于集合$A$或属于集合$B$}的元素所组成的集合，叫做集合$A$与$B$的\textbf{并集}(union)，记作$A \cup B$（读作“$A$并$B$”），即有$A \cup B = \{x|x \in A \mbox{或} x \in B\}$ \\
运算前后集合间的关系：$A \cup B \supseteq B, A \cup B \supseteq A$
\item 设$U$为全集，$A$是$U$的子集．由$U$中所有\textbf{不属于$A$}的元素组成的集合称为集合$A$在全集$U$中的\textbf{补集}(complementary set)，记作$C_UA$，即有$\{x|x \in U \mbox{且} x \notin A\}$，当全集明确时也可以将$C_UA$记作$\overline{A}$\\
集合与其补集具有性质：$\overline{\overline{A}} = A, A \cup \overline{A} = U, A \cap \overline{A} = \varnothing$
\end{enumerate}
\textbf{两个集合的运算本质上是对集合中元素的再分类}。

德摩根律：$\overline{A \cap B}= \overline{A} \cup \overline{B}, \overline{A \cup B}= \overline{A} \cap \overline{B}$

相关结论：
\begin{enumerate}
\item $A \cap B = A \Longleftrightarrow A \subseteq B$，$A \cup B = B \Longleftrightarrow A \subseteq B$
\item $A \cap A = A$, $A \cup A = A$
\item $A \cap \varnothing = \varnothing$, $A \cup \varnothing = A$
\end{enumerate}
