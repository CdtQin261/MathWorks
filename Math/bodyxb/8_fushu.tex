\chapter*{复数}
\section*{复数的概念}
\subsubsection*{复数的概念}
\begin{enumerate}
    \item 从方程的角度来看，负实数能不能开平方就是方程$x^2+a = 0(a > 0)$是否有解，进而换元$t = \dfrac{x}{\sqrt{a}}$后可以归结为$t^2+1 = 0$是否有解. 为了解决这个问题，数学家引入了一个不同于实数的新数$i$，称为\textbf{虚数单位}，并规定$i^2 = -1$
    \item 每一次数系的扩充都与实际需求密切相关，例如为了解决正方形对角线的度量，人们将有理数集扩充到了实数集. \\
    在数集扩充后，在实数集中规定的加法运算，乘法运算均与有理数集中的运算兼容，加法和乘法都满足交换律，乘法对加法都满足分配率.
    \item 在新的数系中，把实数$b$与$i$相乘，结果记作$bi$，把实数$a$与$bi$相加，结果记作$a+bi$. 所有的实数以及$i$都可以写作是$a+bi(a, b \in \mathbf{R})$的形式. \\
    我们将形如$a+bi(a, b \in \mathbf{R})$的数称为\textbf{复数}，全体复数构成的集合$\mathbf{C} = \{z| z = a+bi, a,b \in \mathbf{R}\}$叫做\textbf{复数集}，这样方程$x^2+1 = 0$在复数集中有解$x = i$.
    \item 复数通常用字母$z$表示，即$z = a + bi$，不做特殊说明时均有$a, b \in \mathbf{R}$，其中$a, b$分别叫做复数$z$的\textbf{实部}和\textbf{虚部}，记作$Re$ $z$和$Im$ $z$. 
    \begin{example}
        $3-4i$的实部是3，虚部是$-4$.
    \end{example}
    \item 两个复数相等当且仅当两个复数的实部和虚部分别相等. \\
    \ps 两个复数相等的一个等式，可以转化成其实部虚部相等的一个方程组，最多可以解出两个未知量
    \begin{example}
        $z_1 = (x+y) + i, z_2 = 1 + (x-y)i$，$z_1 = z_2 \Longleftrightarrow \left \{\begin{array}{l}
             x+y = 1 \\
             x-y = 1
        \end{array} \right.$，解得$\left\{\begin{array}{l}
             x = 1 \\
             y = 0
        \end{array}\right.$
    \end{example}
    \item 虚部不为零的复数我们称为\textbf{虚数}，虚部不为零且实部为零的复数我们称为\textbf{纯虚数}. 复数集、实数集、虚数集、纯虚数集之间的关系可以用图\ref{fig:fushuji}表示
    \begin{figure}[h]
        \centering
        \includegraphics[width=0.25\linewidth]{bodyxb//src/fushu1.png}
        \caption{复数集、实数集、虚数集、纯虚数集之间的关系}
        \label{fig:fushuji}
    \end{figure} 
    \item 特别地，$\left\{\begin{array}{l}
         z\mbox{是实数} \Longleftrightarrow b = 0  \\
         z\mbox{是虚数} \Longleftrightarrow b\neq 0 \\
         z\mbox{是纯虚数} \Longleftrightarrow a = 0, b\neq 0
    \end{array}\right.$
\end{enumerate}
例题：
\begin{enumerate}
    \item 若复数$z = 2m^2-3m-2+(m^2-3m+2)i$是纯虚数，实数$m = $\blkda{$-\dfrac{1}{2}$}
    \begin{jieda}
        $\left \{ \begin{array}{l}
             2m^2-3m-2 = 0 \\
             m^2-3m+2 \neq 0
        \end{array} \right.$，解得$m = -\dfrac{1}{2}$或$2$(舍)
    \end{jieda}
    \item 已知$m \in \mathbf{R}$，复数$z = \dfrac{m(m+2)}{m-1} + (m^2+2m-3)i$，当$m$为何值时，
    \begin{enumerate*}
        \item $z$是实数
        \item $z$是虚数
        \item $z$是纯虚数
        \item $z = \dfrac{1}{2} - 4i$
    \end{enumerate*}
    \begin{jieda}
        \begin{enumerate}
            \item $z$是实数$\Longleftrightarrow \left\{\begin{array}{l}
                 m \neq 1 \\
                 m^2+2m-3 = 0
            \end{array}\right.$，解得$m = -3$.
            \item $z$是虚数$\Longleftrightarrow \left\{\begin{array}{l}
                 m \neq 1 \\
                 m^2+2m-3 \neq 0
            \end{array}\right.$，解得$m \in (-\infty, -3) \cup (-3, 1) \cup (1, +\infty)$
            \item $z$是纯虚数$\Longleftrightarrow \left\{\begin{array}{l}
                 \dfrac{m(m+2)}{m-1} = 0 \\
                 m^2+2m-3 \neq 0
            \end{array}\right.$，解得$m \in \{-2, 0\}$
            \item $z = \dfrac{1}{2} - 4i \Longleftrightarrow \left\{\begin{array}{l}
                 \dfrac{m(m+2)}{m-1} = \dfrac{1}{2} \\
                 m^2+2m-3 = -4
            \end{array}\right.$，解得$m = -1$
        \end{enumerate}
    \end{jieda}
    \item 设$x,y,a \in \mathbf{R}$，$z_1 = (x^2-ax)+5i$，$z_2 = (x-4) - (ay^2+4y-1)i$，若对所有$x, y$都有$z_1 \neq z_2$，求$a$的取值范围.
    \begin{jieda}
        \begin{align*}
            & \forall x, y\text{都有} z_1 \neq z_2 \\
            & \Longleftrightarrow \left \{ \begin{array}{l}
             x^2-ax = x-4 \\
             -5 = ay^2+4y-1
            \end{array} \right.\text{两式不同时成立} \\
            & \Longleftrightarrow \left \{ \begin{array}{l}
             a = x+\dfrac{4}{x}-1 \\
             a = -4\left(\dfrac{1}{y^2} - \dfrac{1}{y}\right) 
            \end{array} \right. \text{两式不同时成立} \\
            & \Longleftrightarrow a \text{不在} f(x) = x+\dfrac{4}{x}-1 \text{与} g(x) = -4\left(\dfrac{1}{x^2} - \dfrac{1}{x}\right) \text{值域交集内}
        \end{align*}
        $f(x)$值域为$(-\infty, -5] \cup [3, +\infty)$，而$g(x)$的值域为$\left(-\infty, 1\right]$，故$a \in (-5, +\infty)$
    \end{jieda}
\end{enumerate}
\subsubsection*{复数的几何意义}
\begin{enumerate}
    \item 复数$z = a + bi$可以和有序实数对$(a, b)$建立一一对应的关系，因此可以建立复数与平面直角坐标系的点之间的一一对应关系
    \item 建立了平面直角坐标系表示复数的平面称作复平面，平面直角坐标系的两个轴称为\textbf{实轴}和\textbf{虚轴}，实轴上的点都表示实数，实轴以外的点都对应虚数，虚轴上除原点外的点都表示纯虚数.
    \begin{example}
        \begin{dingautolist}{172}
            \item 复平面内的原点对应着实数0
            \item 复平面内实轴上的点$(2, 0)$表示实数2
            \item 复平面内虚轴上的点$(0, -1)$表示纯虚数$-i$
            \item 复平面内的点$\left(-\dfrac{1}{2}, \dfrac{\sqrt{3}}{2}\right)$表示虚数$-\dfrac{1}{2} + \dfrac{\sqrt{3}}{2}i$
        \end{dingautolist}
    \end{example}
    \item 复平面内的向量与复平面上的点存在一一对应关系，因此复数与复平面内的向量存在一一对应关系
\end{enumerate}
例题：
\begin{enumerate}
    \item 若$\theta \in \left(\dfrac{5\pi}{4}, \dfrac{3\pi}{2}\right)$，则复数$z = cos\theta + sin\theta i$在复平面上所对应的点在第\blkda{三}象限.
\end{enumerate}
\subsubsection*{共轭与模}
\begin{enumerate}
    \item 如果$z_1 = a + bi, z_2 = c+di$，满足$a = c, b = -d$，则称两个复数为共轭复数，记作$z_1 = \overline{z_2}$ \\
    \ps $-z_1 = -a - bi$，复数的相反数与复数的共轭意义不同 \\
    在复平面上，共轭复数所对应的点关于实轴对称，特别地，实数的共轭复数是其本身.
    \item 对于复数$z = a+bi$，将$\sqrt{a^2+b^2}$称作是复数的模，记作$|z|$，复数的模即复数所对应向量的模. 易知$|z| = |\overline{z}|$
\end{enumerate}
例题：
\begin{enumerate}
    \item 设$z \in \mathbf{C}$，则满足条件\begin{enumerate*}
        \item $|z| = 1$
        \item $1 < |z| < 2$
    \end{enumerate*}的点的集合分别是什么图形？
    \begin{jieda}
        \begin{enumerate}
            \item 单位圆
            \item 圆心为原点，半径为1和2组成的圆环.
        \end{enumerate}
    \end{jieda}
    \item 已知$z_1 = m^2 - 2m - 3 + (m^2-4m+3)i (m \in \mathbf{R})$，$z_2 = 5 - 3i$，问$m$为何值时$z_1 = \overline{z_2}$
    \begin{jieda}
        $\overline{z_2} = 5+3i$，因此$z_1 = \overline{z_2}$即$\left \{ \begin{array}{l}
             m^2-2m-3 = 5 \\
             m^2-4m+3 = 3
        \end{array} \right.$，解得$m = 4$
    \end{jieda}
    \item 已知复数$z_1 = 1 + cos\theta + i sin\theta, z_2 = 1 - sin \theta + i cos\theta$，且$|z_1|^2+|z_2|^2 \geq 2$，求$\theta$的取值范围.
    \begin{jieda}
            $|z_1|^2 = (1+cos\theta)^2 + sin^2\theta = 2 + 2cos\theta$，$|z_2|^2 = (1-sin\theta)^2 + cos^2\theta = 2 - 2sin\theta$，故有$4 + 2(cos\theta - sin\theta) \geq 2$，即$sin\theta - cos\theta \leq 1$，即$\sqrt{2}sin\left(\theta - \dfrac{\pi}{4}\right) \leq 1$，故$\theta \in \left[2k\pi - \pi, 2k\pi + \dfrac{\pi}{2}\right], k \in \mathbf{Z}$
    \end{jieda}
\end{enumerate}
\section*{复数的运算}
\subsubsection*{复数的加减运算及其性质}
\begin{enumerate}
    \item $z_1 = a+bi, z_2 = c+di$，则$z_1 \pm z_2 = (a \pm c) + (b \pm d)i$
    \begin{example}
        $(3+4i) + (2-i) = 5+3i$
    \end{example}
    \item $z_1 = a+bi, z_2 = c+di, \vv{OZ_1} = (a, b), \vv{OZ_2} = (c, d)$，则$z_1 \pm z_2$对应着两个向量$\vv{OZ_1} \pm \vv{OZ_2}$
    \item $|z_1 - z_2|$等于两复数在复平面上对应两点$Z_1$与$Z_2$两点间距离
    \item 取共轭复数的运算与复数的加减运算可以交换运算顺序：$\overline{z_1 \pm z_2} = \overline{z_1} \pm \overline{z_2}$
    \item 复数的加减运算满足交换律和结合律：
    \begin{dingautolist}{172}
        \item $z_1 + z_2 = z_2 + z_1$
        \item $(z_1 + z_2) + z_3 = z_1 + (z_2 + z_3)$
    \end{dingautolist}
\end{enumerate}
例题：
\begin{enumerate}
    \item 若复数$z = (2m^2-3mi) - (2m-2i) + (-2+m^2i)$是实数，求实数$m$的值.
    \begin{jieda}
        $z = (2m^2-2m-2) + (m^2-3m+2)i$，$z$为实数$\Longleftrightarrow m^2-3m+2 = 0$，故$m = 1$或$2$.
    \end{jieda}
    \item 已知集合$P = \{\omega | \omega = z + \overline{z}, z \in \mathbf{C}\}, Q = \{\omega | \omega = z - \overline{z}, z \in \mathbf{C}\}$，则$P \cap Q =$\blkda{$\{0\}$}
    \begin{jieda}
        $P$对应着实轴上的点，$Q$对应虚轴上的点，因此两者交集为$\{0\}$，对应着原点.
    \end{jieda}
    \item 已知下列四个命题：
    \begin{dingautolist}{172}
        \item 若两个虚数$z_1, z_2$的和是实数，则$z_1, z_2$是共轭虚数
        \item 若$z_1, z_2$是共轭复数，则$z_1, z_2$的差是纯虚数
        \item 若复数$z - \overline{z} = 0$，则$z$是实数
        \item 若复数$z + \overline{z} = 0$，则$z$是纯虚数
    \end{dingautolist}
    其中正确的命题有\blkda{\ding{172}\ding{174}}
    \begin{jieda}
        \begin{dingautolist}{172}
            \item 正确，和是实数说明$z_1, z_2$的虚部互为相反数，又因为均为虚数，所以是共轭虚数
            \item 错误，若$z_1, z_2 \in \mathbf{R}$，则$z_1 = z_2$，因此$z_1 - z_2 = 0$
            \item 正确，$z - \overline{z} = 0$说明$z$的虚部为0，因此$z$实数
            \item 错误，反例：$z = 0$
        \end{dingautolist}
    \end{jieda}
    \item 已知复数$z_1, z_2$满足$|z_1| = |z_2| = 1$，且$|z_1 - z_2| = \sqrt{2}$，求证$|z_1 + z_2| = \sqrt{2}$
    \begin{jieda}
        易知$z_1, z_2$对应的复平面内的向量$\vv{OZ_1}, \vv{OZ_2}$的终点$Z_1, Z_2$在单位圆上，$|z_1 - z_2|$即$|\vv{Z_1Z_2}| = \sqrt{2}$，即弦$Z_1Z_2$的长度为$\sqrt{2}$，因此$\vv{Z_1}, \vv{Z_2}$的夹角为$\dfrac{\pi}{2}$，因此根据平行四边形法则可知$\vv{OZ_1} + \vv{OZ_2} = \sqrt{2}$，因此$|z_1 + z_2| = \sqrt{2}$
    \end{jieda}
    \item 若复数$z$满足$|z+1-i| = 1$，求$|z|$的最大值和最小值.
    \begin{jieda}
        \begin{enumerate}
            \item 几何意义\\
            $z$在复平面内所对应的点$Z$在以$(-1, 1)$为圆心，$1$为半径的圆上，因此根据几何意义可知$|z| \in [\sqrt{2}-1, \sqrt{2}+1]$，因此最大值为$1+\sqrt{2}$， 最小值为$\sqrt{2}-1$.
            \item 代数求解\\
            设$z = a + bi$，故$|z+1-i| = 1 \Longleftrightarrow (a+1)^2 + (b-1)^2 = 1$，利用三角代换有$a = -1 + cos\theta, b = 1 + sin \theta$，因此$|z| = \sqrt{a^2+b^2} = \sqrt{-2a-2b-1} = \sqrt{-2sin\theta+2cos\theta+3} = \sqrt{-2\sqrt{2}sin\left(\theta - \dfrac{\pi}{4}\right)+3} \in [\sqrt{2}-1, \sqrt{2}+1]$
        \end{enumerate}
    \end{jieda}
    \item 设复数$z$满足$\overline{z} + |z| = 2 - i$，求$z$
    \begin{jieda}
        \begin{enumerate}
            \item 代数求解
            \begin{enumerate}
                \item 方法1 \\
                 $\overline{z} = (2-|z|)-i$，设$z = a + bi$，则$a-bi = 2-i-\sqrt{a^2+b^2}$，即$\left \{ \begin{array}{l}
                 a = 2-\sqrt{a^2+b^2} \\
                 b = 1
                \end{array}\right. $，解得$a = \dfrac{3}{4}, b = 1$，因此$z = \dfrac{3}{4} + i$
                \item 方法2 \\
                $\overline{z} = (2-|z|)-i$，两侧同时取模平方后有$|\overline{z}|^2 = (2-|z|)^2 + 1$，即$5 - 4|z| = 0$，故$|z| = \dfrac{5}{4}$，因此$\overline{z} = \dfrac{3}{4} - i$，故$z = \dfrac{3}{4} + i$
            \end{enumerate}

            \item 几何意义 \\
            设$\overline{z}$在复平面对应的点$Z'$，$A(2, -1)$，则有$|z| = 2 - i - \overline{z} \Longleftrightarrow \left \{\begin{array}{l}
                 |\vv{OZ'}| = |\vv{Z'A}|  \\
                 \vv{Z'A} // \vv{i}
            \end{array} \right. $，因此$Z' \left(\dfrac{3}{4}, -1\right)$，故$z = \dfrac{3}{4} + i$
        \end{enumerate}
    \end{jieda}
    \item $z \in \mathbf{C}$，$2z - 3i + 2|z| - 1 = 0$，求$z$
    \begin{jieda}
        $2z = 1-2|z|+3i$，两侧同时取模平方后有$4|z|^2 = 4|z|^2-4|z|+10$，解得$|z| = \dfrac{5}{2}$，故$2z = -4+3i$，$z = -2 + \dfrac{3}{2}i$
    \end{jieda}
\end{enumerate}
\subsubsection*{复数的乘除法}
\begin{enumerate}
    \item 复数的乘法与多项式的乘法运算类似，只是要在运算的过程中将$i^2$换为$-1$，并把实部和虚部分别合并，即$(a+bi)(c+di) = (ac-bd) + (bc-ad)i$
    \begin{example}
        $(1+i)\cdot(3-2i) = 3 - 2i + 3i -2i^2 = 5+i$
    \end{example}
    \item 特别地$(a+bi)(a-bi) = a^2 - b^2i^2 = a^2+b^2$，即$z \cdot \overline{z} = |z|^2$
    \item 复数的除法是乘法的逆运算，可以用待定系数法求解，即$(x+yi)(a+bi) = c+di$，根据乘法运算法则求解出$x = \dfrac{ac+bd}{c^2+d^2}, y = \dfrac{bc-ad}{c^2+d^2}$
    \item 但通常求复数的除法可以利用共轭复数乘积为常数实现分母实数化的目的. \\
    $\dfrac{a+bi}{c+di} = \dfrac{(a+bi)(c-di)}{(c+di)(c-di)} = \dfrac{(ac+bd) + (bc-ad)i}{c^2+d^2} = \dfrac{ac+bd}{c^2+d^2} + \dfrac{bc-ad}{c^2+d^2}i$ \\
    即$\dfrac{z_1}{z_2} = \dfrac{z_1 \cdot \overline{z_2}}{|z_2|^2}$
    \item 特别地，$\overline{z} = \dfrac{|z|^2}{z}$
    \item 复数的乘法满足交换律、结合率、分配率
    \begin{dingautolist}{172}
        \item $z_1 \cdot z_2 = z_2 \cdot z_1$
        \item $(z_1 \cdot z_2) \cdot z_3 = z_1 \cdot (z_2 \cdot z_3)$
        \item $z_1 \cdot (z_2 + z_3) = z_1 \cdot z_2 + z_1 \cdot z_3$
    \end{dingautolist}
\end{enumerate}
例题：
\begin{enumerate}
    \item 若$a \in \mathbf{R}$，且$\dfrac{2+ai}{1+\sqrt{2}i} = -\sqrt{2}i$，求实数$a$的值
    \begin{jieda}
        $\dfrac{2+ai}{1+\sqrt{2}i} = -\sqrt{2}i$即$2+ai = (1+\sqrt{2}i)(-\sqrt{2}i) = 2-\sqrt{2}i$，因此$a = -\sqrt{2}$
    \end{jieda}
    \item $|z| = 1$，求证：$\dfrac{z}{1+z^2} \in \mathbf{R}$
    \begin{jieda}
        \begin{enumerate}
            \item 先除$z$后代换$\dfrac{1}{z}$ \\
            $\dfrac{z}{1+z^2} = \dfrac{1}{z + \dfrac{1}{z}} = \dfrac{1}{z + \overline{z}}$，因为$z + \overline{z} \in \mathbf{R}$，所以$\dfrac{z}{1+z^2} \in \mathbf{R}$
            \item 先代换1后除 \\
            $\dfrac{z}{1+z^2} = \dfrac{z}{z \cdot \overline{z} + z^2} = \dfrac{1}{\overline{z} + z}$，因为$z + \overline{z} \in \mathbf{R}$，所以$\dfrac{z}{1+z^2} \in \mathbf{R}$
        \end{enumerate}
    \end{jieda}
    \item $z \in \mathbf{C}$，满足$z + \dfrac{4}{z} \in \mathbf{R}$，且$|z-2| = 2$，求$z$
    \begin{jieda}
        \begin{enumerate}
            \item 方法1 \\
            因为$z + \dfrac{4}{z} \in \mathbf{R}$，所以$z + \dfrac{4}{z} = \overline{z} + \dfrac{4}{\overline{z}}$，整理有$z^2\overline{z} + 4\overline{z} = z\overline{z}^2 + 4z$，即$z|z|^2+4\overline{z} = |z|^2\overline{z} + 4z$，因此$(|z|^2-4)(z-\overline{z}) = 0$.
            \item 方法2 \\
            $z + \dfrac{4}{z} = z + 4 \cdot \dfrac{\overline{z}}{|z|^2} \in \mathbf{R} \Longleftrightarrow |z| = 2 \mbox{或} z \in \mathbf{R}$
        \end{enumerate}
        \begin{dingautolist}{172}
            \item $|z| = 2$，结合$|z-2| = 2$，解得$z = 1 \pm \sqrt{3} i$
            \item $z = \overline{z}$，结合$|z-2| = 4$，解得$z = 2$或$0$（舍）
        \end{dingautolist}
    \end{jieda}
    \item $z$为虚数，$\omega = z + \dfrac{1}{z}$，且$-1 < \omega < 2$，求\ding{172}$|z|$，\ding{173}$Re$ $z$的取值范围.
    \begin{jieda}
        \begin{enumerate}
            \item
            \begin{enumerate}
                \item 方法1 \\
                $\omega = z + \dfrac{1}{z} = z + \dfrac{\overline{z}}{|z|^2}$，因为$-1 < \omega < 2$，所以$\omega \in \mathbf{R}$，因此$z$与$\dfrac{\overline{z}}{|z|^2}$的虚部应该互为相反数，即$|z| = 1$
                \item 方法2 \\
                设$z = a+bi, a,b \in \mathbf{R}, b \neq 0$ \\
                $\omega = (a+bi) + \dfrac{1}{a+bi} = (a+bi) + \dfrac{a-bi}{a^2+b^2} = \left(a+\dfrac{a}{a^2+b^2}\right) + \left(b - \dfrac{b}{a^2+b^2}\right)i$ \\
                因为$-1 < \omega < 2$，所以$\omega \in \mathbf{R}$，因此$b - \dfrac{b}{a^2+b^2} = 0$，由于$b \neq 0$，所以$a^2+b^2 = 1$，故$|z| = 1$
            \end{enumerate}
            
            \item $\omega = 2a \in (-1, 2)$，因此$Re$ $z = a \in \left(-\dfrac{1}{2}, 1\right)$
        \end{enumerate}
    \end{jieda}
\end{enumerate}

\subsubsection*{复数的三角表示}
复数$z = a + bi$可以表示为$r(cos\theta + isin\theta)$，称为复数的三角形式，其中$r  = |z| = \sqrt{a^2+b^2}, cos\theta = \dfrac{a}{\sqrt{a^2+b^2}}, sin\theta = \dfrac{b}{\sqrt{a^2+b^2}}$，$\theta$称为复数$z$的辐角，记作$Arg z$，将满足$\theta \in [0, 2\pi)$的辐角$\theta$的值称作是复数$z$的辐角主值，记作$arg z$，因此有$Arg z = 2k\pi + arg z (k \in \mathbf{Z})$ \\
对应地，其共轭复数可以写作：$\overline{z} = r[cos(-\theta) + isin(-\theta)]$，两者模相等，辐角主值互为相反数.

例题：
\begin{enumerate}
    \item 将$2\left(cos\dfrac{\pi}{5}-isin\dfrac{\pi}{5}\right)$表示成三角形式为\blkda[5]{$2\left[cos\left(-\dfrac{\pi}{5}\right)+isin\left(-\dfrac{\pi}{5}\right)\right]$}
\end{enumerate}

\subsubsection*{复数乘除法的性质}
\begin{enumerate}
    \item 复数乘除法的几何意义：
        \begin{dingautolist}{172}
            \item $|z_1 \cdot z_2| = |z_1| \cdot |z_2|, \quad Arg(z_1 \cdot z_2) = Argz_1 + Argz_2$ \\
            \item $\left|\dfrac{z_1}{z_2}\right| = \dfrac{|z_1|}{|z_2|}, \quad Arg\left(\dfrac{z_1}{z_2}\right) = Argz_1 - Argz_2$
        \end{dingautolist}
        证明如下：\\
        设$z_1 = r_1(cos\alpha + i sin\alpha), z_2 = r_2(cos\beta + i sin\beta)$，则有
        \begin{dingautolist}{172}
            \item \begin{align*}
                z_1 \cdot z_2 & = r_1 r_2 (cos\alpha + i sin\alpha) \cdot (cos\beta + i sin\beta) \\
                              & = r_1 r_2 [(cos\alpha cos\beta - sin\alpha sin\beta) + i(sin\alpha cos\beta + cos\alpha sin\beta)] \\
                              & = r_1 r_2[cos(\alpha+\beta) + i sin(\alpha + \beta)]
            \end{align*}
            \item \begin{align*}
                \dfrac{z_1}{z_2} & = \dfrac{z_1 \cdot \overline{z_2}}{|z_2|^2} \\
                                 & = \dfrac{r_1 r_2 \cdot (cos\alpha + i sin\alpha) \cdot (cos\beta - i sin\beta) }{r_2^2} \\
                                 & = \dfrac{r_1}{r_2} \cdot [cos(\alpha - \beta) + i sin(\alpha - \beta)]
            \end{align*}
        \end{dingautolist}
        欧拉公式：$e^{i\theta}=cos\theta+isin\theta$
    \item 取共轭复数的运算与复数的乘除运算可以交换运算顺序：$\overline{z_1 \cdot z_2} = \overline{z_1} \cdot \overline{z_2}$；$\overline{\left(\dfrac{z_1}{z_2}\right)} = \dfrac{\overline{z_1}}{\overline{z_2}}$\\
    证明如下：    
    设$z_1 = r_1(cos\alpha + i sin\alpha), z_2 = r_2(cos\beta + i sin\beta)$，则有$\overline{z_1} = r_1[cos(-\alpha) + i sin(-\alpha)], \overline{z_2} = r_2[cos(-\beta) + i sin(-\beta)]$
    \begin{dingautolist}{172}
        \item 
        $\left \{ \begin{array}{l}
             z_1 \cdot z_2 = r_1 r_2[cos(\alpha+\beta) + i sin(\alpha + \beta)] \\
            \overline{z_1 \cdot z_2} = r_1 r_2[cos(\alpha+\beta) - i sin(\alpha + \beta)] \\
            \overline{z_1} \cdot \overline{z_2} = r_1 r_2[cos(-\alpha-\beta) + i sin(-\alpha - \beta)] = r_1 r_2[cos(\alpha+\beta) - i sin(\alpha + \beta)]
        \end{array} \right. $
        \item 
        $\left \{ \begin{array}{l}
            \dfrac{z_1}{z_2} = \dfrac{r_1}{r_2} \cdot [cos(\alpha - \beta) + i sin(\alpha - \beta)] \\
            \overline{\left(\dfrac{z_1}{z_2}\right)} = \dfrac{r_1}{r_2} \cdot [cos(\alpha - \beta) - i sin(\alpha - \beta)] \\
            \dfrac{\overline{z_1}}{\overline{z_2}} = \dfrac{r_1}{r_2} [cos(-\alpha + \beta) + i sin(-\alpha + \beta)] = \dfrac{r_1}{r_2} [cos(\alpha - \beta) - i sin(\alpha - \beta)]
        \end{array} \right. $
    \end{dingautolist}
\end{enumerate}

\subsubsection*{复数的乘方与开方}
\begin{enumerate}
    \item 实数集中整数的指数幂运算法则在复数集中依然成立，即对于$z, z_1, z_2$以及整数$m, n$有：
    \begin{dingautolist}{172}
        \item $z^0 = 1 (z \neq 0)$
        \item $z^{-n} = \dfrac{1}{z^n}(z \neq 0)$
        \item $z^m \cdot z^n = z^{m+n}$
        \item $(z^m)^n = z^{mn}$
        \item $(z_1 \cdot z_2)^n = z_1^n \cdot z_2^n$
    \end{dingautolist}
    \item 复数的开方是乘方的逆运算，如果$z_1^n = z_2$，则称$z_1$是$z_2$的一个$n$次方根 \\
    \ps 一个复数有$n$个$n$次方根.
\end{enumerate}

\section*{复数集中的方程}
\subsubsection*{实系数一元二次方程}
\begin{enumerate}
    \item 设一元二次方程$ax^2+bx+c = 0 (a,b,c \in \mathbf{R}, a \neq 0)$，因为$a \neq 0$所以原方程可化为$x^2 + \dfrac{b}{a}x + \dfrac{c}{a} = 0$，配方可得$\left(x+\dfrac{b}{2a}\right)^2 = \dfrac{b^2-4ac}{4a^2}$ 
    \begin{dingautolist}{172}
        \item $\Delta = b^2-4ac > 0$时，方程有两个不相等的实根，$x_1 = \dfrac{-b+\sqrt{b^2-4ac}}{2a}, x_2 = \dfrac{-b-\sqrt{b^2-4ac}}{2a}$
        \item $\Delta = b^2-4ac = 0$时，方程有两个相等的实根，$x_1 = x_2 = -\dfrac{b}{2a}$
        \item $\Delta = b^2-4ac < 0$时，方程有两个不相等的虚根，$x_1 = \dfrac{-b+i\sqrt{4ac-b^2}}{2a}, x_2 = \dfrac{-b-i\sqrt{4ac-b^2}}{2a}$
    \end{dingautolist}
    \item 容易验证，韦达定理对于三种情况均成立
    \item 代数基本定理：\textbf{复系数}一元$n$次方程$a_nx^n + a_{n-1}x^{n-1} + ... + a_1x+a_0 = 0, a_i \in C$在复数集中一定有$n$个根：$x_1, x_2, ..., x_n$并且有$a_nx^n + a_{n-1}x^{n-1} + ... + a_1x+a_0 = a_n(x-x_1)(x-x_2) ... (x-x_n)$ \\
    \ps 代数基本定理指出：多项式方程在复数域内一定有解，但并非所有方程在复数域中都一定有解，比如$e^z = 0$在复数域内无解.
    \item 共轭虚根定理：如果虚数$z$是\textbf{实系数}一元$n$次方程$a_nx^n + a_{n-1}x^{n-1} + ... + a_1x+a_0 = 0$的根，则$\overline{z}$也是该方程的根. \\
    证明如下：
    若$z$是$a_nx^n + a_{n-1}x^{n-1} + ... + a_1x+a_0 = 0$的根，则有$a_nz^n + a_{n-1}z^{n-1} + ... + a_1z+a_0 = 0$，故$\overline{a_nz^n + a_{n-1}z^{n-1} + ... + a_1z+a_0} = 0$，即$a_n(\overline{z})^n + a_{n-1}(\overline{z})^{n-1} + ... + a_1(\overline{z})+a_0 = 0$
\end{enumerate}
例题：
\begin{enumerate}
    \item 若 $ 2 - 3\mathrm{i} $ 是方程 $ {x}^{2} - {4x} + a = 0\left( {a \in  \mathrm{R}}\right) $ 的一个根,则其另外一个根是\blkda{$2+3i$}, $a = $ \blkda{13}
    \item 已知关于$x$的方程$x^2-px+1=0(p \in \mathbf{R})$的两个根为$x_1, x_2$，若$|x_1 - x_2| = 1$，求实数$p$的取值集合.
    \begin{jieda}
        $\Delta = p^2 - 4$
        \begin{dingautolist}{172}
            \item $\Delta \geq 0$，即$p \in (-\infty, -2] \cup [2, +\infty)$ \\
            此时$x_1, x_2$为实数， 因此$|x_1 - x_2| = 1 \Longleftrightarrow (x_1 - x_2)^2 = 1 \Longleftrightarrow (x_1+x_2)^2 - 4x_1x_2 = 1$，根据韦达定理有$p^2 - 4 = 1$，因此$p = \pm \sqrt{5}$
            \item $\Delta < 0$，即$p \in (-2, 2)$ \\
            此时$x_1,x_2$为虚数，因此$|x_1 - x_2| = 1 \Longleftrightarrow (x_1 - x_2)^2 = -1 \Longleftrightarrow (x_1+x_2)^2 - 4x_1x_2 = -1$，根据韦达定理有$p^2 - 4 = -1$，因此$p = \pm \sqrt{3}$
        \end{dingautolist}
        综上，$p \in \{\pm \sqrt{3}, \pm \sqrt{5}\}$
    \end{jieda}
    \item 已知$x_1, x_2$是关于$x$的方程$x^2+2x+m=0(m \in \mathbf{R})$的两个根，求$|x_1| + |x_2|$的值.
    \begin{jieda}
        $\Delta = 4 - 4m$
        \begin{dingautolist}{172}
            \item $\Delta \geq 0$，即$m \leq 1$ \\
            此时$x_1, x_2$为实数， $x_1 = -1+\sqrt{1-m}, x_2 = -1-\sqrt{1-m}$，故$|x_1|+|x_2| = \left \{ \begin{array}{ll}
                 2\sqrt{1-m}, &  m < 0 \\
                 2, & m \in [0, 1]
            \end{array} \right.$
            \item $\Delta < 0$，即$m > 1$ \\
            此时$x_1, x_2$为虚数， $x_1 = -1+i\sqrt{m-1}, x_2 = -1-i\sqrt{m-1}$，故$|x_1|+|x_2| = 2\sqrt{m}$
        \end{dingautolist}
        综上$|x_1|+|x_2| = \left \{ \begin{array}{ll}
                 2\sqrt{1-m}, &  m < 0 \\
                 2, & m \in [0, 1] \\
                 2\sqrt{m}, & m > 1
            \end{array} \right.$
    \end{jieda}
\end{enumerate}
