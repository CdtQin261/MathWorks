\chapter*{函数}
\section*{函数的概念}
\subsection*{函数的概念}
\subsubsection*{定义} 
设$D$是一个非空的实数集，如果按照某种确定的对应关系$f$,使对集合$D$中的任意给定的$x$,都有唯一的实数$y$与之对应，就称这个\textbf{对应关系} $f$为集合$D$上的一个\textbf{函数}(function), 记作 $y=f(x), x \in D$，其中，$x$叫做\textbf{自变量}(independent variable), 其取值范围(数集$D$)称为该函数的\textbf{定义域}(domain)。 此时，就称$y$是$x$的函数。当自变量$x$取值$x_0$时，由对应关系$f$所确定的对应于$x_0$的值$y_0$,称为函数在$x_0$处的\textbf{函数值}，记作$y_0=f(x_0)$。所有函数值组成的集合$\{y|y=f(x),x \in D\}$ 称为这个函数的\textbf{值域}(range)。

\subsubsection*{函数的定义域}
\begin{enumerate}
    \item 函数的自然定义域 \\
    在没有特殊说明的情况下，如果只写出反映函数对应关系的表达式而不明显地写出其定义域，此时函数$y=f(x)$的定义域由使得其表达式$f(x)$有意义的全体实数组成。例如$y=\dfrac{1}{x}$的定义域是$(-\infty, 0) \cup (0, +\infty)$。常需要考虑的情形包括：
    \begin{enumerate}
        \item 分式的分母不为0
        \item 偶次方根的被开方数非负
        \item 对数的真数大于0
        \item 在角的终边位于$y$轴的时候，正切函数没有定义
    \end{enumerate}
    \item 复合函数的定义域 
    \begin{enumerate}
        \item 复合函数的定义：一般地，对于两个函数$y=f(u)$和$u=g(x)$,如果通过中间变量$u$, $y$可以表示成$x$的函数，那么称这个函数为函数$y=f(u)$和$u=g(x)$的\textbf{复合函数}(composite function),记作$y=f(g(x))$。例如$y = (x+1)^2$可以看做是$y = u^2$和$u = x+1$复合得到。
        \item 若已知$f(x)$的定义域是$[a, b]$，则求$f(g(x))$的定义域即是求满足$a \leq g(x) \leq b$的$x$的取值范围。
    \end{enumerate}
    \item 实际问题的定义域 \\
    在解决实际问题的过程中应注意函数的定义域要符合实际问题的约束，如以长度作为自变量时，其定义域应满足非负。
\end{enumerate}

\subsubsection*{对应关系}
对应关系的概念完全是形式上的变化
\begin{example}
    若有$f(x) = x+1$，则有$f(x^2) = x^2+1, f(-x) = -x+1, f(x+1) = x+2$
\end{example}
例题：
\begin{enumerate}
    \item 写出函数的解析式：
    \begin{enumerate}
        \item $f(x) = \sqrt{x}, f(x+1) = $ \blkda{$\sqrt{x+1}$}
        \item $f(x) = \dfrac{1}{x}, f(x+1) =$  \blkda{$\dfrac{1}{x+1}$}
        \item $f(x) = \dfrac{1}{x}, f(-x^2) =$  \blkda{$\dfrac{1}{-x^2}$}
        \item $f(x) = x+1, f(-x^2) =$  \blkda{$-x^2+1$}
        \item $f(x) = x+1, f(3x-2) =$  \blkda{$3x-1$}
    \end{enumerate}
    
    \item 已知函数$f(x)=\begin{cases}
    x-3, & x\geq 10\\
    f(f(x+5)), & x<10\\    
    \end{cases}$, 
    则$f(7)$=\blkda{8}.
    \begin{jieda}
    \begin{align*}
    f(7) & =f(f(7+5))=f(f(12))\\
    & =f(12-3)=f(9)\\
    & =f(f(9+5))=f(f(14))\\
    & =f(14-3)=f(11)\\
    & =11-3=8.
    \end{align*}
    更为一般的情形, 分段讨论如下: 当$x\in[8,10)$时, 
    \begin{align*}
    f(x) & = f(f(x+5)) & \hspace*{1cm} x+5\in[13,15)\\
    & = f(x+5-3)=f(x+2) & \hspace*{1cm} x+2\in[10,12)\\
    & = x+2-3=x-1 & 
    \end{align*}
    当$x\in[6,8)$时,

    于是不难得到: 当$x\in[2k,2k+2)$时, $f(x)=f(f(x+5))=f(x+5-3)=f(x+2)$. 其中$k\zz,k\leq 4$. 
    \end{jieda}

    因为$x\in[-2,3]$, 所以$x+1\in[-1,4]$.\\

    \begin{jieda}
        首先求解$f(x)$的定义域。设$g(x) = x^2-x$，故$f(x^2-x)$即$f(g(x))$，因此$f(x)$的定义域即$g(x)$的值域。根据$f(x^2-x)$的定义域是$(0, 2)$可知$-\dfrac{1}{4} \leq g(x) < 2$，即$f(x)$的定义域是$\left[-\dfrac{1}{4}, 2\right)$，因此求解$f(\dfrac{1}{x} - 1)$的定义域即计算$-\dfrac{1}{4} \leq \dfrac{1}{x} - 1 < 2$的解集，解得$x \in \left(\dfrac{1}{3}, \dfrac{4}{3}\right]$
    \end{jieda}

    \begin{dingautolist}{172}
        \item 若 $P \cap  M = \varnothing$ ,则 $f\left( P\right)  \cap  f\left( M\right)  = \varnothing$ ； \item 若 $P \cap  M \neq  \varnothing$ ,则 $f\left( P\right)  \cap  f\left( M\right)  \neq  \varnothing$ ； 
        \item 若 $P \cup  M = \mathbf{R}$ ,则 $f\left( P\right)  \cup  f\left( M\right)  = \mathbf{R}$ ； 
        \item 若 $P \cup  M \neq  \mathbf{R}$ ,则 $f\left( P\right)  \cup  f\left( M\right)  \neq  \mathbf{R}$ 
    \end{dingautolist}
    判断命题的真假并证明.

    \begin{jieda}
        \begin{dingautolist}{172}
            \item 取 $P = \{ 1\} ,M = \{  - 1\}$ ,满足 $P \cap  M = \varnothing$ , 但 $f\left( P\right)  = \{ 1\} ,f\left( M\right)  = \{ 1\} ,f\left( P\right)  \cap  f\left( M\right)  = \{ 1\}$ ,故错误
            \item 若 $P \cap  M \neq  \varnothing$ ,由函数定义可得 $P \cap  M = \{ 0\}$ , 所以 $0 \in  \left\lbrack  {f\left( P\right)  \cap  f\left( M\right) }\right\rbrack   \neq  \varnothing$ ,故正确
            \item 取 $P = \{ x \mid  x \geq  0\} ,M = \{ x \mid  x < 0\}$ ,满足 $P \cup  M = {R}$ , 但 $f\left( P\right)  = \{ x \mid  x \geq  0\} ,f\left( M\right)  = \{ x \mid  x > 0\} ,f\left( P\right)  \cup  f\left( M\right)  \neq  \mathbf{R}$ ,故错误
            \item 假设 $f\left( P\right)  \cup  f\left( M\right)  = \mathbf{R}$
            
            因为$P \cup  M \neq  \mathbf{R}$，所以必然存在 $x \notin  \left( {P \cup  M}\right)$
            
            $x \notin  \left( {P \cup  M}\right) \Longrightarrow x \notin  P \Longrightarrow x \notin  f\left( P\right)$
            
            $\left \{ \begin{array}{l}
                 f\left( P\right)  \cup  f\left( M\right)  = \mathbf{R}  \\
                 x \in \mathbf{R} \\
                 x \notin  f\left( P\right)
            \end{array} \right. \quad \Longrightarrow x \in  f\left( M\right) \Longrightarrow - x \in  M $

            
            $x \notin  \left( {P \cup  M}\right) \Longrightarrow x \notin  M \Longrightarrow -x \notin  f\left( M\right)$

            
            $\left \{ \begin{array}{l}
                 f\left( P\right)  \cup  f\left( M\right)  = \mathbf{R}  \\
                 -x \in \mathbf{R} \\
                 -x \notin  f\left( M\right)
            \end{array} \right. \quad \Longrightarrow -x \in  f\left( P\right) \Longrightarrow - x \in  P $

            $\left \{ \begin{array}{l}
                 - x \in  P  \\
                 - x \in  M
            \end{array} \right. \quad \Longrightarrow -x \in  (P \cap M)$

            若$P \cap M = \varnothing$，此时矛盾，若$P \cap M = \{0\}$，则$x = 0$，则$x \in (P \cap M)$，与$x \notin (P \cup M)$矛盾.

            故原命题正确.
        \end{dingautolist}
    \end{jieda}
\end{enumerate}

\subsubsection*{相等的函数}
两个函数相等要保证其定义域和对应关系都完全一致，否则不是相等的函数
\begin{example}
    以下$f(x)$和$g(x)$不是同一个函数：
    \begin{enumerate}
        \item $f(x) = x, x \in \mathbf{R}$;\quad $g(x) = x, x \neq 0$
        \item $f(x) = x, x \in \mathbf{R}$;\quad $g(x) = x^2, x \in \mathbf{R}$
    \end{enumerate}
\end{example}

        令$t = x+1$，则$x = t-1$，故有$f(t) = (t-1)^2+2(t-1)+2=t^2+1$
    \end{jieda}
    \item 已知$f(\sqrt{x} + 1) = x + 2\sqrt{x}$，求$f(x)$
    \begin{jieda}
        设$t = \sqrt{x}+1 (t \geq 1)$，则$x = (t-1)^2$，故有$f(t) = (t-1)^2 + 2(t-1) = t^2 - 1 (t \geq 1)$ \\
        因此有$f(x) = x^2 - 1 (x \geq 1)$
    \end{jieda} 
    \item 函数$f(x)$满足$2f(x) - f(-x) = x^2$，求函数$f(x)$的解析式
    \begin{jieda}
        $\left \{ \begin{array}{l}
            2f(x) - f(-x) = x^2 \\
            2f(-x) - f(x) = x^2
        \end{array}
        \right.$
        解得$f(x) = x^2$
    \end{jieda}
    \item 已知$2f(x)+f\left(\dfrac{1}{x}\right)=3x$, 则$f(x)$=\blkda[2.5]{$2x - \dfrac{1}{x}$}.
    \begin{jieda}
        $\left \{ \begin{array}{l}
            2f(x)+f\left(\dfrac{1}{x}\right)=3x \\
            2f\left(\dfrac{1}{x}\right)+f(x)=\dfrac{3}{x}
        \end{array}
        \right.$ \\
        上式乘二减下式得到$3f(x) = 6x - \dfrac{3}{x}$，因此$f(x) = 2x - \dfrac{1}{x}$
    \end{jieda}
\end{enumerate}

\subsection*{函数图象的变换}

\begin{enumerate}
    \item 左加右减：$f(x+a)$的图象与$f(x)$的图象向左平移$a$个单位后的图象重合；$f(x-a)$的图象与$f(x)$的图象向右平移$a$个单位后的图象重合($a > 0$)。
    \item 上加下减：$f(x)+a$的图象与$f(x)$的图象向上平移$a$个单位后的图象重合；$f(x)-a$的图象与$f(x)$的图象向下平移$a$个单位后的图象重合($a > 0$)。
    \item 左右伸缩：$f(ax)$的图象可以由$f(x)$的图象沿$x$轴方向，向远离$y$轴拉伸（$0 < a < 1$）或向$y$轴压缩（$a > 1$）得到。
    \item 上下伸缩：$af(x)$的图象可以由$f(x)$的图象沿$y$轴方向，向远离$x$轴拉伸（$a > 1$）或向$x$轴压缩（$0 < a < 1$）得到。
    \item 上下翻折：$|f(x)|$的图象与$f(x)$的图象将$x$轴下方的部分做上下翻折操作后的图象重合。
    \item 左右翻折：$f(|x|)$的图象与$f(x)$的图象将$y$轴右侧的部分做左右翻折操作后的图象重合。
\end{enumerate}

函数图象的变换本质上是函数图象上点的位置变换，点的位置变换分成：水平平移、竖直平移、对称操作，这也对应于上述图象上的变换。
\par 对于$f(x)$上的一个点$(x_0, f(x_0))$，$f(x+b) (b>0)$上与之对应的点是$(x_0-b, f(x_0))$，点$(x_0-b, f(x_0))$是由点$(x_0, f(x_0))$向左平移$b$个单位，故$f(x+b)$的图象是由$f(x)$的图象向左平移$b$个单位得到。例如$f(x) = (x+1)\cdot(x+2)$，则$f(x-1) = x\cdot(x+1)$
\par 更进一步$f(x)$上的一个点$(x_0, f(x_0))$，$f(ax+b)(a > b, b > 0)$上与之对应的点是$\left(\dfrac{x_0-b}{a}, f(x_0)\right)$，点$\left(\dfrac{x_0-b}{a}, f(x_0)\right)$是由点$(x_0, f(x_0))$向左平移$b$个单位，横坐标再向$y$轴压缩为原本的$\dfrac{1}{a}$得到。故$f(x+a)$的图象是由$f(x)$的图象向向左平移$b$个单位，横坐标再向$y$轴压缩为原本的$\dfrac{1}{a}$得到。例如$f(x) = (x+1) \cdot (x+2)$，则$f(2x-1) = (2x)\cdot(2x+1)$ \\

从上述变换中可以看到水平的变换都对应着在自变量$x$上的变换，竖直方向上的变换对应着在函数值$y$上的变换；特别地，$y = f(ax+b)$可以由$y = f(x)$经过以下变换得到：
\begin{dingautolist}{172}
    \item 先平移后伸缩：先将$y = f(x)$的图象向左平移$b$个单位得到$y = f(x+b)$，后将图象上的点纵坐标不变横坐标变为原来的$\dfrac{1}{2}$，得到$y = \dfrac{1}{2x+1}$
    \item 先伸缩后平移：先将$y = f(x)$的图象上的点纵坐标不变横坐标变为原来的$\dfrac{1}{a}$，得到$y = f(ax)$，再将图象向左平移$\dfrac{b}{a}$个单位后得到$y = f(ax+b)$
\end{dingautolist}
\begin{example}
    将$y = \dfrac{1}{x}$变为$y = \dfrac{1}{2x+1}$可以通过\begin{dingautolist}{172}
    \item 先平移后伸缩：先将$y = \dfrac{1}{x}$的图象向左平移1个单位得到$y = \dfrac{1}{x+1}$，后将图象上的点纵坐标不变横坐标变为原来的$\dfrac{1}{a}$，得到$y = f(ax+b)$
    \item 先伸缩后平移：先将$y = \dfrac{1}{x}$的图象上的点纵坐标不变横坐标变为原来的$\dfrac{1}{2}$，得到$y = \dfrac{1}{2x}$，再将图象向左平移$\dfrac{1}{2}$个单位后得到$y = \dfrac{1}{2x+1}$
\end{dingautolist}
\end{example}

\section*{函数的零点}
\subsection*{函数与方程}
对于一般函数$y=f(x)$,我们把使$f(x)=0$的实数$x$叫做函数$y=f(x)$的\textbf{零点}(zero point)。
这样，函数$y=f(x)$的零点就是方程$f(x)=0$的实数解，也就是函数$y=f(x)$的图象与$x$轴的公共点的\textbf{横坐标}。

\subsection*{函数零点存在定理}
如果函数$y = f(x)$在区间$[a, b]$上的图象是一条\textbf{连续不断}的曲线，且有$\mathbf{f(a)f(b)<0}$, 那么函数$y = f(x)$在区间$(a, b)$内\textbf{至少有一个零点}，即存在$c \in (a, b)$, 使得$f(c) = 0$,这个$c$也是方程$f(x)=0$的解。(这是函数存在零点的\textbf{充分不必要条件})

\section*{函数的性质}
\subsection*{函数的奇偶性}
\subsubsection*{定义}
\begin{enumerate}
    \item 对于函数$y=f(x)$,如果对于其定义域$D$中\textbf{任意}给定的实数$x$,都有$-x \in D$,并且$f(-x)=f(x)$, 就称函数$y=f(x)$为\textbf{偶函数}(even function)。
    \item 对于函数$y=f(x)$,如果对于其定义域$D$中\textbf{任意}给定的实数$x$,都有$-x \in D$,并且$f(-x)=-f(x)$, 就称函数$y=f(x)$为\textbf{奇函数}(odd function)。
\end{enumerate}
\subsubsection*{性质推论} 
\begin{enumerate}
    \item 定义域不关于原点对称的函数是非奇非偶函数。
    \item 偶函数的图象关于$y$轴对称。
    \item 奇函数的图象关于原点中心对称。
    \item 如果0在奇函数的定义域内，那么函数图象过原点。\\
    引申结论：如果对称中心横坐标在定义域内，则函数图像必过对称中心
    \item 定义域相同的情况下，两个奇函数相加为奇函数，两个偶函数相加为偶函数。
    \item 任何一个定义域关于原点对称的函数总是可以唯一地拆成一个奇函数和一个偶函数之和：\\
    $f(x)=\dfrac{f(x)-f(-x)}{2}+\dfrac{f(x)+f(-x)}{2}$
    \item 定义域相同的情况下，两个奇函数相乘为偶函数，两个偶函数相乘为偶函数，一个奇函数和一个偶函数相乘为奇函数。  
    \item 对于复合函数，只要内层是偶函数，无论外层函数是什么，复合函数均是偶函数；外层是偶函数，内层是奇函数时复合函数是偶函数；内层函数和外层函数均为奇函数时复合函数为奇函数
\end{enumerate}

\subsection*{函数的单调性}
\subsubsection*{定义}
\begin{enumerate}
    \item 对于定义在$D$上的函数$y=f(x)$，设区间$I$是$D$的一个子集。对于区间$I$上的\textbf{任意}给定的两个自变量的值$x_1, x_2$，当$x_1 < x_2$时，如果总有$f(x_1) \leq f(x_2)$, 就称函数$y=f(x)$在区间$I$上是\textbf{增函数}(increasing function), 如果总有$f(x_1) < f(x_2)$, 就称函数$y=f(x)$在区间$I$上是\textbf{严格增函数}(strictly increasing function)。
    \item 对于定义在$D$上的函数$y=f(x)$，设区间$I$是$D$的一个子集。对于区间$I$上的\textbf{任意}给定的两个自变量的值$x_1, x_2$，当$x_1 < x_2$时，如果总有$f(x_1) \geq f(x_2)$, 就称函数$y=f(x)$在区间$I$上是\textbf{减函数}(decreasing function), 如果总有$f(x_1) > f(x_2)$, 就称函数$y=f(x)$在区间$I$上是\textbf{严格减函数}(strictly decreasing function)。
    \item 如果函数$y = f(x)$在某个区间$I$上是增（减）函数， 那么就称函数$y = f(x)$在区间$I$上是\textbf{单调函数}(monotonic function)，并称区间$I$是函数$y = f(x)$的一个\textbf{单调区间}。
\end{enumerate}
\subsubsection*{性质推论}
\begin{enumerate}
    \item 函数$y=f(x)$在区间$I$上是严格增（减）函数，等价于$\forall x_1, x_2 \in I, \dfrac{f(x_1) - f(x_2)}{x_1 - x_2} > 0 (< 0)$
    \item 闭区间内的严格单调函数的函数值和自变量存在一一对应关系，即一个函数值只对应一个自变量。
    \item 当函数在闭区间上单调时，最值一定在端点处取得。
    \item 复合函数的单调性满足“\textbf{同增异减}”，即内外函数的单调性相同时，复合函数单调递增，内外函数的单调性不同时，复合函数单调递减。\\
    \begin{example}
        设有复合函数$f(g(x))$，设$g(x)$在区间$I$内单调递增(如$g(x) = x+1$)
        \begin{enumerate}
            \item 若$f(u)$在$\{u | u = g(x), x \in I\}$单调递增，则$f(g(x))$在$I$内单调递增。（如$f(u) = u^2, I = [0, 1]$）
            \item 若$f(u)$在$\{u | u = g(x), x \in I\}$单调递减，则$f(g(x))$在$I$内单调递减。（如$f(u) = u^2, I = [-2, -1]$）
        \end{enumerate}
    \end{example}
\end{enumerate}
\subsubsection*{最大值与最小值}
\begin{enumerate}
    \item $\forall x \in I, f(x) \leq M, \exists x_0 \in I, f(x_0) = M \Rightarrow M \mbox{为} f(x)\mbox{在}I\mbox{上最大值}$。
    \item $\forall x \in I, f(x) \geq m, \exists x_0 \in I, f(x_0) = m \Rightarrow m \mbox{为} f(x)\mbox{在}I\mbox{上最小值}$。
\end{enumerate}

\subsection*{函数的周期性}
\subsubsection*{定义}
    如果对于函数$f(x)$，在定义域内的任意一个自变量$x$，都有一个\textbf{非零的常数}$T$，使得$x+T$也在定义域内，并且$f(x) = f(x+T)$。那么我们说$f(x)$是\textbf{周期函数}，$T$为$f(x)$的一个周期。如果在所有的周期中存在一个最小的正数，那么这个周期我们称作最小正周期。如不加说明，周期通常指最小正周期。
\subsubsection*{性质推论} 
\begin{enumerate}
    \item 如果$T$是$f(x)$的周期，那么$k \cdot T (k \in \mathbf{Z} \mbox{且} k \neq 0)$都是$f(x)$的周期
    \item 不是所有周期函数都有最小正周期，例如：常函数$y = c$
    \item 若函数在两个不同位置具有对称性，则其具有周期性
    \begin{dingautolist}{172}
        \item 如果一个函数$f(x)$既关于$x=a$轴对称，又关于$x=b$轴对称，那么$f(x)$是周期函数，$2(b-a)$是其一个周期$(a < b)$ \\
        $f(a-x) = f(a+x), f(b-x) = f(b+x)$ \\
        $\Rightarrow f(b+x) = f(b-x) = f(a+(-a+b-x)) = f(a-(-a+b-x)) = f(2a-b+x))$ \\
        $\Rightarrow f(x) = f(x+2(b-a))$
        \item 如果一个函数$f(x)$既关于$(a, 0)$中心对称，又关于$(b, 0)$中心对称，那么$f(x)$是周期函数，$2(b-a)$是其一个周期$(a < b)$\\
        $f(a-x) = -f(a+x), f(b-x) = -f(b+x)$ \\
        $\Rightarrow f(b+x) = -f(b-x) = -f(a+(-a+b-x)) = f(a-(-a+b-x)) = f(2a-b+x))$ \\
        $\Rightarrow f(x) = f(x+2(b-a))$
        \item 如果一个函数$f(x)$既关于$x=a$轴对称，又关于$(b, 0)$中心对称，那么$f(x)$是周期函数，$4(b-a)$是其一个周期$(a < b)$\\
        $f(a-x) = f(a+x), f(b-x) = -f(b+x)$ \\
        $\Rightarrow f(b+x) = -f(b-x) = -f(a+(-a+b-x)) = -f(a-(-a+b-x)) = -f(2a-b+x))$ \\
        $\Rightarrow f(b+x) = -f(2a-b+x)) = -f(b+(2a-2b+x)) = f(b-(2a-2b+x)) = f(-2a+3b-x)$ \\
        $\Rightarrow f(b+x) = f(-2a+3b-x) = f(a+(-3a+3b-x)) = f(a-(-3a+3b-x)) = f(4a-3b+x)$ \\
        $\Rightarrow f(x) = f(x+4(b-a))$
        \item 如果一个函数$f(x)$既关于$(a, 0)$中心对称，又关于$x=b$轴对称，那么$f(x)$是周期函数，$4(b-a)$是其一个周期$(a < b)$\\
        $f(a-x) = -f(a+x), f(b-x) = f(b+x)$ \\
        $\Rightarrow f(b+x) = f(b-x) = f(a+(-a+b-x)) = -f(a-(-a+b-x)) = -f(2a-b+x))$ \\
        $\Rightarrow f(b+x) = -f(2a-b+x)) = -f(b+(2a-2b+x)) = -f(b-(2a-2b+x)) = -f(-2a+3b-x)$ \\
        $\Rightarrow f(b+x) = -f(-2a+3b-x) = -f(a+(-3a+3b-x)) = f(a-(-3a+3b-x)) = f(4a-3b+x)$ \\
        $\Rightarrow f(x) = f(x+4(b-a))$
    \end{dingautolist}
\end{enumerate}
