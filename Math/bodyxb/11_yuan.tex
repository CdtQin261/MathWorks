\chapter*{解析几何-圆}
\subsection*{圆的标准方程}
\begin{enumerate}
    \item 平面上到定点的距离等于定长（大于零）的点的轨迹（或集合）叫做圆，定点称作是圆的圆心，定长为圆的半径. 因此只要圆心的位置和半径确定了，圆就唯一确定了.
    \item 设圆心坐标为$(a, b)$，圆的半径为$r$，圆上任意一点的坐标为$(x, y)$，则根据两点间距离公式有$\sqrt{(x-a)^2 + (y-b)^2} = r$，变形可得\textbf{圆的标准方程}：$(x-a)^2 + (y-b)^2 = r^2$
    \item 根据圆心和半径就可以写出圆的标准方程，圆的标准方程体现了圆的几何特征
\end{enumerate}
例题：
\begin{enumerate}
    \item 圆心为(1,1)且过原点的圆的方程是\blkda[4]{${\left( x - 1\right) }^{2} + {\left( y - 1\right) }^{2} = 2$}

    \begin{jieda}
        因为圆心为(1,1)且过原点,所以该圆的半径 $r = \sqrt{{1}^{2} + {1}^{2}} = \sqrt{2}$ ,则该圆的方程为 ${\left( x - 1\right) }^{2} + {\left( y - 1\right) }^{2} = 2$ .
    \end{jieda}
    
    \item 求满足下列条件的圆的方程：
    \begin{enumerate*}
        \item 过$A(0, 5), B(1, -2), C(-3, -4)$三点
        \item 过$A(1, 2), B(1, 10)$两点，且与直线$x-2y-1=0$相切
    \end{enumerate*}
    \begin{jieda}
        \begin{enumerate}
            \item 
            \begin{enumerate}
                \item 方法1 \\
                设圆的方程为$(x-a)^2+(y-b)^2 = r^2$ \\
                代入三个点的坐标，得到方程组$\left\{\begin{array}{l}
                     (0-a)^2+(5-b)^2 = r^2  \\
                     (1-a)^2+(-2-b)^2 = r^2  \\
                     (-3-a)^2+(-4-b)^2 = r^2
                \end{array}\right.$，解得$\left\{\begin{array}{l}
                     a = -3  \\
                     b = 1  \\
                     r = 5
                \end{array}\right.$ \\
                故所求圆的标准方程为$(x+3)^2+(y-1)^2=25$
                \item 方法2 \\
                $AB$中点坐标$M\left(\dfrac{1}{2}, \dfrac{3}{2}\right)$，故$AB$中垂线所在直线方程为$y = \dfrac{1}{7}\left(x - \dfrac{1}{2}\right)+\dfrac{3}{2}$ \\
                $AC$中点坐标$M\left(-\dfrac{3}{2}, \dfrac{1}{2}\right)$，故$AC$中垂线所在直线方程为$y = -\dfrac{1}{3}\left(x + \dfrac{3}{2}\right)+\dfrac{1}{2}$ \\
                联立两直线方程得到圆心坐标为$(-3, 1)$，根据两点间距离公式可得半径为$r = 5$
            \end{enumerate}
            \item 因为$AB$的中垂线为直线$y = 6$，因此可设圆心坐标为$(a, 6)$，计算圆心到点$A$距离可知半径$r = \sqrt{(a-1)^2+16}$，根据点到直线距离公式可知半径为$r = \dfrac{|a-13|}{\sqrt{5}}$，因此$\sqrt{(a-1)^2+16} = \dfrac{|a-13|}{\sqrt{5}}$，解得$a = 3$或$-7$
            \begin{dingautolist}{172}
                \item 圆心为$(3, 6)$，则半径为$2\sqrt{5}$，因此圆的方程为$(x-3)^2+(y-6)^2=20$
                \item 圆心为$(-7, 6)$，则半径为$4\sqrt{5}$，因此圆的方程为$(x+7)^2+(y-6)^2=80$
            \end{dingautolist}
        \end{enumerate}
    \end{jieda}
    
    \item 设点 $M$ 在直线 ${2x} + y - 1 = 0$ 上,点(3,0)和(0,1)均在 $\odot  M$ 上,则 $\odot  M$ 的方程为\blkda[4]{${\left( x - 1\right) }^{2} + {\left( y + 1\right) }^{2} = 5$}.
        \begin{jieda}
            \begin{enumerate}
                \item 方法一 \\
                设 $\odot  M$ 的方程为 ${\left( x - a\right) }^{2} + {\left( y - b\right) }^{2} = {r}^{2}$ , 则 $\left\{  \begin{array}{l} {2a} + b - 1 = 0, \\  {\left( 3 - a\right) }^{2} + {b}^{2} = {r}^{2}, \\  {a}^{2} + {\left( 1 - b\right) }^{2} = {r}^{2}, \end{array}\right.$
                
                解得 $\left\{  \begin{array}{l} a = 1, \\  b =  - 1, \\  {r}^{2} = 5, \end{array}\right.$
                
                $\therefore  \odot  M$ 的方程为 ${\left( x - 1\right) }^{2} + {\left( y + 1\right) }^{2} = 5$ .
                \item 方法二 \\
                设 $A\left( {3,0}\right) ,B\left( {0,1}\right) , \odot  M$ 的半径为 $r$，则 ${k}_{AB} = \dfrac{1 - 0}{0 - 3} =  - \dfrac{1}{3},{AB}$ 的中点坐标为 $\left( {\dfrac{3}{2},\dfrac{1}{2}}\right)$ ,
                
                $\therefore {AB}$ 的垂直平分线方程为 $y - \dfrac{1}{2} = 3\left( {x - \dfrac{3}{2}}\right)$ ,即 ${3x} - y - 4 = 0$ .
                
                联立 $\left\{  \begin{array}{l} {3x} - y - 4 = 0, \\  {2x} + y - 1 = 0, \end{array}\right.$ 解得 $\left\{  \begin{array}{l} x = 1, \\  y =  - 1, \end{array}\right.$
                
                $\therefore M\left( {1, - 1}\right)$ ,
                
                $\therefore {r}^{2} = {\left| MA\right| }^{2} = {\left( 3 - 1\right) }^{2} + {\left\lbrack  0 - \left( -1\right) \right\rbrack  }^{2} = 5$ ,
                
                $\therefore  \odot  M$ 的方程为 ${\left( x - 1\right) }^{2} + {\left( y + 1\right) }^{2} = 5$ .
            \end{enumerate}
        \end{jieda}
    \item 设平面上有一条长度为4的线段$AB$，$A(-2, 0), B(2, 0)$，求到$A$距离与到$B$距离之比为2的点的轨迹方程.
    \begin{jieda}
        设轨迹上的点的坐标为$(x, y)$，则$\sqrt{(x+2)^2+y^2} = 2\sqrt{(x-2)^2+y^2}$，平方化简得$\left(x-\dfrac{10}{3}\right)^2 + y^2 = \dfrac{64}{9}$，故轨迹为以$C\left(\dfrac{10}{3}, 0\right)$为圆心，$\dfrac{8}{3}$为半径的圆
    \end{jieda}
\end{enumerate}
\subsection*{圆的一般方程}
\begin{enumerate}
    \item 将圆的标准方程$(x-a)^2+(y-b)^2 = r^2$展开可得$x^2+y^2-2ax-2by+a^2+b^2-r^2 = 0$，由此可见任意一个圆的方程均可以写成：$x^2+y^2+Dx+Ey+F = 0$的形式，其中$D,E,F$均为实常数.
    \item 对于任意一个$x^2+y^2+Dx+Ey+F = 0$形式的方程，我们可以将其配方整理得$\left(x + \dfrac{D}{2}\right)^2+\left(y+\dfrac{E}{2}\right)^2 = \dfrac{D^2+E^2-4F}{4}$
    \begin{dingautolist}{172}
        \item 当$\dfrac{D^2+E^2-4F}{4} > 0$时，$x^2+y^2+Dx+Ey+F = 0$表示圆心为$\left(-\dfrac{D}{2}, -\dfrac{E}{2}\right)$，半径为$\dfrac{\sqrt{D^2+E^2-4F}}{2}$的圆
        \item 当$\dfrac{D^2+E^2-4F}{4} = 0$时，$x^2+y^2+Dx+Ey+F = 0$对应点$\left(-\dfrac{D}{2}, -\dfrac{E}{2}\right)$
        \item 当$\dfrac{D^2+E^2-4F}{4} < 0$时，$x^2+y^2+Dx+Ey+F = 0$不表示任何图形.
    \end{dingautolist}
    $\dfrac{D^2+E^2-4F}{4} > 0$时，我们将$x^2+y^2+Dx+Ey+F = 0$叫做圆的一般方程，将$\dfrac{D^2+E^2-4F}{4}$称作是方程的判别式
    \item 圆的一般方程体现的是圆的代数特征，任何一个圆的方程都是一个特殊的二元二次方程.
    \item 特别地，若已知$A(x_1, y_2), B(x_2, y_2)$，则以$AB$为直径的圆上任意一点$M(x, y)$满足$\vv{MA} \cdot \vv{MB} = 0$，故以$AB$为直径的圆的方程可以表示为$(x-x_1)(x-x_2) + (y-y_1)(y-y_2) = 0$
\end{enumerate}
例题：
\begin{enumerate}
    \item 若曲线 $C : {x}^{2} + {y}^{2} + {2ax} - {4ay} - {10a} = 0$ 表示圆,则实数 $a$ 的取值范围为\blkda[4]{$\left( {-\infty , - 2}\right)  \cup  \left( {0, + \infty }\right)$}
    \begin{jieda}
        由 ${x}^{2} + {y}^{2} + {2ax} - {4ay} - {10a} = 0$ ，得 ${\left( x + a\right) }^{2} + {\left( y - 2a\right) }^{2} = 5{a}^{2} + {10a}$ ,
        
        由该曲线表示圆,可知 $5{a}^{2} + {10a} > 0$ ,解得 $a > 0$ 或 $a <  - 2$ .
    \end{jieda} 
    \item 已知 $a \in  \mathbf{R}$ ,方程 ${a}^{2}{x}^{2} + \left( {a + 2}\right) {y}^{2} + {4x} + {8y} + {5a} = 0$ 表示圆,则圆心坐标为\blkda{$\left( {-2, - 4}\right)$},半径为\blkda{5}.
    \begin{jieda}
        由圆的一般方程的形式知, $a + 2 = {a}^{2}$ ,解得 $a = 2$ 或 $a =  - 1$ .
        \begin{dingautolist}{172}
            \item 当 $a = 2$ 时,该方程可化为 ${x}^{2} + {y}^{2} + x + {2y} + \dfrac{5}{2} = 0$ ,
            
            $\because {D}^{2} + {E}^{2} - {4F} = {1}^{2} + {2}^{2} - 4 \times  \dfrac{5}{2} < 0$ ,
            
            $\therefore a = 2$ 不符合题意;
            \item 当 $a =  - 1$ 时,方程可化为 ${x}^{2} + {y}^{2} + {4x} + {8y} - 5 = 0$ ,
            
            即 ${\left( x + 2\right) }^{2} + {\left( y + 4\right) }^{2} = {25}$ ,
            
            $\therefore$ 圆心坐标为(-2, - 4),半径为 5 .
        \end{dingautolist}        
    \end{jieda}
    \item 已知$\triangle ABC$的顶点坐标为$A(2, -1), B(6, 3), C(-2, 3)$，求$\triangle ABC$的外接圆方程
    \begin{jieda}
        设圆的方程为$x^2+y^2+Dx + Ey+F = 0$，代入三个顶点坐标可得方程组：\\
        $\left\{\begin{array}{l}
             5+2D-E+F = 0  \\
             45+6D+3E+F=0 \\
             13-2D+3E+F = 0
        \end{array} \right.$，\\
        解得$D =-4, E = -6, F = -3$，此时$D^2+E^2-4F > 0$，因此外接圆的方程为$x^2+y^2-4x-6y-3=0$
    \end{jieda}
    \item 设方程$x^2+y^2-2(m+3)x+2(1-4m^2)y+16m^4+9=0$表示一个圆
    \begin{enumerate}
        \item 求实数$m$的取值范围
        \item 当$m$取何值时，圆的半径最大？并求出半径的最大值
        \item 求圆心的轨迹方程
    \end{enumerate}
    \begin{jieda}
        \begin{enumerate}
            \item 方程$x^2+y^2-2(m+3)x+2(1-4m^2)y+16m^4+9=0$表示一个圆即$4(m+3)^2+4(1-4m^2)^2-4(16m^4+9)>0$，即$(m^2+6m+9)+(1-8m^2+16m^4)-(16m^4+9) > 0$，即$-7m^2+6m+1>0$，解得$m \in \left(-\dfrac{1}{7}, 1\right)$
            \item 方程$x^2+y^2-2(m+3)x+2(1-4m^2)y+16m^4+9=0$表示一个圆心为$\left((m+3), (4m^2-1)\right)$，$\sqrt{-7m^2+6m+1}$为半径的圆，因此当$m = \dfrac{3}{7}$时圆的半径最大为$\dfrac{4\sqrt{7}}{7}$
            \item 圆心为$\left((m+3), (4m^2-1)\right)$，故圆心的参数方程为$\left\{\begin{array}{l}
                 x = m+3 \\
                 y = 4m^2-1
            \end{array}\right.$，因此圆心轨迹方程为$y = 4(x-3)^2-1, x \in \left(\dfrac{20}{7}, 4\right)$
        \end{enumerate}
    \end{jieda}
\end{enumerate}
\subsection*{点和圆的位置关系}
\begin{enumerate}
    \item 平面上的点和圆的位置关系有三种：点在圆内，点在圆上，点在圆外
    \item 借助点的坐标和圆的标准方程，通过代数计算判断它们的位置关系，设点$P(x_0, y_0)$与圆$C:(x-a)^2 + (y-b)^2 = r^2$，若点$P$到圆心$C$的距离为$d$，则$d = \sqrt{(x_0-a)^2 + (y_0-b)^2}$，故有：
    \begin{dingautolist}{172}
        \item 点$P$在圆外$\Longleftrightarrow (x_0-a)^2 + (y_0-b)^2 > r^2$
        \item 点$P$在圆上$\Longleftrightarrow (x_0-a)^2 + (y_0-b)^2 = r^2$
        \item 点$P$在圆内$\Longleftrightarrow (x_0-a)^2 + (y_0-b)^2 < r^2$
    \end{dingautolist}
\end{enumerate}
例题：
\begin{enumerate}
    \item 下列结论正确的有\blkda{\ding{172}\ding{174}\ding{175}}
    \begin{dingautolist}{172}
        \item 确定圆的几何要素是圆心与半径
        \item  ${\left( x - 2\right) }^{2} + {\left( y + 1\right) }^{2} = {a}^{2}\left( {a \neq  0}\right)$ 表示以(2,1)为圆心, $a$ 为半径的圆. 
        \item 方程 $A{x}^{2} + {Bxy} + C{y}^{2} + {Dx} + {Ey} + F = 0$ 表示圆的充要条件是 $A = C \neq  0,B = 0,{D}^{2} + {E}^{2} - {4AF} > 0.$
        \item 若点 $M\left( {{x}_{0},{y}_{0}}\right)$ 在圆 ${x}^{2} + {y}^{2} + {Dx} + {Ey} + F = 0$ 外,则 ${x}_{0}^{2} + {y}_{0}^{2} + D{x}_{0} + E{y}_{0} + F > 0.$
    \end{dingautolist}
    \item 判断点$M_1(6, -6)$、$M_2(-5, -1)$、$M_3(3, 1)$与圆$C: (x-2)^2+(y+3)^2 = 25$的位置关系
    \begin{jieda}
        $M_1$在圆上，$M_2$在圆外，$M_3$在圆内.
    \end{jieda}
    \item 已知圆$C: x^2+y^2-2x-4y-20 = 0$和直线$l: (2m+1)x + (m+1)y-7m-4 = 0$，求证：直线$l$与圆$C$恒有两个交点.
    \begin{jieda}
        将圆写成标准方程为$(x-1)^2+(y-2)^2 = 25$，观察可知直线$l$恒过点$M(3, 1)$，而$M$到圆心$(1, 2)$的距离为$\sqrt{5}$，故点$M$在圆内部，因此直线$l$与圆$C$恒有两个交点.
    \end{jieda}
\end{enumerate}
\subsection*{直线和圆的位置关系}
\begin{enumerate}
    \item 直线和圆有三种位置关系：相交、相切、相离
    \item 设直线$L$方程为：$Ax+By+C = 0$，圆$C$的方程为：$(x-a)^2+(y-b)^2 = r^2$ \\
    \begin{center}
        \begin{tabular}{c|c|c|c}
            \hline
             & 图形 & 代数角度 & 几何角度 \\
             \hline
            相交 & \includegraphics[width=0.1\textwidth]{bodyxb/src/01972b2d-adb6-7aca-bab1-e684ea51e946_0_1107_647_108_103_0.jpg} & $\left \{\begin{array}{l}
                      Ax+By+C = 0  \\
                      (x-a)^2+(y-b)^2 = r^2
                 \end{array} \right.$有两组解 & $d = \dfrac{|Aa+Bb+C|}{\sqrt{A^2+B^2}} < r$ \\
            相切 & \includegraphics[width=0.1\textwidth]{bodyxb/src/01972b2d-adb6-7aca-bab1-e684ea51e946_0_922_646_124_108_0.jpg}& $\left \{\begin{array}{l}
                      Ax+By+C = 0  \\
                      (x-a)^2+(y-b)^2 = r^2
                 \end{array} \right.$有一组解 & $d = \dfrac{|Aa+Bb+C|}{\sqrt{A^2+B^2}} = r$ \\
            相离 & \includegraphics[width=0.1\textwidth]{bodyxb/src/01972b2d-adb6-7aca-bab1-e684ea51e946_0_738_646_142_108_0.jpg} & $\left \{\begin{array}{l}
                      Ax+By+C = 0  \\
                      (x-a)^2+(y-b)^2 = r^2
                 \end{array} \right.$无解 & $d = \dfrac{|Aa+Bb+C|}{\sqrt{A^2+B^2}} > r$ \\
             \hline
        \end{tabular}
    \end{center}
    \item 垂径定理：当直线$L$与圆$C$相交的情况下，设直线被圆所截弦长为$l$，则有$\left(\dfrac{l}{2}\right)^2+d^2 = r^2$，即$l = 2\sqrt{r^2-d^2}$
\end{enumerate}
例题：
\begin{enumerate}
    \item 若直线$mx+ny = 4$和圆$C: x^2+y^2 = 4$没有公共点，试判断$P(m,n)$与圆$C$的位置关系.
    \begin{jieda}
        $d = \dfrac{4}{\sqrt{m^2+n^2}} > 2$，故$\sqrt{m^2+n^2} \in (0, 2)$，因此$P$到圆心的距离小于半径，因此$P$在圆内.
    \end{jieda}
    \item 已知圆$C: x^2+y^2 = 2$和直线$l: y = x + b$，当$b$为何值时，
    \begin{enumerate}
        \item 直线$l$与圆$C$有两个公共点
        \item 直线$l$与圆$C$有一个公共点
        \item 直线$l$与圆$C$没有公共点
    \end{enumerate}
    \begin{jieda}
        直线与圆的距离$d = \dfrac{\sqrt{2}|b|}{2}$，圆的半径为$r = \sqrt{2}$，因此
        \begin{dingautolist}{172}
            \item $b \in (-2, 2)$时，直线与圆有两个公共点
            \item $b \in \{-2, 2\}$时，直线与圆有一个公共点
            \item $b \in (-\infty, -2) \cup (2, +\infty)$时，直线与圆没有公共点
        \end{dingautolist}
    \end{jieda}
    
    \item 直线 ${kx} - y + 2 - k = 0$ 与圆 ${x}^{2} + {y}^{2} - {2x} - 8 = 0$ 的位置关系为\dotfill{(\kh{C})}
    \xx{相交、相切或相离}{相交或相切}{相交}{相切}
    \begin{jieda}
        \begin{enumerate}
            \item 方法1 \\
            直线 ${kx} - y + 2 - k = 0$ 的方程可化为 $k\left( {x - 1}\right)  - \left( {y - 2}\right)  = 0$ ,该直线恒过定点(1,2).
            
            因为 ${1}^{2} + {2}^{2} - 2 \times  1 - 8 < 0$ ，所以点(1,2)在圆 ${x}^{2} + {y}^{2} - {2x} - 8 = 0$ 的内部，所以直线 ${kx} - y + 2 - k = 0$ 与圆 ${x}^{2} + {y}^{2} - {2x} - 8 = 0$ 相交.

            \item 方法2 \\
            圆的方程可化为 ${\left( x - 1\right) }^{2} + {y}^{2} = {3}^{2}$，所以圆的圆心为$(1,0)$，半径为 $3$ . 
            
            圆心到直线 ${kx} - y + 2 - k = 0$ 的距离为$\dfrac{\left| k + 2 - k\right| }{\sqrt{1 + {k}^{2}}} = \dfrac{2}{\sqrt{1 + {k}^{2}}} \leq  2 < 3$ ,所以直线与圆相交.
        \end{enumerate}
    \end{jieda}
    \item 已知圆$C$的圆心在直线$l_1: 2x-y=0$上，且圆$C$与$x$轴相切，直线$l_2: 3x-4y=0$被圆$C$截得的弦长为$2\sqrt{3}$，求圆$C$的标准方程
    \begin{jieda}
        设圆心坐标为$(a, 2a)$，则圆的标准方程为$(x-a)^2+(x-2a)^2 = 4a^2$，圆心到$l_2$的距离为$d = |a|$，因此根据垂径定理可知$3+a^2 = (2a)^2$，因此$|a| = 1$，故圆的标注方程为$(x-1)^2+(x-2)^2 = 4$或$(x+1)^2+(x+2)^2 = 4$
    \end{jieda}
    \item 已知圆$C:(x+1)^2+y^2=r^2 (r>0)$
    \begin{enumerate}
        \item 当$r = 2$时，过点$P(1, 3)$作圆的切线，求切线$l$的方程
        \item 已知圆$C$上只有两个点到直线$l': 4x-3y-6 = 0$的距离为1，求$r$的取值范围.
    \end{enumerate}
    \begin{jieda}
        \begin{enumerate}
            \item 易知点$P$在圆外，因此过$P$有两条切线
            \begin{dingautolist}{172}
                \item 若切线斜率不存在，过$P$斜率不存在的直线为$x = 1$，经检验该直线为圆的切线
