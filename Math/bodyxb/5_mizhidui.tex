\chapter* {幂函数、指数函数和对数函数}
\subsection*{幂与指数}
\begin{enumerate}
    \item 正整数指数幂的定义和运算性质 \\
        如果$a$是一个实数，$n$是一个正整数，那么称$a^n = \underbrace{a \cdot a \cdot a \cdot ... \cdot a}_{n \textsf{\mbox{个}} a}$ 为$a$的$n$次幂。
        正整数指数幂具有如下\textbf{运算性质}：
        \begin{enumerate}
            \item $a^{m+n} = a^m \cdot a^n$
            \item $a^{m \cdot n} = (a^m)^n$
            \item $a^n \cdot b^n = (ab)^n$
        \end{enumerate}
    \item 整数指数幂 \\
        在保障三条运算性质成立的前提下，\textbf{当$a \neq 0$时}，我们可以做如下定义：
        \begin{enumerate}
            \item $a^0 = 1$
            \item $a^{-n} = \dfrac{1}{a^n}$
        \end{enumerate}
        首先根据$a^{n+0} = a^n \cdot a^0$可知应有$a^0 = 1$，其次根据$a^0 = a^n \cdot a^{-n}$可知应有$a^{-n} = \dfrac{1}{a^n}$。可以验证在这样的定义下，三条运算性质依然是成立的。\\
        注：我们认为$0^0$没有定义。
    \item 有理数指数幂
        \begin{enumerate}
            \item $n$次方根：一般地，如果$n$为\textbf{大于１的整数}，且有$x^n = a$，那么$x$叫做$a$的\textbf{$n$次方根}。我们用$\sqrt[n]{a}$表示$a$的$n$次根式，其中$a$称为\textbf{被开方数}，$n$称为\textbf{根指数}。
            \begin{enumerate}
                \item 奇次方根：当$n$为奇数的时候，正数的$n$次方根是一个正数，负数的$n$次方根是一个负数。一个数的奇次方根是\textbf{唯一确定的}。例如：$\sqrt[5]{32} = 2, \sqrt[3]{-27} = -3$。
                \item 偶次方根：当$n$为偶数的时候，\textbf{正数的$n$次方根有两个值，他们互为相反数}，$\sqrt[n]{a}$或者$+\sqrt[n]{a}$表示正的偶次方根，而$-\sqrt[n]{a}$表示负的偶次方根。两者可以合写为$\pm \sqrt[n]{a}$ 。例如：$\sqrt[4]{16} = 2, \pm \sqrt[2]{25} = \pm 5$。\textbf{负数没有偶次方根}。
                \item 特别指出：0的任意次方根都是0，即 $0 = \sqrt[n]{0}$
            \end{enumerate}
            \item 正数的有理数指数幂 \\
                首先考虑定义$a^{\frac{1}{n}}, n > 1, n \in \mathbf{Z}$。 根据$a = a^{1} = (a^{\frac{1}{n}})^n$，可以定义$a^{\frac{1}{n}} = \sqrt[n]{a}$ 。可以验证当$n=1$时定义依然成立。\\
                因此在保障三条运算性质成立的前提下，我们可以做如下定义：$a^{\frac{m}{n}} = \sqrt[n]{a^m}$, 其中$a > 0$, $m$为整数，$n$为正整数。可以验证，当$m,n$不互素的时候，$a^{\frac{m}{n}} = a^{\frac{m'}{n'}}$，其中$\dfrac{m}{n} = \dfrac{m'}{n'}$且$m',n'$互素。\\
                由此可以定义\textbf{正数的有理数指数幂}：$s = \dfrac{m}{n}$为有理数，定义$a^s = a^{\frac{m}{n}}$（$m,n$互素）。\\
                可以推导出开方和乘方的先后顺序不影响幂的值：即$a^{\frac{m}{n}} = \sqrt[n]{a^m} = (\sqrt[n]{a})^m$ \\
                推导如下：$\sqrt[n]{a^m} = (a^m)^{\frac{1}{n}} = (a^m)^{\frac{m}{m \cdot n}} = ((a^m)^{\frac{1}{m \cdot n}})^m = (a^\frac{1}{n})^m = (\sqrt[n]{a})^m$。\\
                正数的有理数指数幂具有性质(\textbf{幂的基本不等式})：$a^s > 1$， 其中$a > 1, s > 0, s \in \mathbf{Q}$ \\
                对于负数，由于负数没有偶次方根，因此对于一个负数$a$，并非所有的有理数$s$都可以使得$a^s$有意义。因此我们一般不考虑负数的有理指数幂。
        \end{enumerate}
    \item 正数的无理数指数幂 \\
        在上述对正数的有理数指数幂定义的基础上，可以证明：对任意一个正数$a$与任意一个无理数$s$，可确定一个唯一的实数，记作$a^s$，使得上述有理数指数幂的三条运算性质与幂的基本不等式对所有的实数$a^s$都成立。我们把这个实数$a^s$定义为$a$的$s$次幂。 \\
        理解：任意一个正数$a$与任意一个无理数$s$都对应唯一的一个实数$a^s$，实数$a^s$定义为$a$的$s$次幂，且满足运算性质。 \\
        证明思路：无理数可以由有理数夹逼趋近，对应的正数的无理数指数幂也可以由正数的有理数指数幂夹逼得到。
    \item 小结 \\
        正数的实数指数幂总是有意义的，具有三条运算性质和一个不等关系。
        \begin{enumerate}
            \item $a^{m+n} = a^m \cdot a^n$, 其中$a > 0, m \in \mathbf{R}, n \in \mathbf{R}$
            \item $a^{m \cdot n} = (a^m)^n$, 其中$a > 0, m \in \mathbf{R}, n \in \mathbf{R}$
            \item $a^n \cdot b^n = (ab)^n$, 其中$a > 0, b > 0, n \in \mathbf{R}$
            \item $a^s > 1$， 其中$a > 1, s > 0, s \in \mathbf{R}$
        \end{enumerate}
\end{enumerate}

\subsection*{对数}
\subsubsection*{对数的定义}
在$a > 0, a \neq 1, N > 0$的条件下，唯一满足$a^x = N$的数$x$称为以$a$为底，$N$的对数，记作$x = log_aN$，其中$N$称为真数。
\begin{equation}
    a^x = N \quad \Longleftrightarrow \quad x = log_aN \nonumber
\end{equation}
对数运算是指数运算的\textbf{逆运算}：$a^{log_aN} = N$\\
\textbf{指数式和对数式是同一个数量关系的不同表达形式}，两者可以互化。 \\
自然对数和常用对数 \\
以10为底的对数称为常用对数，$log_{10}N$通常记为$lgN$ \\ 
以e为底的对数称为自然对数，$log_{e}N$通常记为$lnN$。(注：$e$是一个无理数，其值约为2.71828...) \\
例题：
\begin{enumerate}
    \item 求下列各式的值
    \begin{enumerate}
        \item $log_28$
        \item $log_2\sqrt{2}$
        \item $log_{10}0.00001$
    \end{enumerate}
    \begin{jieda}
        \begin{enumerate}
            \item 因为$2^3 = 8$, 所以$log_28=3$
            \item 因为$2^{\frac{1}{2}} = \sqrt{2}$, 所以$log_2\sqrt{2} = \dfrac{1}{2}$
            \item 因为$10^{-5} = 0.00001$, 所以$log_{10}0.00001 = -5$
        \end{enumerate}
    \end{jieda}
    \item 求下列各式中$x$的值
    \begin{enumerate}
        \item $log_2x = -1$
        \item $log_{\frac{1}{2}}x = 3$
        \item $lnx = -1$
        \item $log_{64}x = -\dfrac{2}{3}$
        \item $log_x8 = 6$
        \item $lg100 = x$
        \item $-ln(e^2) = x$
    \end{enumerate}
    \begin{jieda}
        \begin{enumerate}
            \item $log_2x = -1, x = \dfrac{1}{2}$
            \item $log_{\frac{1}{2}}x = 3, x = \dfrac{1}{8}$
            \item $lnx = -1, x = \dfrac{1}{e}$
            \item $log_{64}x = -\dfrac{2}{3}, x = \dfrac{1}{16}$
            \item $log_x8 = 6, x = \sqrt{2}$
            \item $lg100 = x, x = 2$
            \item $-ln(e^2) = x, x = -2$
        \end{enumerate}
    \end{jieda}
    \item 将下列指数式化为对数式，将对数式化为指数式
    \begin{enumerate}
        \item $5^4 = 625$
        \item $2^{-6} = \dfrac{1}{64}$
        \item $\left(\dfrac{1}{3}\right)^m = 5.73$
        \item $log_{\frac{1}{2}}16 = -4$
        \item $lg0.01=-2$
        \item $ln10 = 2.303$
    \end{enumerate}
    \begin{jieda}
        \begin{enumerate}
            \item $5^4 = 625 \Longleftrightarrow log_5{625} = 4$
            \item $2^{-6} = \dfrac{1}{64} \Longleftrightarrow log_2{\dfrac{1}{64}} = -6$
            \item $(\dfrac{1}{3})^m = 5.73 \Longleftrightarrow log_{\frac{1}{3}}5.73 = m$
            \item $log_{\frac{1}{2}}16 = -4 \Longleftrightarrow (\dfrac{1}{2})^{-4} = 16$
            \item $lg0.01=-2 \Longleftrightarrow 10^{-2} = 0.01$
            \item $ln10 = 2.303 \Longleftrightarrow e^{2.303} = 10$
        \end{enumerate}
    \end{jieda}
\end{enumerate}
\subsection*{对数的运算律}
\begin{enumerate}
    \item 对数的运算律
        \begin{enumerate}
            \item $a^{m+n} = a^m \cdot a^n \Longleftrightarrow log_aM + log_aN = log_a(M \cdot N) $, \\
            其中$a > 0, a \neq 1, M>0, N > 0, m = log_aM \in \mathbf{R}, n = log_aN \in \mathbf{R}$
            \item $a^{m-n} = \dfrac{a^m}{a^n} \Longleftrightarrow log_aM - log_aN = log_a(\dfrac{M}{N}) $ , \\
            其中$a > 0, a \neq 1, M>0, N > 0, m = log_aM \in \mathbf{R}, n = log_aN \in \mathbf{R}$
            \item $a^{m \cdot n} = (a^m)^n  \Longleftrightarrow  nlog_aM = log_a(M^n)$, \\
            其中$a > 0, a \neq 1, M>0, m = log_aM \in \mathbf{R}, n \in \mathbf{R}$
        \end{enumerate}
    \item 换底公式 \\
        $log_bN = \dfrac{log_aN}{log_ab}$ \\
        设$x = log_bN$, 则$b^x = N$。 两侧以$a$为底取对数，得$log_a(b^x) = log_aN$, 即$x \cdot log_ab = log_aN$。 代换$x$，故$log_bN = \dfrac{log_aN}{log_ab}$
        换底公式允许我们制作对数表，并利用对数表中固定底的对数值来计算以其他数为底的对数值。例如：$log_23 = \dfrac{ln3}{ln2}$ \\
        由换底公式可以推出$log_ab = \dfrac{log_bb}{log_ba} = \dfrac{1}{log_ba}$
\end{enumerate}
例题：
\begin{enumerate}
    \item 计算下列各式的值
    \begin{enumerate}
        \item $log_3{(9^4 \times 3^2)}$
        \item $log_3{(\sqrt[5]{9})}$
        \item $2 log_5{10} + log_5{3} - log_5{12}$
        \item $lg\sqrt[5]{100}$
        \item $log_2{(4^7 \times 2^5)}$
    \end{enumerate}
    \begin{jieda}
        \begin{enumerate}
            \item $log_3{(9^4 \times 3^2)} = log_3(9^4) + log_3(3^2) = log_3(3^8) + log_3(3^2) = 8+2 = 10$
            \item $log_3{(\sqrt[5]{9})} = log_3(9^{\frac{1}{5}}) = log_3(3^{\frac{2}{5}}) = \dfrac{2}{5}$
            \item $2log_5{10} + log_5{3} - log_5{12} = log_5\left(\dfrac{10^2 \times 3}{12}\right) = log_5(25) = 2$
            \item $lg\sqrt[5]{100} = lg(100^{\frac{1}{5}}) = lg(10^{\frac{2}{5}}) = \dfrac{2}{5}$
            \item $log_2{(4^7 \times 2^5)} = log_2(4^7) + log_2(2^5) = log_2(2^{14}) + log_2(2^5) = 14+5 = 19$
        \end{enumerate}
    \end{jieda}
    \item 用$lnx, lny, lnz$表示$ln{\dfrac{x^2\sqrt{y}}{\sqrt[3]{z}}}$
    \begin{jieda}
        $ln{\dfrac{x^2\sqrt{y}}{\sqrt[3]{z}}} = ln(x^2\sqrt{y}) - ln(\sqrt[3]{z}) = ln{x^2} + ln{\sqrt{y}} - ln(\sqrt[3]{z}) = 2lnx + \dfrac{lny}{2} - \dfrac{lnz}{3}$
    \end{jieda}
    \item 求下列各式的值
    \begin{enumerate}
        \item $log_2{25} \times log_3{4} \times \log_5{9}$
        \item $\dfrac{1}{log_2{3}} + \dfrac{lg13.5}{lg3}$
    \end{enumerate}
    \begin{jieda}
        \begin{enumerate}
            \item $log_2{25} \times log_3{4} \times \log_5{9} = \dfrac{log_5{25}}{log_5{2}} \times \dfrac{log_5{4}}{log_5{3}} \times \log_5{9} = \dfrac{2 \times 2 \times log_5{2} \times 2 \times log_5{3}}{log_5{2} \times log_53} = 8$
            \item $\dfrac{1}{log_2{3}} + \dfrac{lg13.5}{lg3} = \dfrac{log_3{2}}{log_3{3}} + log_3{13.5} = log_3{27} = 3$
        \end{enumerate}
    \end{jieda}
    \item 已知$lg2 = a, 10^b = 3$，试用$a, b$表示$log_{24}15$
    \begin{jieda}
        $a = \dfrac{1}{1+log_25}$，$b = \dfrac{log_23}{log_2{10}} = \dfrac{log_23}{log_25+1}$，$log_{24}15 = \dfrac{log_215}{log_224} = \dfrac{log_23 + log_25}{3+log_23}$ \\

    \end{jieda}
    \item 已知$log_35 = a, log_57 = b$，试用$a, b$表示$log_{75}63$
    \begin{jieda}
        $a = \dfrac{1}{log_53}, b = log_57$，$log_{75}63 = \dfrac{log_57+2log_53}{2+log_53} = \dfrac{b+\dfrac{2}{a}}{2 + \dfrac{1}{a}} = \dfrac{2+ab}{1+2a}$
    \end{jieda}
    \item 已知$2^{6a} = 3^{3b} = 6^{2c}$，试建立$a, b, c$之间的关系式
    \begin{jieda}

    \end{jieda}
\end{enumerate}
\subsection*{幂函数}
\begin{enumerate}
    \item 定义 \\
        当指数$a$固定，等式$y = x^a$ 确定了变量$y$随变量$x$变化的规律，称为指数为$a$的\textbf{幂函数} (power function)
        \begin{example}
            \begin{enumerate}
                \item 如果张红以1元/kg的价格购买了某种蔬菜$w$ kg, 那么她需要支付$p = w$元，这里$p$是$w$的函数
                \item 如果正方形的边长为$a$,那么正方形的面积$S = a^2$，这里$S$是$a$的函数
                \item 如果立方体的棱长为$b$,那么立方体的体积$V = b^3$，这里$V$是$b$的函数
                \item 如果一个正方形场地的面积为$S$,那么这个正方形的边长$c = \sqrt{S}$, 即$c = S^{\frac{1}{2}}$，这里$c$是$S$的函数
                \item 如果某人$t$秒内骑车行进了$1$ km,那么他骑车的平均速度$v = \dfrac{1}{t}$ km/s, 即$v = t^{-1}$，这里$v$是$t$的函数
            \end{enumerate}
        \end{example}
    \item 幂函数的图象参考图 \ref{fig:power_function}

    \item 性质

\subsection*{指数函数}
\begin{enumerate}
    \item 定义 \\
    当底数$a$固定，且$a>0, a \neq 1$时，等式$y=a^x$ 确定了变量$y$随变量$x$变化的规律，称为底为$a$的\textbf{指数函数}(exponential function). \\
    \ps 只有$a>0$才能保证对于所有的$x$，$a^x$都有意义。而$a=1$的特例不必研究，因此有$a>0, a \neq 1$

    \begin{example}
        \begin{enumerate}
            \item 一张纸对折一次，由1层变为2层，再对折一次，由2层变为4层，...，对折$x$次后，层数$y$与对折次数$x$的函数关系为$y=2^x$
            \item 古生物体内的$^{14}C$随着时间而衰减，每过5730年，$^{14}C$的含量就衰减为原来的一半。如果设古生物死亡的时候体内的$^{14}C$含量为1，古生物体内剩余的$^{14}C$含量$y$与古生物死亡的时间$x$的函数关系为$y = \left(\dfrac{1}{\sqrt[5730]{2}}\right)^x$，即$y = 2^{-\frac{x}{5730}}$.
        \end{enumerate}
    \end{example}

\subsection*{反函数}
    \begin{enumerate}
        \item 定义 \\
        对于函数$y=f(x), x \in D$，记其值域为 $f(D)=\{y|y=f(x), x \in D\}$．如果对$f(D)$中的任意给定的一个值$y$，在$D$中满足$f(x) = y$的$x$值只有一个，那么由此得到的$x$关于$y$的函数叫做$y=f(x), x \in D$的\textbf{反函数}(inverse function)，记作$x = f^{-1}(y), y \in f(D)$。由于自变量习惯上常用$x$表示，而函数值常用$y$表示，因此通常把该函数改写为$y = f^{-1}(x), x \in f(D)$
        \item 理解：反函数的本质是用“原函数的因变量”作为自变量去表示“原函数的自变量”。若这两者存在一一对应关系，则满足函数的定义。
        \begin{example}
            $y = 2x$和$y = \dfrac{x}{2}$
        \end{example}
        \item 求解反函数的步骤：
        \begin{enumerate}
            \item 明确原函数的定义域和值域
            \item 用原函数的$y$表示$x$
            \item 交换变量名
            \item 确认符合函数定义，明确反函数的定义域和值域
        \end{enumerate}
        \item 性质
        \begin{enumerate}
            \item 若点$(a, b)$在原函数的图象上，则点$(b, a)$在反函数的图象上.
            \item 原函数的定义域和值域是反函数的值域和定义域
            \item 函数$f_1$是函数$f_2$的反函数，则函数$f_2$是函数$f_1$的反函数，两者互为反函数
            \item 原函数与反函数的图象关于$y=x$对称
            \item 具有反函数的函数在其定义域内任意一个连续的区间内一定是严格的单调函数
            \item 原函数和反函数在对应的区间内的单调性相同
        \end{enumerate}
    \end{enumerate}

\subsection*{对数函数}

        当底数$a$固定，且$a > 0, a \neq 1$时，$x$以$a$为底的对数$y = log_ax$确定了变量$y$随变量$x$变化的规律，称为底为$a$的\textbf{对数函数} (logarithmic function).

        设$h(t) = \dfrac{1-t}{t(t+1)}$，易知$h(t)$分子分母均为正，且分子单调递减，分母单调递增，故在$\left[\dfrac{1}{2}, 1\right]$内单调递减，故$a \geq h\left(\dfrac{1}{2}\right) = \dfrac{2}{3}$
    \end{enumerate}
    \end{jieda}
    
    \item 已知奇函数 $f\left( x\right)$ 与偶函数 $g\left( x\right)$ 的定义域都是 $\mathbf{R}$ ,且满足 $f\left( x\right)  + g\left( x\right)  = {2}^{x}$ .若 $f\left( {{\log }_{2}m}\right)  + f\left( {2{\log }_{2}m - 1}\right)  \leq  1 - 3{\log }_{2}m$ ,求实数 $m$ 的取值范围.
    \begin{jieda}
        $f(x) = \dfrac{2^x-2^{-x}}{2}$，为奇函数，增函数.

        设$t = {\log }_{2}m, t\in \mathbf{R}$，$f\left( {{\log }_{2}m}\right)  + f\left( {2{\log }_{2}m - 1}\right)  \leq  1 - 3{\log }_{2}m$即$f(t) + f(2t-1) \leq -(3t-1)$，即$f(t)+t + f(2t-1)+(2t-1) \leq 0$

        设$g(x) = f(x)+x$，则$g(x)$为奇函数，增函数.$f(t)+t + f(2t-1)+(2t-1) \leq 0$即$g(t)+g(2t-1) \leq 0$，故$3t-1 \leq 0$，即$t \leq \dfrac{1}{3}$，故$m \in (0, \sqrt[3]{2}]$
    \end{jieda}
\end{enumerate}
\subsection*{指对方程}
例题：
\begin{enumerate}
    \item 方程$2^{2+x}-2^{2-x}=15$的解集是\blkda{$\{2\}$}. 
    \item 方程$9^x+6^x=2^{2x+1}$的解集是\blkda{$\{0\}$}. 
    \item 方程$2^{\left| {x+1} \right|}=3$的解集是\blkda[3]{$\left\{\log_2\frac{3}{2},\log_2\frac{1}{6}\right\}$}.
    \begin{jieda}
    $y = 2^{\left| {x+1} \right|}$ 关于$x = -1$对称，两根之和为$-2$
    \end{jieda}
    \item 方程$log_2(9-2^x) = 3-x$的解集是\blkda{$\{0, 3\}$}.
    \begin{jieda}
        转化为$9-2^x = 2^{3-x}$，即$9 = 2^x + 8 \cdot 2^{-x}$，故$2^x = 1$或$8$，$x \in \{0, 3\}$
    \end{jieda}
    \item 方程$4^x+4^{-x}-5(2^x+2^{-x})+6=0$的解集是\blkda[6]{$\{\log_2(2+\sqrt{3}), \log_2(2-\sqrt{3})\}$}.
    \begin{jieda}
    令$t=2^x+2^{-x}$, 则$t^2-5t+4=0$, 故$t=1$或4. \\
    而$2^x+2^{-x}=1$无解, 解$2^x+2^{-x}=4$得$x=\log_2(2\pm\sqrt{3})$. 
    \end{jieda}
    \item 方程$\log_4(2-x)=\log_2(x-1)-1$的解是\blkda{$2\sqrt{2}-1$}. 
    \begin{jieda}
    注意到$x\in(1,2)$, 故原方程
    $\iff\dfrac{1}{2}\log_2(2-x)=\log_2(x-1)-1$, \\
    即$\log_2(2-x)+2=2\log_2(x-1)$. \\
    于是$4(2-x)=(x-1)^2\land x\in(1,2)$, 解得$x=2\sqrt{2}-1$. 
    \end{jieda}
    
    \item 方程$3^{\sqrt{5x-2y}}+5^{\sqrt{y-5}}=2$的解为\blkda[2]{$x=2,y=5$}. 
    \begin{jieda}
    $3^{\sqrt{5x-2y}}\geq 1, 5^{\sqrt{y-5}}\geq 1$, 故原方程等价于$3^{\sqrt{5x-2y}}=5^{\sqrt{y-5}}=1$. \\
    解得$x=2,y=5$. 
    \end{jieda}
    \item 解方程: 
    \begin{enumerate}[(1)] 
    \item $\log_4(13-3x)\cdot\log_{(x-1)}2=1$; 
    \item $\log_2(4^x+4)=x+\log_2(2^{x+1}-3)$. 
    \item $x^{\lg{x}}=\frac{x^3}{100}$; 
    \item $10^{\lg^2x}+x^{\lg{x}}=20$. 
    \end{enumerate}
    \begin{jieda}
    \begin{enumerate}[(1)]
    \item $\iff \log_4(13-3x) = \log_2(x-1)\iff\log_2(13-3x) = \log_2((x-1)^2)$. \\
    故$(x-1)^2=13-3x\land x-1>0\land x-1\neq 1$, 解得$x=3$. 
    \item $\iff\log_2(4^x+4)=\log_22^x+\log_2(2^{x+1}-3)\iff 4^x+4=2^x(2^{x+1}-3)$. \\
    于是$(2^x+1)(2^x-4)=0$, 故$2^x=4$, 即$x=2$. 
    \item 两边取常用对数, 得$\lg^2{x}=3\lg{x}-2$, 解得$\lg{x}=1$或$2$, 即$x=10$或100. 
    
    \item 首先$x>0$, 于是$x=10^{\lg{x}}$. \\
    故方程可化为$10^{\lg^2x}+(10^{\lg{x}})^{\lg{x}}=20$, 即$2\cdot10^{\lg^2x}=20$. \\
    故$\lg^2x=1$, 即$x=\dfrac{1}{10}$或10. 
    \end{enumerate}
    \end{jieda}
     \item 若方程$\lg(2x)\lg(3x)+a^2=0$有两个不相等的实数根, 求实数$a$的取值范围, 并求方程的两根的积. 

    \begin{jieda}
    $\iff -a^2=(\lg2+\lg{x})(\lg3+\lg{x})\iff -a^2=\lg^2x+\lg6\cdot\lg{x}+\lg2\lg3$. \\
    于是$-a^2>-\left(\dfrac{\lg\dfrac{3}{2}}{2}\right)^2$, 故$-\dfrac{\lg\dfrac{3}{2}}{2}<a<\dfrac{\lg\dfrac{3}{2}}{2}$. \\
    此时$\lg{x_1}+\lg{x_2}=-\lg6$, 即$x_1x_2=\dfrac{1}{6}$. 
    \end{jieda}
    \item 函数$f(x)=\log_2(1+3^x+9^x\cdot a)$当$x\leq1$时总有意义, 则实数$a$的取值范围是\blkda[2]{$\left(-\dfrac{4}{9},+\infty\right)$}
    \begin{jieda}
    问题等价于: 对于任意$x\leq 1$, $1+3^x+9^x\cdot a>0$恒成立. \\
    于是$a>\dfrac{-3^x-1}{9^x}=-t^2-t$, 其中$t=\left(\dfrac{1}{3}\right)^x\in\left[\dfrac{1}{3},+\infty\right)$. \\
    而$-t^2-t=-\left(t+\dfrac{1}{2}\right)^2+\dfrac{1}{4}$, 在$\left[\dfrac{1}{3},+\infty\right)$上的最大值为$-\dfrac{4}{9}$, 故$a>-\dfrac{4}{9}$. 
    \end{jieda}
    
\end{enumerate}
