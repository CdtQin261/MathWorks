    \item 下列向量中, 真命题是\blkda{\ding{172}}. (填序号)
    \begin{dingautolist}{172}
        \item 若 $A$ 、 $B$ 、 $C$ 、 $D$ 在一条直线上，则 $\vv{AB}$ 与 $\vv{CD}$ 是共线向量
        \item 若 $A$ 、 $B$ 、 $C$ 、 $D$ 不在一条直线上，则 $\vv{AB}$ 与 $\vv{CD}$ 不是共线向量
        \item 向量 $\vv{AB}$ 与 $\vv{CD}$ 是共线向量，则 $A\text{ 、 }B\text{ 、 }C\text{ 、 }D$ 四点必在一条直线上

    \end{dingautolist}
    \begin{jieda}
        对于\ding{172} ，若 $A\text{ 、 }B\text{ 、 }C\text{ 、 }D$ 在一条直线上，则 $\vv{AB}$ 与 $\vv{CD}$ 是共线向量，故\ding{172} 正确；

        对于\ding{173} ，若 $A\text{ 、 }B\text{ 、 }C\text{ 、 }D$ 构成平行四边形时， $A\text{ 、 }B\text{ 、 }C\text{ 、 }D$ 不在一条直线上，但是 $\vv{AB}$ 与 $\vv{CD}$ 是共线向量，故\ding{173} 不正确;

        对于\ding{174} ，若 $A$ 、 $B$ 、 $C$ 、 $D$ 构成平行四边形时， $A$ 、 $B$ 、 $C$ 、 $D$ 不在一条直线上，但是 $\vv{AB}$ 与 $\vv{CD}$ 是共线向量，故\ding{174} 不正确;

        对于\ding{175} ,若 $A\text{ 、 }B\text{ 、 }C\text{ 、 }D$ 构成平行四边形时, $A\text{ 、 }B\text{ 、 }C$ 不在一条直线上,但是 $\vv{AB}$ 与 $\vv{CD}$ 是共线向量,故\ding{175} 不正确;
    \end{jieda}

    \item 下列关于空间向量的命题中，正确的序号是\blkda{\ding{173} \ding{175}}.
    \begin{dingautolist}{172}
        \item 若两个空间向量相等，则它们的起点相同，终点也相同
        \item $\left| \vv{a}\right|  = \left| \vv{b}\right|$ 是向量 $\vv{a} = \vv{b}$ 的必要非充分条件
        \item 向量 $\vv{a}$ 、 $\vv{b}$ 相等的充要条件是 $\left\{  \begin{array}{l} \left| \vv{a}\right|  = \left| \vv{b}\right| \\  \vv{a} \parallel  \vv{b} \end{array}\right.$
        \item 若 $A\text{ 、 }B\text{ 、 }C\text{ 、 }D$ 不共线，则 $\vv{AB} = \vv{DC}$ 是四边形 ${ABCD}$ 为平行四边形的充要条件.
    \end{dingautolist}

    \begin{jieda}
        向量相等只需满足方向相同且模相等即可, 故\ding{172} 错误;

        根据相等向量的概念可知,若 $\vv{a} = \vv{b}$ ,则 $\left| \vv{a}\right|  = \left| \vv{b}\right|$ ,但 $\left| \vv{a}\right|  = \left| \vv{b}\right|$ ,有可能 $\vv{a}\text{ 、 }\vv{b}$ 的方向不同,故 $\left| \vv{a}\right|  = \left| \vv{b}\right|$ 是向量 $\vv{a} = \vv{b}$ 的必要非充分条件, \ding{173} 正确；

        当 $\vv{a}$ 、 $\vv{b}$ 为相反向量时,显然满足 $\left\{  \begin{array}{l} \left| \vv{a}\right|  = \left| \vv{b}\right| \\  \vv{a} \parallel  \vv{b} \end{array}\right.$ ,故\ding{174} 错误；

        因为 $A\text{ 、 }B\text{ 、 }C\text{ 、 }D$ 不共线,所以由 $\vv{AB} = \vv{DC}$ ,可知 ${AB} = {DC}$ 且 ${AB}//{DC}$ ,所以四边形 ${ABCD}$ 为平行四边形,反之,若四边形 ${ABCD}$ 为平行四边形,则由平行四边形的性质可得 $\vv{AB} = \vv{DC}$ ,故\ding{175} 正确.
    \end{jieda}

\subsection*{根据空间向量共线求参数}
\begin{enumerate}
    \item 在空间四边形 ${OABC}$ 中, $M$ 为 ${OA}$ 中点, $N$ 为 ${BC}$ 的中点,若 $\vv{OG} = \dfrac{1}{3}\vv{OA} + \dfrac{x}{3}\vv{OB} + \dfrac{x}{3}\vv{OC}$ ,则使 $G\text{ 、 }M\text{ 、 }N$ 三点共线的 $x$ 的值是\blkda{$\dfrac{1}{2}$}.

    \begin{center}
    \includegraphics[width=0.15\textwidth]{bodyxb/src/2_1165_585_350_264_0.jpg}
    \end{center}

    \begin{jieda}
        $\vv{OA} = 2\vv{OM},\vv{ON} = \dfrac{1}{2}\vv{OB} + \dfrac{1}{2}\vv{OC}$ ,则 $2\vv{ON} = \vv{OB} + \vv{OC}$ ,

        $\therefore \vv{OG} = \dfrac{1}{3}\vv{OA} + \dfrac{x}{3}\vv{OB} + \dfrac{x}{3}\vv{OC} = \dfrac{2}{3}\vv{OM} + \dfrac{x}{3}\left( {\vv{OB} + \vv{OC}}\right)  = \dfrac{2}{3}\vv{OM} + \dfrac{2x}{3}\vv{ON}$ ,

        $\because G,N,M$ 三点共线, $\therefore \dfrac{2}{3} + \dfrac{2x}{3} = 1,\therefore x = \dfrac{1}{2}$.
    \end{jieda}
    
    \item 设 $\vv{{e}_{1}},\vv{{e}_{2}}$ 是两个不共线的空间向量,若 $\vv{AB} = 2\vv{{e}_{1}} - \vv{{e}_{2}},\vv{BC} = 3\vv{{e}_{1}} + 3\vv{{e}_{2}},\vv{CD} = \vv{{e}_{1}} + k\vv{{e}_{2}}$ ,且 $A,C,D$ 三点共线, 则实数 $k$ 的值为\blkda{$\dfrac{2}{5}$}.
    \begin{jieda}
        $\because \vv{AB} = 2\vv{{e}_{1}} - \vv{{e}_{2}},\vv{BC} = 3\vv{{e}_{1}} + 3\vv{{e}_{2}},\vv{CD} = \vv{{e}_{1}} + k\vv{{e}_{2}},\therefore \vv{AC} = \vv{AB} + \vv{BC} = 5\vv{{e}_{1}} + 2\vv{{e}_{2}}$，又 $\because A,C,D$ 三点共线, $\therefore \vv{AC}//\vv{CD},\because \vv{{e}_{1}},\vv{{e}_{2}}$ 不共线, $\therefore \vv{AC} = 5\vv{CD},\therefore 2 - {5k} = 0,\therefore k = \dfrac{2}{5}$.
    \end{jieda}
    
    \item 设向量 ${\vv{e}}_{1},{\vv{e}}_{2},{\vv{\mathbf{e}}}_{3}$ 不共面,已知 $\vv{AB} =  - 3{\vv{e}}_{1} - {\vv{e}}_{2} + 2{\vv{e}}_{3},\vv{BC} = {\vv{e}}_{1} + \lambda {\vv{e}}_{2} - 6{\vv{e}}_{3},\vv{CD} = 4{\vv{e}}_{1} + 2{\vv{e}}_{2} + 8{\vv{e}}_{3}$ ,若 $A,C,D$ 三点共线,则 $\lambda  =$ \blkda{0}.
    \begin{jieda}
        因为 $\vv{AB} =  - 3{\vv{e}}_{1} - {\vv{e}}_{2} + 2{\vv{e}}_{3},\vv{BC} = {\vv{e}}_{1} + \lambda {\vv{e}}_{2} - 6{\vv{e}}_{3},\vv{CD} = 4{\vv{e}}_{1} + 2{\vv{e}}_{2} + 8{\vv{e}}_{3}$ ,

        所以 $\vv{AC} = \vv{AB} + \vv{BC} =  - 2{\vv{e}}_{1} + \left( {\lambda  - 1}\right) {\vv{e}}_{2} - 4{\vv{e}}_{3}$ .

        因为 $A,C,D$ 三点共线,所以存在唯一的实数 $y$ ,使得 $\vv{CD} = y\vv{AC}$ ,

        即 $4{\vv{e}}_{1} + 2{\vv{e}}_{2} + 8{\vv{e}}_{3} = y\left\lbrack  {-2{\vv{e}}_{1} + \left( {\lambda  - 1}\right) {\vv{e}}_{2} - 4{\vv{e}}_{3}}\right\rbrack$，即 $\left\{  \begin{array}{l} 4 =  - {2y} \\  2 = y\left( {\lambda  - 1}\right) \\  8 =  - {4y} \end{array}\right.$ ,解得 $\left\{  \begin{array}{l} \lambda  = 0 \\  y =  - 2 \end{array}\right.$.
    \end{jieda}
    
    \item 已知向量 $\vv{a},\vv{b}$ ， $\vv{c}$ 是两两不平行的向量，且 $\left( {\vv{a} + 3\vv{b}}\right) //\left( {\vv{b} + \vv{c}}\right)$ ， $\left( {2\vv{a} + k\vv{c}}\right) //\left( {\vv{b} + \vv{c}}\right)$ ，则 $k$ 的值为\blkda{$-6$}.
\end{enumerate}
\subsection*{四点共面}
\begin{enumerate}
    \item $O$ 为空间中任意一点, $A,B,C$ 三点不共线,且 $\vv{OP} = \dfrac{3}{4}\vv{OA} + \dfrac{1}{8}\vv{OB} + t\vv{OC}$ ,若 $P,A,B,C$ 四点共面,则实数 $t =$ \blkda{$\dfrac{1}{8}$}.
    \begin{jieda}
        $\dfrac{3}{4} + \dfrac{1}{8} + t = 1$ ,解得 $t = \dfrac{1}{8}$.
    \end{jieda}
    
    \item 已知 $A,B,C,D$ 四点共面，且任意三点不共线， $O$ 为平面 ${ABCD}$ 外任意一点，若 $\vv{OA} = \dfrac{1}{5}\vv{OB} + \dfrac{2}{5}\vv{OC} + \lambda \vv{OD}$ ，则 $\lambda  =$ \blkda{$\dfrac{2}{5}$}.
    \begin{jieda}
        $\dfrac{1}{5} + \dfrac{2}{5} + \lambda  = 1 \Longrightarrow  \lambda  = \dfrac{2}{5}$
    \end{jieda}

    \item 已知 $A,B,C,D$ 四点共面，且任意三点不共线， $O$ 为平面 ${ABCD}$ 外任意一点，若 $\vv{BA} = \dfrac{1}{5}\vv{OB} + \dfrac{2}{5}\vv{OC} + \lambda \vv{OD}$ ，则 $\lambda  =$ \blkda{$-\dfrac{3}{5}$}.
    \begin{jieda}
        $\vv{BA} = \dfrac{1}{5}\vv{OB} + \dfrac{2}{5}\vv{OC} + \lambda \vv{OD} \Longleftrightarrow \vv{OA} = \dfrac{6}{5}\vv{OB} + \dfrac{2}{5}\vv{OC} + \lambda \vv{OD}$，故$\dfrac{6}{5} + \dfrac{2}{5} + \lambda  = 1 \Longrightarrow  \lambda  = -\dfrac{3}{5}$
    \end{jieda}
    
    \item $O$ 不与 $A,B,C,D$ 共面，并且 ${ABCD}$ 四点在一个平面上， $2\vv{OD} = x\vv{OA} + y\vv{OB} + \vv{OC}\left( {x,y > 0}\right)$ ，则 $\dfrac{1}{x} + \dfrac{9}{y}$ 的最小值为\blkda{16}.
    \begin{jieda}
        由题设 $2\vv{OD} - x\vv{OA} - y\vv{OB} = \vv{OC},O$ 不与 $A,B,C,D$ 共面,且 ${ABCD}$ 四点共面，所以 $2 - x - y = 1$ ,可得 $x + y = 1$ ,且 $x,y > 0$ ,

        所以 $\dfrac{1}{x} + \dfrac{9}{y} = \left( {\dfrac{1}{x} + \dfrac{9}{y}}\right) \left( {x + y}\right)  = {10} + \dfrac{y}{x} + \dfrac{9x}{y} \geq  {10} + 2\sqrt{\dfrac{y}{x} \cdot  \dfrac{9x}{y}} = {16}$，当且仅当 $x = \dfrac{1}{4},y = \dfrac{3}{4}$ 时取等号.
    \end{jieda}

\end{enumerate}
