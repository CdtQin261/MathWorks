\subsection*{柱体的体积、表面积}
\begin{enumerate}
    \item 在正四棱柱 ${ABCD} - {A}_{1}{B}_{1}{C}_{1}{D}_{1}$ 中， ${BD} = {4\sqrt{2}},{D{B}_{1}} = 9$ ，则该正四棱柱的体积为\blkda{112}.
    \begin{jieda}
        因为 ${BD} = 4\sqrt{2}$ 且四边形 ${ABCD}$ 为正方形,故 ${BA} = 4$ ,而 $D{B}_{1} = 9$ ,故 $B{B}_{1}^{2} + B{D}^{2} = {81}$ ,故 $B{B}_{1} = 7$ ,故所求体积为 $7 \times  {16} = {112}$
    \end{jieda}

    \item 若一个长方体其一顶点的三个面的面积分别为 $\sqrt{2}$ 、 $\sqrt{3}$ 、 $\sqrt{6}$ ，则这个长方体的对角线的长是\blkda{$\sqrt{6}$}.
    \begin{jieda}
        棱长：$1, \sqrt{2}, \sqrt{3}$
    \end{jieda}
    
    \item 一个正四棱柱底面边长为 2,高为 1,则它的表面积是\blkda{16}.
    \begin{jieda}
        因为一个正四棱柱底面边长为 2,高为 1,则它的表面积是 $2 \times  {2}^{2} + 4 \times  2 \times  1 = {16}$ .
    
    \item 在正四棱柱 ${ABCD} - {A}_{1}{B}_{1}{C}_{1}{D}_{1}$ 中， ${BD} = {4\sqrt{2}}$ ， $\mathbf{D}{B}_{1} = \sqrt{41}$ ，则该正四棱柱的表面积为\blkda{80}.
    \begin{jieda}
        因为四边形 ${ABCD}$ 为正方形, ${BD} = 4\sqrt{2}$ ,故 ${BA} = 4$ ,而 $D{B}_{1} = \sqrt{41}$ ,故 $B{B}_{1}^{2} + B{D}^{2} = B{B}_{1}^{2} + {\left( 4\sqrt{2}\right) }^{2} = {41}$ ,故 $B{B}_{1} = 3$ , 故正四棱柱的表面积为 $2 \times  {4}^{2} + 4 \times  4 \times  3 = {80}$ .
    \end{jieda}
    \end{jieda}
    
    \item 正三棱柱的侧面积为 ${54}{{{cm}}}^{2}$ ，体积为 ${45}\sqrt{3}{{{cm}}}^{3}$ ，则此棱柱的高为\blkda{${45}\sqrt{3}$}cm，底面边长为\blkda{$\dfrac{9}{5}$}cm. 
    \begin{jieda}
        设底面边长为 $a{{cm}}$ ,高为 $h{{cm}}$ ,则 ${3ah} = {54}$ ,且 $\dfrac{\sqrt{3}}{4}{a}^{2}h = {45}\sqrt{3}$ ,解得 $a = {10},h = \dfrac{9}{5}$.
    \end{jieda}
    
    \item 一个直三棱柱形容器中盛有水，侧棱 $A{A}_{1} = {18}$ ，底面 $\bigtriangleup  {ABC}$ 边 ${AB}$ 上的高为 $h$ . 当底面 ${ABC}$ 水平放置时水面高度为 16 (如图 \ding{172} ) .当侧面 $A{A}_{1}{B}_{1}B$ 水平放置时 (如图 \ding{173} ),水面高度为\blkda{$\dfrac{2}{3}h$}

    \begin{center}
    \includegraphics[width=0.4\textwidth]{bodyxb/src/0_254_1658_174_107_0.jpg}
    \end{center}
    \begin{jieda}
        设底面 $\bigtriangleup  {ABC}$ 的面积为 ${S}$ ，当底面 ${ABC}$ 水平放置时水面高度为 16，所以水的体积为 $V = {16S}$ ，

        设侧面 $A{A}_{1}{B}_{1}B$ 水平放置时,水呈四棱柱体,设四棱柱体的底面梯形的面积为 ${S}^{\prime }$ ,

        则水的体积为 $V = {18}{S}^{\prime }$ ,所以 ${16S} = {18}{S}^{\prime }$ ,所以 $\dfrac{{S}^{\prime }}{S} = \dfrac{16}{18} = \dfrac{8}{9}$ ,

    \end{jieda}

    \item 底面是菱形的直棱柱，若对角面的面积为 ${50}\sqrt{2}c{m}^{2}$ 和 ${10}\sqrt{14}c{m}^{2}$ ，而底边长为 ${8cm}$ ， 则其对角线的长等于\blkda[3]{$15cm, 9cm$}.
    \begin{jieda}
        设底面菱形对角线分别为$a, b$，高为$h$（单位$cm$），则有$\left\{\begin{array}{l}
             ah = 50\sqrt{2} \\

             a^2+b^2 = 256
        \end{array}\right.$，解得$\left\{\begin{array}{l}
             a = 10\sqrt{2} \\
             b = 2\sqrt{14} \\
             h = 5
        \end{array}\right.$，故对角线长度分别为$15cm, 9cm$.
    \end{jieda}
    
    \item 已知圆柱的底面积为$9\pi$，侧面积为$16\pi$，则该圆柱的体积为\blkda{$24\pi$}.
    \begin{jieda}
        设圆柱的底面半径为 $r$ ,高为 $h$ ,由题意可知 $\left\{  \begin{array}{l} \pi {r}^{2} = {9\pi } \\  {2\pi rh} = {16\pi } \end{array}\right.$ ,解得 $r = 3,h = \dfrac{8}{3}$ .

        所以该圆柱的体积为 $\pi {r}^{2}h = \pi  \times  9 \times  \dfrac{8}{3} = {24\pi }$ .
    \end{jieda}

    \item 已知圆柱的底面积为 ${9\pi }$ ，表面积为 ${36\pi }$ ，则该圆柱的体积为\blkda{$27\pi$}.
    \begin{jieda}  
        设圆柱的底面圆的半径为 $r$ ,高为 $h$ ,由题意可得 $\left\{  \begin{array}{l} \pi {r}^{2} = {9\pi } \\  {2\pi rh} + 2 \times 9\pi = {18\pi } \end{array}\right.$ ,解得 $\left\{  \begin{array}{l} r = 3 \\  h = 3 \end{array}\right.$ ,所以圆柱的体积 $V = \pi {r}^{2}h = {27\pi }$.
    \end{jieda}
\end{enumerate}

\subsection*{球体的体积、表面积、截面性质}
\begin{enumerate}
    \item 已知一球的体积为 $\dfrac{\pi }{6}$ ,则该球的表面积为\blkda{$\pi$}
    \begin{jieda}
        由球的体积公式可得: $\dfrac{4}{3}\pi {R}^{3} = \dfrac{\pi }{6} \Rightarrow  {R}^{3} = \dfrac{1}{8} \Rightarrow  R = \dfrac{1}{2}$ ,所以再由球的表面积公式可得: ${4\pi }{R}^{2} = {4\pi } \times  {\left( \dfrac{1}{2}\right) }^{2} = \pi$
    \end{jieda}
    \item 已知球 $O$ 的体积为 ${36\pi }{{{cm}}}^{3}$ ,则球 $O$ 的表面积为\blkda{$36\pi$} ${{{cm}}}^{2}$ .
    \begin{jieda}
        设球 $O$ 的半径长为 $R{{cm}}$ ,则该球的体积为 $\dfrac{4}{3}\pi {R}^{3} = {36\pi }$ ,解得 $R = 3$ ,所以,球 $O$ 的表面积为 ${4\pi } \times  {3}^{2} = {36\pi }{{{cm}}}^{2}$ .
    \end{jieda}
    \item 若球的表面积扩大为原来的 2 倍，则体积是原来的\blkda{$(\sqrt{2})^3$}倍.
    \begin{jieda}
        $\dfrac{r_1}{r_2} = \sqrt{\dfrac{S_1}{S_2}} = \sqrt{2}$，$\dfrac{V_1}{V_2} = \left(\dfrac{r_1}{r_2}\right)^3 = (\sqrt{2})^3$
    \end{jieda}

    \item 一球的体积和表面积在数值上相等，则该球的半径数值为\blkda{3}.
    \begin{jieda}
        $\dfrac{4}{3}\pi R^3 = 4\pi R^2$，$R = 3$
    \end{jieda}
    \item 一个底面半径为 $2{{cm}}$ 的圆柱形容器内盛有足量的水,能放入一个半径为 $1{{cm}}$ 的实心铁球,沉入水底后,水未溢出容器,则水面升高了\blkda{$\dfrac{1}{3}$}cm.
    \begin{jieda}
        设水面升高了 $x{{cm}}$ ,由题意知 $\pi  \cdot  {2}^{2} \cdot  x = \dfrac{4}{3}\pi  \cdot  {1}^{3}$ ,解得: $x = \dfrac{1}{3}$ .
    \end{jieda}
    \item 若一个半径为 $\sqrt{3}$ 的球与一个高为 1 的圆柱表面积相等,则该圆柱的侧面积为\blkda{$4\pi$}.
    \begin{jieda}
        设球的半径为 $R$ ,圆柱的底面半径为 $r$ ,高为 $h = 1$ .
        由题 ${S}_{\text{球 }} = {4\pi }{R}^{2} = {12\pi }$ ,故 ${S}_{\text{圆柱 }} = {2\pi }{r}^{2} + {2\pi rh} = {12\pi }$ ,即 ${r}^{2} + r = 6$ ,故 $r = 2$ (负根舍去),
        所以 ${S}_{\text{侧 }} = {2\pi rh} = {4\pi }$ .
    \end{jieda}
    \item 已知圆柱的底面圆的半径与球的半径相等,若圆柱的表面积与球的表面积也相等,则圆柱的体积 ${V}_{1}$ 与球的体积 ${V}_{2}$之比 ${V}_{1} : {V}_{2} =$ \blkda{$\dfrac{3}{4}$}.
    \begin{jieda}
        设圆柱的底面圆和球的半径为 $r$ ,圆柱的高为 $h$ ,
        则由题意得, ${2\pi }{r}^{2} + {2\pi rh} = {4\pi }{r}^{2}$ ,则 $h = r$ ,则 $\dfrac{{V}_{1}}{{V}_{2}} = \dfrac{\pi {r}^{2}h}{\dfrac{4}{3}\pi {r}^{3}} = \dfrac{3}{4}$ .
    \end{jieda}
    
    \item 现有一个圆锥与一个球, 它们的表面积相等, 圆锥的母线长与球的直径相等, 则圆锥的底面直径与母线长的比值为\blkda{$\sqrt{5} - 1$}.
    \begin{jieda}
        设该圆锥的底面半径为 $r$ ,母线长为 $l$ ,则其表面积为 ${\pi rl} + \pi {r}^{2}$ ,
        球的表面积为 ${4\pi }{\left( \dfrac{l}{2}\right) }^{2} = \pi {l}^{2}$ ,所以 ${\pi rl} + \pi {r}^{2} = \pi {l}^{2}$ ,
        即 ${\left( \dfrac{r}{l}\right) }^{2} + \dfrac{r}{l} - 1 = 0$ ,解得 $\dfrac{r}{l} = \dfrac{\sqrt{5} - 1}{2}$ (负值舍去),

    \end{jieda}
    
    \item 已知一平面截球 $O$ 所得截面圆的半径为 2,且球心 $O$ 到截面圆所在平面的距离为 1,则该球的体积为\blkda{$\dfrac{{20}\sqrt{5}\pi }{3}$}.
    \begin{jieda}

    \end{jieda}
    
    \item 已知圆柱的两个底面的圆周在同一个球的球面上,圆柱的高和底面直径均为 $\sqrt{2}$ ,则这个球的体积为\blkda{$\dfrac{4\pi }{3}$}.
    \begin{jieda}

    \end{jieda}

\end{enumerate}
