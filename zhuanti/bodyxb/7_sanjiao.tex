    \item 三角函数的概念
    \begin{enumerate}
        \item 下列各组角中，终边相同的是\dotfill{(\kh{C})}
        \xx{$\dfrac{k}{2}\pi$与$k\pi + \dfrac{\pi}{2} (k \in \mathbf{Z})$}
        {$k\pi \pm \dfrac{\pi}{3}$与$\dfrac{k\pi}{3} (k \in \mathbf{Z})$}
        {$(2k+1)\pi$与$(4k \pm 1)\pi (k \in \mathbf{Z})$}
        {$k\pi + \dfrac{\pi}{6}$与$k\pi \pm \dfrac{\pi}{6} (k \in \mathbf{Z})$}
        \begin{jieda}
            \begin{enumerate}[A.]
                \item 第一项可以取$\{0, \dfrac{\pi}{2}, \pi, \dfrac{3\pi}{2} \}$及其终边相同角，而第二项可取到$\{\dfrac{\pi}{2}, \dfrac{3\pi}{2}\}$及其终边相同角
                \item 第一项可以取到$\{\dfrac{\pi}{3}, \dfrac{2\pi}{3}, \dfrac{4\pi}{3},\dfrac{5\pi}{3}\}$及其终边相同角，而第二项可取到$\{0, \dfrac{\pi}{3}, \dfrac{2\pi}{3}, \pi, \dfrac{4\pi}{3},\dfrac{5\pi}{3}\}$及其终边相同角
                \item 第一项可以取$\{\pi\}$及其终边相同角，第二项也是可以取到$\{\pi\}$及其终边相同角
                \item 第一项可以取$\{\dfrac{\pi}{6}, \dfrac{7\pi}{6}\}$及其终边相同角，而第二项可取到$\{\dfrac{\pi}{6}, \dfrac{5\pi}{6}, \dfrac{7\pi}{6}，\dfrac{11\pi}{6}\}$及其终边相同角
            \end{enumerate}
        \end{jieda}
        \item 若$\alpha, \beta$均为第一象限角，则“$\alpha < \beta$”是“$sin\alpha < sin\beta$”的\blkda[3]{既非充分也非必要}条件.
        \begin{jieda}
            若条件改为$\alpha, \beta$均属于$(0, \dfrac{\pi}{2})$则“$\alpha < \beta$”是“$sin\alpha < sin\beta$”的充要条件.
        \end{jieda}
        \item $\dfrac{sinx}{|sinx|} + \dfrac{|cosx|}{cosx} + \dfrac{tanx}{|tanx|} + \dfrac{|cotx|}{cotx}$的取值范围是\blkda[2]{$\{-2, 0, 4\}$}
        \begin{jieda}
            $x$在第一象限，则原式取值为4，$x$在第二象限，则原式取值为$-2$，$x$在第三象限，则原式取值为0，$x$在第四象限，则原式取值为$-2$
        \end{jieda}
        \item 若$x$为三角形内角，则当$x = $\blkda[2]{$\dfrac{\pi}{4}$或$\dfrac{\pi}{2}$}时，$\dfrac{sin\dfrac{x}{2}}{1-tanx}$无意义.
        \begin{jieda}
            当$x = \dfrac{\pi}{4}$时$tanx = 1$，分母为零，当$x = \dfrac{\pi}{2}$时$tanx$无意义.
        \end{jieda}
        \item 若$1+sin\theta\sqrt{1 - cos^2\theta} + cos\theta\sqrt{1-sin^2\theta} = 0$，则$\theta$的取值范围是\blkda[5]{$\left[2k\pi + \pi, 2k\pi + \dfrac{3\pi}{2}\right], k \in \mathbf{Z}$}
        \begin{jieda}
            $1 + sin\theta |sin\theta| + cos\theta |cos\theta| = 0$，故必有$sin\theta \leq 0$且$cos\theta \leq 0$ \\
            因此$\theta \in [2k\pi + \pi, 2k\pi + \dfrac{3\pi}{2}], k \in \mathbf{Z}$
        \end{jieda}
        \item 若$0 < \alpha < \dfrac{\pi}{2}$，$lg(1+cos\alpha) = m, lg\left(\dfrac{1}{1-cos\alpha}\right) = n$则$lg(sin\alpha) = $\blkda{$\dfrac{m-n}{2}$}(用$m, n$表示)
        \begin{jieda}
            首先进行指对转化，由$1+cos\alpha = 10^m, 1-cos\alpha = 10^{-n}$ \\
            要得到$sin\alpha$，应考虑两式相乘，得到$sin^2\alpha = 10^{m-n}$ \\
            再两侧取对数后有$2lg(sin\alpha) = m - n$，故$lg(sin\alpha) = \dfrac{m-n}{2}$
        \end{jieda}
        \item 若$\sqrt{\dfrac{1-sinx}{1+sinx}} = \dfrac{sinx - 1}{cosx}$，则$x$的取值范围是\blkda[5]{$(2k\pi+\dfrac{\pi}{2}, 2k\pi+\dfrac{3\pi}{2}), k \in \mathbf{Z}$}
        \begin{jieda}
            $\sqrt{\dfrac{1-sinx}{1+sinx}} = \sqrt{\dfrac{(1-sinx)^2}{cos^2x}} = \dfrac{1-sinx}{|cosx|}$，因此有$\dfrac{1-sinx}{|cosx|} = \dfrac{sinx - 1}{cosx}$，故$cosx < 0$
        \end{jieda}
        \item 若$\alpha \in (0, 2\pi)$，则满足$\sqrt{\dfrac{1+cos\alpha}{1-cos\alpha}} - \sqrt{\dfrac{1-cos\alpha}{1+cos\alpha}} = 2cot\alpha$的角$\alpha$的取值范围是\blkda[3]{$(0, \pi) \cup \{\dfrac{3\pi}{2}\}$}
        \begin{jieda}
            $\sqrt{\dfrac{1+cos\alpha}{1-cos\alpha}} - \sqrt{\dfrac{1-cos\alpha}{1+cos\alpha}} = \sqrt{\dfrac{(1+cos\alpha)^2}{sin^2\alpha}} - \sqrt{\dfrac{(1-cos\alpha)^2}{sin^2\alpha}} = \dfrac{2cos\alpha}{|sin\alpha|}$ \\
            因为$\dfrac{2cos\alpha}{|sin\alpha|} = 2cot\alpha$，所以$sin\alpha > 0$或$cos\alpha = 0$，又因为$\alpha \in (0, 2\pi)$，因此有$\alpha \in (0, \pi) \cup \{\dfrac{3\pi}{2}\}$
        \end{jieda}
        \item 已知$tan\theta = \dfrac{sin\alpha - cos\alpha}{sin\alpha + cos\alpha}$（$\alpha, \theta$都是锐角），则$\dfrac{sin\alpha - cos\alpha}{sin \theta} = $\blkda{$\sqrt{2}$}
        \begin{jieda}
            由$tan\theta = \dfrac{sin\alpha - cos\alpha}{sin\alpha + cos\alpha}$，且$\alpha, \theta$都是锐角知$sin\alpha - cos\alpha > 0, sin\alpha + cos\alpha > 0$，因此可得$\left \{ \begin{array}{l}
                sin\theta = \dfrac{sin\alpha - cos\alpha}{\sqrt{2}} \\
                cos\theta = \dfrac{sin\alpha + cos\alpha}{\sqrt{2}} 
            \end{array} \right.$，故$\dfrac{sin\alpha - cos\alpha}{sin \theta} = \sqrt{2}$
        \end{jieda}
    \end{enumerate}

        \end{jieda}
        \item 求$y = \dfrac{sinx \cdot cosx}{1 + sinx + cosx}$的值域.

        \end{jieda}
