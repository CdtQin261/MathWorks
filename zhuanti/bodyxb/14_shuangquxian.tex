\chapter*{双曲线专题练习}
\begin{enumerate}
    \item 定义
    \begin{enumerate}
        \item \begin{enumerate}
            \item 当实数 $k$ 为何值时,方程 $\dfrac{{x}^{2}}{1 - k} - \dfrac{{y}^{2}}{\left| k\right|  - 3} =  - 1$ 表示的曲线是双曲线?
            \item 当实数 $k$ 为何值时,方程 $\dfrac{{x}^{2}}{{25} - k} + \dfrac{{y}^{2}}{9 - k} = 1$ 表示的曲线是椭圆或双曲线? 并指出它们的共同特征.
        \end{enumerate}
        \begin{jieda}
            \begin{enumerate}
                \item 方程可以化为$\dfrac{{y}^{2}}{\left| k\right|  - 3} - \dfrac{{x}^{2}}{1 - k} = 1$
                \begin{dingautolist}{172}
                    \item 当$\left\{\begin{array}{l}
                     \left| k\right|  - 3 > 0  \\
                     1 - k > 0
                    \end{array}\right.$，即$k \in (-\infty, -3)$时，该方程表示的曲线是焦点在$y$轴的双曲线.
                    \item 当$\left\{\begin{array}{l}
                     \left| k\right|  - 3 < 0  \\
                     1 - k < 0
                    \end{array}\right.$，即$k \in (1, 3)$时，该方程表示的曲线是焦点在$x$轴的双曲线.
                \end{dingautolist}
                
                \item 只要满足$25-k>0$或$9-k>0$，即$k \in (-\infty, 9)$方程所表示的曲线就是椭圆或者双曲线

                因为$25-k > 9-k$，因此无论方程表示椭圆还是双曲线，都是有$a^2 = 25-k, b^2 = |9-k|$，因此$c = 4$，即焦点为$(4, 0)$和$(-4, 0)$
            \end{enumerate}
        \end{jieda}
        
        \item “ ${mn} < 0$ ” 是 “方程 $m{x}^{2} + n{y}^{2} = 1$ 表示双曲线” 的\dotfill{(\kh{C})}
        \xx{充分不必要条件}{必要不充分条件}{充要条件}{既不充分也不必要条件}

        \begin{jieda}
            因为方程 $m{x}^{2} + n{y}^{2} = 1$ 表示双曲线,所以 ${mn} < 0$，又当 ${mn} < 0$ 时,方程 $m{x}^{2} + n{y}^{2} = 1$ 表示双曲线，因此 “ ${mn} < 0$ ” 是 “方程 $m{x}^{2} + n{y}^{2} = 1$ 表示双曲线” 的充要条件.
        \end{jieda}

        \begin{jieda}
            \begin{center}
            \includegraphics[width=0.2\textwidth]{bodyxb/src/bo_d1chk83ef24c73d2s6gg_2_677_634_290_211_0.jpg}
            \end{center}
            
            如图,设 $\bigtriangleup  {ABC}$ 与圆的切点分别为 $D,E,F$ , 则有 $\left| {AD}\right|  = \left| {AE}\right|  = 5,\left| {BF}\right|  = \left| {BE}\right|  = 1,\left| {CD}\right|  = \left| {CF}\right|$，所以 $\left| {CA}\right|  - \left| {CB}\right|  = 5 - 1 = 4$ .
            
            根据双曲线定义,所求轨迹是以 $A,B$ 为焦点,实轴长为 4 的双曲线的右支 (右顶点除外)，即 $c = 3,a = 2$ ,又 ${c}^{2} = {a}^{2} + {b}^{2}$ ,所以 ${b}^{2} = 5$，所以顶点 $C$ 的轨迹方程为 $\dfrac{{x}^{2}}{4} - \dfrac{{y}^{2}}{5} = 1\left( {x > 2}\right)$ .
        \end{jieda}

        \item 在相距 2000 米的两个观察站 $A$ 、 $B$ 先后听到远处传来的爆炸声,已知 $A$ 站听到爆炸声的时间比 $B$ 站早 4 秒,声速是 340 米/秒,请判断爆炸点可能在什么样的曲线上,并建立适当的平面直角坐标系求出该曲线的方程.
        \begin{jieda}
            根据双曲线的定义可知爆炸点处在双曲线的一支上.

            以$AB$中点为原点建立平面直角坐标系，$A(-1000, 0), B(1000, 0)$，则$c = 1000, 2a = 1360$，因此曲线方程为$\dfrac{x^2}{462400} - \dfrac{y^2}{537600} = 1(x < 0)$
        \end{jieda}

        \item 已知 ${F}_{1},{F}_{2}$ 为双曲线 $C : \dfrac{{y}^{2}}{3} - {x}^{2} = 1$ 的两个焦点, $P$ 为双曲线 $C$ 上任意一点,则$\left| {\vv{P{F}_{1}} + \vv{P{F}_{2}}}\right|$的取值范围是\blkda[2]{$[2\sqrt{3},+\infty)$}
        \begin{jieda}
            设 $P\left( {x,y}\right)$是双曲线上一点，$\left| {\vv{P{F}_{1}} + \vv{P{F}_{2}}}\right|  = \left| {2\vv{PO}}\right| = 2\left| \vv{PO}\right|$
            
            $\left| {PO}\right|  = \sqrt{{x}^{2} + {y}^{2}} = \sqrt{{x}^{2} + \left( {3{x}^{2} + 3}\right) } = \sqrt{3 + 4{x}^{2}} \geq  \sqrt{3}$ (当且仅当 $x = 0$ 时取等号)，故$\left| {\vv{P{F}_{1}} + \vv{P{F}_{2}}}\right| \geq  2\sqrt{3}$
        \end{jieda}
    \end{enumerate}
    \item 求标准方程
    \begin{enumerate}
        \item 求满足下列条件的双曲线的标准方程:
        \begin{dingautolist}{172}
            \item 经过点 $\left( {\sqrt{6},0}\right) ,\left( {3,2}\right)$
            \item 焦点为 $\left( {0, -5}\right) ,\left( {0,5}\right)$ ,经过点 $\left( {\dfrac{4\sqrt{3}}{3},2\sqrt{3}}\right)$
            \item $a = b$ ,经过点$(3, -1)$
            \item 经过两点 $\left( {3, - 4\sqrt{2}}\right)$ 和 $\left( {\dfrac{9}{4},5}\right)$
        \end{dingautolist}
        \begin{jieda}
            \begin{dingautolist}{172}
                \item 因为经过点$(\sqrt{6}, 0)$，所以$a = \sqrt{6}$，设双曲线的标准方程为$\dfrac{x^2}{6} - \dfrac{y^2}{b^2} = 1$，代入$(3, 2)$，解得$b^2 = 8$，因此双曲线的标准方程为$\dfrac{x^2}{6} - \dfrac{y^2}{8} = 1$
                \item 根据焦点坐标可知$c = 5$
                
                计算焦点到$\left( {\dfrac{4\sqrt{3}}{3},2\sqrt{3}}\right)$的距离之差为6，故$a = 3$

                因此双曲线的标准方程为$\dfrac{y^2}{9} - \dfrac{x^2}{16} = 1$
                \item 根据已知条件可以判断出焦点位于$x$轴上，设双曲线的标准方程为$\dfrac{x^2}{a^2} - \dfrac{y^2}{a^2} = 1$，代入点$(3, -1)$，得到$a^2 = 8$，因此双曲线的标准方程为$\dfrac{x^2}{8} - \dfrac{y^2}{8} = 1$
                \item 根据题意,可设双曲线的方程为 $A{x}^{2} - B{y}^{2} = 1\left( {{AB} > 0}\right)$ . 因为双曲线过 $\left( {3, - 4\sqrt{2}}\right)$ 和 $\left( {\dfrac{9}{4},5}\right)$ 两点,所以 $\left\{  \begin{array}{l} {9A} - {32B} = 1, \\  {\dfrac{81}{16}A} - {25B} = 1, \end{array}\right.$ 解得 $\left\{  \begin{array}{l} A =  - \dfrac{1}{9}, \\  B =  - \dfrac{1}{16}. \end{array}\right.$
            
                因此,双曲线的标准方程为 $\dfrac{{y}^{2}}{16} - \dfrac{{x}^{2}}{9} = 1$ .
                \end{dingautolist}
        \end{jieda}
        \item 求满足下列条件的双曲线的标准方程:
        \begin{dingautolist}{172}
            \item 焦点 ${F}_{1}\left( {-2,0}\right) \text{ 、 }{F}_{2}\left( {2,0}\right)$ ,一个顶点为$(1,0)$
            \item 一个焦点为$(0,3)$,离心率为 3
            \item 一条渐近线为 ${2x} - {3y} = 0$ ,且过点$(1, -1)$
            \item 焦点在$x$轴上，过点 $\left( {\sqrt{2},\sqrt{3}}\right)$ ,且离心率为 2
        \end{dingautolist}
        \begin{jieda}
            \begin{dingautolist}{172}
                \item $c = 2, a = 1$，实轴在$x$轴上，$b = \sqrt{3}$，故双曲线方程为$x^2 - \dfrac{y^2}{3} = 1$
                \item $c = 3, a = 1$，实轴在$y$轴上，$b = 2\sqrt{2}$，故双曲线方程为$y^2 - \dfrac{x^2}{8} = 1$
                \item 双曲线的另一条渐近线为$y = -\dfrac{2}{3}x$，因为$(1， -1)$在两渐近线下方区域，因此实轴在$y$轴上，因此$\dfrac{a}{b} = \dfrac{2}{3}$，设双曲线方程为$\dfrac{y^2}{4k}-\dfrac{x^2}{9k} = 1$，代入$(1, -1)$，故$k = \dfrac{1}{4} - \dfrac{1}{9} = \dfrac{5}{36}$，因此双曲线的标准方程为$\dfrac{y^2}{\dfrac{5}{9}}-\dfrac{x^2}{\dfrac{5}{4}} = 1$
                \item 由 $e = \dfrac{c}{a} = 2$，得 $c = {2a}, b = \sqrt{{c}^{2} - {a}^{2}} = \sqrt{3}a$，则双曲线的方程为 $\dfrac{{x}^{2}}{{a}^{2}} - \dfrac{{y}^{2}}{3{a}^{2}} = 1$ 
                
                将点 $\left( {\sqrt{2},\sqrt{3}}\right)$ 的坐标代入双曲线的方程可得 $\dfrac{2}{{a}^{2}} - \dfrac{3}{3{a}^{2}} = \dfrac{1}{{a}^{2}} = 1$，解得 $a = 1$ ,故 $b = \sqrt{3}$
                
                因此双曲线的标准方程为 ${x}^{2} - \dfrac{{y}^{2}}{3} = 1$ .
            \end{dingautolist}
        \end{jieda}
        \item 已知双曲线 $\dfrac{{x}^{2}}{{a}^{2}} - \dfrac{{y}^{2}}{{b}^{2}} = 1\left( {a > 0,b > 0}\right)$ 的离心率为 2,左焦点到渐近线的距离为 $2\sqrt{3}$ ,则双曲线的方程为\blkda[3]{$\dfrac{{x}^{2}}{4} - \dfrac{{y}^{2}}{12} = 1$}
        \begin{jieda}
            易知双曲线 $\dfrac{{x}^{2}}{{a}^{2}} - \dfrac{{y}^{2}}{{b}^{2}} = 1\left( {a > 0,b > 0}\right)$ 的渐近线方程为 ${ay} =  \pm  {bx}$ ,由 $C$ 的左焦点$(-c,0)$到其渐近线的距离是 $2\sqrt{3}$ , 可得 $\dfrac{bc}{\sqrt{{a}^{2} + {b}^{2}}} = b = 2\sqrt{3}$ ,则 ${b}^{2} = {12}$ ,
            
            由双曲线 $\dfrac{{x}^{2}}{{a}^{2}} - \dfrac{{y}^{2}}{{b}^{2}} = 1\left( {a > 0,b > 0}\right)$ 的离心率为 2,得 $e = \dfrac{c}{a} = 2$ ,又 ${c}^{2} = {a}^{2} + {b}^{2}$ ,
            
            解得 $a = 2,c = 4$ ,
            
            则双曲线的方程为 $\dfrac{{x}^{2}}{4} - \dfrac{{y}^{2}}{12} = 1$ .
        \end{jieda}
    \end{enumerate}
    \item 双曲线的几何性质
    \begin{enumerate}
        \item 渐近线
        \begin{enumerate}
            \item 已知双曲线 $\dfrac{{x}^{2}}{{a}^{2}} - \dfrac{{y}^{2}}{{b}^{2}} = 1\left( {a > 0,b > 0}\right)$ 的离心率为 2,则该双曲线的渐近线方程为\blkda[2]{$y =  \pm  \sqrt{3}x$}
            \begin{jieda}
                因为双曲线 $\dfrac{{x}^{2}}{{a}^{2}} - \dfrac{{y}^{2}}{{b}^{2}} = 1$ 的离心率为 2,则 $2 = \sqrt{1 + {\left( \dfrac{b}{a}\right) }^{2}}$ ,解得 $\dfrac{b}{a} = \sqrt{3}$ 
                
                故双曲线的渐近线方程为 $y =  \pm  \sqrt{3}x$ .
            \end{jieda}
            \item 点$(4,0)$到双曲线 $\dfrac{{x}^{2}}{3} - {y}^{2} = 1$ 的一条渐近线的距离为\blkda{2}
            \begin{jieda}
                由已知可得双曲线的渐近线方程为 $y =  \pm \dfrac{\sqrt{3}}{3}x$ .
                
                所以,点$(4,0)$到 $y = \dfrac{\sqrt{3}}{3}x$ ,即 $x - \sqrt{3}y = 0$ 的距离 $d = \dfrac{\left| 4\right| }{\sqrt{{1}^{2} + {\left( -\sqrt{3}\right) }^{2}}} = 2$ 
            \end{jieda}
            \item 已知双曲线 ${y}^{2} + \dfrac{{x}^{2}}{m} = 1$ 的渐近线方程为 $y =  \pm  \dfrac{\sqrt{3}}{3}x$ ,则 $m =$\blkda{$-3$}
            
            \begin{jieda}
                依题意得 $m < 0$，故渐近线方程为${y}^{2} - \dfrac{{x}^{2}}{-m} = 0$ ,得 $y =  \pm  \dfrac{1}{\sqrt{-m}}x$ ,则 $\pm  \dfrac{1}{\sqrt{-m}} =  \pm  \dfrac{\sqrt{3}}{3}$ ,解得 $m =  - 3$ .
            \end{jieda}
            \item 若双曲线经过点 $\left( {1,\sqrt{3}}\right)$ ,其渐近线方程为 $y =  \pm  {2x}$ ,则双曲线的方程是\blkda[3]{$4{x}^{2} - {y}^{2} = 1$}.
            
            \begin{jieda}
                由题可设双曲线方程为 $4{x}^{2} - {y}^{2} = \lambda \left( {\lambda  \neq  0}\right)$ 
                
                由双曲线经过点 $\left( {1,\sqrt{3}}\right)$，代入得$\lambda  = 4 \times  {1}^{2} - {\left( \sqrt{3}\right) }^{2} = 1$
                
                因此双曲线方程为 $4{x}^{2} - {y}^{2} = 1$ .
            \end{jieda}
            \item 已知双曲线的渐近线方程为 $y =  \pm  \dfrac{\sqrt{2}}{2}x$ ,实轴长为 4,则该双曲线的标准方程为\blkda[5]{$\dfrac{{x}^{2}}{4} - \dfrac{{y}^{2}}{2} = 1$ 或 $\dfrac{{y}^{2}}{4} - \dfrac{{x}^{2}}{8} = 1$}
            \begin{jieda}
                设双曲线方程为 $\dfrac{{x}^{2}}{2m} - \dfrac{{y}^{2}}{m} = 1\left( {m \neq  0}\right)$，$\because {2a} = 4,\therefore {a}^{2} = 4$ 
                \begin{dingautolist}{172}
                    \item 当 $m > 0$ 时, ${2m} = 4,m = 2$
                    \item 当 $m < 0$ 时, $- m = 4,m =  - 4$
                \end{dingautolist}
                
                故所求双曲线的标准方程为 $\dfrac{{x}^{2}}{4} - \dfrac{{y}^{2}}{2} = 1$ 或 $\dfrac{{y}^{2}}{4} - \dfrac{{x}^{2}}{8} = 1$ .
            \end{jieda}

            \item 已知双曲线 $C : \dfrac{{x}^{2}}{3} - {y}^{2} = 1,O$ 为坐标原点, $F$ 为 $C$ 的右焦点,过 $F$ 的直线与 $C$ 的两条渐近线的交点分别为 $M\text{ 、 }N$ . 若 $\bigtriangleup {OMN}$ 为直角三角形,则 $\left| {MN}\right|  =$ \blkda{3}
            \begin{jieda}
                根据题意,可知其渐近线的斜率为 $\pm  \dfrac{\sqrt{3}}{3}$ ,且右焦点为 $F\left( {2,0}\right)$ ,
                
                从而得到 $\angle {FON} = {30}^{ \circ  }$ ,所以直线 ${MN}$ 的倾斜角为 ${60}^{ \circ  }$ 或 ${120}^{ \circ  }$ ,
                
                根据双曲线的对称性,设其倾斜角为 ${60}^{ \circ  }$ ,
                
                可以得出直线 ${MN}$ 的方程为 $y = \sqrt{3}\left( {x - 2}\right)$ ,
                
                分别与两条渐近线 $y = \dfrac{\sqrt{3}}{3}x$ 和 $y =  - \dfrac{\sqrt{3}}{3}x$ 联立,
                
                求得 $M\left( {3,\sqrt{3}}\right) ,N\left( {\dfrac{3}{2}, - \dfrac{\sqrt{3}}{2}}\right)$ ,
                
                所以 $\left| {MN}\right|  = \sqrt{{\left( 3 - \dfrac{3}{2}\right) }^{2} + {\left( \sqrt{3} + \dfrac{\sqrt{3}}{2}\right) }^{2}} = 3$
            \end{jieda}

            \item 已知双曲线 $C : \dfrac{{x}^{2}}{{a}^{2}} - \dfrac{{y}^{2}}{{b}^{2}} = 1\left( {a > 0,b > 0}\right)$ 的左、右焦点分别为 ${F}_{1},{F}_{2}$ ,过 ${F}_{1}$ 作一条渐近线的垂线交双曲线 $C$ 的左支于点 $P$ ,已知 $\dfrac{\left| P{F}_{1}\right| }{\left| P{F}_{2}\right| } = \dfrac{2}{5}$ ,则双曲线 $C$ 的渐近线方程为\blkda[2]{$y =  \pm  {2x}$}.
            \begin{jieda}
                根据题意画出图象如下:
                
                \begin{center}
                \includegraphics[width=0.25\textwidth]{bodyxb/src/bo_d2a13mv7aajc738pd0qg_0_245_868_408_307_0.jpg}
                \end{center}
                
                由 $\dfrac{\left| P{F}_{1}\right| }{\left| P{F}_{2}\right| } = \dfrac{2}{5}$ 得 $\left| {P{F}_{1}}\right|  = \dfrac{2}{5}\left| {P{F}_{2}}\right|$ ,又 $\left| {P{F}_{2}}\right|  - \left| {P{F}_{1}}\right|  = {2a}$ ,所以 $\left| {P{F}_{2}}\right|  = \dfrac{10a}{3},\left| {P{F}_{1}}\right|  = \dfrac{4a}{3}$
                
                $\cos \angle P{F}_{1}{F}_{2} = \dfrac{b}{c}$，由余弦定理得 ${\left| P{F}_{2}\right| }^{2} = {\left| P{F}_{1}\right| }^{2} + {\left| {F}_{1}{F}_{2}\right| }^{2} - 2\left| {P{F}_{1}}\right| \left| {{F}_{1}{F}_{2}}\right| \cos \angle P{F}_{1}{F}_{2}$ ,
                
                即 $\dfrac{{100}{a}^{2}}{9} = \dfrac{{16}{a}^{2}}{9} + 4{c}^{2} - \dfrac{16ab}{3}$ ,化简得 $3{\left( \dfrac{b}{a}\right) }^{2} - \dfrac{4b}{a} - 4 = 0$ ,即 $\left( {\dfrac{b}{a} - 2}\right) \left( {\dfrac{3b}{a} + 2}\right)  = 0$
                
                解得 $\dfrac{b}{a} = 2$ 或 $\dfrac{b}{a} =  - \dfrac{2}{3}$ ,因为 $a > 0,b > 0$ ,所以 $\dfrac{b}{a} = 2$ .
                
                则双曲线 $C$ 的渐近线方程为 $y =  \pm  {2x}$ .
            \end{jieda}

            \item 设 ${A}$ 为双曲线 $\Gamma  : \dfrac{{x}^{2}}{{a}^{2}} - \dfrac{{y}^{2}}{{b}^{2}} = 1\left( {a > 0,b > 0}\right)$ 的一个实轴顶点, $B,C$ 为 $\Gamma$ 的渐近线上的两点, 满足 $\vv{BC} = 4\vv{AC},\left| {AC}\right|  = a$ ,则 $\Gamma$ 的渐近线方程是\blkda[3]{$y =  \pm  \sqrt{2}x$ }
            \begin{jieda}
                作图如下:
                
                \begin{center}
                \includegraphics[width=0.25\textwidth]{bodyxb/src/bo_d2a0hlbef24c73avno0g_2_242_1520_323_355_0.jpg}
                \end{center}

                \begin{dingautolist}{172}
                    \item 法1

                    设$C\left(m, \dfrac{b}{a}m\right), B\left(n, -\dfrac{b}{a}n\right)$，根据$\vv{BC} = 4\vv{AC}$可知$n = 3m$，故$B\left(m, -\dfrac{3b}{a}m\right)$，易知$\vv{OA} = \dfrac{1}{4}\vv{OB} + \dfrac{3}{4}\vv{OC}$，因此$\dfrac{3}{4}m+\dfrac{3}{4}m = -a$，即$m = -\dfrac{2}{3}a$，故$C\left(-\dfrac{2}{3}a, -\dfrac{2}{3}b\right)$，又因为$|AC| = a$，故有$\sqrt{\left(\dfrac{a}{3}\right)^2+\dfrac{4}{9}b^2} = a$，可得$b^2 = 2a^2$，所以 $\Gamma$ 的渐近线方程是 $y =  \pm  \sqrt{2}x$ .
                    \item 法2

                    依题意, ${OA}$ 为 $\angle {COB}$ 的角平分线,且 $\left| {CB}\right|  = 4\left| {OA}\right|  = 4\left| {CA}\right|  = {4a}$ ,
                    
                    设 $\left| {OC}\right|  = m$ ,由角平分线定理可得: $\dfrac{\left| OB\right| }{\left| OC\right| } = \dfrac{\left| AB\right| }{\left| AC\right| } = 3$ ,则 $\left| {OB}\right|  = {3m}$ ;
                    
                    在 $\bigtriangleup {OAC}$ 中,由余弦定理 $\cos \angle {OCA} = \dfrac{{\left| AC\right| }^{2} + {\left| CO\right| }^{2} - {\left| OA\right| }^{2}}{2\left| {AC}\right| \left| {CO}\right| } = \dfrac{{m}^{2}}{2am} = \dfrac{m}{2a}$ ;
                    
                    在 $\bigtriangleup {OBC}$ 中,由余弦定理可得, ${\left| OB\right| }^{2} = {\left| OC\right| }^{2} + {\left| BC\right| }^{2} - 2\left| {OC}\right|  \cdot  \left| {BC}\right| \cos \angle {OCA}$ ,
                    
                    即 $9{m}^{2} = {m}^{2} + {16}{a}^{2} - 2 \times  m \times  {4a} \times  \dfrac{m}{2a}$ ,解得 $\dfrac{m}{a} = \dfrac{2\sqrt{3}}{3}$ .
                    
                    故 $\cos \angle {COA} = \cos \angle {OCA} = \dfrac{m}{2a} = \dfrac{\sqrt{3}}{3},\tan \angle {COA} = \sqrt{2}$ ,
                    
                    所以 $\Gamma$ 的渐近线方程是 $y =  \pm  \sqrt{2}x$ .
                \end{dingautolist}

            \ps 求双曲线的渐近线方程, 常见方法:
            \begin{dingautolist}{172}
                \item 得到 $a,b$ 的齐次式,从而得解
                \item 根据几何关系求得其中一个渐近线的倾斜角 (或斜率), 从而得解
            \end{dingautolist}
            \end{jieda}
        \end{enumerate}
