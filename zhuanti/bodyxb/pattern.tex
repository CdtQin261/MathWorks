    已知 $ sin \alpha+ cos \alpha=\dfrac{1}{5}(0<\alpha<\pi)$, 求: 
    \begin{enumerate}
        \item 
        \begin{enumerate}
            \item  $ sin \alpha- cos \alpha$;
            \item  $ sin ^3 \alpha+ cos ^3 \alpha$;
            \item $ sin ^4 \alpha+ cos ^4 \alpha$.
        \end{enumerate}
        \begin{jieda}
            \begin{enumerate}
                \item $(sin\alpha+cos\alpha)^2 = 1 + 2sin\alpha cos\alpha = \dfrac{1}{25}$，因此$sin\alpha cos\alpha = -\dfrac{12}{25}$，根据$\alpha \in (0, \pi)$，所以$sin\alpha > 0$，因此$cos\alpha < 0$，即$\alpha$是第二象限角. \\
                因此有$(sin\alpha - cos\alpha)^2 = 1 - 2sin\alpha cos\alpha = \dfrac{49}{25}$ \\
                故$sin\alpha - cos\alpha = \pm \dfrac{7}{5}$, 根据$\alpha$是第二象限角，故$sin\alpha - cos\alpha = \dfrac{7}{5}$ \\
                \ps 可以根据$sin\alpha + cos\alpha$在不同象限的值域直接判断出来$\alpha$是第二象限角.
                \item $sin^3 \alpha + cos^3 \alpha = (sin\alpha + cos\alpha)(sin^2 \alpha + cos^2 \alpha - sin\alpha cos\alpha) = (\dfrac{1}{5})(1 + \dfrac{12}{25}) = \dfrac{37}{125}$ \\
                \ps 也可以在第一问的结果上构建方程组$\left \{ \begin{array}{l}
                    sin\alpha + cos\alpha = \dfrac{1}{5} \\
                    sin\alpha - cos\alpha = \dfrac{7}{5} 
                \end{array}\right.$解得$\left \{ \begin{array}{l}
                    sin\alpha = \dfrac{4}{5} \\
                    cos\alpha = -\dfrac{3}{5} 
                \end{array}\right.$后代入计算（下同）
                \item $ sin ^4 \alpha+ cos ^4 \alpha = ( sin ^2 \alpha+ cos ^2 \alpha)^2 - 2(sin \alpha cos\alpha)^2 = 1 - 2 \times (-\dfrac{12}{25})^2 = \dfrac{337}{625}$
            \end{enumerate}
        \end{jieda}
        
        \item 求$y = \dfrac{sinx \cdot cosx}{1 + sinx + cosx}$的值域.
        \begin{jieda}
            设$t = sinx+cosx, t \in [-\sqrt{2}, -1) \cup (-1, \sqrt{2}]$，则$sinx \cdot cosx = \dfrac{t^2-1}{2}$，故$y = \dfrac{t^2-1}{2(1+t)} = \dfrac{t-1}{2}$，故值域为$\left[\dfrac{-\sqrt{2}-1}{2}, -1\right) \cup \left(-1, \dfrac{\sqrt{2}-1}{2}\right]$
        \end{jieda}
    \end{enumerate}

        \end{jieda}
        \item 已知函数$f(x) = -cotx$，$f(\alpha - \beta) = -2$，$f(\alpha) = -3$且$\alpha \in (0, \dfrac{\pi}{2})$，$\beta \in (\dfrac{\pi}{2}, \pi)$，则$2\alpha - \beta = $\blkda{$-\dfrac{3\pi}{4}$}

            $tan(2\alpha - \beta) = tan(\alpha + \alpha - \beta) = \dfrac{tan\alpha + tan(\alpha-\beta)}{1-tan\alpha \cdot tan(\alpha - \beta)} = 1$，故$2\alpha - \beta = -\dfrac{3\pi}{4}$
        \end{jieda}

        \begin{jieda}
            \begin{enumerate}
                \item 方法1 \\
                $\dfrac{1}{cos2\alpha} + tan2\alpha = \dfrac{1+sin2\alpha}{cos2\alpha} = \dfrac{(sin\alpha+cos\alpha)^2}{cos^2\alpha - sin^2\alpha} = \dfrac{(1+tan\alpha)^2}{1 - tan^2\alpha} = \dfrac{1+tan\alpha}{1-tan\alpha} = 2025$
                \item 方法2 \\
                $\dfrac{1}{cos2\alpha} + tan2\alpha = \dfrac{1+sin2\alpha}{cos2\alpha} = \dfrac{1+cos(\dfrac{\pi}{2} - 2\alpha)}{sin(\dfrac{\pi}{2} - 2\alpha)} = \dfrac{1}{tan(\dfrac{\pi}{4} - \alpha)} = \dfrac{1+tan\alpha}{1-tan\alpha} = 2025$
            \end{enumerate}
        \end{jieda}
        \item 若方程$x^2+4ax+3a+1 = 0 (a>1)$的两根分别是$tan\alpha, tan\beta$，且$\alpha, \beta \in (-\dfrac{\pi}{2}, \dfrac{\pi}{2})$，求$tan\left(\dfrac{\alpha+\beta}{2}\right)$.
        \begin{jieda}
            根据韦达定理有$\left \{ \begin{array}{l}
                 tan\alpha + tan\beta = -4a  \\
                 tan\alpha \cdot tan\beta = 3a+1 
            \end{array}\right.$，因为$a > 1$，所以$tan\alpha, tan\beta < 0$，因此$\alpha, \beta \in (-\dfrac{\pi}{2}, 0)$ \\
            根据和角公式有$tan(\alpha + \beta) = \dfrac{tan\alpha + tan\beta}{1 - tan\alpha \cdot tan\beta} = \dfrac{4}{3}$，故$cos(\alpha+\beta) = -\dfrac{3}{5}, sin(\alpha + \beta) = -\dfrac{4}{5}$，故$tan\left(\dfrac{\alpha+\beta}{2}\right) = \dfrac{sin(\alpha+\beta)}{1+cos(\alpha + \beta)} = -2$. \\
            \ps 判别式大于零影响$a$的范围，但是对于本题$tan(\alpha + \beta)$的值不含$a$，所以对解题没影响.
        \end{jieda}
    \end{enumerate}
