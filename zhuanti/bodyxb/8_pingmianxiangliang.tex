    \item 平面向量及其运算的概念
    \begin{enumerate}
        \item 给出以下四个命题：
        \ding{172} 对于实数$p$和向量$\vv{a}, \vv{b}$恒有$p(\vv{a} - \vv{b}) = p\vv{a} - p\vv{b}$；
        \ding{173}对于实数$p, q$和向量$\vv{a}$，恒有$(p-q)\vv{a} = p\vv{a} - q\vv{a}$;
        \ding{174}对于实数$p$和向量$\vv{a}, \vv{b}$，若$p\vv{a} = p\vv{b}$，则$\vv{a} = \vv{b}$；
        \ding{175}对于实数$p, q$和向量$\vv{a}, \vv{a} \neq \vv{0}$，若$p \vv{a} \neq q \vv{a}$，则$p \neq q$. 
        其中真命题是\blkda{\ding{172}\ding{173}\ding{175}}
        \begin{jieda}
            \begin{dingautolist}{172}
                \item 由数乘运算对向量的分配率可知为真命题
                \item 由数乘运算对实数的分配率可知为真命题
                \item 反例：$p = 0, \vv{a} \neq \vv{b}$
                \item 由数乘运算的定义可知为真命题
            \end{dingautolist}
        \end{jieda}
        \item 给出以下四个命题：
        \ding{172} $\vv{AB} + \vv{BA} = \vv{0}$；
        \ding{173} $\vv{AB} \cdot \vv{0} = \vv{0}$;
        \ding{174} $\vv{AB} - \vv{AC} = \vv{BC}$；
        \ding{175} $0 \cdot \vv{AB} = 0$. 其中真命题是\blkda{\ding{172}}
        \begin{jieda}
            \begin{dingautolist}{172}
                \item 符合相反向量的性质
                \item 等式右侧应为$0$
                \item 等式右侧应为$\vv{CB}$
                \item 等式右侧应为$\vv{0}$
            \end{dingautolist}
        \end{jieda}
        \item 平面向量$\vv{a}, \vv{b}$共线的充要条件是\dotfill{(\kh{D})}
        \xx{$\vv{a}, \vv{b}$方向相同}{$\vv{a}, \vv{b}$中至少有一个是零向量}{存在$\lambda \in \mathbf{R}$使得$\vv{b} = \lambda \vv{a}$}{存在不完全为零的实数$\lambda_1, \lambda_2$使得$\lambda_1 \vv{a} + \lambda_2 \vv{b} = \vv{0}$}
        \begin{jieda}
            对于C，若$\vv{a} = \vv{0}, \vv{b} \neq \vv{0}$，则两者共线但C不成立，因此C是两向量共线的充分非必要条件.
        \end{jieda}
        \item 设$\vv{a}, \vv{b}$都是非零向量，下列四个条件中，使得$\dfrac{\vv{a}}{|\vv{a}|} = \dfrac{\vv{b}}{|\vv{b}|}$成立的充分条件是\dotfill{(\kh{C})}
        \xx{$\vv{a} = -\vv{b}$}{$\vv{a} // \vv{b}$}{$\vv{a} = 2\vv{b}$}{$\vv{a} // \vv{b}$且$|\vv{a}| = |\vv{b}|$}
        \begin{jieda}
            $\dfrac{\vv{a}}{|\vv{a}|} = \dfrac{\vv{b}}{|\vv{b}|}$即两向量同向，故选C.
        \end{jieda}
          \item $P$ 是 $\bigtriangleup  {ABC}$ 所在平面上一点，满足 $\left| {\vv{PB} - \vv{PC}}\right|  - \left| {\vv{PB} + \vv{PC} - 2\vv{PA}}\right|  = 0$ ，则 $\bigtriangleup  {ABC}$ 的形状是\dotfill{(\kh{A})}
        \xx{直角三角形}{等腰三角形}{等腰直角三角形}{等边三角形}
        \begin{jieda}
            因为 $\left| {\vv{PB} - \vv{PC}}\right|  - \left| {\vv{PB} + \vv{PC} - 2\vv{PA}}\right|  = 0$ ,
            
            即 $\left| \vv{CB}\right|  - \left| {\left( {\vv{PB} - \vv{PA}}\right)  + \left( {\vv{PC} - \vv{PA}}\right) }\right|  = 0$ ,
            
            即 $\left| \vv{CB}\right|  = \left| {\vv{AB} + \vv{AC}}\right|$ ,
            
            即 $\left| {\vv{AB} - \vv{AC}}\right|  = \left| {\vv{AB} + \vv{AC}}\right|$ ,
            
            所以 ${\left( \vv{AB} - \vv{AC}\right) }^{2} = {\left( \vv{AB} + \vv{AC}\right) }^{2}$ ,
            
            即 ${\vv{AB}}^{2} - 2\vv{AB} \cdot  \vv{AC} + {\vv{AC}}^{2} = {\vv{AB}}^{2} + 2\vv{AB} \cdot  \vv{AC} + {\vv{AC}}^{2}$ ,
            
            所以 $\vv{AB} \cdot  \vv{AC} = 0$ ,所以 $\vv{AB} \bot  \vv{AC}$ ,即 ${AB} \bot  {AC}$ ,
            
            所以 $\bigtriangleup  {ABC}$ 为直角三角形.
        \end{jieda}
    \end{enumerate}
