\begin{enumerate}
    \item 直线的倾角与斜率
    \begin{enumerate}
        \item 在 $x$ 轴与 $y$ 轴上截距分别为 $-2,2$ 的直线的倾斜角为\blkda{$\ang{45}$}
        \item 已知直线 ${l}_{1} : \sqrt{3}x + y = 0$ 与直线 ${l}_{2} : {kx} - y + 1 = 0$ ,若直线 ${l}_{1}$ 与直线 ${l}_{2}$ 的夹角是 ${60}^{ \circ  }$ ,则 $k = $\blkda{0 或 $\sqrt{3}$}
        \begin{jieda}
            直线 ${l}_{1} : \sqrt{3}x + y = 0$ 的斜率为 ${k}_{1} =  - \sqrt{3}$，所以直线 ${l}_{1}$ 的倾斜角为 ${120}^{ \circ  }$.
            
            要使直线 ${l}_{1}$ 与直线 ${l}_{2}$ 的夹角是 ${60}^{ \circ  }$，只需直线 ${l}_{2}$ 的倾斜角为 ${0}^{ \circ  }$ 或 ${60}^{ \circ  }$，即$k = 0$ 或 $\sqrt{3}$
        \end{jieda}
        \item 若将直线 $l$ 沿 $x$ 轴正方向平移 3 个单位长度,再沿 $y$ 轴负方向平移 2 个单位长度,又回到了原来的位置,则 $l$ 的斜率是\blkda{$- \dfrac{2}{3}$}
        \begin{jieda}
            由题意可知直线 $l$ 的斜率存在且不为 0
            
            设直线 $l$ 的方程为 $y = {kx} + b\left( {k \neq  0}\right)$ , 则平移后直线的方程为 $y = k\left( {x - 3}\right)  + b - 2 = \left( {{kx} + b}\right)  + \left( {-{3k} - 2}\right)$ ,
            
            可得 ${kx} + b = \left( {{kx} + b}\right)  + \left( {-{3k} - 2}\right)$ , 即 $k =  - \dfrac{2}{3}$ .

            \ps 题干也可以翻译成$(3, -2)$是直线的一个方向向量，因此斜率应为$- \dfrac{2}{3}$
        \end{jieda}
        \item 直线 ${2x}\cos \alpha  - y - 3 = 0\left( {\alpha  \in  \left\lbrack  {\dfrac{\pi }{6},\dfrac{\pi }{3}}\right\rbrack  }\right)$ 的倾斜角的变化范围是\blkda{$\left\lbrack  {\dfrac{\pi }{4},\dfrac{\pi }{3}}\right\rbrack$}
        \begin{jieda}
            直线 ${2x}\cos \alpha  - y - 3 = 0$ 的斜率 $k = 2\cos \alpha$ .

            由于 $\alpha  \in  \left\lbrack  {\dfrac{\pi }{6},\dfrac{\pi }{3}}\right\rbrack$ ,所以 $\dfrac{1}{2} \leq  \cos \alpha  \leq  \dfrac{\sqrt{3}}{2}$，因此 $k = 2\cos \alpha  \in  \left\lbrack  {1,\sqrt{3}}\right\rbrack$ .
            
            设直线的倾斜角为 $\theta$ ,则有 $\tan \theta  \in  \left\lbrack  {1,\sqrt{3}}\right\rbrack$. 由于 $\theta  \in  \lbrack 0,\pi )$，所以 $\theta  \in  \left\lbrack  {\dfrac{\pi }{4},\dfrac{\pi }{3}}\right\rbrack$ ,即倾斜角的变化范围是 $\left\lbrack  {\dfrac{\pi }{4},\dfrac{\pi }{3}}\right\rbrack$ .
        \end{jieda}
        \item 若直线 $l$ 过点 $P\left( {1,0}\right)$ ,且与以 $A\left( {2,1}\right) ,B\left( {0,\sqrt{3}}\right)$ 为端点的线段有公共点,则直线 $l$ 的斜率的取值范围是\blkda[4]{$\left( -\infty , - \sqrt{3}\right\rbrack \cup \left \lbrack 1, + \infty \right)$}
        \begin{jieda}
            如图,当直线 $l$ 过点 $B$ 时,设直线 $l$ 的斜率为 ${k}_{1}$ ,则 ${k}_{1} = \dfrac{\sqrt{3} - 0}{0 - 1} =  - \sqrt{3}$ ; 当直线 $l$ 过点 $A$ 时,设直线 $l$ 的斜率为 ${k}_{2}$ ,则 ${k}_{2} = \dfrac{1 - 0}{2 - 1} = 1$ ,所以要使直线 $l$ 与线段 ${AB}$ 有公共点,则直线 $l$ 的斜率的取值范围是 $\left( -\infty , - \sqrt{3}\right\rbrack \cup \left \lbrack 1, + \infty \right)$ .

            \end{center}
        \end{jieda}
        \item 直线 $x + \left( {{m}^{2} + 1}\right) y + {m}^{2} = 0\left( {m \in  \mathbf{R}}\right)$ 的倾斜角的最小值是\blkda{$\dfrac{3\pi }{4}$}.
        \begin{jieda}
            直线可化为 $y =  - \dfrac{1}{{m}^{2} + 1}x - \dfrac{{m}^{2}}{{m}^{2} + 1}$ .
            
            因为${m}^{2} \geq  0,\therefore {m}^{2} + 1 \geq  1$，所以 $0 < \dfrac{1}{{m}^{2} + 1} \leq  1$，即$- 1 \leq   - \dfrac{1}{{m}^{2} + 1} < 0$ .
            
            故所求倾斜角的最小值是 $\dfrac{3\pi }{4}$ .
        \end{jieda}
    \end{enumerate}
    \item 求直线方程
    \begin{enumerate}
        \item 求符合下列条件的直线方程:
        \begin{enumerate}
            \item 直线过点 $A\left( {-1, - 3}\right)$ ,且斜率为 $- \dfrac{1}{4}$ ;
            \item 直线过点 $P\left( {-1,0}\right)$ 且与直线 ${l}_{1} : \sqrt{3}x - y + 2 = 0$ 的夹角为 $\dfrac{\pi }{6}$.
            \item 直线过点(2,1),且横截距为纵截距的两倍；
        \end{enumerate}
        \begin{jieda}
            \begin{enumerate}
                \item $\because$ 所求直线过点 $A\left( {-1, - 3}\right)$ ,且斜率为 $- \dfrac{1}{4}$ ,

                $\therefore y + 3 =  - \dfrac{1}{4}\left( {x + 1}\right)$ ,即 $x + {4y} + {13} = 0$ .

                \item 直线 ${l}_{1}$ 的倾斜角 $\beta  \in  \lbrack 0,\pi )$ 且 $\tan \beta  = \sqrt{3}$，则 $\beta  = \dfrac{\pi }{3}$ ,
                    
                因为所求直线与直线 ${l}_{1}$ 的夹角为 $\dfrac{\pi }{6}$，所以所求直线的倾斜角为 $\dfrac{\pi }{6}$ 或 $\dfrac{\pi }{2}$ ,

                \begin{dingautolist}{172}
                    \item 当所求直线的倾斜角为 $\dfrac{\pi }{2}$ 时,直线为 $x =  - 1$，即$x+1=0$ ;
                    \item 当所求直线的倾斜角为 $\dfrac{\pi }{6}$ 时,直线为 $y = \dfrac{\sqrt{3}}{3}\left( {x + 1}\right)$，即$x - \sqrt{3}y + 1 = 0$ .
                \end{dingautolist}
                综上,所求直线为 $x + 1 = 0$ 或 $x - \sqrt{3}y + 1 = 0$ .
                \item 
                \begin{dingautolist}{172}
                    \item 当横截距与纵截距都为 0 时,可设直线方程为 $y = {kx}$, 代入点(2,1), 解得 $k = \dfrac{1}{2}$ ,
                    故直线方程为 $y = \dfrac{1}{2}x$ ,即 $x - {2y} = 0$ ;
                    \item 当横截距与纵截距都不为 0 时,可设直线方程为 $\dfrac{x}{a} + \dfrac{y}{b} = 1$ ,
                          由题意可得 $\left\{  \begin{array}{l} \dfrac{2}{a} + \dfrac{1}{b} = 1, \\  a = {2b}, \end{array}\right.$ 解得 $\left\{  \begin{array}{l} a = 4, \\  b = 2, \end{array}\right.$                        
                        $\therefore$ 直线方程为 $\dfrac{x}{4} + \dfrac{y}{2} = 1$ ,即 $x + {2y} - 4 = 0$ ;

                \end{dingautolist}
                综上,所求直线方程为 $x - {2y} = 0$ 或 $x + {2y} - 4 = 0$ .
            \end{enumerate}
        \end{jieda}
        \item 若直线过点 $A\left( {1,2}\right)$ ,且在两坐标轴上截距的绝对值相等，求直线 $l$ 的方程
        \begin{jieda}
            \begin{dingautolist}{172}
                \item 当直线经过原点时,斜率为 $k = \dfrac{2 - 0}{1 - 0} = 2$，所求的直线方程为 $y = {2x}$ ,即 ${2x} - y = 0$ ;
                \item 当直线不过原点时，设所求的直线方程为 $x \pm  y = a$，把点 $A\left( {1,2}\right)$ 代入可得 $1 - 2 = a$ 或 $1 + 2 = a$，求得 $a =  - 1$ 或 $a = 3$ ,故所求的直线方程为 $x - y + 1 = 0$ 或 $x + y - 3 = 0$ .    
            \end{dingautolist}
            综上,所求的直线方程为 ${2x} - y = 0,x - y + 1 = 0$ 或 $x + y - 3 = 0$ .
        \end{jieda}
        \item 在 $\bigtriangleup {ABC}$ 中,已知点 $A\left( {5, - 2}\right) ,B\left( {7,3}\right)$ ,且 ${AC}$ 边的中点 $M$ 在 $y$ 轴上, ${BC}$ 边的中点 $N$ 在 $x$ 轴上,则 ${MN}$ 所在直线的方程为\blkda[3]{${5x} - {2y} - 5 = 0$}
        \begin{jieda}
            设 $C\left( {x,y}\right) ,M\left( {0,m}\right) ,N\left( {n,0}\right)$ ,
            
            因为 $A\left( {5, - 2}\right) ,B\left( {7,3}\right)$ , 所以根据中点坐标公式可得 $\left\{  {\begin{array}{l} \dfrac{x + 5}{2} = 0, \\  \dfrac{y - 2}{2} = m \end{array}\text{ 且 }\left\{  \begin{array}{l} \dfrac{x + 7}{2} = n, \\  \dfrac{y + 3}{2} = 0, \end{array}\right. }\right.$
            
            解得 $x =  - 5,y =  - 3,m =  - \dfrac{5}{2},n = 1$, 即 $C\left( {-5, - 3}\right) ,M\left( {0, - \dfrac{5}{2}}\right) ,N\left( {1,0}\right)$ ,
            
            所以 ${MN}$ 所在直线的方程为 $\dfrac{y + \dfrac{5}{2}}{\dfrac{5}{2}} = \dfrac{x}{1}$, 即 ${5x} - {2y} - 5 = 0$ .
        \end{jieda}
        \item 已知直线 $l$ 过点 $M\left( {2,1}\right)$ ,且分别与 $x$ 轴的正半轴、 $y$ 轴的正半轴交于 $A,B$ 两点, $O$ 为原点
        \begin{enumerate}
            \item 当 $\bigtriangleup  {AOB}$ 面积最小时,求直线 $l$ 的方程.
            \item 当 $\left| {OA}\right|  + \left| {OB}\right|$ 取最小值时，求直线 $l$ 的方程.
            \item 当 $\left| {MA}\right|  \cdot  \left| {MB}\right|$ 取得最小值时，求直线 $l$ 的方程.
            \item 当 ${\left| MA\right| }^{2} + {\left| MB\right| }^{2}$ 取得最小值时，求直线 $l$ 的方程.
        \end{enumerate}
        \begin{jieda}
            因为题干中给出与坐标轴的两个交点的条件，因此考虑设截距式方程：设$A(a, 0), B(0, b)$，$a > 0,b > 0$，则直线 $l : \dfrac{x}{a} + \dfrac{y}{b} = 1$，因为直线 $l$ 过点 $M\left( {2,1}\right)$，所以 $\dfrac{2}{a} + \dfrac{1}{b} = 1$，根据当前约束条件的形式可知求最值应考虑利用平均值不等式.

            \begin{enumerate}
                \item $S_{\bigtriangleup  {AOB}} = \dfrac{ab}{2}$，根据平均值不等式有$\dfrac{ab}{2} = \dfrac{1}{2}\left(\dfrac{2}{a} + \dfrac{1}{b}\right)\cdot ab = \dfrac{a+2b}{2} \geq \sqrt{2ab}$，解得$ab \geq 8$
            
                故 ${S}_{\bigtriangleup {AOB}}$ 的最小值为 $\dfrac{1}{2} \times  {ab} = \dfrac{1}{2} \times  8 = 4$, 当且仅当 $\dfrac{2}{a} = \dfrac{1}{b} = \dfrac{1}{2}$ 时取等号，此时 $a = 4,b = 2$ ,故直线 $l$ 的方程为 $\dfrac{x}{4} + \dfrac{y}{2} = 1$，即 $x + {2y} - 4 = 0$ .

                \item $\left| {OA}\right|  + \left| {OB}\right| = a + b = (a+b)\cdot\left(\dfrac{2}{a} + \dfrac{1}{b}\right) = 2+1+\dfrac{2b}{a}+\dfrac{a}{b} \geq 3+2\sqrt{2}$当且仅当 $a = 2 + \sqrt{2},b = 1 + \sqrt{2}$ 时,等号成立，所以当 $\left| {OA}\right|  + \left| {OB}\right|$ 取最小值时,直线$l$的方程为$\dfrac{x}{2+\sqrt{2}} + \dfrac{y}{1+\sqrt{2}} = 1$，即$x + \sqrt{2}y = 2 + \sqrt{2}$.

                \item 
                \begin{dingautolist}{172}
                    \item $\left| {MA}\right|  \cdot  \left| {MB}\right|  = \left| \vv{MA}\right|  \cdot  \left| \vv{MB}\right| =  - \vv{MA} \cdot  \vv{MB} =  - \left( {a - 2, - 1}\right)  \cdot  \left( {-2,b - 1}\right) = 2\left( {a - 2}\right)  + b - 1 = {2a} + b - 5 = \left( {{2a} + b}\right) \left( {\dfrac{2}{a} + \dfrac{1}{b}}\right)  - 5 = 2\left( {\dfrac{b}{a} + \dfrac{a}{b}}\right)  \geq  4$
                
                    当且仅当 $a = b = 3$ 时取等号,此时直线 $l$ 的方程为 $x + y - 3 = 0$.

                    \item 设$\angle MAO$为$\theta$，则$|MA| = \dfrac{1}{sin\theta}, |MB| = \dfrac{2}{cos\theta}$，故$|MA|\cdot|MB| = \dfrac{4}{sin2\theta} \geq 4$，当$\theta = \dfrac{\pi}{4}$时取等，此时直线 $l$ 的方程为 $x + y - 3 = 0$.
                \end{dingautolist}
                

                \item 
                \begin{dingautolist}{172}
                    \item ${\left| MA\right| }^{2} + {\left| MB\right| }^{2} = {\left( a - 2\right) }^{2} + {\left( 0 - 1\right) }^{2} + {\left( 0 - 2\right) }^{2} + {\left( b - 1\right) }^{2} = (a-2)^2+(b-1)^2+5$

                    根据$\dfrac{2}{a} + \dfrac{1}{b} = 1$可知$b = \dfrac{a}{a-2}$，故${\left| MA\right| }^{2} + {\left| MB\right| }^{2} = (a-2)^2+\dfrac{4}{(a-2)^2}+5 \geq 9$，当且仅当$a = 2+\sqrt{2}$时取等号，此时$b = 1+\sqrt{2}$，直线$l$的方程为$\dfrac{x}{2+\sqrt{2}} + \dfrac{y}{1+\sqrt{2}} = 1$，即$x + \sqrt{2}y = 2 + \sqrt{2}$.

                    \item 设$\angle MAO$为$\theta$，则$|MA| = \dfrac{1}{sin\theta}, |MB| = \dfrac{2}{cos\theta}$，故$|MA|^2+|MB|^2 = \dfrac{1}{sin^2\theta} + \dfrac{4}{cos^2\theta} = 5 + 4tan^2\theta + \dfrac{1}{tan^2\theta}$，当$tan\theta = \sqrt{2}$时取等号，此时$a = 2+\sqrt{2}$，$b = 1+\sqrt{2}$，直线$l$的方程为$\dfrac{x}{2+\sqrt{2}} + \dfrac{y}{1+\sqrt{2}} = 1$，即$x + \sqrt{2}y = 2 + \sqrt{2}$.
                \end{dingautolist}
                
            \end{enumerate}
        \end{jieda}
    \end{enumerate}
    \item 直线的位置
    \begin{enumerate}
        \item 若直线 $l$ 的方程 $y =  - \dfrac{a}{b}x - \dfrac{c}{b}$ 中, ${ab} > 0,{ac} < 0$ ,则此直线必不经过\dotfill{(\kh{C})}
        \xx{第一象限}{第二象限}{第三象限}{第四象限}
        \begin{jieda}
            由 $y =  - \dfrac{a}{b}x - \dfrac{c}{b},{ab} > 0,{ac} < 0$知直线 $l$ 的斜率 $k =  - \dfrac{a}{b} < 0$ ,在 $y$ 轴上的截距 $- \dfrac{c}{b} > 0$

            所以此直线必不经过第三象限.
        \end{jieda}
        \item 直线 $l$ 的方程为 $\left( {a + 1}\right) x + y + 3 - a = 0\left( {a \in  \mathbf{R}}\right)$ ,直线 $l$ 过定点\blkda{$\left( {1, - 4}\right)$},若直线 $l$ 不经过第三象限,则实数 $a$ 的取值范围是\blkda{$\lbrack 3, + \infty )$}.
        \begin{jieda}
            直线 $l : \left( {a + 1}\right) x + y + 3 - a = 0$ 可化为 $a\left( {x - 1}\right)  + x + y + 3 = 0$，令 $\left\{  \begin{array}{l} x - 1 = 0, \\  x + y + 3 = 0, \end{array}\right.$ 解得 $\left\{  \begin{array}{l} x = 1, \\  y =  - 4, \end{array}\right.$，故直线 $l$ 过定点(1, - 4),

            直线 $l$ 可化为 $y =  - \left( {a + 1}\right) x + a - 3$，又直线 $l$ 不经过第三象限，$\therefore \left\{  \begin{array}{l}  - \left( {a + 1}\right)  < 0 \\  a - 3 \geq  0. \end{array} \right.$，解得 $a \geq  3$.
        \end{jieda}
        \item 已知点 $A\left( {2,4}\right)$ , $B\left( {4,2}\right)$ ,直线 $l : y = {kx} - 2$ ,则直线 $l$ 经过定点\blkda{$(0, -2)$}，若直线 $l$ 与线段 ${AB}$ 有公共点,则 $k$ 的取值范围是\blkda{$[1, 3]$}.

        \item 已知直线 $y = \left( {3 - {2k}}\right) x - 6$ 不经过第一象限,则 $k$ 的取值范围为\blkda{$\left\lbrack \dfrac{3}{2}, + \infty \right)$}.
        \begin{jieda}
            由题意知,需满足它在 $y$ 轴上的截距不大于零,且斜率不大于零,则 $\left\{  \begin{array}{l}  - 6 \leq  0, \\  3 - {2k} \leq  0, \end{array}\right.$ 得 $k \geq  \dfrac{3}{2}$ .
        \end{jieda}

    \item 两直线的平行与垂直
    \begin{enumerate}
        \item 若 ${l}_{1} : {3x} - {my} - 1 = 0$ 与 ${l}_{2} : {3\left( {m + 2}\right) }x - {3y} + 1 = 0$ 是两条不同的直线,则 “ $m = 1$ ” 是 “ ${l}_{1}//{l}_{2}$ ”的\dotfill{(\kh{C})}
        \xx{充分不必要条件}{必要不充分条件}{充要条件}{既不充分也不必要条件}
        \begin{jieda}

            而当 $m =  - 3$ 时, ${l}_{1},{l}_{2}$ 重合,故舍去，则 “ $m = 1$ ” 是 “ ${l}_{1}//{l}_{2}$ ” 的充要条件.
        \end{jieda} 

        \item 已知直线 ${l}_{1} : {ax} + {2y} - a + 1 = 0,{l}_{2} : {3x} + \left( {a - 1}\right) y - 2 = 0$ ,则 “ $a =  - 2$ ” 是 “ ${l}_{1}//{l}_{2}$ ” 的\dotfill{(\kh{C})}
        \xx{充分不必要条件}{必要不充分条件}{充要条件}{既不充分也不必要条件}
        \begin{jieda}
            当 $a =  - 2$ ,直线 ${l}_{1} : {2x} - {2y} - 3 = 0,{l}_{2} : {3x} - {3y} - 2 = 0$ ,此时 ${l}_{1}//{l}_{2}$ ,故 “ $a =  - 2$ ” 是 “ ${l}_{1}//{l}_{2}$ ” 的充分条件

            由 ${l}_{1}//{l}_{2}$ ,得 $\left\{  \begin{array}{l} a\left( {a - 1}\right)  - 2 \times  3 = 0 \\   - {2a} - 3\left( {1 - a}\right)  \neq  0 \end{array}\right.$ ,解得 $a =  - 2$ ,故 “ $a =  - 2$ ” 是 “ ${l}_{1}//{l}_{2}$ ” 的必要条件

        \item 已知两直线 ${l}_{1} : \left( {m - 1}\right) x - {6y} - 2 = 0,{l}_{2} : {mx} + y + 1 = 0$ ,若 ${l}_{1} \bot  {l}_{2}$ ,则 $m =$\blkda{3或$-2$}；若 ${l}_{1}//{l}_{2}$ ,则 $m =$\blkda{$\dfrac{1}{7}$}.
        \begin{jieda}

        \item 已知直线 ${l}_{1} : {ax} + \left( {a - 1}\right) y + 3 = 0,{l}_{2} : {2x} + {ay} - 1 = 0$ ,若 ${l}_{1}\bot {l}_{2}$ ,则实数 $a = $\blkda{ 0 或 $-1$}

        \item 已知直线 ${l}_{1}$ 经过点 $A\left( {2,a - 1}\right) ,B\left( {a,4}\right)$ ,且与直线 ${l}_{2} : {2x} + y - 3 = 0$ 平行,则 $a=$\blkda{$-1$}
