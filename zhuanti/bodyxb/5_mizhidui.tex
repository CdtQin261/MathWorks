        \begin{jieda}
            $log_2(x+y) = log_2(xy)$，故$x+y = xy$，即$x = \dfrac{y}{y-1}$，$y = \dfrac{x}{x-1}$，其中$x > 1, y > 1$.\\
            $\dfrac{4x}{x-1} + \dfrac{9y}{y-1} = \dfrac{4x}{x-1} + 9x = 9(x-1) + \dfrac{4}{x-1}+13 \geq 25$，在$x = \dfrac{5}{3}, y = \dfrac{5}{2}$时取等号.
        \end{jieda}
    \end{enumerate}
    \item 指对方程
    \begin{enumerate}
        \item 基础运算
        \begin{enumerate}
            \item $2^{3x} = \dfrac{1}{4}$
            \begin{jieda}
                原方程等价于$2^{3x} = 2^{-2}$，故$3x = -2$，因此有$x = -\dfrac{2}{3}$
            \end{jieda}
            \item $log_2(x-2) = log_2(x^2-x-5)$
            \begin{jieda}
                原方程等价于 $\left \{ \begin{array}{ll}
                     x-2 = x^2-x-5  \\
                     x - 2 > 0 \\
                     x^2-x-5 > 0
                \end{array}\right.$ 解得$x = 3$
            \end{jieda}
        \end{enumerate}
        \item 换元
        \begin{enumerate}
            \item $9^x-4 \cdot 3^x - 45 = 0$
            \begin{jieda}
                换元$t = 3^x, t > 0$，故$t^2-4t - 45 = 0$，故$t = 9$或$-5$（舍），故$x = 2$
            \end{jieda}
            \item $9^x + 9^{-x} - 4(3^x+3^{-x}) + 5 = 0$
            \begin{jieda}
                换元$t = 3^x+3^{-x}, t \geq 2$，故$9^x + 9^{-x} = t^2-2$，故$t^2-4t+3 = 0$，解得$t =3$或$1$（舍），故$3^x = \dfrac{3\pm\sqrt{5}}{2}$，即$x = log_3{\dfrac{3-\sqrt{5}}{2}}$或$log_3{\dfrac{3+\sqrt{5}}{2}}$
            \end{jieda}
            \item $(log_3x)^2+log_{\frac{1}{3}}(x^2)-3 = 0$
            \begin{jieda}
                $(log_3x)^2+log_{\frac{1}{3}}(x^2)-3 = 0$ \\
                首先统一底数，原方程等价于$(log_3x)^2-2log_3x-3 = 0$\\
                换元$t = log_3x, t \in \mathbf{R}$，故$t^2 - 2t - 3 = 0$，解得$t = -1$或$3$，因此$x = \dfrac{1}{3}$或$27$
            \end{jieda}
            \item $log_3(3^x-1) \cdot log_3(3^{x-1}-\dfrac{1}{3})=2$
            \begin{jieda}
                原方程等价于$log_3(3^x-1) \cdot log_3(\dfrac{1}{3}(3^{x}-1))=2$，即$log_3(3^x-1) \cdot (log_3(3^{x}-1)-1)=2$ \\
                换元$t = log_3(3^x-1), t \in \mathbf{R}$，故$t(t-1) = 2$，解得$t = 2$或$-1$ \\
                因此$3^x - 1 = 9$或$\dfrac{1}{3}$，故$x = log_310$或$log_3\dfrac{4}{3}$ 
            \end{jieda}
        \end{enumerate}
        \item 指对转化
        \begin{enumerate}
            \item $x^{lgx} = \dfrac{x^3}{100}$
            \begin{jieda}
                原方程等价于$lgx \cdot lgx = 3lgx - 2$ \\
                换元$t = lgx$，故$t^2 -3t+2 = 0$，解得$t = 1$或$2$，故$x = 10$或$100$
            \end{jieda}
            \item $3 \cdot  {4}^{x+ 2}  = 4 \cdot  {3}^{{x}^{2}}$
            \begin{jieda}
                原方程等价于${4}^{x+1}  = {3}^{{x}^{2}-1}$
    
                两侧同时取对数有$(x+1)log_34 = x^2 - 1$
    
                合并同类项有$(x+1)(x-1-log_34) = 0$
    
                故$x = -1$或$log_312$.
            \end{jieda}
        \end{enumerate}
        \item 换底
        \begin{enumerate}
            \item $9^x + 4^x = \dfrac{5}{2} \cdot 6^x$
            \begin{jieda}
                原方程等价于$(\dfrac{3}{2})^x + (\dfrac{2}{3})^x = \dfrac{5}{2}$，换元$t = (\dfrac{3}{2})^x, t > 0$，故$t + \dfrac{1}{t} = \dfrac{5}{2}$，故$t = 2$或$\dfrac{1}{2}$，故$x = log_{\frac{3}{2}}2$或$-log_{\frac{3}{2}}2$
            \end{jieda}
            \item $3^x + 4^x = 5^x$
            \begin{jieda}
                原方程等价于$(\dfrac{3}{5})^x + (\dfrac{4}{5})^x = 1$ \\
                因为$y = (\dfrac{3}{5})^x + (\dfrac{4}{5})^x $单调递减，因此原方程只有一根$x = 2$
            \end{jieda}
            \item $2 \cdot x^{lg2} \cdot 2^{lgx} - 7x^{lg2} - 4 = 0$
            \begin{jieda}
                首先统一底数有$2 \cdot x^{lg2} \cdot x^{lg2} - 7x^{lg2} - 4 = 0$ \\
                换元$t = x^{lg2} = 2^{lgx}, t > 0$，因此$2t^2-7t-4 = 0$，解得$t = 4$或$-\dfrac{1}{2}$（舍）\\
                故$x = 100$
            \end{jieda}
        \end{enumerate}
    \end{enumerate}
    \item 反函数的定点问题
    \begin{enumerate}
        \item 设函数$f(x)$存在反函数$f^{-1}(x)$，且函数$y = x - f(x)$的图像过点$(1, 2)$，则函数$y = f^{-1}(x) - x$的图像一定过点\blkda{$(-1, 2)$}
        \begin{jieda}
            $y = x - f(x)$，代入$x = 1, y = 2$，则$f(1) = -1$，故$f^{-1}(-1) = 1$. $x = -1$代入$y = f^{-1}(x) - x$，$y = 2$，故过$(-1, 2)$ \\
            \ps 反函数过定点问题本质是$f(x)$过$(a, b)$，则反函数过$(b, a)$，在此基础上，通过代入特值解决问题。
        \end{jieda}
        \item 已知 $f(x)=\dfrac{a-x}{x-a-1}$, 若函数 $y=f(x)$ 的反函数图像的对称中心是 $(-1,3)$, 则实数 $a$ 的值是\blkda{$2$}.
        \begin{jieda}
            由反函数的对称中心知$f(x)$的对称中心为$(3, -1)$，因此$a+1 = 3$，故$a = 2$
        \end{jieda}
    \end{enumerate}
    \item 反函数平移问题
    \begin{enumerate}
        \item 如果函数$y = f(x), x \in \mathbf{R}$的反函数为$y = f^{-1}(x), x \in \mathbf{R}$，则函数$y = f^{-1}(x-1)(x\in \mathbf{R})$的反函数为\blkda[3]{$y = f(x)+1$}.
        \begin{jieda}
            反函数和原函数一个向下平移，令一个向左平移相同长度（长度为负，方向取反），两者可保持关于$y = x$对称.
        \end{jieda}
        \item 已知 $f(x)=\sqrt{x-1}$, 函数 $y=f(x)$ 的反函数为 $y=f^{-1}(x)$, 若 $f^{-1}(x)-a=f(x+a)$有实数根, 则实数 $a$ 的取值范围为\blkda{$\left[\dfrac{3}{4}, +\infty\right)$}.
        \begin{jieda}
            $f^{-1}(x) = x^2 + 1 (x \geq 0)$. \\
            设$g(x) = f(x+a)$，则$g^{-1}(x) = f^{-1}(x)-a$，因为$g(x)$与$g^{-1}(x)$均为增函数，因此其交点必然在$y = x$上。由图象可知，$a$的取值范围形式应为$[m, +\infty)$，$m$为边界情况对应的值.边界情况为$x^2 + 1 - a = x$只有一解，此时$a = \dfrac{3}{4}$，故$a \in \left[\dfrac{3}{4}, +\infty\right)$ 
        \end{jieda}
    \end{enumerate}
\end{enumerate}
