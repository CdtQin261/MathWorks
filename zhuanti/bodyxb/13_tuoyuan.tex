    \item 定义

        \item 已知椭圆 $\dfrac{{x}^{2}}{4} + \dfrac{{y}^{2}}{3} = 1$ 的两个焦点为 ${F}_{1},{F}_{2}$ ,过 ${F}_{2}$ 的直线交椭圆于 $M,N$ 两点,则 $\bigtriangleup {F}_{1}{MN}$ 的周长为\blkda{8}
        \item 已知 $\bigtriangleup {ABC}$ 的周长为 ${12},B\left( {0, - 2}\right) ,C\left( {0,2}\right)$ ,则顶点 $A$ 的轨迹方程为\dotfill{(\kh{A})}

    \item 标准方程
    \begin{enumerate}
        \item \begin{enumerate}
            \item 已知椭圆的两个焦点分别为 ${F}_{1}\left( {0,2}\right) ,{F}_{2}\left( {0, - 2}\right) ,P$ 为椭圆上任意一点, 若 $\left| {{F}_{1}{F}_{2}}\right|$ 是 $\left| {P{F}_{1}}\right| ,\left| {P{F}_{2}}\right|$ 的等差中项,则此椭圆的标准方程为\blkda[3]{$\dfrac{{y}^{2}}{16} + \dfrac{{x}^{2}}{12} = 1$}
            \item 已知椭圆的中心在原点,以坐标轴为对称轴,且经过两点 ${P}_{1}\left( {\sqrt{6},1}\right) ,{P}_{2}\left( {-\sqrt{3}, - \sqrt{2}}\right)$ , 则该椭圆的方程为\blkda[3]{$\dfrac{{x}^{2}}{9} + \dfrac{{y}^{2}}{3} = 1$}
        \end{enumerate}
        \begin{jieda}
            \begin{enumerate}
                \item 由题意 $\left| {P{F}_{1}}\right|  + \left| {P{F}_{2}}\right|  = 2\left| {{F}_{1}{F}_{2}}\right|  = 8 = {2a}$ ,故 $a = 4$ ,又 $c = 2$ ,则 $b = 2\sqrt{3}$ ,

                焦点在 $y$ 轴上,故椭圆的标准方程为 $\dfrac{{y}^{2}}{16} + \dfrac{{x}^{2}}{12} = 1$ .
                \item 设椭圆的方程为 $m{x}^{2} + n{y}^{2} = 1\left( {m > 0,n > 0\text{ ,且 }m \neq  n}\right)$ . 将 ${P}_{1},{P}_{2}$ 代入方程, 得 $\left\{  \begin{array}{l} {6m} + n = 1, \\  {3m} + {2n} = 1, \end{array}\right.$
                解得 $\left\{  \begin{array}{l} m = \dfrac{1}{9}, \\  n = \dfrac{1}{3}. \end{array}\right.$
                所以椭圆的方程为 $\dfrac{{x}^{2}}{9} + \dfrac{{y}^{2}}{3} = 1$ .
            \end{enumerate}
            \ps 根据条件求椭圆方程的主要方法
            \begin{dingautolist}{172}
                \item 定义法:根据题目所给条件确定动点的轨迹满足椭圆的定义，找寻$a,b,c$，写出标准方程.
                \item 待定系数法: 根据题目所给的条件确定椭圆中的 $a,b$ . 当不知焦点在哪一个坐标轴上时,一般可设所求椭圆的方程为 $m{x}^{2} + n{y}^{2} = 1\left( {m > 0,n > 0,m \neq  n}\right)$ ,不必考虑焦点位置,用待定系数法求出 $m,n$ 的值即可.
            \end{dingautolist}
        \end{jieda}

                \item 根据条件可知$c = 3$

                    \item 若焦点在$x$轴上，则$b = 4$，$a^2 = 16+9 = 25$，故标准方程为$\dfrac{x^2}{25} + \dfrac{y^2}{16} = 1$
                    \item 若焦点在$y$轴上，则$a = 4$，$b^2 = 16-9 = 7$，故标准方程为$\dfrac{x^2}{7} + \dfrac{y^2}{16} = 1$
                \end{dingautolist}
                \item 根据椭圆的定义有$2a = \sqrt{\dfrac{1}{4}+\dfrac{9}{4}} + \sqrt{\dfrac{81}{4}+\dfrac{9}{4}} = 2\sqrt{10}$，因此$a^2 = 10, b^2 = 6$，故标准方程为$\dfrac{x^2}{10} + \dfrac{y^2}{6} = 1$
                \item

                    \item 若焦点在$y$轴上，则设标准方程为$\dfrac{x^2}{b^2} + \dfrac{y^2}{a^2} = 1$，代入$a = 3b$，得到$\dfrac{x^2}{b^2} + \dfrac{y^2}{9b^2} = 1$，代入$(3, 0)$，得到$b = 3$，因此$a = 9$，标准方程为$\dfrac{x^2}{9}+\dfrac{y^2}{81} = 1$
                \end{dingautolist}
            \end{enumerate}

        \item 求满足下列条件的椭圆的标准方程:

            \item 焦点在 $x$ 轴上,且一个焦点与短轴两端点的连线互相垂直,且此焦点与长轴较近端点的距离为 $\sqrt{10} - \sqrt{5}$ ；

                \item 将$4{x}^{2} + 9{y}^{2} = {36}$化为标准形式为$\dfrac{x^2}{9}+\dfrac{y^2}{4} = 1$，故其焦点为$(-\sqrt{5}, 0), (\sqrt{5}, 0)$\\

                \item 根据几何关系可知$b = c, a-c = \sqrt{10}-\sqrt{5}$，因此$a = \sqrt{10}, b = c = \sqrt{5}$，故椭圆的标准方程为$\dfrac{x^2}{10}+\dfrac{y^2}{5} = 1$

        \item 设椭圆的中心是坐标原点,长轴在 $x$ 轴上,且离心率等于 $\dfrac{\sqrt{3}}{2}$ ,已知点 $P\left( {0,\dfrac{3}{2}}\right)$ 到这个椭圆上的点的最远距离是 $\sqrt{7}$ ,求这个椭圆方程.
        \begin{jieda}
            $\dfrac{c}{a} = \dfrac{\sqrt{3}}{2}$，故$a = 2b$，因此可设椭圆的标准方程为$\dfrac{x^2}{4k}+\dfrac{y^2}{k} = 1$ \\
            因为点 $P\left( {0,\dfrac{3}{2}}\right)$ 到这个椭圆上的点的最远距离是 $\sqrt{7}$，所以$x^2+\left(y-\dfrac{3}{2}\right)^2 \leq 7$恒成立，且等号可以取到，将椭圆的标准方程代入得$4k-4y^2+\left(y-\dfrac{3}{2}\right)^2 \leq 7$，整理得$4k-7 \leq 3y^2+3y-\dfrac{9}{4}$，故$4k-7$就是$3y^2+3y-\dfrac{9}{4}(y \in [-\sqrt{k}, \sqrt{k}])$的最小值.$y = -\dfrac{1}{2}$时$3y^2+3y-\dfrac{9}{4}$取得最小值$-3$，故$k = 1$，经检验满足要求，因此椭圆的标准方程为$\dfrac{x^2}{4}+y^2 = 1$
        \end{jieda}

        \end{jieda}

    \item 椭圆的性质
    \begin{enumerate}
        \item 若椭圆 ${x}^{2} + m{y}^{2} = 1$ 的焦点在 $y$ 轴上,长轴长是短轴长的两倍,则实数 $m$ 的值是\blkda{$\dfrac{1}{4}$}
        \begin{jieda}
            因为焦点在$y$轴上，故$a^2 = \dfrac{1}{m}, b^2 = 1$，又因为$a = 2b$，所以$\dfrac{1}{m} = 4$，解得$m = \dfrac{1}{4}$
        \end{jieda}
        \item 若椭圆的方程为 $a{x}^{2} + b{y}^{2} + {ab} = 0\left( {a < b < 0}\right)$ ,则其焦点坐标为\blkda[2]{$(0, \pm \sqrt{b-a})$}.
        \begin{jieda}
            将方程化为标准形式有$\dfrac{x^2}{-b}+\dfrac{y^2}{-a} = 1$，因为$a < b < 0$，所以焦点在$y$轴上，为$(0, \pm \sqrt{b-a})$
        \end{jieda}
        \item 若椭圆 $5{x}^{2} - k{y}^{2} = 5$ 的一个焦点是$(0,2)$,则该椭圆的离心率 $e =$ \blkda{$\dfrac{2\sqrt{5}}{5}$}
        \begin{jieda}
            将方程化为标准形式有$x^2 + \dfrac{y^2}{\left(-\dfrac{5}{k}\right)} = 1$，因为焦点为$(0, 2)$，所以$c = 2, b = 1$，故$a^2 = 5$，因此$a = \sqrt{5}$， $e = \dfrac{2}{\sqrt{5}} = \dfrac{2\sqrt{5}}{5}$
        \end{jieda}
        \item 设 ${F}_{1},{F}_{2}$ 是椭圆 $E : \dfrac{{x}^{2}}{{a}^{2}} + \dfrac{{y}^{2}}{{b}^{2}} = 1\left( {a > b > 0}\right)$ 的左、右焦点,过点 ${F}_{1}$ 且斜率为 $\dfrac{\sqrt{3}}{3}$ 的直线交椭圆于点 $P$ ,若 $2\angle P{F}_{1}{F}_{2} = \angle P{F}_{2}{F}_{1}$ ,则椭圆 $E$ 的离心率为\blkda{$\sqrt{3}-1$}
        \begin{jieda}
            由题干可知$\angle PF_1F_2 = \ang{30}, \angle PF_2F_1 = \ang{60}$，因此$PF_1 \perp PF_2$，所以根据几何关系有$F_1F_2 = 2c, PF_2 = c, PF_1 = \sqrt{3}c$，故$2a = (\sqrt{3}+1)c$，故$e = \dfrac{c}{a} = \dfrac{2}{\sqrt{3}+1} = \sqrt{3}-1$
        \end{jieda}
        
        \item 古希腊数学家阿波罗尼奥斯在研究圆锥曲线时发现了椭圆的光学性质:从椭圆的一个焦点射出的光线, 经椭圆反射, 其反射光线必经过椭圆的另一焦点. 设椭圆方程 $\dfrac{{x}^{2}}{{a}^{2}} + \dfrac{{y}^{2}}{{b}^{2}} = 1\left( {a > b > 0}\right) ,{F}_{1}$ 、 ${F}_{2}$ 为其左、右焦点,若从右焦点 ${F}_{2}$ 发出的光线经椭圆上的点 ${A}$ 和点 $B$ 反射后,满足 ${AB} \bot  {AD},\cos \angle {ABC} = \dfrac{4}{5}$ ,则该椭圆的离心率为\blkda{$\dfrac{\sqrt{2}}{2}$}.

            由题意, 可作图如下:

            则 $\cos \angle {AB}{F}_{1} = \dfrac{4}{5} = \dfrac{\left| AB\right| }{\left| B{F}_{1}\right| },\sin \angle {AB}{F}_{1} = \sqrt{1 - {\cos }^{2}\angle {AB}{F}_{1}} = \dfrac{3}{5} = \dfrac{\left| A{F}_{1}\right| }{\left| B{F}_{1}\right| }$ ,即 $\left| {AB}\right|  : \left| {A{F}_{1}}\right|  : \left| {B{F}_{1}}\right|  = 4 : 3 : 5$ , 可设 $\left| {AB}\right|  = {4k}$ , $\left| {A{F}_{1}}\right|  = {3k}$ , $\left| {B{F}_{1}}\right|  = {5k}$ ,

            \item 若 $\angle {F}_{1}P{F}_{2} = {60}^{ \circ  }$ ,求 $\bigtriangleup P{F}_{1}{F}_{2}$ 的面积
        \end{enumerate*}
        \begin{jieda}
            \begin{enumerate}
                \item 易知动点$P$的轨迹满足椭圆的定义，且有$a = 2c = 2$.因此$P$ 的轨迹方程为$\dfrac{x^2}{4} + \dfrac{y^2}{3} = 1$
                \item 设$|PF_1| = t$，则$|PF_2| = 4-t$，根据余弦定理有$4 = t^2 + (4-t)^2 - t(4-t)$，即$4 = 16 - 3t(4-t)$，因此$t(4-t) = 4$，故$S = \dfrac{1}{2} \times t(4-t) \times \dfrac{\sqrt{3}}{2} = \sqrt{3}$
            \end{enumerate}

        \item 设点 $P$ 为椭圆 $C : \dfrac{{x}^{2}}{{a}^{2}} + \dfrac{{y}^{2}}{b^2} = 1\left( {a > b > 0}\right)$ 上一点, ${F}_{1},{F}_{2}$ 分别为 $C$ 的左、右焦点,且 $\angle {F}_{1}P{F}_{2} = \theta$ , 则 $\bigtriangleup  P{F}_{1}{F}_{2}$ 的面积为\blkda{$b^2tan\dfrac{\theta}{2}$}（用$a, b, \theta$表示）
        \begin{jieda}
            设$|PF_1| = t$，则$|PF_2| = 2a-t$，根据余弦定理有$4c^2 = t^2 + (2a-t)^2 - 2t(2a-t)cos\theta$，即$4c^2 = 4a^2 - 2(1+cos\theta)t(2a-t)$，因此$t(2a-t) = \dfrac{2b^2}{1+cos\theta}$，故$S = \dfrac{1}{2} \times t(4-t) \times sin\theta = b^2tan\dfrac{\theta}{2}$
        \end{jieda}
        \item 已知 ${F}_{1},{F}_{2}$ 为椭圆 $\dfrac{{x}^{2}}{{a}^{2}} + \dfrac{{y}^{2}}{{b}^{2}} = 1\left( {a > b > 0}\right)$ 的左、右焦点,椭圆的离心率为 $\dfrac{1}{2},M$ 为椭圆上一动点,则 $\angle {F}_{1}M{F}_{2}$ 的最大值为\blkda{$\dfrac{\pi}{3}$}
