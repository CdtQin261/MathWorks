\subsection*{引例}

设椭圆 $C : \dfrac{{x}^{2}}{{a}^{2}} + \dfrac{{y}^{2}}{{b}^{2}} = 1\left( {a > b > 0}\right)$ 过点 $M\left( {\sqrt{2},1}\right)$ ，且左焦点为 ${F}_{1}\left( {-\sqrt{2},0}\right)$
\begin{enumerate}[(1)]
    \item 求椭圆 $C$ 的方程
    \item 当过点 $P\left( {4,1}\right)$ 的动直线 $l$ 与椭圆 $C$ 相交于两不同点 $A,B$ 时,在线段 ${AB}$ 上取点 $Q$ , 满足 $\left| \vv{AP}\right|  \cdot  \left| \vv{QB}\right|  = \left| \vv{AQ}\right|  \cdot  \left| \vv{PB}\right|$ ,证明: 点 $Q$ 总在某定直线上
\end{enumerate}
\begin{jieda}
    \begin{enumerate}[(1)]
        \item $\left\{\begin{array}{l}
             \dfrac{2}{{a}^{2}} + \dfrac{1}{{b}^{2}} = 1  \\
             {c}^{2} = 2 \\
             {c}^{2} = {a}^{2} - {b}^{2}
        \end{array}\right.$  , 解得${a}^{2} = 4,{b}^{2} = 2$ ,所求椭圆方程为 $\dfrac{{x}^{2}}{4} + \dfrac{{y}^{2}}{2} = 1$

        \item 设点 ${Q}\text{ 、 }{A}\text{ 、 }{B}$ 的坐标分别为 $\left( x_0, y_0\right) ,\left( {{x}_{1},{y}_{1}}\right) ,\left( {{x}_{2},{y}_{2}}\right)$

        由题设知 $\left| \vv{AP}\right| ,\left| \vv{PB}\right| ,\left| \vv{AQ}\right| ,\left| \vv{QB}\right|$ 均不为零，则根据三角形相似可以将线段长度乘积转换为坐标差的乘积，即$\left| \vv{AP}\right|  \cdot  \left| \vv{QB}\right|  = \left| \vv{AQ}\right|  \cdot  \left| \vv{PB}\right| \Longleftrightarrow  (4-x_1) \cdot (x_2-x_0) = (x_0-x_1) \cdot (4-x_2) \Longleftrightarrow (4+x_0)(x_1+x_2) - 2x_1x_2-8x_0 = 0$

        易知直线斜率存在，设直线方程为 $y - 1 = k\left( {x - 4}\right)$ ,联立椭圆方程 $\dfrac{{x}^{2}}{4} + \dfrac{{y}^{2}}{2} = 1$，得 $\left( {1 + 2{k}^{2}}\right) {x}^{2} - \left( {{16}{k}^{2} - {4k}}\right) x + {32}{k}^{2} - {16k} - 2 = 0.$

        当$\Delta > 0$时由韦达定理, $\left\{\begin{array}{l}
             {x}_{1} + {x}_{2} = \dfrac{{16}{k}^{2} - {4k}}{1 + 2{k}^{2}}  \\
             {x}_{1}{x}_{2} = \dfrac{{32}{k}^{2} - {16k} - 2}{1 + 2{k}^{2}}
        \end{array}\right.$
        
        代入$(4+x_0)(x_1+x_2) - 2x_1x_2-8x_0 = 0$，可得$k = \dfrac{1-2x_0}{x_0-4}$，代入直线方程, 得$2{x}_{0} + {y}_{0} - 2 = 0$，即点 $Q\left( {{x}_{0},{y}_{0}}\right)$ 在定直线 ${2x} + y - 2 = 0$ 上.
    \end{enumerate}
    \ps 将线段长度乘积（比值）根据三角形相似转化为坐标差的乘积（比值）是代数表达的关键.
\end{jieda}

\subsection*{调和点列与调和线束}
\subsubsection*{调和点列}
\begin{dingautolist}{172}
    \item $A,B,C,D$共线，若满足$\dfrac{2}{AC} = \dfrac{1}{AB} + \dfrac{1}{AD}$，则称其为调和点列.
    \item 其等价定义为$B,D$分$\vv{AC}$的比互为相反数，
    
    即$\left\{\begin{array}{l}
         \vv{AB} = \lambda \vv{BC}  \\
         \vv{AD} = \mu \vv{DC} \\
         \lambda + \mu = 0
    \end{array}\right.$，将上式变形消参有$\left\{\begin{array}{l}
         \vv{AB} = \dfrac{\lambda}{1+\lambda} \vv{AC} \\
         \vv{AD} = \dfrac{\lambda}{\lambda-1} \vv{AC} \\
    \end{array}\right.$，取模取倒数即是$\dfrac{2}{AC} = \dfrac{1}{AB} + \dfrac{1}{AD}$.
    \item 对$\left\{\begin{array}{l}
         \vv{AB} = \lambda \vv{BC}  \\
         \vv{AD} = \mu \vv{DC} \\
         \lambda + \mu = 0
    \end{array}\right.$取模交叉相乘并消参可以得到调和点列的另一个常见表达形式：$|AD| \cdot |BC| = |AB| \cdot |CD|$

    上面的例题中的$A,Q,B,P$即为调和点列.
    \item 特别地，当$B$为$A, C$中点时，$A,B,C,\infty$为调和点列
\end{dingautolist}

\subsection*{极点与极线的定义与代数定义}
\begin{dingautolist}{172}
    \item 定义：不在二次曲线上的一点 $P$ 作直线 $l$ 交二次曲线于 $M$ 、 $N$ 两点，则在 $l$ 上有且只有一点 $Q$ ,使得 $P\text{、}Q\text{、}M\text{、}N$ 构成一调和点列) 当 $l$ 绕着 $P$ 旋转时, $Q$ 的轨迹是一条直线 $p$ (或一部分),这条直线 $p$ 叫做点 $P$ 关于二次曲线的极线，而 $P$ 叫做 $p$ 关于该曲线的极点.

    反之，连接极点$P$和极线上的点$Q$，和二次曲线交于$M,N$，则$P,M,Q,N$为调和点列.

    \begin{center}
        \includegraphics[width=0.85\linewidth]{bodyxb//src/jixiandingyi.png}
    \end{center}
    若$P$在二次曲线上，则定义该处的切线为极线(此时是上面的极限情况).
    \item 代数定义：对于圆锥曲线 $A{x}^{2} + B{y}^{2} + {Dx} + {Ey} + F = 0$ (圆、椭圆、双曲线、抛物线) 上的任意一点 $P\left( {{x}_{0},{y}_{0}}\right)$ ,将曲线方程中的  ${x}^{2}$  替换成  ${x}_{0}x$ 、 ${y}^{2}$  替换成  ${y}_{0}y$ 、 $x$  替换成  $\dfrac{{x}_{0} + x}{2}$ 、 $y$  替换成  $\dfrac{{y}_{0} + y}{2}$ ,就得到一个直线的方程,这条直线就称点 $P$ 的 “极线”. 反之, $P$ 称作这条直线的 “极点”.
    \begin{enumerate}
        \item 对于椭圆 $\dfrac{{x}^{2}}{{a}^{2}} + \dfrac{{y}^{2}}{{b}^{2}} = 1$ ,与点 $P\left( {{x}_{0},{y}_{0}}\right)$ 对应的极线方程为 $\dfrac{{x}_{0}x}{{a}^{2}} + \dfrac{{y}_{0}y}{{b}^{2}} = 1$
        \item 对于双曲线 $\dfrac{{x}^{2}}{{a}^{2}} - \dfrac{{y}^{2}}{{b}^{2}} = 1$ ,与点 $P\left( {{x}_{0},{y}_{0}}\right)$ 对应的极线方程为 $\dfrac{{x}_{0}x}{{a}^{2}} - \dfrac{{y}_{0}y}{{b}^{2}} = 1$ 
        \item 对于抛物线 ${y}^{2} = {2px}$ ,与点 $P\left( {{x}_{0},{y}_{0}}\right)$ 对应的极线方程为 ${y}_{0}y = p\left( {{x}_{0} + x}\right)$
    \end{enumerate}
\end{dingautolist}

\subsubsection*{配极原则}
对于同一条二次曲线 $C$，如果点 $P$ 的极线经过点 $Q$ ,那么点 $Q$的极线经过点 $P$ .

以椭圆 $C : \dfrac{{x}^{2}}{{a}^{2}} + \dfrac{{y}^{2}}{{b}^{2}} = 1\left( {a > b > 0}\right)$ 为例，证明配极原则
\begin{jieda}
    设 $P\left( {{x}_{1},{y}_{1}}\right) ,Q\left( {{x}_{2},{y}_{2}}\right)$ ,则点 $P$ 关于 $C$ 的极线为直线 $p : \dfrac{{x}_{1}x}{{a}^{2}} + \dfrac{{y}_{1}y}{{b}^{2}} = 1$ ,
    点 $Q$ 关于 $C$ 的极线为直线 $q : \dfrac{{x}_{2}x}{{a}^{2}} + \dfrac{{y}_{2}y}{{b}^{2}} = 1$ ,
    因为点 $Q$ 在直线 $p$ 上, 故$\dfrac{{x}_{1}{x}_{2}}{{a}^{2}} + \dfrac{{y}_{1}{y}_{2}}{{b}^{2}} = 1$，因此点 $P$ 在直线 $q$ 上.
\end{jieda}

\subsubsection*{极点与极线的位置关系}
\begin{dingautolist}{172}
    \item 对于同一条二次曲线 $C$，如果点 $P$可以作两条切线，切点分别为$A,B$，则$P$的极线为$AB$
    \item 对于同一条二次曲线 $C$，如果点 $P$做不出任何一条切线（不考虑双曲线的对称中心），则$P$的极线为与$C$相离
\end{dingautolist}
\begin{jieda}
    \begin{dingautolist}{172}
        \item 因为$PA, PB$是切线，所以根据极线的定义可知$P$在$A$的极线上，$P$在$B$的极线上，故根据配极原则可知$A,B$在$P$的极线上，故$AB$为$P$的极线.
        \item 假设$P$的极线和$C$有公共点$Q$，则$Q$的极线是以$Q$为切点的切线，根据配极原则可知$P$在$Q$的极线上，矛盾，故$P$的极线为与$C$相离.
    \end{dingautolist}
\end{jieda}
