\chapter*{抛物线专题练习}
\begin{enumerate}
    \item 抛物线定义的应用
    \begin{enumerate}
        \item 抛物线 ${x}^{2} = \dfrac{1}{4}y$ 的准线方程为\blkda[3]{$y =  - \dfrac{1}{16}$}
        \begin{jieda}
            由抛物线的标准方程可得,抛物线的焦点位于 $y$ 轴正半轴上,焦点坐标为 $\left( {0,\dfrac{1}{16}}\right)$ ,准线方程为 $y =  - \dfrac{1}{16}$ .
        \end{jieda}
        \item 已知抛物线 ${x}^{2} = {2py}\left( {p > 0}\right)$ 上的一点 $M\left( {{x}_{0},1}\right)$ 到其焦点的距离为 2,则该抛物线的焦点到其准线的距离为\blkda{2}
        \begin{jieda}
            由题可知,抛物线准线为 $y =  - \dfrac{p}{2}$ ,可得 $1 + \dfrac{p}{2} = 2$ ,解得 $p = 2$ ,所以该抛物线的焦点到其准线的距离为 $p = 2$ .
        \end{jieda}
        \item 已知抛物线 $y = m{x}^{2}\left( {m > 0}\right)$ 上的点 $\left( {{x}_{0},2}\right)$ 到该抛物线焦点 $F$ 的距离为 $\dfrac{11}{4}$ ,则 $m$ 等于\blkda{$\dfrac{1}{3}$}
        \begin{jieda}
            由题意知,抛物线 $y = m{x}^{2}\left( {m > 0}\right)$ 的准线方程为 $y =  - \dfrac{1}{4m}$
            
            根据抛物线的定义,可得点 $\left( {{x}_{0},2}\right)$ 到焦点 $F$ 的距离等于到准线 $y =  - \dfrac{1}{4m}$ 的距离，可得 $2 + \dfrac{1}{4m} = \dfrac{11}{4}$ ,解得 $m = \dfrac{1}{3}$ .
        \end{jieda}
        \item 已知点 $F$ 为抛物线 ${y}^{2} = {2px}\left( {p > 0}\right)$ 的焦点,点 $P$ 在抛物线上且横坐标为8, $O$ 为坐标原点,若 $\bigtriangleup {OFP}$ 的面积为 $2\sqrt{2}$ ,则该抛物线的准线方程为\blkda{$x =  - 1$}
        \begin{jieda}
            抛物线 ${y}^{2} = {2px}\left( {p > 0}\right)$ 的焦点 $F\left( {\dfrac{p}{2},0}\right)$
            
            将点 $P$ 的横坐标代入抛物线得 ${y}^{2} = {16p}$ ,可得 $y =  \pm  4\sqrt{p}$ ,不妨令 $P\left( {8,4\sqrt{p}}\right)$，则 ${S}_{\bigtriangleup {OFP}} = \dfrac{1}{2} \times  \dfrac{p}{2} \times  4\sqrt{p} = p\sqrt{p} = 2\sqrt{2}$ ,解得 $p = 2$ ,
            
            则抛物线方程为 ${y}^{2} = {4x}$ ,其准线方程为 $x =  - 1$ .
        \end{jieda} 
        \item 已知 $O$ 为坐标原点,抛物线 $C : {y}^{2} = {2px}\left( {p > 0}\right)$ 的焦点为 $F,P$ 为 $C$ 上一点, ${PF}$与 $x$ 轴垂直, $Q$ 为 $x$ 轴上一点,且 ${PQ}\bot {OP}$ . 若 $\left| {FQ}\right|  = 6$ ,则 $C$ 的准线方程为\blkda{$x =  - \dfrac{3}{2}$}.
        
        \begin{jieda}
            由题易得 $\left| {OF}\right|  = \dfrac{p}{2},\left| {PF}\right|  = p,\angle {OPF} = \angle {PQF}$，所以 $\tan \angle {OPF} = \tan \angle {PQF}$，所以 $\dfrac{\left| OF\right| }{\left| PF\right| } = \dfrac{\left| PF\right| }{\left| FQ\right| }$，即 $\dfrac{\dfrac{p}{2}}{p} = \dfrac{p}{6}$ ,
            
            解得 $p = 3\left( {p = 0\text{ 舍去 }}\right)$ ,所以 $C$ 的准线方程为 $x =  - \dfrac{3}{2}$ .
        \end{jieda}
        \item 已知抛物线 $C : {y}^{2} = {4x}$ ,焦点为 $F$ ,点 $M$ 为抛物线 $C$ 上的点,且 $\left| {FM}\right|  = 6$ ,则 $M$ 的横坐标是\blkda{5}, 作 ${MN}\bot x$ 轴于 $N$ ,则 ${S}_{\bigtriangleup {FMN}} =$ \blkda{$4\sqrt{5}$}.
        \begin{jieda}
            因为抛物线的方程为 ${y}^{2} = {4x}$，故 $p = 2$ 且 $F\left( {1,0}\right)$，因为 $\left| {FM}\right|  = 6$ ,所以 ${x}_{M} + \dfrac{p}{2} = 6$，解得 ${x}_{M} = 5$，故 ${y}_{M} =  \pm  2\sqrt{5}$，所以 ${S}_{\bigtriangleup {FMN}} = \dfrac{1}{2} \times  \left( {5 - 1}\right)  \times  2\sqrt{5} = 4\sqrt{5}$ .
        \end{jieda}
        \item 设 $M$ 是抛物线 ${y}^{2} = {4x}$ 上的一点, $F$ 是抛物线的焦点, $O$ 是坐标原点,若 $\angle {OFM} = {120}^{ \circ  }$ ,则 $\left| {FM}\right|$ 等于\blkda{4}
        \begin{jieda}

            易知 $\left| {EF}\right|  = 2$ ,则 $\left| {FM}\right|  = \left| {FN}\right|  = 2\left| {EF}\right|  = 4$ .

        \item 已知 $F$ 是抛物线 ${y}^{2} = {16x}$ 的焦点, $M$ 是抛物线上一点, ${FM}$ 的延长线交 $y$ 轴于点 $N$ ,若 ${3\vv{FM}} = 2\vv{MN}$ ,则 $\left| {FN}\right|  =$ \blkda{16}.

            易知焦点 $F$ 的坐标为(4,0),准线 $l$ 的方程为 $x =  - 4$

        \item 已知抛物线 $C : {y}^{2} = {2px}\left( {p > 0}\right)$ 的焦点为 $F$ ,准线为 $l,A,B$ 是抛物线 $C$ 上的两个动点,且 ${AF}\bot {AB}$，$\angle {ABF} = {30}^{ \circ  }$ ,设线段 ${AB}$ 的中点 $M$ 在准线 $l$ 上的射影为点 $N$ ,则 $\dfrac{\left| MN\right| }{\left| AB\right| }$ 的值是\blkda{$\dfrac{\sqrt{3}}{2}$}.

            在梯形 ${ABED}$ 中, $2\left| {MN}\right|  = \left| {AD}\right|  + \left| {BE}\right|  = a + b$.

        \end{jieda}

        \end{jieda}
            
        \item 若 $P$ 是抛物线 ${y}^{2} = {8x}$ 上的动点, $P$ 到 $y$ 轴的距离为 ${d}_{1}$ ,到圆 $C : {\left( x + 3\right) }^{2} + {\left( y - 3\right) }^{2} = 4$ 上动点 $Q$ 的距离为 ${d}_{2}$ ,则 ${d}_{1} + {d}_{2}$ 的最小值为\blkda{$\sqrt{34} - 4$}.

        \item 分别求满足下列条件的抛物线的标准方程.

                所以 $\left| {AC}\right|  = \sqrt{1+\dfrac{1}{t^2}}|y_1-y_3| = \sqrt{\left( {1 + \dfrac{1}{{t}^{2}}}\right) \left\lbrack  {{\left( {y}_{1} + {y}_{3}\right) }^{2} - 4{y}_{1}{y}_{3}}\right\rbrack  }$
                
                $= \sqrt{\left( {1 + \dfrac{1}{{t}^{2}}}\right)  \cdot  \dfrac{{16} + {16}{t}^{2}{x}_{1} + {16t}{y}_{1}}{{t}^{2}}} = \sqrt{\left( {1 + \dfrac{1}{{t}^{2}}}\right)  \cdot  \dfrac{{16} + 4{t}^{2}{y}_{1}^{2} + {16t}{y}_{1}}{{t}^{2}}}$
                
                $= \dfrac{2\sqrt{{t}^{2} + 1}}{{t}^{2}} \cdot  \left| {t{y}_{1} + 2}\right| = \dfrac{2\sqrt{{t}^{2} + 1}}{{t}^{2}} \cdot  \left( {t{y}_{1} + 2}\right) .$
                
                同理可得 $\left| {BD}\right|  = \dfrac{2\sqrt{{t}^{2} + 1}}{{t}^{2}} \cdot  \left( {t{y}_{2} + 2}\right)$ ,
                
                所以 $\left| {AC}\right|  + \left| {BD}\right|  = \dfrac{2\sqrt{{t}^{2} + 1}}{{t}^{2}} \cdot  \left\lbrack  {t\left( {{y}_{1} + {y}_{2}}\right)  + 4}\right\rbrack   = \dfrac{8\sqrt{{t}^{2} + 1}}{{t}^{2}}\left( {{t}^{2} + 1}\right)  = 8\sqrt{\dfrac{{\left( {t}^{2} + 1\right) }^{3}}{{t}^{4}}}$ ,
                
                令 $f\left( x\right)  = \dfrac{{\left( x + 1\right) }^{3}}{{x}^{2}},x > 0$ ,则 ${f}^{\prime }\left( x\right)  = \dfrac{{\left( x + 1\right) }^{2}\left( {x - 2}\right) }{{x}^{3}},x > 0$ ,
                
                所以当 $x \in  \left( {0,2}\right)$ 时, ${f}^{\prime }\left( x\right)  < 0,f\left( x\right)$ 单调递减；当 $x \in  \left( {2, + \infty }\right)$ 时, ${f}^{\prime }\left( x\right)  > 0,f\left( x\right)$ 单调递增.
                
                所以当 $x = 2$ 时, $f\left( x\right)$ 取得最小值,即当 $t =  \pm  \sqrt{2}$ 时, $\left| {AC}\right|  + \left| {BD}\right|$ 的最小值为 ${12}\sqrt{3}$ .
            \end{enumerate}
