    \item 设函数$f(x)=\dfrac{1}{\sqrt{ax^2+ax+1}}$分别满足下列条件, 求实数$a$的取值范围: 
    \begin{enumerate}
    \item 定义域是$\mathbf{R}$; 
    \item 值域是$(0,+\infty)$. 
    \end{enumerate}

        \begin{enumerate}
        \item $ax^2+ax+1>0$恒成立. \\

        故$a\geq 4$. 
        \end{enumerate}
    \end{jieda}
\end{enumerate}
\item 抽象函数的定义域
\begin{enumerate}
    \item 已知函数$y = f(x)$的定义域是$[-1, 1]$，则函数$y = f\left(\dfrac{1}{x}\right)$的定义域是 \blkda[3]{$(-\infty, -1] \cup [1, +\infty)$}
    \item 已知$y = f(2x+1)$的定义域为$(1, 3]$，则$y = f(x+1)$的定义域为\blkda{$(2, 6]$}. 
    \begin{jieda}
        因为$y = f(2x+1)$的定义域为$(1, 3]$，所以$y = f(x)$的定义域是$(3,7]$，故$y = f(x+1)$的定义域为$(2, 6]$. \\
        \ps 此类题目的桥梁是\textbf{$f$括号内的取值范围是一致的}.
    \end{jieda}
    \item 若函数$f(2x-1)$的定义域为$[-3, 1]$，则$y = \dfrac{f(3-4x)}{\sqrt{x-1}}$的定义域为\blkda{$\left(1, \dfrac{5}{2}\right]$}
    \begin{jieda}
        $f(2x-1)$的定义域为$[-3, 1] \Longleftrightarrow f(x)$的定义域为$[-7, 1]$，故应有$\left\{\begin{array}{l}
             3-4x \in [-7, 1] \\
             x - 1 > 0
        \end{array}\right.$，解集为$\left(1, \dfrac{5}{2}\right]$.
    \end{jieda}
    \item 已知$f(x)=x+2, x\in[1,4]$, 则$y=f(x^2)+f^2(x)$的最大值是\blkda{22}. 
    \begin{jieda}
    先看定义域: $f(x)$的定义域为$[1,4]$，因此$f(x^2)$中$x^2\in[1,4]$, 故$f(x^2)$的定义域为$x\in[-2,2]$. 而$f^2(x)$的定义域与$f(x)$的定义域相同.\\
    两者相加，定义域取交集，因此$y=f(x^2)+f^2(x)$的定义域为$[1, 2]$ \\
    于是$y=x^2+2+(x+2)^2=2x^2+4x+6=2(x+1)^2+4\in[12,22]$. 
    \end{jieda}
\end{enumerate}

\item 奇偶性的概念
\begin{enumerate}
    \item $f(x) = ax^2+bx$是定义在$[a-1, 2a]$上的偶函数，则$a+b = $\blkda{$\dfrac{1}{3}$}
    \begin{jieda}
        定义域对称可以推出$a = \dfrac{1}{3}$，多项式函数的性质推出$b = 0$.
    \end{jieda}
    \item “$a+b=0$”是“函数$f(x)=x\abs{x+a}+b$是奇函数”的\blkda[2]{必要不充分}条件
    \begin{jieda}
    奇函数若在$x=0$处有定义, 则此处的函数值为零, 即$f(0)=0$, 于是$b=0$. \\
    故$f(x)=x\abs{x+a}$. 于是由$f(x)$为奇函数，不难得到$a=0$, 比如$f(-a)+f(a)=0$. 
    \end{jieda}
    \item 已知函数$f(x) = \left \{ \begin{array}{ll}
        x^2 + x & x \leq 0 \\
        ax^2 + bx & x > 0
    \end{array}\right.$为奇函数，则$a+b = $\blkda{0}
    \begin{jieda}
        代入特值有$f(1) = a+b$，$f(-1) = 0$，因为$f(x)$是奇函数，所以有$a+b = 0$ \\
        \ps 根据奇偶性求参数可以考虑代入特值（必要条件），但最后需要检验是否充要（本题只问$a+b$并不涉及）.
    \end{jieda}
    \item 已知函数$f(x)=ax^5+bx^3$, 且$f(-2)=10$, 则$f(2)$=\blkda{$-10$}
    \begin{jieda}
        奇函数相加仍然是奇函数.
    \end{jieda}
    \item 设$f(x)$和$g(x)$分别是$\mathbb{R}$上的偶函数和奇函数, 则下列结论恒成立的是\dotfill(\kh{A})
    \xx{$f(x)+\abs{g(x)}$是偶函数}{$f(x)-\abs{g(x)}$是奇函数}{$\abs{f(x)}+g(x)$是偶函数}{$\abs{f(x)}-g(x)$是奇函数}
    \begin{jieda}
    $g(x)$为奇函数，则$|g(x)|$为偶函数，偶函数$+$偶函数还是偶函数，故$A, B$都是偶函数，$A$正确，$B$错误。$f(x)$是偶函数，$|f(x)|$还是偶函数。奇函数$+$偶函数的结果是不确定的（如果两个函数都不是“既是奇函数又是偶函数”的话，结果必然非奇非偶函数）。故$C,D$都错.
    \end{jieda}
    \item 如果 “若 $p$ ,则 $q$ ” 和 “若 $q$ ,则 $p$ ” 中有且仅有一个真命题命题,称 $p$ 与 $q$ 具有 “ $U-$关系”.已知函数 $y = f\left( x\right)$ 的定义域为 $\mathbf{R},p : y = f\left( x\right)$ 为偶函数,则 $p$ 与下列选项中的 $q$ 具有 “$U-$关系” 的为\dotfill{(\kh{C})}
    \xx{$q$ : 对任意 $x \in  \mathbf{R}$ 都有 $- f\left( x\right)  \geq  f\left( x\right)$}{$q$ : 对任意 $x \in  \mathbf{R}$ 都有 $f\left( {-x}\right)  \geq  f\left( x\right)$}{$q$ : 对任意 $x \in  \mathbf{R}$ 都有 $f\left( {-x}\right)  = \left| {f\left( x\right) }\right|$}{$q$ : 对任意 $x \in  \mathbf{R}$ 都有 $f\left( {-x}\right)  = f\left( \left| x\right| \right)$}
    \begin{jieda}
        $q_A: f(x) \leq 0$，$p$是$q_A$的既非充分也非必要条件.

        $q_B$迭代可得$f(x) \geq f(-x)$，故$q_B: f(x) = f(-x)$，因此$p$是$q_A$的充要条件.

        $q_C$迭代可得$f(x) = |f(-x)|$，因此$f(x) \geq 0$，故$f(-x) = f(x)$，故$q_C: f(-x) = f(x)$且$f(x) \geq 0$，故$p$是$q_C$的充分非必要条件.

        $q_D$迭代可得$f(x) = f(|-x|) = f(|x|)$，故$q_D: f(x) = f(-x)$，因此$p$是$q_A$的充要条件.
    \end{jieda}

    \item 已知函数$f(x) = x^3+2x$，若$m > 0, n>0$且有$f(2m)+f(n-1) = f(0)$，则$\dfrac{1}{m}+\dfrac{2}{n}$的最小值为\blkda{8}
    \begin{jieda}
        $f(x)$为奇函数（单调递增），因此$f(2m)+f(n-1) = f(0) \Longleftrightarrow 2m+n=1$，故$\dfrac{1}{m}+\dfrac{2}{n} = \left(\dfrac{1}{m}+\dfrac{2}{n}\right) \cdot (2m+n) = 4+\dfrac{n}{m}+\dfrac{4m}{n} \geq 8$，当$n = \dfrac{1}{2}, m = \dfrac{1}{4}$时取最小值.
    \end{jieda}
\end{enumerate}

\item 反比例型函数的对称中心
\begin{enumerate}
    \item 函数$y=\dfrac{3-x}{1+2x}$的对称中心是\blkda{$\left(-\dfrac{1}{2},-\dfrac{1}{2}\right)$}, 值域是\blkda[2]{$\left\{t\mid t\neq-\dfrac{1}{2}\right\}$}. 
    \begin{jieda}
        反比例函数对称中心的横坐标是分母为零的值，纵坐标是分离出的常数.
    \end{jieda}
    \item 若函数$f(x) = \dfrac{ax-1}{x-1}$的图象关于点$(1, 2)$对称，则$a = $\blkda{2}
    \item 函数$f(x) = \dfrac{x}{ax+b}$（$a, b$是非零常数），满足$f(2) = 1$，且方程$f(x) = x$仅有一个解.
    \begin{enumerate}
        \item 求$a, b$的值
        \item 是否存在常数$m$，使得对于定义域内任意的$x$，$f(x) + f(m-x) = 4$恒成立
        \item 在平面直角坐标系中，求定点$A(-3, 1)$到此函数上任意一点$P$的距离$|AP|$的最小值.
    \end{enumerate}
    \begin{jieda}
        \begin{enumerate*}
            \item 由$f(2) = 1$，$\dfrac{2}{2a+b} = 1$，故$2a+b = 2$；$f(x) = x$即$\dfrac{x}{ax+b} = x$，即$(\dfrac{1}{ax+b}-1)x = 0$，容易判断其有两个解$x_1  = 0, x_2 = \dfrac{1-b}{a}$.
            
            因为$ab \neq 0$，所以要符合题意只可能是$b = 1$，两根重合.进一步推得$a = \dfrac{1}{2}$ \\
            \item $f(x) = \dfrac{2x}{x+2}$，为反比例型函数，其对称中心为$(-2, 2)$，因此$f(x)$满足$f(x) + f(-4-x) = 4$. 由此可知$m = -4$ \\
            \item 设$P\left(x_0, \dfrac{2x_0}{x_0+2}\right)$，故$|AP| = \sqrt{(x_0+3)^2+(\dfrac{2x_0}{x_0+2}-1)^2}   = \sqrt{(x_0+3)^2+(1-\dfrac{4}{x_0+2})^2}$. 设$t = x_0+2 (t \neq 0)$，则$|AP| = \sqrt{(t+1)^2+(1-\dfrac{4}{t})^2} = \sqrt{(t-\dfrac{4}{t})^2+2(t-\dfrac{4}{t})+10}$，设$k = t - \dfrac{4}{t}, k \in \mathbf{R}$，$|AP| = \sqrt{k^2+2k+10} = \sqrt{(k+1)^2+9} \geq 3$（当$x = \dfrac{-5\pm\sqrt{17}}{2}$时取得最小值）
        \end{enumerate*}
    \end{jieda}
\end{enumerate}

    \end{jieda}

    \begin{jieda}

    \begin{jieda}

    \item 已知$y=f(x)$是定义在$\bR$上的函数, 方程$f(2013+x)f(2015-x)=0$恰好有7个解, 则这7个解的和为\blkda{$7$}.

    \begin{enumerate}
        \item 互对称角度 \\
        $y = f(2013+x)$与$y = f(2015-x)$两者关于$x = 1$对称，若$f(2013+x)$有零点$x_1$，则$f(2025-x)$有零点$x_2 = 2-x_1$，满足$x_1+x_2 = 2$

    \end{enumerate}

            2 - |x| & x \leq 2 \\
            (x-2)^2 & x > 2

        根据对称性绘制图像可知当$b \in \left\{\dfrac{7}{4}\right\} \cup (2, +\infty)$时满足要求. 
    \end{jieda}
\end{enumerate}
\item 单调性的概念及应用
\begin{enumerate}
    \item 设$f(x), g(x)$都是单调函数, 则如下四个命题中, 真命题的序号是\blkda{\ding{172},\ding{173}}: 
    \begin{dingautolist}{172}
    \item 若$f(x)$是严格增函数, $g(x)$是严格增函数, 则$f(x)+g(x)$是严格增函数; 
    \item 若$f(x)$是严格增函数, $g(x)$是严格减函数, 则$f(x)-g(x)$是严格增函数; 
    \item 若$f(x)$是严格增函数, $g(x)$是严格增函数, 则$f(x)\cdot g(x)$是严格增函数; 
    \item 若$f(x)$是严格增函数, $g(x)$是严格减函数, 则$\dfrac{f(x)}{g(x)}$是严格减函数. 
    \end{dingautolist}
    \item 已知函数$f(x) = \left\{  
     \begin{array}{ll}  
        x^2+x, & 0 < x < 2, \\  
        -2x+8, & x \geq 2.
     \end{array}
     \right.$ 若$f(a) = f(a+2)$, 则$f\left(\dfrac{1}{a}\right)$的值是\blkda{2}
     \begin{jieda}
         根据题意可知$0 < a < 2$，否则$a > 2, a+2 > 2$，由单调性可知无法满足$f(a) = f(a+2)$,因此有$a^2 + a = -2(a+2)+8$。解得$a = 1$(舍去-4)，故$f(\dfrac{1}{a}) = f(1) = 2$
     \end{jieda}
     \item 若函数$f(x)=\abs{x+1}+\abs{x-a}$在$[2,+\infty)$上是严格增函数, 则实数$a$的取值范围是\blkda{$a\leq 2$}.
     \item 已知函数$f(x)$是定义在$(0, +\infty)$上的单调函数，且对$x \in (0, +\infty)$都有$f\left(f(x) - \dfrac{4}{x}\right) = 4$，则$f(x) = $\blkda{$2+\dfrac{4}{x}$}
     \begin{jieda}
        由于函数$f(x)$是定义在$(0, +\infty)$的单调函数，则必有$f(x) - \dfrac{4}{x}$为常数才可以使得$f(f(x) - \dfrac{4}{x}) = 4$，故可以设$f(x) = c + \dfrac{4}{x}$，因此有$f(c) = c + \dfrac{4}{c} = 4$，解得$c = 2$，因此$f(x) = 2+\dfrac{4}{x}$
     \end{jieda}
     \item 已知奇函数$f(x)$在$(-\infty,0)$上是严格减函数, $f(2)=0$, 则不等式$(x-1)f(x+1)>0$的解集为\dotfill(\kh{C})
     \xx{$(-2,-1)\cup(1,2)$}{$(-3,1)\cup(2,+\infty)$}{$(-3,-1)$}{$(-2,0)\cup(2,+\infty)$}
\end{enumerate}
