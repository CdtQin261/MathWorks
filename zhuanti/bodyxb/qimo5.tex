    \item $\frac{1-2\log_{xy}\left(z\right)\cdot\log_{xz}\left(y\right)\cdot\log_{yz}\left(x\right)}{\log_{xy}\left(z\right)\cdot\log_{xz}\left(y\right)+\log_{xz}\left(y\right)\cdot\log_{yz}\left(x\right)+\log_{yz}\left(x\right)\cdot\log_{xy}\left(z\right)} = $\blkda{1}

        考虑换底，$log_{xy}z = \frac{log_xz}{log_xy+1}, log_{yz}x = \frac{1}{log_xy+log_xz}, log_{zx}y = \frac{log_xy}{log_xz + 1}$\\

    \end{jieda}

    \item 当$x \in (0, +\infty)$时，幂函数$f(x) = (m^2-m-1)x^{-5m-3}$为增函数，则实数$m =$\blkda{$-1$}
    \begin{jieda}

    \item 已知定义在$R^{+}$上的函数$f(x) = \left \{\begin{array}{ll}

    \item 已知$f(x) = x^2 + \frac{1}{x^2} + k (x + \frac{1}{x})+11, x \in (0, +\infty)$，若$f(x) = 0$有两个解，求实数$k$的取值范围.
    \begin{jieda}
        首先考虑换元$t = x + \frac{1}{x}$，$f(x) = g(t) = t^2+k\cdot t + 9$，当$t = 2$时，只对应一个$x$，当$t > 2$时，对应着两个$x$. 故若$f(x)$在$(0, +\infty)$，有两个零点可以转化为$g(t)$在$(2, +\infty)$有一个零点，进一步转化为$y = -k$与$y = t + \frac{9}{t}$在$(2, +\infty)$有一个交点. 因此$k \in (-\infty, -\frac{13}{2}) \cup \{-6\}$ \\
        \ps \textbf{换元后及时判断新元的取值范围，并且分析清楚换元前后的对应关系！}
    \end{jieda}

    \item 若关于$x$的方程$\frac{|x|}{x+6} = kx^2$有四个不同的实数解，求实数$k$的取值范围.
    \begin{jieda}
        易知必有一根为0 \\
        参变分离有$\frac{1}{k} = \frac{(x+6)x^2}{|x|} = \left \{ \begin{array}{ll}
            \frac{(x+6)x^2}{x}, & x > 0 \\
            -\frac{(x+6)x^2}{x},  & x < 0
        \end{array}\right.$
        由于原方程必有一根为0，故可以转化为方程$\frac{1}{k} = \left \{ \begin{array}{ll}
            (x+6)x, & x > 0 \\
            -(x+6)x,  & x < 0
        \end{array}\right.$有三个不等实根.因此$\frac{1}{k} \in (0, 9)$，即$k \in (\frac{1}{9}, +\infty)$
    \end{jieda}

    \item 关于$x$的方程$k\cdot9^x-k\cdot3^{x+1}+6(k-5)=0$在$[0,2]$上有唯一解, 求实数$k$的取值范围. 
    \begin{jieda}
    令$t=3^x\in[1,9]$, $t$与$x$一一对应.于是关于$t$的方程$kt^2-3kt+6(k-5)=0$在$[1,9]$上有唯一解. \\
    于是$\frac{30}{k}=t^2-3t+6$. 结合图像可得$\frac{30}{k}\in(4,60]\cup\{\frac{15}{4}\}$, 即$k\in[\frac{1}{2},\frac{15}{2})\cup\{8\}$. \\

    \begin{jieda}
        换元$t = 2^x - 1$，$t \in (-1, +\infty)$，$t$与$x$一一对应. 故原方程有三个解即$t^2 - 2|t| + k = 0$有三个根，即$y = k$与$y = 2|t| - t^2$有三个交点.函数$y = 2|t| - t^2$在$(-1, 0)$单调递减，值域为$(0, 1)$，在$(0, 1)$单调递增，值域为$(0, 1)$，在$[1, +\infty)$单调递减，值域为$(-\infty, 1]$，故$k \in (0, 1)$时两函数有三个交点.
    \end{jieda}

    \item 已知二次函数 $f(x)=-\frac{1}{2} x^2+x$，是否存在实数 $m, n(m<n)$, 使得 $f(x)$ 的定义域为 $[m, n]$, 值域为 $[k m, k n](k>1)$？若存在则求出所有
    \begin{jieda}
    $f(x)=-\frac{1}{2} x^2+x=-\frac{1}{2} \times(x-1)^2+\frac{1}{2} \leq \frac{1}{2}$
    $\therefore k n \leq \frac{1}{2}, \therefore m<n<\frac{1}{2k}$
    又 $\because$ 函数的对称轴为 $x=1$
    $\therefore f(x)$ 在 $[m, n]$ 单调递增则有 $f(m)=k m, f(n)=k n$
    故$m, n$是$-\frac{1}{2}x^2 + x = kx$的两根，故$m = 2(1-k), n = 0$ \\
    \ps 根据值域判断函数在定义域内的单调性是简化，解决问题的突破口。\\
    常规解决方法是分析单调性，根据最值列方程组.
    \end{jieda}

    \item 若函数$f(x)$在区间$[a, b]$上的值域恰是$[\frac{1}{b}, \frac{1}{a}]$，则称区间$[a, b]$是$f(x)$的一个倒域区间. 求函数$g(x) = \left \{ \begin{array}{ll}
        -x^2+2x & 0 \leq x \leq 2 \\
        x^2 + 2x & -2 \leq x < 0
    \end{array}\right.$的所有倒域区间.
    \begin{jieda}
        $g(x)$的定义域为$[-2, 2]$，$g(x)$值域为$[-1, 1]$，故有$-2 \leq a < b \leq 2, -1 \leq \frac{1}{b} < \frac{1}{a} \leq 1$，解得$1 \leq a < b \leq 2$或$-2 \leq a < b \leq -1$.
        \begin{dingautolist}{172}
            \item  $1 \leq a < b \leq 2$，此时$g(x)$单调递减，有$\left \{ \begin{array}{l}
                 -a^2 + 2a = \frac{1}{a}  \\
                 -b^2 + 2b = \frac{1}{b} 
            \end{array} \right.$，即$a, b$是$-x^2 + 2x = \frac{1}{x}$的两根，即$a, b$是$-x^3 + 2x^2 -1 = 0$的两根，可以看出有一根为$1$，即$a, b$是$(x-1)(-x^2+x+1) = 0$的两根，故$a = 1, b = \frac{1+\sqrt{5}}{2}$
            \item $-2 \leq a < b \leq -1$，此种情况和前一种对称，可得$a = -\frac{1+\sqrt{5}}{2}, b = -1$
        \end{dingautolist}
        因此两个倒域区间为$[1, \frac{1+\sqrt{5}}{2}]$和$[-\frac{1+\sqrt{5}}{2}, -1]$
    \end{jieda}

    \end{vspace_}
    \item 若函数$f(x)$在区间$[m, n]$上的值域恰是$[m, n]$，则称区间$[m, n]$是$f(x)$的一个等域区间. 若区间$[m,n]$是函数$y = \frac{(a+1)x - \frac{1}{a}}{ax}(a \neq 0)$的一个等域区间.求$n-m$的最大值.
    \begin{jieda}
        $\frac{(a+1)x - \frac{1}{a}}{ax} = \frac{a+1}{a} - \frac{1}{a^2x}$，故函数为$y = \frac{a+1}{a} - \frac{1}{a^2x} (a \neq 0)$，易知函数的定义域为$(-\infty, 0) \cup (0, +\infty)$，故必有$m < n < 0$或$0 < m < n$. 故易知函数在$[m, n]$内严格单调递增，故$\left \{ \begin{array}{l}
                \frac{a+1}{a} - \frac{1}{a^2m} = m \\
                \frac{a+1}{a} - \frac{1}{a^2n} = n
            \end{array} \right.$，即$m, n$是$a^2x^2 - (a+1)ax + 1 = 0$的两根. 由韦达定理有$\left \{ \begin{array}{l}
                m+n = \frac{a+1}{a}\\
                m \cdot n = \frac{1}{a^2}
            \end{array} \right.$，由$\Delta > 0$得$((a+1)^2 - 4)a^2 > 0$，故$a \in (-\infty, -3]\cup [1, +\infty)$ \\
            $n - m = \sqrt{(m+n)^2 - 4mn} = \sqrt{(\frac{a+1}{a})^2 - \frac{4}{a^2}} = \sqrt{\frac{a^2+2a-3}{a^2}} \leq \frac{2\sqrt{3}}{3}$，当$a = 3$时取等号.
    \end{jieda}
