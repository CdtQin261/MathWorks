    \item 若入射光线所在的直线为 ${l}_{1} : x - y + 1 = 0$ 射入,被镜面直线 ${l}_{2} : {2x} + y - 4 = 0$ 反射,则反射光线所在的直线 $l$ 的方程是\blkda[3]{$x-7y+13=0$}.

    \item “ $a =  - 2$ ” 是 “直线 $x + {ay} + 3 = 0$ 与 ${ax} + {4y} + 6 = 0$ 平行” 的\dotfill{(\kh{A})}

            \item 当 $a =  - 2$ 时,此时直线 $x - {2y} + 3 = 0$ 与 $- {2x} + {4y} + 6 = 0$ 平行

    \end{jieda}
    \item 在平面直角坐标系 ${xOy}$ 中,若动点 $P\left( {a,b}\right)$ 到两直线 ${l}_{1}$ : $y = x$ 和 ${l}_{2} : y =  - x + 2$ 的距离之和为 $2\sqrt{2}$ ,则 ${a}^{2} + {b}^{2}$ 的最大值是\blkda{18}.

        \item 当 ${MA} \cdot  {MB}$ 取得最小值时

            \item $A\left( {a,0}\right) ,B\left( {0,b}\right) ,a > 0,b > 0,\dfrac{2}{a} + \dfrac{1}{b} = 1$，所以 ${MA} \cdot  {MB} = \left| \vv{MA}\right|  \cdot  \left| \vv{MB}\right|  =  - \vv{MA} \cdot  \vv{MB} =  - \left( {a - 2, - 1}\right)  \cdot  \left( {-2,b - 1}\right) = 2\left( {a - 2}\right)  + b - 1 = {2a} + b - 5 = \left( {{2a} + b}\right) \left( {\dfrac{2}{a} + \dfrac{1}{b}}\right)  - 5 = 2\left( {\dfrac{b}{a} + \dfrac{a}{b}}\right)  \geq  4$，当且仅当 $a = b = 3$ 时取等号，此时直线 $l$ 的方程为 $x + y - 3 = 0$ .
            \item 设直线 $l$ 的方程为 $\dfrac{x}{a} + \dfrac{y}{b} = 1$ ,则 $\dfrac{2}{a} + \dfrac{1}{b} = 1$，所以 $a + b = \left( {a + b}\right) \left( {\dfrac{2}{a} + \dfrac{1}{b}}\right)  = 3 + \dfrac{2b}{a} + \dfrac{a}{b} \geq  3 + 2\sqrt{\dfrac{{2b} \cdot  a}{a \cdot  b}} = 3 + 2\sqrt{2}$ ,即 $a + b \geq  3 + 2\sqrt{2}$，当且仅当 $\left\{  \begin{array}{l} a = b, \\  \dfrac{2}{a} + \dfrac{1}{b} = 1, \end{array}\right.$ 即 $\left\{  \begin{array}{l} a = 2 + \sqrt{2}, \\  b = \sqrt{2} + 1 \end{array}\right.$ 时取 $a =$ ”,此时直线 $l$ 的方程为 $x + \sqrt{2}y - 2 - \sqrt{2} = 0$ .
        \end{enumerate}
    \end{jieda}
\end{enumerate}

            \item 当斜率存在时，设切线为$y = k(x-3)+4$，故原点到切线的距离$d = \dfrac{|4-3k|}{\sqrt{1+k^2}} = 3$，解得$k = \dfrac{7}{24}$，故切线方程为$7x-24y+75=0$

    \end{jieda}
    
    \item 已知动点 $P$ 在抛物线 ${y}^{2} = {8x}$ 上,过点 $P$ 引圆 ${\left( x - 5\right) }^{2} + {y}^{2} = 4$ 的切线,切点分别为 $A,B$ ,则 $\left| {AB}\right|$ 的最小值为\blkda{$\dfrac{2\sqrt{30}}{3}$}

                故$A\left(\dfrac{3}{\sqrt{1+3k_1^2}}, \dfrac{3k_1}{\sqrt{1+3k_1^2}}\right)$，$B\left(\dfrac{3\sqrt{3}k_1}{\sqrt{1+3k_1^2}}, \dfrac{-\sqrt{3}}{\sqrt{1+3k_1^2}}\right)$

                故$\vv{OA} \cdot \vv{OB} = \dfrac{6\sqrt{3}k_1}{1+3k_1^2} = \dfrac{6\sqrt{3}}{3k_1+\dfrac{1}{k_1}} \in (0, 3]$

                根据对称性可知$k_1 < 0$时，范围相同

                故根据对称性可知$x_A < 0$时$\vv{OA} \cdot \vv{OB} \in \left\lbrack -3,0\right)$，故$\vv{OA} \cdot \vv{OB} \in \left\lbrack  {-3,0)\cup (0,3}\right\rbrack$

                \item 方法2

                设 $A\left( {{x}_{1},{y}_{1}}\right) ,B\left( {{x}_{2},{y}_{2}}\right)$
                \begin{dingautolist}{172}
                    \item 当直线 $l$ 的斜率不存在时, ${x}_{2} = {x}_{1},{y}_{2} =  - {y}_{1}$ ,则 ${k}_{1}{k}_{2} =  - \dfrac{{y}_{1}^{2}}{{x}_{1}^{2}} =  - \dfrac{1}{3}$ ,又 $\dfrac{{x}_{1}^{2}}{9} + \dfrac{{y}_{1}^{2}}{3} = 1$ ,解得 ${x}_{1}^{2} = \dfrac{9}{2}$ ,则 $\vv{OA} \cdot  \vv{OB} = {x}_{1}^{2} - {y}_{1}^{2} = \dfrac{2}{3}{x}_{1}^{2} = 3$ .
                    \item 当直线 $l$ 的斜率存在时,设 $l : y = {kx} + t$，联立直线和椭圆方程有 $\left\{  \begin{array}{l} y = {kx} + t, \\  \dfrac{x^2}{9} + \dfrac{{y}^{2}}{3} = 1, \end{array}\right.$
                    
                    消去 $y$ 得 $\left( {1 + 3{k}^{2}}\right) {x}^{2} + {6ktx} + 3{t}^{2} - 9 = 0$，根据题意可得$\Delta  = {12}\left( {3 + 9{k}^{2} - {t}^{2}}\right)  > 0$ ,
                    
                    根据韦达定理有$\left\{  {\begin{array}{l} {x}_{1} + {x}_{2} = \dfrac{-{6kt}}{1 + 3{k}^{2}}, \\  {x}_{1}{x}_{2} = \dfrac{3{t}^{2} - 9}{1 + 3{k}^{2}}, \end{array}}\right.$
                    
                    ${k}_{1}{k}_{2} = \dfrac{{y}_{1}{y}_{2}}{{x}_{1}{x}_{2}} =  - \dfrac{1}{3}$，代入直线方程可得$\dfrac{{y}_{1}{y}_{2}}{{x}_{1}{x}_{2}} = \dfrac{\left( {k{x}_{1} + t}\right) \left( {k{x}_{2} + t}\right) }{{x}_{1}{x}_{2}} = {k}^{2} + \dfrac{{kt}\left( {{x}_{1} + {x}_{2}}\right)  + {t}^{2}}{{x}_{1}{x}_{2}} = {k}^{2} + \dfrac{\dfrac{-6{k}^{2}{t}^{2}}{1 + 3{k}^{2}} + {t}^{2}}{\dfrac{3{t}^{2} - 9}{1 + 3{k}^{2}}} = \dfrac{{t}^{2} - 9{k}^{2}}{3{t}^{2} - 9} =  - \dfrac{1}{3},$
                    
                    整理得 $2{t}^{2} = 9{k}^{2} + 3$且$t^2 \neq 3$，故${t}^{2} \geq \dfrac{3}{2}$且$t^2 \neq 3$，经检验$\Delta  > 0$ 恒成立.
                    
                    $\vv{OA} \cdot  \vv{OB} = {x}_{1}{x}_{2} + {y}_{1}{y}_{2} = {x}_{1}{x}_{2} - \dfrac{1}{3}{x}_{1}{x}_{2} = \dfrac{2}{3}{x}_{1}{x}_{2} = \dfrac{2}{3} \cdot  \dfrac{3{t}^{2} - 9}{1 + 3{k}^{2}} = 3 \cdot  \dfrac{{t}^{2} - 3}{{t}^{2}} = 3\left( {1 - \dfrac{3}{{t}^{2}}}\right) \in  \lbrack  - 3,0) \cup  \left( {0,3}\right)$ .
                \end{dingautolist}
                综上, $\vv{OA} \cdot  \vv{OB}$ 的取值范围为 $\left\lbrack  {-3,0)\cup (0,3}\right\rbrack$ .
            \end{enumerate}
        \end{enumerate}
    \end{jieda}
\end{enumerate}
