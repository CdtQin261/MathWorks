\chapter*{三角函数}
\subsection*{对称轴、对称中心}
\begin{enumerate}
    \item 函数$y = sin\left(2x+\dfrac{\pi}{3}\right)$的图象的对称轴方程是\blkda[3.5]{$x = \dfrac{k\pi}{2}+\dfrac{\pi}{12}, k \in \mathbf{Z}$}，对称中心的坐标是\blkda[3.5]{$\left(\dfrac{k\pi}{2}-\dfrac{\pi}{6}, 0\right), k \in \mathbf{Z}$}
    
    \item 已知函数$f(x) = sin\left(\omega x + \dfrac{\pi}{3}\right) (\omega > 0)$，若将$f(x)$的图象向左平移$\dfrac{\pi}{2}$个单位后关于$y$轴对称，则$\omega$的最小值为\blkda{$\dfrac{1}{3}$}
    \begin{jieda}
        代入$x = \dfrac{\pi}{2}$，$f\left(\dfrac{\pi}{2}\right) = sin\left(\dfrac{\omega \pi}{2} + \dfrac{\pi}{3}\right) = \pm 1$，因此$\dfrac{\omega \pi}{2} + \dfrac{\pi}{3} = k\pi + \dfrac{\pi}{2} (k \in \mathbf{Z})$，即$\omega = 2k + \dfrac{1}{3} (k \in \mathbf{Z})$，又因为$\omega > 0$，因此当$k = 0$时$\omega$取最小值$\dfrac{1}{3}$.
    \end{jieda}

    \item (2024$\cdot$闵行区一模$\cdot$7)若将函数 $y = \sin \left( {{2x} + \varphi }\right) \left( {0 < \varphi  < \pi }\right)$ 的图像向右平移 $\dfrac{\pi }{3}$ 个单位,得到的图像所对应的函数为奇函数,则 $\varphi  =$ \blkda[2]{$\dfrac{2\pi }{3}$}.
    \begin{jieda}
        原函数有对称中心$\left(-\dfrac{\pi}{3}, 0\right)$，故$2 \times \left(-\dfrac{\pi}{3}\right) + \varphi = k\pi, k \in \mathbf{Z}$，故$\varphi = \dfrac{2\pi}{3} + k\pi, k \in \mathbf{Z}$，又因为$0 < \varphi  < \pi $，所以$\varphi = \dfrac{2\pi}{3}$
    \end{jieda}

    \item 已知函数$f(x) = sin\omega x+acos\omega x (\omega > 0)$的最小正周期为$2\pi$
    \begin{enumerate}
        \item 求$\omega$的值
        \item 已知直线$x = -\dfrac{\pi}{4}$是函数$f(x)$的一条对称轴，求$f(x)$的最大值与最小值
    \end{enumerate}
    \begin{jieda}
        \begin{enumerate}
            \item $f(x) = \sqrt{a^2+1}sin(\omega x + \varphi)$（$cos\varphi = \dfrac{1}{\sqrt{a^2+1}}, sin\varphi = \dfrac{a}{\sqrt{a^2+1}}$）$\dfrac{2\pi}{\omega} = T = 2\pi$，故$\omega = 1$

            \item 因为$x = -\dfrac{\pi}{4}$是函数的一条对称轴，因此$f(0) = f(-\dfrac{\pi}{2})$，故$a = -1$，经检验$f(x) = sinx - cosx$符合要求，故$f(x)$的最大值为$\sqrt{2}$，最小值为$-\sqrt{2}$，当$x = 2k\pi - \dfrac{\pi}{4}, k \in \mathbf{Z}$的时候取得最小值，当$x = 2k\pi + \dfrac{\pi}{4}, k \in \mathbf{Z}$的时候取得最大值
        \end{enumerate}
    \end{jieda}
\end{enumerate}
\subsection*{求值域}
\begin{enumerate}
    \item $f(x) = sin^2x + 2cosx \cdot sinx + 3cos^2x - 2, x \in \left(0, \dfrac{\pi}{6}\right)$的值域为\blkda{$(1, \sqrt{2}]$} 
    \begin{jieda}
        $f(x) = sin2x + 2cos^2x -1 = sin2x + cos2x = \sqrt{2}sin\left(2x+\dfrac{\pi}{4}\right)$ \\
        $x \in \left(0, \dfrac{\pi}{6}\right)$，$2x+\dfrac{\pi}{4} \in \left(\dfrac{\pi}{4}, \dfrac{7\pi}{12}\right)$，因此$f(x) \in (1, \sqrt{2}]$
    \end{jieda}
    \item $f(x) = \dfrac{sin^2x + sin2x}{cos2x}$在$x \in \left(0, \dfrac{\pi}{4}\right)$的值域为\blkda{$(0, +\infty)$}
    \begin{jieda}
        $f(x) = \dfrac{sin^2x + 2sinxcosx}{cos^2x-sin^2x} = \dfrac{tan^2x+2tanx}{1-tan^2x} = -1 + \dfrac{2tanx + 1}{1-tan^2x}$

        令$t = 2tanx+1, t \in (1,3)$，$f(x) = g(t) = -1 + \dfrac{t}{1-\left(\dfrac{t-1}{2}\right)^2} = -1 + \dfrac{4t}{-t^2 + 2t +3} = -1 + \dfrac{4}{-t+\dfrac{3}{t}+2}$，$-t + \dfrac{3}{t} + 2 \in \left(0, 4\right)$，故$f(x) \in \left(0, +\infty\right)$
    \end{jieda} 
    \item 已知关于$x$的方程$sin^2x+cosx+a = 0$有解，求$a$的范围.
    \begin{jieda}
        $a+1 = cos^2x - cosx \in \left[-\dfrac{1}{4}, 2\right]$，故$a \in \left[-\dfrac{5}{4}, 1\right]$
    \end{jieda}
    \item $y=1+\sin{x}+\cos{x}+\sin{x}\cos{x}$的值域是\blkda[3]{$\left[0,\dfrac{3}{2}+\sqrt{2}\right]$}. 
    \begin{jieda}
    $y=1+t+\dfrac{t^2-1}{2}$, 其中$t=\sin{x}+\cos{x}\in[-\sqrt{2},\sqrt{2}]$. 
    \end{jieda}

    \item (2024$\cdot$嘉定区二模$\cdot$10)已知 $f\left( x\right)  = \dfrac{2}{\sin x} + \dfrac{2}{\cos x},x \in  \left( {0,\dfrac{\pi }{2}}\right)$ ，则函数 $y = f\left( x\right)$ 的最小值为\blkda[2]{$4\sqrt{2}$}.
    \begin{jieda}
        $f\left( x\right)  = \dfrac{2}{\sin x} + \dfrac{2}{\cos x} = \dfrac{2\left( {\sin x + \cos x}\right) }{\sin x\cos x}$ ,令 $t = \sin x +  \cos x,t \in  (1,\sqrt{2}\rbrack ,\dfrac{2t}{\dfrac{{t}^{2} - 1}{2}} = \dfrac{4t}{{t}^{2} - 1} = \dfrac{4}{t - \dfrac{1}{t}}$ ,当 $t = \sqrt{2}$ 时 $f\left( x\right)$ 取得最小值 $4\sqrt{2}$.
    \end{jieda}
\end{enumerate}

        故$sint_1$可以取到$1$，故$sint_2$应可以取到$-1$，故$\omega \geq \dfrac{3}{4}$，经检验$\omega = \dfrac{3}{4}$满足要求.
    \end{jieda}
    \item 若存在实数 $\varphi$ ,使函数 $f\left( x\right)  = \cos \left( {{\omega x} + \varphi }\right)  - \dfrac{1}{2}\left( {\omega  > 0}\right)$ 在 $x \in  \left\lbrack  {\pi ,{3\pi }}\right\rbrack$ 上有且仅有 2 个零点,则 $\omega$ 的取值范围\blkda{$\left[\dfrac{1}{3}, \dfrac{5}{3}\right)$}
    \begin{jieda}
        令$t = \omega x + \varphi$，故$cos t  = \dfrac{1}{2}$在$t \in [\omega \pi + \varphi, 3\omega \pi + \varphi]$恰有两个解

        $cost = \dfrac{1}{2}$的解集为$\{x| x = \pm \dfrac{\pi}{3} + 2k\pi, k \in \mathbf{Z}\}$，
        
        故存在长度为$2\omega \pi$的区间与$\{x| x = \pm \dfrac{\pi}{3} + 2k\pi, k \in \mathbf{Z}\}$有且仅有两个公共元素，数形结合可知$2\omega\pi \in \left[\dfrac{2\pi}{3}, \dfrac{10\pi}{3}\right)$，即$\omega \in \left[\dfrac{1}{3}, \dfrac{5}{3}\right)$
    \end{jieda}
\end{enumerate}
\subsection*{两个周期函数相加}
如果两个周期函数相加,一般来说,它们的和函数的周期会是这两个周期的公倍数(如果存在公倍数的话)。

对于周期函数,如果 ${T}_{1}$ 和 ${T}_{2}$ 的比值是有理数，那么它们存在公共周期，但周期是多少并不确定：经典反例$y = f(x), y = -f(x)$，周期相同，相加后没有最小正周期.

对于周期函数,如果 ${T}_{1}$ 和 ${T}_{2}$ 的比值是无理数,通常和函数不是周期函数。


\begin{enumerate}
     \item (2024$\cdot$嘉定区二模$\cdot$16)已知函数 $y = f\left( x\right) \left( {x \in  \mathbf{R}}\right)$ 的最小正周期是 ${T}_{1}$ ，函数 $y = g\left( x\right) \left( {x \in  \mathbf{R}}\right)$ 的最小正周期是 ${T}_{2}$ ,且 ${T}_{1} = k{T}_{2}\left( {k > 1}\right)$ ,对于命题甲: 函数 $y = f\left( x\right)  + g\left( x\right) \left( {x \in  \mathbf{R}}\right)$ 可能不是周期函数; 命题乙: 若函数 $y = f\left( x\right)  + g\left( x\right) \left( {x \in  \mathbf{R}}\right)$ 的最小正周期是 ${T}_{3}$ ,则 ${T}_{3} \geq  {T}_{1}$. 下列选项正确的是 \dotfill{(\kh{C})}
    \xx{甲和乙均为真命题}{甲和乙均为假命题}{甲为真命题且乙为假命题}{甲为假命题且乙为真命题}
