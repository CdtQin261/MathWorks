\subsection*{不等式求解}
\begin{enumerate}
    \item 已知不等式$|x+3| \leq 5x$和$\dfrac{x^2(x+3)(x^2-x+1)}{x^2 - 5x + 4} > 0$中有且仅有一个成立，则实数$x$的范围是\blkda[4]{$(-3, 0) \cup \left(0, \dfrac{3}{4}\right) \cup [1, 4]$ }
    \begin{jieda}
        第一个不等式的解集为$\left[\dfrac{3}{4}, +\infty\right)$，第二个不等式的解集为$(-3, 0) \cup (0, 1) \cup (4, +\infty)$

        因此第一个不等式成立第二个不等式不成立时$x \in [1, 4]$，第二个不等式成立，第一个不等式不成立时$(-3, 0) \cup \left(0, \dfrac{3}{4}\right)$

        两者取并集，得到$(-3, 0) \cup \left(0, \dfrac{3}{4}\right) \cup [1, 4]$ 
    \end{jieda}

    \item 已知集合$M = \left\{x \left| \dfrac{ax- 5}{x^2-a} < 0\right. \right\}$，若$3 \in M, 5 \notin M$，则实数$a$的取值范围是\blkda[3]{$\left[1, \dfrac{5}{3}\right) \cup (9, 25]$}
    \begin{jieda}
        \begin{enumerate}
            \item $3 \in M \Longrightarrow \dfrac{3a - 5}{9-a} < 0$
            \item $5 \notin M \Longrightarrow \dfrac{5a - 5}{25-a} \geq 0$ 或 $a = 25$(此时$\dfrac{5a - 5}{25-a}$无意义，满足条件）
        \end{enumerate}
        求解分式不等式：
        \begin{enumerate}
            \item $\dfrac{3a - 5}{9-a} < 0 \Longrightarrow (3a-5)(9-a) < 0 \Longrightarrow a \in \left(-\infty, \dfrac{5}{3}\right) \cup (9, +\infty)$
            \item $\dfrac{5a - 5}{25-a} \geq 0 \Longrightarrow (5a-5)(25-a) \geq 0, a \neq 25 \Longrightarrow a \in [1, 25)$
        \end{enumerate}
    \end{jieda}

    \item 若不等式$(2a-b)x +3a - 4b < 0$的解集是$\left(\dfrac{9}{4}, +\infty\right)$，则不等式$(a-4b)x + 2a - 3b > 0$的解集是\blkda[2]{$\left(-\dfrac{8}{19}, +\infty\right)$}
    \begin{jieda}
        从解集形式可知\ding{172} $2a-b < 0$，\ding{173}$(2a-b)x +3a - 4b = 0$的根是$\dfrac{9}{4}$，即$a = \dfrac{5}{6}b$，故$(a-4b)x + 2a - 3b > 0$可转化为$\left(-\dfrac{19}{5}a\right)x - \left(\dfrac{8}{5}a\right) > 0$，因为$2a-b < 0$，所以$a < 0$，故可以进一步转化为$\dfrac{19}{5}x + \dfrac{8}{5} < 0$，解得$x \in \left(-\dfrac{8}{19}, +\infty\right)$
    \end{jieda}

    \item 若 ${\log }_{2026}x + {2026}^{x} > {2026}$ ,则 $x$ 的取值范围为\blkda{$\left( {1, + \infty }\right)$}

    \item 已知不等式组$ \left\{ \begin{array}{l}
         x^2-x+a-a^2 < 0  \\
         x + 2a > 1
    \end{array} \right.$的整数解恰好有两个，$a$的取值范围是\blkda[4]{$(1, 2]$}
    \begin{jieda}
        设函数$f(x) = x^2 - x$，则第一个不等式即$f(x) < f(a)$，根据函数的性质可知不等式即$\left|x-\dfrac{1}{2}\right| < \left|a - \dfrac{1}{2}\right|$，故$\dfrac{1}{2}-\left|a - \dfrac{1}{2}\right| < x < \dfrac{1}{2} + \left|a - \dfrac{1}{2}\right|$，故第一个不等式解集为$(1 - a, a)$或$(a, 1-a)$，第二个不等式解集为$(1-2a, +\infty)$
        \begin{dingautolist}{172}
            \item $a \in [0, 1]$，此时第一个不等式中不含有两个整数解，不符合要求.
            \item $a < 0$，第一个不等式解集为$(a, 1-a)$，而$1-a < 1-2a$，故不等式组解集为空集，不符合要求.
            \item $a > 1$，第一个不等式解集为$(1 - a, a)$，故不等式解集为$(1 - a, a)$，当$a \in (1, 2]$满足要求.
        \end{dingautolist}
    \end{jieda}
    \item 若关于$x$的不等式$k|x| > |x - 2|$恰有4个整数解，则实数$k$的取值范围是\blkda[4]{$\left(\dfrac{3}{5}, \dfrac{2}{3}\right]$}
    \begin{jieda}
        易知$x = 0$不是不等式的解，故$k|x| > |x - 2|$进行参变分离，同解变形$k > \left|\dfrac{x-2}{x}\right|$，即$k > \left|1 - \dfrac{2}{x}\right|$ \\
        由函数$f(x) = \left|1 - \dfrac{2}{x}\right|$的图像可知整根的集合为$\{2, 3, 4, 5\}$，故$f(5) < k \leq f(6)$，即$k \in \left(\dfrac{3}{5}, \dfrac{2}{3}\right]$
        \ps \textbf{解决整根问题的第一步往往是确认整根的集合}
    \end{jieda}
\end{enumerate}

\subsection*{平均值不等式、三角不等式}
\begin{enumerate}
    \item 已知 $a > 0,b > 0$ ,且 ${ab} = a - b + 3$ ,则 $a + b$ 的最小值为\blkda{$2\sqrt{2}$}.
    \begin{jieda}
        因为 ${ab} = a - b + 3$ ,解得: $b = \dfrac{a + 3}{a + 1} = 1 + \dfrac{2}{a + 1}$，则 $a + b = a + 1 + \dfrac{2}{a + 1} \geq  2\sqrt{2}$
        
        当且仅当 $a = \sqrt{2} - 1,b = \sqrt{2} + 1$ 时，等号成立
    \end{jieda}

    \item 已知$x > 0, y > 0, 4x+2y-xy=0$，则$2x+y$的最小值为\blkda{16}
    \begin{jieda}
        \begin{enumerate}
            \item 消元法 \\
            首先根据题目中的等式用$x$来表示$y$，得到$y = \dfrac{4x}{x-2} (x > 2)$ \\
            目标式子变为$2x+\dfrac{4x}{x-2} = 2x + 4 + \dfrac{8}{x-2} = 8 + 2(x-2) + \dfrac{8}{x-2} \geq 8 + 2 \sqrt{2(x-2) \cdot \dfrac{8}{x-2}} \\ = 16$
            \item 常数法 \\
            首先对题目中的等式进行变换，等式两边同时除以$xy$得到$\dfrac{2}{x} + \dfrac{4}{y} = 1$，所以有$(2x+y) = (2x+y)(\dfrac{2}{x} + \dfrac{4}{y}) = \dfrac{8x}{y} + \dfrac{2y}{x} + 8 \geq 2\sqrt{\dfrac{8x}{y} \cdot \dfrac{2y}{x}} + 8 = 16$
        \end{enumerate}
    \end{jieda}

    \item 若正实数 $a$ 、 $b$ 满足 $a^2+b^2 = ab + 1$，求$ab$最大值.
    \begin{jieda}
        $ab + 1 = a^2+b^2 \geq 2ab$，$ab \leq 1$（$a = b = 1$时取等）
    \end{jieda}

    \item 已知正数$x, y$满足$x+y = 5$，则$\dfrac{1}{x+2} + \dfrac{1}{y+2}$的最小值是 \blkda{$\dfrac{4}{9}$ }
    \begin{jieda}
        $x+y=5 \Longrightarrow (x+2)+(y+2) = 9$，因此$\dfrac{1}{x+2}+\dfrac{1}{y+2} = \left(\dfrac{1}{x+2}+\dfrac{1}{y+2}\right) \cdot ((x+2)+(y+2)) \cdot \dfrac{1}{9} = \dfrac{1}{9} \left(2 + \dfrac{x+2}{y+2} + \dfrac{y+2}{x+2}\right) \geq \dfrac{4}{9}$
    \end{jieda}

    \item 当 $0 < m < \dfrac{1}{2}$ 时,若 $\dfrac{1}{m} + \dfrac{2}{1 - {2m}} \geq  {k}^{2} - {2k}$ 恒成立,则实数 $k$ 的取值范围为\blkda[4]{$[-2,4]$}.
    \begin{jieda}
        由于 $0 < m < \dfrac{1}{2}$ ,所以 $\dfrac{1}{2} \cdot  {2m} \cdot  \left( {1 - {2m}}\right)  \leq  \dfrac{1}{2} \cdot  {\left\lbrack  \dfrac{{2m} + \left( {1 - {2m}}\right) }{2}\right\rbrack  }^{2} = \dfrac{1}{8}$ ,当且仅当 ${2m} = 1 - {2m}$ ,即 $m = \dfrac{1}{4}$ 时取等号,所以 $\dfrac{1}{m} + \dfrac{2}{1 - {2m}} = \dfrac{1}{m\left( {1 - {2m}}\right) } \geq  8$ ,因为 $\dfrac{1}{m} +$  $\dfrac{2}{1 - {2m}} \geq  {k}^{2} - {2k}$ 恒成立,所以 ${k}^{2} - {2k} - 8 \leq  0$ ,所以 $- 2 \leq  k \leq  4$
    \end{jieda}

    \item 已知函数 $f\left( x\right)  = \left| {x - {a}^{2}}\right|  + \left| {x - {2a} + 1}\right|$
    \begin{enumerate}
        \item 当 $a = 2$ 时,求不等式 $f\left( x\right)  \geq  4$ 的解集
        \item 若 $f\left( x\right)  \geq  4$恒成立 ,求 $a$ 的取值范围
    \end{enumerate}
    \begin{jieda}
        \begin{enumerate}
        \item 当 $a = 2$ 时, $f\left( x\right)  = \left| {x - 4}\right|  + \left| {x - 3}\right| = \left\{\begin{array}{ll}
             7 - {2x}, & x \leq  3  \\
             4, & 3 < x < 4 \\
             {2x} - 7, & x \geq  4
        \end{array}\right.$ .
        \begin{dingautolist}{172}
            \item $x \leq  3$，$f\left( x\right)  \geq  4 \Longleftrightarrow x \leq  \dfrac{3}{2}$
            \item $3 < x < 4$，无解
            \item $x \geq  4$，$f\left( x\right)  \geq  4 \Longleftrightarrow x \geq  \dfrac{11}{2}$
        \end{dingautolist}
        
        综上所述: $f\left( x\right)  \geq  4$ 的解集为 $\left\{  x \left |  x \leq  \dfrac{3}{2}\right. \right.$ 或 $\left. {x \geq  \dfrac{11}{2}}\right\}$ .
        \item $f\left( x\right)  = \left| {x - {a}^{2}}\right|  + \left| {x - {2a} + 1}\right|  \geq  \left| {\left( {x - {a}^{2}}\right)  - \left( {x - {2a} + 1}\right) }\right|  = \left| {-{a}^{2} + {2a} - 1}\right|  = {\left( a - 1\right) }^{2}$ (当且仅当 ${2a} - 1 \leq  x \leq  {a}^{2}$ 时取等号),
        
        $\therefore {\left( a - 1\right) }^{2} \geq  4$ ,解得: $a \leq   - 1$ 或 $a \geq  3$
    \end{enumerate}
    \end{jieda}
\end{enumerate}
\subsection*{恒成立问题}
\begin{enumerate}
    \item 若存在 $x \in  \left\lbrack  {1,3}\right\rbrack$ ,使不等式 ${x}^{2} - {2ax} + a + 2 \leq  0$ 成立,则 $a$ 的取值范围为\blkda[4]{$\lbrack 2, + \infty )$}.
    \begin{jieda}
        由 ${x}^{2} - {2ax} + a + 2 \leq  0 \Longrightarrow  {x}^{2} + 2 \leq  a\left( {{2x} - 1}\right)$

        因为 $x \in  \left\lbrack  {1,3}\right\rbrack$ ,所以 ${2x} - 1 \in  \left\lbrack  {1,5}\right\rbrack$ ,令 $t = {2x} - 1 \in  \left\lbrack  {1,5}\right\rbrack  ,x = \dfrac{t + 1}{2}$ ,
        
        由 ${x}^{2} + 2 \leq  a\left( {{2x} - 1}\right)  \Longrightarrow  a \geq  \dfrac{{x}^{2} + 2}{{2x} - 1} = \dfrac{\dfrac{1}{4}\left( {{t}^{2} + {2t} + 9}\right) }{t} = \dfrac{1}{4}\left( {t + \dfrac{9}{t} + 2}\right)$ ,
        
        构造函数 $g\left( t\right)  = \dfrac{1}{4}\left( {t + \dfrac{9}{t} + 2}\right)  \geq  \dfrac{1}{4}\left( {2\sqrt{t \cdot  \dfrac{9}{t}} + 2}\right)  = 2$ ,
        
        即 $g{\left( t\right) }_{\min } = 2$ ,当且仅当 $t = 3 \in  \left\lbrack  {1,5}\right\rbrack$ 时取等号，所以 $a \geq  g{\left( t\right) }_{\min } = 2$
    \end{jieda} 

    \item 已知函数 $f\left( x\right)  = {\left( \dfrac{1}{2}\right) }^{x}$ 与 $g\left( x\right)  = {\log }_{4}\left( {{x}^{2} - {2ax} + 4}\right) \left( {a > 0}\right)$ ,若对任意的 ${x}_{1} \in  \left( {0,1}\right)$ ,都存在 ${x}_{2} \in  \left\lbrack  {0,2}\right\rbrack$ ,使得 $f\left( {x}_{1}\right)  = g\left( {x}_{2}\right)$ ,则实数 $a$ 的取值范围是\blkda{$[\sqrt{2}, +\infty)$}
    \begin{jieda}
        问题转化为$f(x)$在$(0, 1)$上的值域是$g(x)$在$[0, 2]$上值域的子集. \\
        $f(x)$在$(0, 1)$上的值域为$\left(\dfrac{1}{2}, 1\right)$，因此内层函数$h(x) = {x}^{2} - {2ax} + 4$在$[0, 2]$上的值域要包含$(2, 4)$，因此内层函数对称轴为直线$x = a$
        \begin{dingautolist}{172}
            \item $a \in (0, 2]$ \\
            此时$h(x)$在$[0, a]$上单调减，在$[a, 2]$上单调增，因此当$h(a) \leq 2$时满足要求，即$4-a^2\leq2$，解得$a \in [\sqrt{2}, 2]$
            \item $a > 2$ \\
            此时$h(x)$在$[0, 2]$上单调减，因此当$h(2) \leq 2$时满足要求，即$0 < 8-4a \leq 2$，在$a > 2$时恒成立.
        \end{dingautolist}
        综上，$a \in [\sqrt{2}, +\infty)$
    \end{jieda}

    \item 设 $k \in  \mathbf{R}$ ,函数 $y = f\left( x\right)$ 和 $y = g\left( x\right)$ 的表达式分别为$f\left( x\right)  = e^{x} - {kx},g\left( x\right)  = {x}^{2} + \left( {k - 2}\right) x + \dfrac{1}{4}$ ,若对任意的实数 ${x}_{0}$ ,皆有 $f\left( {x}_{0}\right)  \cdot  g\left( {x}_{0}\right)  \geq  0$ 成立,则 $k$ 的取值范围是\blkda{$[1,e]$}.
    \begin{jieda}
        $f\left( {x}_{0}\right)  \cdot  g\left( {x}_{0}\right)  \geq  0$即$\left[e^{x_0} - {kx_0}\right] \cdot \left[x_0^{2} + \left( {k - 2}\right) x_0 + \dfrac{1}{4}\right] \geq 0$

        \begin{dingautolist}{172}
            \item 易知$x_0 = 0$时不等式成立
            \item 当$x_0 \neq 0$时，即$\left[\dfrac{e^{x_0}}{x_0} - k\right] \cdot \left[x_0 + \left( {k - 2}\right) + \dfrac{1}{4x_0}\right] \geq 0$
            
            即$\left[k - \dfrac{e^{x_0}}{x_0}\right] \cdot \left[k - \left(2 - \left(x_0 + \dfrac{1}{4x_0}\right)\right)\right] \leq 0$

            \begin{center}
            \includegraphics[width=0.2\textwidth]{bodyxb/src/2026putuoyimo12.png}
            \end{center}

            当$x_0 < 0$，$\dfrac{e^x_0}{x_0} \in (-\infty, 0)$，$2 - \left(x_0 + \dfrac{1}{4x_0}\right) \in [3, +\infty)$

            当$x_0 > 0$，$\dfrac{e^x_0}{x_0} \in [e, +\infty)$，$2 - \left(x_0 + \dfrac{1}{4x_0}\right) \in (-\infty, 1]$

            故$k \in [1, e]$
        \end{dingautolist}
    \end{jieda}
\end{enumerate}
\subsection*{一元二次方程根的分布}
\begin{enumerate}
    \item 已知函数$f(x) = x^2 + 2mx + 2m + 3$恰有一个零点在区间$(0, 2)$内，求实数$m$的取值范围.
    \begin{jieda}
        \begin{dingautolist}{172}
            \item 通解法 \\
            $f(x)$对称轴为$x = -m$，$f(0) = 2m+3$，$f(2) = 7+6m$，$\Delta = 4m^2 - 8m - 12$
            \begin{enumerate}[1.]
                \item 符合函数零点存在定理，此时有$f(0) \cdot f(2) < 0$，即$m \in (-\dfrac{3}{2}, -\dfrac{7}{6})$
                \item 二次函数与$x$轴相切在区间内，此时有$\Delta = 0$，且$-m \in (0, 2)$即$m = -1$
                \item 左端点函数值为0，此时有$m = -\dfrac{3}{2}$，此时$f(2) < 0$，$f(x)$在$(0, 2)$内先减后增，故不存在零点.
                \item 右端点函数值为0，此时有$m = -\dfrac{7}{6}$，此时$f(0) > 0$，$f(x)$在$(0, 2)$内先减后增，存在唯一零点.
            \end{enumerate}
            综上，$m \in (-\dfrac{3}{2}, -\dfrac{7}{6}] \cup \{-1\}$
            \item 参变分离 \\
            $f(x)$有一个零点在区间$(0, 2)$内，即方程$m = \dfrac{-3-x^2}{2x+2}$在$(0, 2)$内有一根，即方程$m = \dfrac{1}{2} \cdot \dfrac{-3-(t-1)^2}{t}$在$t \in (1, 3)$内有一根，即方程$m = \dfrac{1}{2} \cdot [-(t+\dfrac{4}{t})+2]$在$t \in (1, 3)$内有一根，即函数$y = m$与函数$y = \dfrac{1}{2} \cdot [-(t+\dfrac{4}{t})+2]$在$t \in (1, 3)$内只有一个交点.\\
            $y = \dfrac{1}{2} \cdot [-(t+\dfrac{4}{t})+2]$在$(1, 2)$单调增，在$(2, 3)$单调减，因此当$m \in (-\dfrac{3}{2}, -\dfrac{7}{6}] \cup \{-1\}$时满足要求.
        \end{dingautolist}
        \ps 二次函数在区间内只有一个零点的通解法对应着\textbf{四种情况}：\begin{enumerate*}[1.]
            \item 符合函数零点存在定理
            \item $\Delta = 0$且对称轴在区间内
            \item 左端点为函数的一个零点（应根据区间开闭情况单独检验）
            \item 右端点为函数的一个零点（应根据区间开闭情况单独检验）
        \end{enumerate*} \\
        参变分离会将问题转化为动直线和定曲线只有唯一交点，要\textbf{注意相切的情况和端点情况}。\\
        这种情况参变分离和通解法都没有明显优势.
    \end{jieda}

    \item 若关于$x$的方程$-x^2+2x = -mx-m$在$(0, 2)$上恰有两个不等实根，求$m$的取值范围.
    \begin{jieda}
        \begin{dingautolist}{172}
            \item 通解法 \\
            “方程$g(x) = -mx-m$在$(0, 2)$上恰有两个不等实根”即“方程$-x^2+(2+m)x+m = 0$在$(0, 2)$上恰有两个不等实根”，设$h(x) = -x^2+(2+m)x+m$，对称轴为$x = \dfrac{2+m}{2}$，$\Delta = (2+m)^2 + 4m$，$h(0) = m$，$h(2) = 3m$ \\
            $h(x) = 0$在$(0, 2)$上恰有两个不等实根的充要条件是$\left \{\begin{array}{l}
                \Delta > 0  \\
                \dfrac{2+m}{2} \in (0, 2) \\
                h(0) < 0 \\
                h(2) < 0
            \end{array} \right.$，解得$m \in (2\sqrt{3}-4, 0)$
            \item 参变分离 \\
            “方程$g(x) = -mx-m$在$(0, 2)$上恰有两个不等实根”即“方程$m = \dfrac{x^2-2x}{x+1}$在$(0, 2)$上恰有两个不等实根”，即“函数$y = m$与函数$y = t + \dfrac{3}{t} - 4$在$(1, 3)$上恰有两个交点” \\
            $y = t + \dfrac{3}{t} - 4$在$(1, \sqrt{3})$单调减，在$[\sqrt{3}, 3)$上单调增，因此当$m \in (2\sqrt{3}-4, 0)$时满足要求.
        \end{dingautolist}
    \end{jieda}
    \item 若关于$x$的方程$\lg(ax)\cdot\lg(ax^2)=4$有两个小于1的正根$\alpha,\beta$, 
    且满足$\abs{\lg\alpha-\lg\beta}\leq 2\sqrt{3}$, 求实数$a$的取值范围. 
    \begin{jieda}
    $ax>0\land ax^2>0$, 故$a>0,x>0$. \\
    于是原方程可化为$(\lg{a}+\lg{x})(\lg{a}+2\lg{x})=4$. \\
    令$t=\lg{x}$, $t<0$, 则$2t^2+3t\lg{a}+\lg^2a-4=0$有两个负根.\\
    于是$\begin{cases}
    \Delta\geq 0\\
    t_1+t_2<0\\
    t_1t_2>0\\
    \abs{t_1-t_2}\leq 2\sqrt{3}
    \end{cases}$, 
    解得$a\in(100,10000]$. 
    \end{jieda}
\end{enumerate}
