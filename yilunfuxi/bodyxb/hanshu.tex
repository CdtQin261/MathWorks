\subsection*{二次函数根据值域寻找定义域}
\begin{enumerate}
    \item 已知函数$f(x) = x^2$的值域为$[0, 36]$，则下列区间中可以作为$f(x)$的定义域的有\blkda{\ding{173}\ding{174}\ding{175}} \\
    \ding{172}$[2, 6]$，\ding{173}$[-2, 6]$，\ding{174}$[-6, 6]$，\ding{175}$[0, 6]$
    \item 若函数$f(x)=x^2-3x-4$的定义域为$[0,m]$, 值域为$\left[-\dfrac{25}{4},-4\right]$, 则实数$m$的取值范围是\blkda{$\left[\dfrac{3}{2},3\right]$}. 
    \begin{jieda}
    $f(x)=x^2-3x-4=(x-\dfrac{3}{2})^2-\dfrac{25}{4}$，因此$x = \dfrac{3}{2}$是函数值$-\dfrac{25}{4}$对应的唯一自变量，必然要在定义域内. 又因为$f(0)=f(3)=-4$, 是$f(x)=-4$唯二的解. 于是由二次函数的图像易得$m\in\left[\dfrac{3}{2},3\right]$. 
    \end{jieda}
    \item 若一系列函数的解析式相同, 值域相同, 但其定义域不同, 则称这些函数为``同族函数'', 
    那么函数解析式为$f(x)=x^2$, 值域为$\{1,4\}$的``同族函数''共有\blkda{$9$}个.
    \begin{jieda}
    $1,-1,\pm1\to 1$, $2,-2,\pm2\to 4$, 于是从值域的角度, $x$的组合共有$3\times3=9$种选择. \\
    \ps 关键是要清楚一个$y$对应几个$x$.
    \end{jieda} 
\end{enumerate}

    \item 值域是$(0,+\infty)$.

        \item $ax^2+ax+1>0$恒成立. \\

        $a=0$显然不符合条件; \\
        $a\neq 0$时, $a>0$且$\Delta\geq0$, 即$a\geq 4$. \\
        故$a\geq 4$. 
        \end{enumerate}
    \end{jieda}

\subsection*{抽象函数定义域}
\begin{enumerate}
    \item 已知函数$y = f(x)$的定义域是$[-1, 1]$，则函数$y = f\left(\dfrac{1}{x}\right)$的定义域是 \blkda[3]{$(-\infty, -1] \cup [1, +\infty)$}
    \item 已知$y = f(2x+1)$的定义域为$(1, 3]$，则$y = f(x+1)$的定义域为\blkda{$(2, 6]$}. 
    \begin{jieda}
        因为$y = f(2x+1)$的定义域为$(1, 3]$，所以$y = f(x)$的定义域是$(3,7]$，故$y = f(x+1)$的定义域为$(2, 6]$. \\
        \ps 此类题目的桥梁是\textbf{$f$括号内的取值范围是一致的}.
    \end{jieda}
    \item 若函数$f(2x-1)$的定义域为$[-3, 1]$，则$y = \dfrac{f(3-4x)}{\sqrt{x-1}}$的定义域为\blkda{$\left(1, \dfrac{5}{2}\right]$}
    \begin{jieda}
        $f(2x-1)$的定义域为$[-3, 1] \Longleftrightarrow f(x)$的定义域为$[-7, 1]$，故应有$\left\{\begin{array}{l}
             3-4x \in [-7, 1] \\
             x - 1 > 0
        \end{array}\right.$，解集为$\left(1, \dfrac{5}{2}\right]$.
    \end{jieda}
    \item 已知$f(x)=x+2, x\in[1,4]$, 则$y=f(x^2)+f^2(x)$的最大值是\blkda{22}. 
    \begin{jieda}
    先看定义域: $f(x)$的定义域为$[1,4]$，因此$f(x^2)$中$x^2\in[1,4]$, 故$f(x^2)$的定义域为$x\in[-2,2]$. 而$f^2(x)$的定义域与$f(x)$的定义域相同.\\
    两者相加，定义域取交集，因此$y=f(x^2)+f^2(x)$的定义域为$[1, 2]$ \\
    于是$y=x^2+2+(x+2)^2=2x^2+4x+6=2(x+1)^2+4\in[12,22]$. 
    \end{jieda}
\end{enumerate}
\subsection*{求解析式}
\begin{enumerate}
    \item 函数$f(x)$满足$f(x+1) = x^2+2x+2$，求函数$f(x)$的解析式
    \begin{jieda}
        令$t = x+1$，则$x = t-1$，故有$f(t) = (t-1)^2+2(t-1)+2=t^2+1$
    \end{jieda}
    \item 已知$f(\sqrt{x} + 1) = x + 2\sqrt{x}$，求$f(x)$
    \begin{jieda}
        设$t = \sqrt{x}+1 (t \geq 1)$，则$x = (t-1)^2$，故有$f(t) = (t-1)^2 + 2(t-1) = t^2 - 1 (t \geq 1)$ \\
        因此有$f(x) = x^2 - 1 (x \geq 1)$
    \end{jieda} 
    \item 函数$f(x)$满足$2f(x) - f(-x) = x^2$，求函数$f(x)$的解析式
    \begin{jieda}
        $\left \{ \begin{array}{l}
            2f(x) - f(-x) = x^2 \\
            2f(-x) - f(x) = x^2
        \end{array}
        \right.$
        解得$f(x) = x^2$
    \end{jieda}
    \item 已知$2f(x)+f\left(\dfrac{1}{x}\right)=3x$, 则$f(x)$=\blkda[2.5]{$2x - \dfrac{1}{x}$}.
    \begin{jieda}
        $\left \{ \begin{array}{l}
            2f(x)+f\left(\dfrac{1}{x}\right)=3x \\
            2f\left(\dfrac{1}{x}\right)+f(x)=\dfrac{3}{x}
        \end{array}
        \right.$ \\
        上式乘二减下式得到$3f(x) = 6x - \dfrac{3}{x}$，因此$f(x) = 2x - \dfrac{1}{x}$
    \end{jieda}
\end{enumerate}

    \begin{jieda}
        $f\left( x\right)  = \sqrt{1 - {x}^{2}}, - \dfrac{1}{2} \leq  x \leq  \dfrac{1}{2}$ 的图象是圆心角$\dfrac{\pi}{3}$，半径为1的圆弧，根据其函数图像可知$\theta \in \left\lbrack  {0,\dfrac{\pi }{3}}\right\rbrack   \cup  \left\lbrack  {\dfrac{2\pi }{3},\pi }\right\rbrack$
    \end{jieda}

    \item 在平面直角坐标系中,将函数 $y = f\left( x\right)$ 的图像绕坐标原点 $O$ 逆时针旋转 $\dfrac{\pi }{4}$ 后,所得曲线仍然是某个函数的图像,则称函数 $y = f\left( x\right)$ 为 “ $G$ 函数”. 若函数 $f\left( x\right)  = m{x}^{3} + {x}^{2} + {2x} + 1$ 为 “ $G$ 函数”. 则实数 $m$ 的取值范围是\blkda{$\left\lbrack  {\dfrac{1}{3}, + \infty }\right)$}.
    \begin{jieda}
        依题意可得 $f\left( x\right)  = m{x}^{3} + {x}^{2} + {2x} + 1$ 与函数 $y = x + t\left( {t \in  {R}}\right)$ 至多只有一个交点,
        由 $m{x}^{3} + {x}^{2} + {2x} + 1 = x + t$ ,
        即关于 $x$ 的方程 $m{x}^{3} + {x}^{2} + x + 1 = t\left( {t \in  \mathbf{R}}\right)$ 最多只有一根,
        即 $y = m{x}^{3} + {x}^{2} + x + 1$ 与 $y = t\left( {t \in  \mathbf{R}}\right)$ 最多只有一个交点,
        令 $g\left( x\right)  = m{x}^{3} + {x}^{2} + x + 1$ ,则 $g\left( x\right)$ 在 ${R}$ 上单调,
        又 ${g}^{\prime }\left( x\right)  = {3m}{x}^{2} + {2x} + 1$ ,
        当 $m = 0$ 时 ${g}^{\prime }\left( x\right)  = {2x} + 1$ ,则当 $x <  - \dfrac{1}{2}$ 时 ${g}^{\prime }\left( x\right)  < 0$ ,当 $x >  - \dfrac{1}{2}$ 时 ${g}^{\prime }\left( x\right)  > 0$ ,
        即 $g\left( x\right)$ 在 $\left( {-\infty , - \dfrac{1}{2}}\right)$ 上单调递减,在 $\left( {-\dfrac{1}{2}, + \infty }\right)$ 上单调递增,不符合题意;
        当 $m \neq  0$ 时,因为 ${g}^{\prime }\left( 0\right)  = 1 > 0$ ,又 $g\left( x\right)$ 在 ${R}$ 上单调,所以 ${g}^{\prime }\left( x\right)  \geq  0$ 恒成立,
        所以 $\left\{  \begin{array}{l} {3m} > 0 \\  \Delta  = {2}^{2} - 4 \times  {3m} \leq  0 \end{array}\right.$ ,解得 $m \geq  \dfrac{1}{3}$ ;
        综上可得实数 $m$ 的取值范围是 $\left\lbrack  {\dfrac{1}{3}, + \infty }\right)$ .
    \end{jieda}
\end{enumerate}

\subsection*{二次函数零点的分布}
\begin{enumerate}
    \item 已知函数$f(x) = x^2 + 2mx + 2m + 3$
    \begin{enumerate}
        \item 若$f(x)$至少有一个零点在区间$\mathbf{R}$内，求实数$m$的取值范围.
        \item 若$f(x)$至少有一个零点在区间$(0, 2)$内，求实数$m$的取值范围.
        \item 若$f(x)$恰有两个零点在区间$(0, 2)$内，求实数$m$的取值范围.
        \item 若$f(x)$恰有一个零点在区间$(0, 2)$内，求实数$m$的取值范围.
        \item 若$f(x)$恰有一个零点在区间$(0, +\infty)$内，求实数$m$的取值范围.
    \end{enumerate}
    \begin{jieda}
    \begin{enumerate}
        \item $\Delta = 4m^2 - 4(2m+3) = 4m^2 - 8m - 12 \geq 0$，$m \in \left(-\infty, -1\left] \cup \right[3, +\infty\right)$
        \item 
        \begin{dingautolist}{172}
            \item 分对称轴位置讨论 \\
            $f(x)$对称轴为$x = -m$，$f(0) = 2m+3$，$f(2) = 7+6m$，$\Delta = 4m^2 - 8m - 12$
            \begin{enumerate}[1.]
                \item 若$-m \leq 0$，则$f(x)$在$(0, 2)$单调增，故“$f(x)$在$(0, 2)$上有零点”即$f(0) < 0, f(2) > 0$，此种情况无解.
                \item $-m \geq 2$，则$f(x)$在$(0, 2)$单调减，故“$f(x)$在$(0, 2)$上有零点”即$f(0) > 0, f(2) < 0$，此种情况无解.
                \item $-m \in (0, 2)$，则$f(x)$在$(0, 2)$先减后增，“$f(x)$在$(0, 2)$上有零点”即$\Delta \geq 0$，且有$f(0) > 0$或$f(2) > 0$，此种情况解得$m \in (-\dfrac{3}{2}, -1]$
            \end{enumerate}
            综上，$m \in (-\dfrac{3}{2}, -1]$
            \item 参变分离
            
            $f(x)$至少有一个零点在区间$(0, 2)$内，即$f(x) = 0$在$(0, 2)$内有根，即方程$m = \dfrac{-3-x^2}{2x+2}$在$(0, 2)$内有根，即方程$m = \dfrac{1}{2} \cdot \dfrac{-3-(t-1)^2}{t}$在$t \in (1, 3)$内有根，即方程$m = \dfrac{1}{2} \cdot [-(t+\dfrac{4}{t})+2]$在$t \in (1, 3)$内有根.\\
            因此$m$的取值范围即$y = \dfrac{1}{2} \cdot [-(t+\dfrac{4}{t})+2]$在$t \in (1, 3)$内的值域. \\
            $y = \dfrac{1}{2} \cdot [-(t+\dfrac{4}{t})+2]$在$(1, 2)$单调增，在$(2, 3)$单调减，因此值域为$(-\dfrac{3}{2}, -1]$，故$m \in (-\dfrac{3}{2}, -1]$
        \end{dingautolist}
        \item 
        \begin{dingautolist}{172}
            \item 通解法
            
            $\left\{\begin{array}{l}
             \Delta = 4m^2 - 8m - 12 > 0 \\
             -m \in (0, 2) \\
             f(2) = 7+6m > 0 \\
             f(0) = 2m+3 > 0
            \end{array}\right.$，$m \in \left(-\dfrac{7}{6}, -1\right)$

            \item 参变分离

            $y = \dfrac{1}{2} \cdot [-(t+\dfrac{4}{t})+2]$在$(1, 2)$单调增，在$(2, 3)$单调减，因此当$m \in \left(-\dfrac{7}{6}, -1\right)$时满足要求.
        \end{dingautolist}
        
        \item

            \end{enumerate}
            综上，$m \in (-\dfrac{3}{2}, -\dfrac{7}{6}] \cup \{-1\}$
            \item 参变分离

                \item 二次函数与$x$轴相切在区间内，此时有$\Delta = 0$，且$-m \in (0, +\infty)$即$m = -1$
