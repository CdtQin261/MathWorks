\subsection*{三角函数}
\begin{enumerate}
    \item 已知函数 $y=\sin x+a \cos x$ 的图象关于 $x=\dfrac{5}{3} \pi$ 对称，则函数 $y=a \sin x+\cos x$的图象的一条对称轴是\dotfill(\kh{C})
    \xx
    {$x=\dfrac{1}{3} \pi$}
    {$x=\dfrac{2}{3} \pi$}
    {$x=\dfrac{11}{6} \pi$}
    {$x=\pi$}
    \begin{jieda}
        设$f(x) = \sin x+a \cos x$，则$f\left(\dfrac{\pi}{2} - x\right) = a \sin x+\cos x$，因为$x=\dfrac{5}{3} \pi$是$f(x)$的对称轴，所以$x = -\dfrac{7\pi}{6}$是$f\left(\dfrac{\pi}{2} - x\right)$的对称轴，又因为$f(x), f\left(\dfrac{\pi}{2} - x\right) $的周期都是$2\pi$，所以$x = k\pi - \dfrac{7\pi}{6}$是$f\left(\dfrac{\pi}{2} - x\right) $的对称轴，$k = 3$对应$C.$
    \end{jieda}

    \item 函数$y = sin^2x + sinx + 1$的值域为\blkda{$\left[\dfrac{3}{4}, 3\right]$}
    
    \item 函数$f(x) = sin^2x + 2sinx cosx + 3cos^2x$的值域是\blkda[3]{$[2-\sqrt{2}, 2+\sqrt{2}]$}
    
    \item $y=\dfrac{\sin{x}\cos{x}}{1+\sin{x}+\cos{x}}$的值域是\blkda[4.5]{$\left[\dfrac{-\sqrt{2}-1}{2},-1\right)\cup\left(-1,\dfrac{\sqrt{2}-1}{2}\right]$}. 
    \begin{jieda}
    $y=\dfrac{t-1}{2}$, 其中$t=\sin{x}+\cos{x}\in[-\sqrt{2},-1)\cup(-1,\sqrt{2}]$. 
    \end{jieda}
    
    \item 将函数$y = sinx$的图像上的所有点向左平移$\dfrac{\pi}{3}$个长度，再把图像上各点的横坐标变为原来的2倍，则最后得到的图像对应的函数解析式为\blkda[3.5]{$y = sin(\dfrac{x}{2}+\dfrac{\pi}{3})$}
    \begin{jieda}
        平移后函数解析式为$y = sin(x + \dfrac{\pi}{3})$，伸缩变换后函数解析式为$y = sin(\dfrac{x}{2}+\dfrac{\pi}{3})$.
    \end{jieda}

    \item (2025$\cdot$黄浦区一模$\cdot$18)已知 $f\left( x\right)  = \sin x$.
    \begin{enumerate}
        \item 求函数 $y = f\left( x\right)  \cdot  f\left( {\dfrac{\pi }{2} - x}\right)$ 的最小正周期
        \item 求函数 $y = f\left( {{2x} + \dfrac{\pi }{3}}\right) ,x \in  \left\lbrack  {0,\dfrac{\pi }{2}}\right\rbrack$ 的单调减区间.
    \end{enumerate}
    \begin{jieda}
        \begin{enumerate}
            \item 由 $f\left( x\right)  = \sin x$ ,得 $f\left( {\dfrac{\pi }{2} - x}\right)  = \sin \left( {\dfrac{\pi }{2} - x}\right)  =  \cos x$ ,则函数 $y = f\left( x\right)  \cdot  f\left( {\dfrac{\pi }{2} - x}\right)  = \sin x \cdot  \cos x = \dfrac{1}{2}\sin {2x}$ , 故最小正周期为 $\dfrac{2\pi }{2} = \pi$
            
            \item 由 $f\left( x\right)  = \sin x$ ，得 $y = f\left( {{2x} + \dfrac{\pi }{3}}\right)  = \sin \left( {{2x} + \dfrac{\pi }{3}}\right) ,x \in  \left\lbrack  {0,\dfrac{\pi }{2}}\right\rbrack$ ; 由 $0 \leq  x \leq  \dfrac{\pi }{2}$ ,得 $\dfrac{\pi }{3} \leq  {2x} + \dfrac{\pi }{3} \leq  \dfrac{4\pi }{3}$ ,令 $\dfrac{\pi }{2} \leq  {2x} +  \dfrac{\pi }{3} \leq  \dfrac{4\pi }{3}$ ,解得 $\dfrac{\pi }{12} \leq  x \leq  \dfrac{\pi }{2}$ ,故单调减区间为 $\left\lbrack  {\dfrac{\pi }{12},\dfrac{\pi }{2}}\right\rbrack$ .
        \end{enumerate}
    \end{jieda}

    \item 函数 $y = \sin \left( {{\omega x} - \dfrac{\pi }{6}}\right) \left( {\omega  > 0}\right)$ 在 $\left\lbrack  {0,\pi }\right\rbrack$ 上有且仅有 3 个零点,则实数 $\omega$ 的取值范围是\blkda{$\left[\dfrac{13}{6}, \dfrac{19}{6}\right)$}
    

    \item 已知函数$f(x) = sin\left(\omega x + \dfrac{\pi}{3}\right) (\omega > 0)$，若$f(x)$在$\left(\dfrac{\pi}{3}, \pi\right)$上恰有一个零点，则$\omega$的取值范围是\blkda[3]{$\left(\dfrac{2}{3}, \dfrac{5}{3}\right] \cup \left[2, \dfrac{8}{3}\right]$}
        \begin{jieda}
            换元$t = \omega x + \dfrac{\pi}{3}$，$t$与$x$一一对应，
            则问题转化为$g(t) = sint$在$\left(\dfrac{\omega \pi}{3} + \dfrac{\pi}{3}, \omega \pi + \dfrac{\pi}{3} \right)$上恰有一个零点 \\
            
            因为正弦函数的零点为$k\pi, k \in \mathbf{Z}$，所以问题可以进一步转化为：\\
            $\left(\dfrac{\omega \pi}{3}, \omega \pi\right)$与$\left\{\alpha | \alpha = k\pi - \dfrac{\pi}{3}, k \in \mathbf{Z}\right\}$的交集只有一个元素 \\

            通过化简进一步转化为$\left(\dfrac{\omega}{3}, \omega\right)$与$\left\{\alpha | \alpha = k - \dfrac{1}{3}, k \in \mathbf{Z}\right\}$的交集只有一个元素
            
            $\left(\dfrac{\omega}{3}, \omega \right)$是一个长度为$\dfrac{2\omega}{3}$的区间不存在周期，而$\left\{\alpha | \alpha = k\pi - \dfrac{1}{3}, k \in \mathbf{Z}\right\}$存在周期，下面考虑数形结合
            \begin{dingautolist}{172}
                \item 交集中元素为$\dfrac{2}{3}$，结合图像计算边界条件可知$\left \{ \begin{array}{l}
                    0 < \dfrac{\omega}{3} < \dfrac{2}{3} \\
                    \dfrac{2}{3} < \omega \leq \dfrac{5}{3} 
                \end{array} \right.$，解得$\omega$范围是$\left(\dfrac{2}{3}, \dfrac{5}{3}\right]$
                \item 交集中元素为$\dfrac{5}{3}$，结合图像计算边界条件可知$\left \{ \begin{array}{l}
                    \dfrac{2}{3} \leq \dfrac{\omega}{3} < \dfrac{5}{3} \\
                    \dfrac{5}{3} < \omega  \leq \dfrac{8}{3} 
                \end{array} \right.$，解得$\omega$范围是$\left[2, \dfrac{8}{3}\right]$
                \item 交集中元素为$\dfrac{8}{3}$，结合图像计算边界条件可知$\left \{ \begin{array}{l}
                    \dfrac{5}{3} \leq \dfrac{\omega}{3} < \dfrac{8}{3} \\
                    \dfrac{8}{3} < \omega \leq \dfrac{11}{3} 
                \end{array} \right.$，此种情况无解，后续情况均无解.
            \end{dingautolist}
        \end{jieda}

    \item 已知函数 $f\left( x\right)  = \sin \left( {{\omega x} + \varphi }\right)$ ，如图 $A,B$ 是直线 $y = \dfrac{\sqrt{3}}{2}$ 与曲线 $y = f\left( x\right)$ 的两个交点， $f\left( \dfrac{2\pi }{3}\right)  = 0$ 且 $\left| {AB}\right|  = \dfrac{\pi }{12}$ ，则 $f\left( {2023\pi }\right)  =$\blkda{$- \dfrac{\sqrt{3}}{2}$}

    \begin{center}
    \includegraphics[width=0.2\textwidth]{bodyxb/src/0_1221_1854_374_312_0.jpg}
    \end{center}
    \begin{jieda}
        不妨设 $\omega  > 0,A\left( {{x}_{1},\dfrac{\sqrt{3}}{2}}\right) ,B\left( {{x}_{2},\dfrac{\sqrt{3}}{2}}\right)$

        可得 $\sin \left( {\omega {x}_{1} + \varphi }\right)  = \sin \left( {\omega {x}_{2} + \varphi }\right)  = \dfrac{\sqrt{3}}{2},\sin \left( {\dfrac{2\pi }{3}\omega  + \varphi }\right)  = 0$ ,

        由图可知 $A,B,\left( {\dfrac{2\pi }{3},0}\right)$ 在一个周期内,

        则 $\omega {x}_{1} + \varphi  = \dfrac{\pi }{3} + {2k\pi },\omega {x}_{2} + \varphi  = \dfrac{2\pi }{3} + {2k\pi },\dfrac{2\pi }{3}\omega  + \varphi  = {2\pi } + {2k\pi },k \in  \mathbf{Z}$ ,

        又因为 $\left| {AB}\right|  = \dfrac{\pi }{12}$ ,即 ${x}_{2} - {x}_{1} = \dfrac{\pi }{12}$ ,可得 $\omega {x}_{2} - \omega {x}_{1} = \dfrac{\omega \pi }{12} = \dfrac{\pi }{3}$ ,解得 $\omega  = 4$ ,

        则 $\dfrac{2\pi }{3} \times  4 + \varphi  = {2\pi } + {2k\pi },k \in  \mathbf{Z}$ ,解得 $\varphi  =  - \dfrac{2\pi }{3} + {2k\pi },k \in  \mathbf{Z}$ ,

        所以 $f\left( x\right)  = \sin \left( {{4x} - \dfrac{2\pi }{3} + {2k\pi }}\right)  = \sin \left( {{4x} - \dfrac{2\pi }{3}}\right) ,k \in  \mathbf{Z}$ ,

        可知 $f\left( x\right)$ 的最小正周期 $T = \dfrac{2\pi }{4} = \dfrac{\pi }{2}$ ,

        所以 $f\left( {2023\pi }\right)  = f\left( {{2046} \times  \dfrac{\pi }{2}}\right)  = f\left( 0\right)  = \sin \left( {-\dfrac{2\pi }{3}}\right)  =  - \sin \dfrac{2\pi }{3} =  - \dfrac{\sqrt{3}}{2}$ .
    \end{jieda}

    \item 已知函数 $f\left( x\right)  = \sin {\omega x} + \sqrt{3}\cos {\omega x}$ ,若存在 $0 < {x}_{1} < {x}_{2} < \dfrac{\pi }{2}$ ,使得 $f\left( {x}_{1}\right)  - f\left( {x}_{2}\right) = -4$,则正数$\omega$的取值范围是\blkda{$\left(\dfrac{13}{3}, +\infty\right)$}
    \begin{jieda}
        $f(x) = 2sin(\omega x + \dfrac{\pi}{3})$，令$t = \omega x + \dfrac{\pi}{3}$，则存在 $\dfrac{\pi}{3} < {t}_{1} < {t}_{2} < \dfrac{\omega\pi}{2}+\dfrac{\pi}{3}$，使得$sint_1 = -1, sint_2 = 1$，故$t_1 = \dfrac{3\pi}{2}, t_2 = \dfrac{5\pi}{2} + 2k\pi, k \in \mathbf{N}$，故$\dfrac{\omega\pi}{2}+\dfrac{\pi}{3} > \dfrac{5\pi}{2}$，故$\omega \in \left(\dfrac{13}{3}, +\infty\right)$
    \end{jieda}
\end{enumerate}
