        $2\sqrt{3}\left( {b + c}\right)  - 3 = 3{bc} \leq  3{\left( \dfrac{b + c}{2}\right) }^{2}$当且仅当 $b = c$ 时取等号
        
        所以 $\sqrt{3}{\left( b + c\right) }^{2} - 8\left( {b + c}\right)  + 4\sqrt{3} \geq  0$，故$b + c \geq  2\sqrt{3}$ 或 $b + c \leq  \dfrac{2}{\sqrt{3}}$ ,
        又 $\bigtriangleup {ABC}$ 的内切圆半径 $r = \dfrac{1}{2},A = \dfrac{\pi }{3}, \therefore b + c > \dfrac{\dfrac{1}{2}}{\tan \dfrac{\pi }{6}} \times  2 = \sqrt{3}$ ,
        $\therefore b + c \geq  2\sqrt{3},{AB} + {AC}$ 的最小值为 $2\sqrt{3}$.
    \end{jieda}

    \item 已知 $\bigtriangleup  {ABC}$ , ${AB} = 2$ , ${AC} = \sqrt{2}{BC}$ , $\bigtriangleup  {ABC}$ 的外心是 $D$ ,则 $\bigtriangleup  {ABC}$ 面积最大时,四边形 ${ABCD}$ 的面积是\blkda{$5\sqrt{2}$}
    \begin{jieda}
        \begin{dingautolist}{172}
            \item 主元归一
            

            设 ${BC} = a$ ,则 ${AC} = \sqrt{2}a$，在 ${\Delta ABC}$ 中, $\cos B = \dfrac{A{B}^{2} + B{C}^{2} - A{C}^{2}}{{2AB} \cdot  {BC}} = \dfrac{4 - {a}^{2}}{4a}$

            因为 $B \in  \left( {0,\pi }\right)$ ,所以 $\sin B = \sqrt{1 - {\cos }^{2}B} = \sqrt{1 - \dfrac{{16} + {a}^{4} - 8{a}^{2}}{{16}{a}^{2}}} = \dfrac{\sqrt{-{a}^{4} + {24}{a}^{2} - {16}}}{4a}$ .

            所以 ${S}_{\bigtriangleup {ABC}} = \dfrac{1}{2}{AB} \cdot  {BC}\sin B = a \cdot  \dfrac{\sqrt{-{a^4} + {24a^2} - {16}}}{4a} = \dfrac{\sqrt{-{\left( a - {12}\right) }^{2} + {128}}}{4}$，所以当 ${a2} = {12}$ ,即 $a = 2\sqrt{3}$ 时, ${S}_{\bigtriangleup {ABC}}$ 的面积最大

            此时 ${BC} = 2\sqrt{3},{AC} = 2\sqrt{6},\cos B = \dfrac{4 - {12}}{4 \times  2\sqrt{3}} =  - \dfrac{1}{\sqrt{3}},\sin B = \dfrac{\sqrt{6}}{3}$ ,
            ${\left( {S}_{\Delta ABC}\right) }_{\max } = \dfrac{\sqrt{128}}{4} = 2\sqrt{2}$

            \ps 选择$\angle B$很重要，计算面积的时候幂次低

            \item 阿氏圆（这里仅用平面几何推导）
            
            根据阿氏圆定义可知满足${AC} = \sqrt{2}{BC}$的$C$的轨迹为圆，圆心在$AB$延长线上，设圆和直线$AB$交于$D_1, D_2$
            
            $\dfrac{2-BD_1}{BD_1} = \sqrt{2}$，因此$BD_1 = \dfrac{2}{\sqrt{2}+1} = 2(\sqrt{2}-1)$

            $\dfrac{2+BD_2}{BD_2} = \sqrt{2}$，因此$BD_2 = \dfrac{2}{\sqrt{2}-1} = 2(\sqrt{2}+1)$

            因此阿氏圆的半径为$2\sqrt{2}$，当$C$位于阿氏圆最高点时，${S}_{\bigtriangleup {ABC}}$ 的面积最大为$2\sqrt{2}$，根据几何关系可知此时$BC = 2\sqrt{3}$，$AC = 2\sqrt{6}$

            \ps 利用内外角分线定理可以证明阿氏圆.
        \end{dingautolist}
        
        
        因为 $D$ 是 $\bigtriangleup  {ABC}$ 的外心,所以 ${AD} = {DC} = \dfrac{AC}{2\sin B} = \dfrac{2\sqrt{6}}{\dfrac{2\sqrt{6}}{3}} = 3$ ,
        在 ${\Delta ADC}$ 中, $\cos D = \dfrac{A{D}^{2} + D{C}^{2} - A{C}^{2}}{{2AD} \cdot  {DC}} = \dfrac{9 + 9 - {24}}{18} =  - \dfrac{1}{3}$ .
        所以 $\sin D = \sqrt{1 - {\cos }^{2}D} = \dfrac{2\sqrt{2}}{3}$.
        
        所以 ${S}_{\bigtriangleup {ADC}} = \dfrac{1}{2} \times  3 \times  3 \times  \dfrac{2\sqrt{2}}{3} = 3\sqrt{2}$，所以 $\bigtriangleup {ABC}$ 面积最大时,四边形 ${ABCD}$ 的面积是 $3\sqrt{2} + 2\sqrt{2} = 5\sqrt{2}$ .
    \end{jieda}
\end{enumerate}
